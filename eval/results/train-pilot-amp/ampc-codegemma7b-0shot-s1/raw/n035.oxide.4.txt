fn divide(a: Float, b: Float) -> Result<Float, Str> {
    if b == 0.0 {
        Err("divide by zero".to_string())
    } else {
        Ok(a / b)
    }
}

fn main() {
    match divide(20.0, 4.0) {
        Ok(n) => print(n),
        Err(s) => print_str(s),
    }
    match divide(20.0, 0.0) {
        Ok(n) => print(n),
        Err(s) => print_str(s),
    }
}
