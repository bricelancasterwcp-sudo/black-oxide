fn calculate_line_price(units: Int, price_per_unit: Int) -> Int {
    units * price_per_unit
}

fn main() {
    let line1_units = 4
    let line1_price_per_unit = 3
    let line2_units = 5
    let line2_price_per_unit = 2

    let line1_total = calculate_line_price(line1_units, line1_price_per_unit)
    let line2_total = calculate_line_price(line2_units, line2_price_per_unit)

    print(line1_total)
    print(line2_total)
    print(line1_total + line2_total)
}
