```rust
fn main() {
    let fruits = vec!["fig", "banana", "kiwi"];
    let mut max_len = 0;
    let mut max_name = "";

    for fruit in &fruits {
        if fruit.len() > max_len {
            max_len = fruit.len();
            max_name = fruit;
        }
    }

    println!("The name with the most letters is '{}'", max_name);
    println!("It has {} letters.", max_len);
}
```
