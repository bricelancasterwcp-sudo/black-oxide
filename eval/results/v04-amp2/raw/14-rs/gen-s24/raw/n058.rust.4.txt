fn main() {
    let word = "coffee";
    let mut seen = [false; 26];
    for c in word.chars() {
        if let Some(index) = c.to_ascii_lowercase().wrapping_sub('a' as u8) as usize {
            if !seen[index] {
                seen[index] = true;
                println!("{}", c);
            }
        }
    }
}
