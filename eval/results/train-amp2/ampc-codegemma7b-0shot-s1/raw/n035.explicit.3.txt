fn divide(a: Float, b: Float) -> Result<Float, String> {
    if b == 0.0 {
        Err("Division by zero!".to_string())
    } else {
        Ok(a / b)
    }
}

fn main() {
    let result = divide(20.0, 4.0).unwrap_or_else(println)
    print(result)
    let result = divide(20.0, 0.0).unwrap_or_else(println)
    print(result)
}

    print("\n")
}

fn print(n: Int) {
    if n >= 10 {
        print(n / 10)
        print(n % 10)
    } else {
        print(n)
    }
}

fn print(f: Float) {
    let s = int_to_str(trunc(f))
    print(s)
    if f - trunc(f) >= 0.5 {
        print(".5")
    } else if f - trunc(f) >= 0.25 {
        print(".25")
    } else if f - trunc(f) >= 0.125 {
        print(".125")
    } else {
        print(".0")
    }
}

fn int_to_str(n: Int) -> Str {
    if n >= 10 {
        let s = int_to_str(n / 10)
        return concat(s, int_to_str(n % 10))
    } else {
        return str_len("0") + n
    }
}

fn str_len(s: &Str) -> Str {
    let mut len = 0
    for _ in s.chars() {
        len = len + 1
    }
    return len
}

fn concat(a: &Str, b: &Str) -> Str {
    let mut result = a.to_string()
    result.push_str(&b)
    return result
}
