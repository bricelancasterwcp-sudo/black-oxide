fn count_positives(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count = count + 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        match max {
            Some(current_max) => {
                if x > current_max {
                    max = Some(x)
                }
            },
            None => {
                max = Some(x)
            }
        }
    }
    max
}

fn main() {
    let values = vec(3, 8, -2, 12, 7)
    let positives_count = count_positives(clone(values))
    print(positives_count)
    match find_max(values) {
        Some(max_value) => print(max_value),
        None => print_str("No values")
    }
}
