use std::io::{stdin, Read};

fn main() {
    let mut text = String::new();
    stdin().read_to_string(&mut text).unwrap();

    let mut sum = 0;
    let mut count = 0;

    for c in text.chars() {
        if c.is_digit(10) {
            sum += c.to_digit(10).unwrap() as i32;
            count += 1;
        }
    }

    println!("Sum of digits: {}", sum);
    println!("Number of digits: {}", count);
}
