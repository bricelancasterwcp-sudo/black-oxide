fn reverse_digits(n: Int) -> Str {
    let mut digits = Vec<Int>()
    let mut num = n

    while num > 0 {
        let digit = num % 10
        digits = push(digits, digit)
        num = num / 10
    }

    let mut result = Str("")
    for d in digits {
        result = concat(result, int_to_str(d))
    }

    result
}

fn main() {
    print(reverse_digits(1234))
}
