fn main() {
    let mut multiples = vec()
    for i in range(1, 7) {
        multiples = push(multiples, i * 4)
    }
    let total = fold(multiples, 0, add)
    print(total)
}

fn fold(vec: Vec<Int>, init: Int, f: Fn<Int, Int, Int>) -> Int {
    let mut acc = init
    for x in vec {
        acc = f(acc, x, 0)
    }
    acc
}

fn add(a: Int, b: Int, _: Int) -> Int {
    a + b
}
