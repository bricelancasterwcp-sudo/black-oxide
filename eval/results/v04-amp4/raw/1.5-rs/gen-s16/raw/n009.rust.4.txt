```rust
fn main() {
    let num = 1234;
    let mut reversed_num = 0;

    while num > 0 {
        let digit = num % 10; // Capture the current digit
        reversed_num = reversed_num * 10 + digit; // Build the reversed number
        num /= 10; // Remove the last digit from `num`
    }

    println!("Reversed number: {}", reversed_num);
}
```
