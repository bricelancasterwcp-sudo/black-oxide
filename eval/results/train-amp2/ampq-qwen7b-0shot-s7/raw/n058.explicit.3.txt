fn main() {
    let word = "coffee"
    let mut seen = vec()
    for ch in word.chars() {
        if !seen.contains(&ch) {
            let count = word.matches(ch).count()
            if count > 1 {
                print_str(&ch)
                push(seen, ch)
            }
        }
    }
    drop word
    drop seen
}
