```rust
struct Queue {
    items: Vec<Int>,
}

impl Queue {
    fn new() -> Self {
        Queue { items: vec![] }
    }

    fn enqueue(&mut self, item: Int) {
        self.items.push(item);
    }

    fn dequeue(&mut self) -> Option<Int> {
        self.items.pop()
    }

    fn first(&self) -> Option<&Int> {
        self.items.first()
    }

    fn last(&self) -> Option<&Int> {
        self.items.last()
    }
}

fn main() {
    let mut q = Queue::new();
    for i in 1..=5 {
        q.enqueue(i);
    }

    if let Some(first) = q.first() {
        if let Some(last) = q.last() {
            q.enqueue(*last);
            q.dequeue();
            q.enqueue(*first);
            q.dequeue();

            while let Some(item) = q.dequeue() {
                print!("{}", item);
                println!();
            }
        }
    } else {
        println!("Queue is empty");
    }
}
```
