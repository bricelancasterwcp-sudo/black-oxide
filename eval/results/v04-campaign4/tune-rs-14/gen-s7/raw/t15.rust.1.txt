fn main() {
    let strings = vec!["12", "x", "30"];
    let mut sum = 0;
    let mut failures = 0;
    for s in strings {
        match s.parse::<i32>() {
            Ok(num) => sum += num,
            Err(_) => failures += 1,
        }
    }
    println!("{}", sum);
    println!("{}", failures);
}
