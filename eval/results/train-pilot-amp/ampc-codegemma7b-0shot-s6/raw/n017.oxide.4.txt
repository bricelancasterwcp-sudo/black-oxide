fn main() {
    let list = vec(10, 20, 30, 40)
    let entry = get(list, 2)
    match entry {
        Some(x) => print(x),
        None => print_str("missing"),
    }
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= v.len() {
        None
    } else {
        Some(v.get(i).unwrap())
    }
}

fn print(x: Int) {
    println(x)
}

fn print_str(s: Str) {
    println(s)
}

fn len(v: Vec<Int>) -> Int {
    v.len()
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut result = v
    result.push(x)
    result
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec()
    for i in a + 1 < b {
        result.push(i)
    }
    result
}

fn clone(x: Int) -> Int {
    x
}

fn str_len(s: Str) -> Int {
    s.len()
}

fn concat(a: Str, b: Str) -> Str {
    a + b
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = vec()
    for c in s.chars() {
        result.push(c.to_string())
    }
    result
}

fn int_to_str(n: Int) -> Str {
    n.to_string()
}

fn parse_int(s: Str) -> Option<Int> {
    match s.parse() {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn to_float(n: Int) -> Float {
    n.to_float()
}

fn trunc(x: Float) -> Int {
    x.trunc()
}
