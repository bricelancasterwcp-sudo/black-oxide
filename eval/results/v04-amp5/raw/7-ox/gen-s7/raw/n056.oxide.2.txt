fn main() {
    let nums = vec(7, 'x', 9, '?');
    let mut sum = 0;
    let mut count_not_num = 0;

    for x in nums {
        match x {
            '0'..='9' => sum += parse_int(concat("0", &x)).unwrap(),
            _ => count_not_num += 1
        }
    }

    print(sum);
    print(count_not_num);
}
