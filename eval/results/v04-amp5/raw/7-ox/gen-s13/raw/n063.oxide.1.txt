```rust
struct Line { items: Vec<(Int, Int)> }

fn total_cost(line: &Line) -> Int {
    line.items.iter().map(|&(price, count)| price * count).sum()
}

fn main() {
    let line1 = Line { items: vec![(3, 4), (2, 5)] };
    print(total_cost(&line1));
    let line2 = Line { items: vec![(3, 2), (2, 3)] };
    print(total_cost(&line2));
    print(total_cost(&Line { items: [&line1.items, &line2.items].concat() }));
}
```
