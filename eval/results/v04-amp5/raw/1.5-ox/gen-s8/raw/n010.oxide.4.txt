fn main() {
    let num = 91;
    let mut divisors = Vec::new();
    
    for i in 2..=((num as f64).sqrt() as usize + 1) {
        if num % i == 0 {
            divisors.push(i);
            if i != (num as f64).sqrt() as usize {
                divisors.push(num / i);
            }
        }
    }

    if divisors.len() > 2 {
        print_str("yes");
    } else {
        print_str("no");
    }
}
