struct Square { side: Int }
struct Circle { radius: Float }

fn area_of_square(s: Square, pi: Float) -> Float {
    pi * s.side.to_float() * s.side.to_float()
}

fn area_of_circle(c: Circle, pi: Float) -> Float {
    pi * c.radius * c.radius
}

fn main() {
    let square = Square { side: 4 }
    let circle = Circle { radius: 3.0 }
    print(area_of_square(square, 3.14))
}
