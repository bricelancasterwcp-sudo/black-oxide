struct Item { name: Str, unit_price: Int, quantity: Int }
fn main() {
    let pen = Item { name: "pen", unit_price: 3, quantity: 4 }
    let pad = Item { name: "pad", unit_price: 9, quantity: 2 }
    let ink = Item { name: "ink", unit_price: 5, quantity: 3 }
    let total_cost = pen.unit_price * pen.quantity + pad.unit_price * pad.quantity + ink.unit_price * ink.quantity
    print(total_cost)
    let mut max_cost_item = ""
    let mut max_cost = 0
    if pen.unit_price * pen.quantity > max_cost {
        max_cost_item = pen.name
        max_cost = pen.unit_price * pen.quantity
    }
    if pad.unit_price * pad.quantity > max_cost {
        max_cost_item = pad.name
        max_cost = pad.unit_price * pad.quantity
    }
    if ink.unit_price * ink.quantity > max_cost {
        max_cost_item = ink.name
        max_cost = ink.unit_price * ink.quantity
    }
    print(max_cost_item)
}
