use std::io;

fn main() {
    let mut vowels = 0;
    let stdin = io::stdin();
    for line in stdin.lines() {
        for c in line.unwrap().chars() {
            if matches!(c, 'a' | 'A' | 'e' | 'E' | 'i' | 'I' | 'o' | 'O' | 'u' | 'U') {
                vowels += 1;
            }
        }
    }
    println!("Vowels: {}", vowels);
}
