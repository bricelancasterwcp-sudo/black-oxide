fn count_vowels_and_consonants(s: Str) -> (Int, Int) {
    let vowels = "aeiouAEIOU";
    let mut vowel_count = 0;
    let mut consonant_count = 0;

    for c in chars(s) {
        if contains(vowels, c) {
            vowel_count += 1;
        } else if c.is_alpha() {
            consonant_count += 1;
        }
    }

    (vowel_count, consonant_count)
}

fn main() {
    let input = "keyboard";
    let (vowels, consonants) = count_vowels_and_consonants(input);
    print(vowels);
    print(consonants);
}
