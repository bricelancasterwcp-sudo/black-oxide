fn main() {
    let shapes = vec(Shape::Rectangle(Rect { width: 3, height: 4 }), Shape::Triangle(Tri { a: 3, b: 4, c: 5 }), Shape::Rectangle(Rect { width: 2, height: 5 }))
    for shape in shapes {
        match shape {
            Shape::Rectangle(rect) => print(perimeter_rect(rect)),
            Shape::Triangle(tri) => print(perimeter_tri(tri)),
        }
    }
}
enum Shape {
    Rectangle(Rect),
    Triangle(Tri),
}
struct Rect { width: Int, height: Int }
struct Tri { a: Int, b: Int, c: Int }
fn perimeter_rect(r: Rect) -> Int {
    2 * (r.width + r.height)
}
fn perimeter_tri(t: Tri) -> Int {
    t.a + t.b + t.c
}
