struct Int { value: Int }

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return False
    }
    let mut i = 2
    while i * i <= n {
        if n % i == 0 {
            return False
        }
        i = i + 1
    }
    True
}

fn main() {
    let n = Int { value: 91 }
    if is_prime(n) {
        print_str("yes")
    } else {
        print_str("no")
    }
}
