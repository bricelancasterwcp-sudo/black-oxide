fn main() {
    let batch = vec(9, 2, 12, 5, 16)
    for x in &batch {
        if x < 10 {
            print(x)
        }
    }
    let kept = filter(batch, x || x < 10)
    print_str("\nKept entries: ")
    for x in &kept {
        print(x)
    }
    drop kept
}
