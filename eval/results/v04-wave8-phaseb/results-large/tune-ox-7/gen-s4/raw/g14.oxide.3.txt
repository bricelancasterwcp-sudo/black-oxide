struct Reading { label: Str, values: Vec<Int> }
fn main() {
    let r = Reading { label: "lab", values: vec(4, 4, 4, 7, 7, 2, 9, 9, 9, 9, 1, 1, 5) }
    print(len(r.values))
    let (longest_stretch, value) = longest_consecutive(r.values)
    print(longest_stretch)
    print(value)
    print(count_unique(r.values))
    let first_repetition_pos = find_first_repetition(r.values)
    match first_repetition_pos {
        Some(pos) => print(pos),
        None => print_str("no repetition"),
    }
    let never_decreases = is_never_decreasing(r.values)
    print(never_decreases)
}
fn longest_consecutive(v: Vec<Int>) -> (Int, Int) {
    let mut max_stretch = 0
    let mut current_stretch = 1
    let mut last_value = -1
    for x in v {
        if x == last_value {
            current_stretch += 1
        } else {
            if current_stretch > max_stretch {
                max_stretch = current_stretch
            }
            current_stretch = 1
        }
        last_value = x
    }
    if current_stretch > max_stretch {
        max_stretch = current_stretch
    }
    (max_stretch, last_value)
}
fn count_unique(v: Vec<Int>) -> Int {
    let mut seen = vec()
    for x in v {
        if !contains(seen, x) {
            seen = push(seen, x)
        }
    }
    len(seen)
}
fn find_first_repetition(v: Vec<Int>) -> Option<Int> {
    let mut seen = vec()
    for (i, x) in enumerate(v) {
        if contains(seen, x) {
            return Some(i)
        }
        seen = push(seen, x)
    }
    None
}
fn is_never_decreasing(v: Vec<Int>) -> Bool {
    for i in range(1, len(v)) {
        if v[i] < v[i - 1] {
            return false
        }
    }
    true
}
