fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut max_name = &names[0];
    let mut max_count = max_name.len();

    for name in names.iter() {
        if name.len() > max_count {
            max_name = name;
            max_count = name.len();
        }
    }

    println!("{} {}", max_name, max_count);
}
