fn digit_sum(s: &Str) -> Int {
    let mut sum = 0
    let mut count = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            sum = sum + int_to_str(parse_int(c).unwrap()).unwrap().parse().unwrap()
            count = count + 1
        }
    }
    print(sum)
    print(count)
    Unit
}

fn main() {
    let text = "a1b22"
    digit_sum(&text)
}
