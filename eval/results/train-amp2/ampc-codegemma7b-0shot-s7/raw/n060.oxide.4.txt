fn main() {
    let a1b22 = "a1b22"
    let mut digits = 0
    let mut sum = 0
    for c in a1b22.chars() {
        if c.is_digit(10) {
            digits = digits + 1
            sum = sum + c.to_digit(10).unwrap()
        }
    }
    print(digits)
    print(sum)
}
