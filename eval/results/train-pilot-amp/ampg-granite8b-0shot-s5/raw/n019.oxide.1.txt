struct List {
    head: Int,
    tail: Option<List>,
}

fn main() {
    let l1 = List { head: 1, tail: Some(List { head: 2, tail: Some(List { head: 3, tail: None }) }) }
    let l2 = List { head: 4, tail: Some(List { head: 5, tail: None }) }
    let l3 = concat(l1, l2)
    let count = len(l3)
    print(count)
}

fn concat(l1: List, l2: List) -> List {
    match l1.tail {
        None => List { head: l1.head, tail: Some(l2) },
        Some(r1) => List { head: l1.head, tail: Some(concat(r1, l2)) },
    }
}

fn len(l: List) -> Int {
    match l.tail {
        None => 1,
        Some(r) => 1 + len(r),
    }
}

fn int_to_str(n: Int) -> Str {
    let mut v = vec()
    let mut n_copy = n
    while n_copy > 0 {
        let d = trunc(n_copy % 10.0)
        let s = get(range(48, 58), d + 1)?
        v = push(v, s)
        n_copy = trunc(n_copy / 10.0)
    }
    let s = concat(v, vec("".to_string()))
    s
}

    if i >= 0 && i < len(v) {
        Some(get_helper(v, i))
    } else {

    }

fn get_helper(v: Vec<Int>, i: Int) -> Int {
    if i == 0 {
        return head(v)
    } else {
        return get_helper(tail(v)?, i - 1)
    }
}

}
