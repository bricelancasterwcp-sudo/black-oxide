fn main() {
    let mut total = 0
    let mut failures = 0
    for s in vec("12", "x", "30") {
        match parse_int(s) {
            Some(n) => {
                total = total + n
            },
            None => {
                failures = failures + 1
            }
        }
    }
    print(total)
    print(failures)
}
