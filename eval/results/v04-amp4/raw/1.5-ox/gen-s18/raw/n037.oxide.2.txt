struct Rectangle {
    width: Int,
    height: Int,
}

impl Rectangle {
    fn area(&self) -> Int {
        self.width * self.height
    }
}

fn main() {
    let rect = Rectangle { width: 2, height: 3 };
    rect.width = 5; // fix the move error by cloning or reassigning
    println!("{}", rect.area());
}
