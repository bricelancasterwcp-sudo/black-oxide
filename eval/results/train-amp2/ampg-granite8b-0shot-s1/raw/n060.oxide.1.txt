    let b = 10

    print(a1b22(a, b))

fn a1b22(a: Int, b: Int) -> Int {
    let mut c = 0
    let mut d = 0
    for i in range(a, b) {
        c = c + i
        if is_digit(i) {
            d = d + 1
        }
    }
    c + d
}

fn is_digit(x: Int) -> Bool {
    x >= 48 && x <= 57
}

    let mut v = Vec { data: 0, len: 0 }

fn push(mutexisting: Vec<T>, x: T) -> Vec<T> {
    let len = existing.len
    existing.len = len + 1
    if len == 0 {
        existing.data = alloc(1)
    } else {
        let prev = get(existing.data, len - 1)?
        existing.data = alloc(len + 1)
        set(existing.data, len, prev)
    }
    set(existing.data, len, x)
    existing
}

fn len(v: Vec<T>) -> Int {
    v.len
}

fn get(v: Vec<T>, i: Int) -> Option<T> {
    if i >= v.len {
        None
    } else {
        Some(get_unchecked(v.data, i))
    }
}
