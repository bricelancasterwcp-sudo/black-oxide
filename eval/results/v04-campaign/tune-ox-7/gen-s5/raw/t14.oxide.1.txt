fn main() {
    let s = "mississippi"
    print(count_chars(s, 's'))
    print(distinct_chars(s))
}
fn count_chars(s: Str, c: Char) -> Int {
    let mut count = 0
    for ch in chars(s) {
        if ch == c {
            count = count + 1
        }
    }
    count
}
fn distinct_chars(s: Str) -> Int {
    let mut chars_set = HashSet()
    for ch in chars(s) {
        chars_set.insert(ch)
    }
    chars_set.len()
}
