fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    match second(vec(7)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    
    match second(vec(8)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    
    match second(vec(9)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
