struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 5 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    match get(v, 1) {
        Some(n) => Some(n + 1),
        None => None,
    }
}

fn main() {
    let readings = vec![6, 11, 3, 14, 9];
    let count = first_big(readings);
    println!("{} readings exceed 5", count);

    match second(readings) {
        Some(n) => print_str(&n.to_string()),
        None => print_str("too short"),
    }
}
