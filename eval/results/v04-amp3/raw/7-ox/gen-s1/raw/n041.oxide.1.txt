fn sort(v: Vec<Int>) -> Vec<Int> {
    vec().push(v[0]).push(v[1]).push(v[2]).push(v[3])
}

fn main() {
    let measurements = vec(9, 2, 7, 4);
    let sorted = sort(measurements);
    print(sorted[0]);
    print_str("\n");
    print(sorted[1]);
    print_str("\n");
    print(sorted[2]);
    print_str("\n");
    print(sorted[3]);
}
