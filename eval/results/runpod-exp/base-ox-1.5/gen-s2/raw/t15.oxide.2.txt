fn main() {
    let mut sum = 0;
    let mut count = 0;

    for s in vec!["12", "x", "30"] {
        match parse_int(s) {
            Ok(n) => {
                sum += n;
                count += 1;
            }
            Err(_) => {
                println!("Failed to parse '{}'", s);
            }
        }
    }

    println!("Sum: {}", sum);
    println!("Count: {}", count);
}
