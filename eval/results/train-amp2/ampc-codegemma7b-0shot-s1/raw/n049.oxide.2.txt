fn smallest_and_gap(figures: Vec<Int>) -> Result<Vec<Int>, String> {
    if figures.is_empty() {
        return Err("Input vector is empty.".to_string())
    }
    let mut smallest = figures.get(0).copied().unwrap_or(0)
    let mut largest = figures.get(0).copied().unwrap_or(0)
    for figure in figures {
        if figure < smallest {
            smallest = figure
        }
        if figure > largest {
            largest = figure
        }
    }
    Ok(vec(smallest, largest - smallest))
}
