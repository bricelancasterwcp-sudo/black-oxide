fn main() {
    let list = [7, 4, 9, 10];

    let mut odd_sum = 0;

    for &item in list.iter() {
        if item % 2 != 0 {
            println!("{} doubled: {}", item, item * 2);
            odd_sum += item;
        }
    }

    // Adjusted the print statement to correctly add up even numbers
    let total_even = list.iter().filter(|&&x| x % 2 == 0).sum::<i32>();
    println!("Total of even entries: {}", total_even);
}
