fn count_digits_and_sum(s: Str) -> (Int, Int) {
    let mut sum = 0;
    let mut count = 0;
    for ch in chars(s) {
        if parse_int(ch).is_some() {
            let digit = parse_int(ch).unwrap_or(0);
            sum += digit;
            count += 1;
        }
    }
    (sum, count)
}

fn main() {
    let input = "a1b22";
    let (total, num_digits) = count_digits_and_sum(input);
    print(total);
    print(num_digits);
}
