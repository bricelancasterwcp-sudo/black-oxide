struct Reading { label: Str, values: Vec<Int> }

fn sort_ascending(v: Vec<Int>) -> Vec<Int> {
    let mut sorted = v
    for i in range(0, sorted.len()) {
        for j in range(i + 1, sorted.len()) {
            if get(sorted, i) > get(sorted, j) {
                let temp = get(sorted, i)
                sorted = set(sorted, i, get(sorted, j))
                sorted = set(sorted, j, temp)
            }
        }
    }
    sorted
}

fn set(v: Vec<Int>, i: Int, x: Int) -> Vec<Int> {
    let mut v = v
    v = push(v, x)
    v = concat(v, vec())
    v
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 9), 2), 7), 4) }
    let sorted_values = sort_ascending(r.values)
    for x in sorted_values {
        print(x)
    }
}
