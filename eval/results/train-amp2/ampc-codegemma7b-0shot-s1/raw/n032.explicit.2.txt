fn add_numbers(p: &Point) -> Int {
    p.x + p.y
}

fn main() {
    let p1 = Point { x: 1, y: 5 }
    let p2 = Point { x: 10, y: 5 }
    print(add_numbers(&p1) + add_numbers(&p2))
    drop p1
    drop p2
}
