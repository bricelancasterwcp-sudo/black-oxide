```rust
fn main() {
    let numbers: Vec<Option<Int>> = vec![Some(7), None, Some(x), None, Some(9), None];
    let mut total = 0;
    let mut count = 0;

    for number in &numbers {
        if let Some(n) = number {
            total += n;
            count += 1;
        }
    }

    println!("Total: {}", total);
    println!("Count of ones: {}", count);
}
```
