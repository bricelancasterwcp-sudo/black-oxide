```rust
fn main() {
    let mut signal = "green";
    while true {
        match signal {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => panic!("unexpected signal: {}", signal),
        }
        // Simulate arrival of signals
        if let Some(new_signal) = std::thread::sleep(std::time::Duration::from_secs(1)).then(|| {
            match &signal[..] {
                "green" => "red",
                "yellow" => "green",
                _ => "yellow",
            }
        }) {
            signal = new_signal;
        }
    }
}
```
