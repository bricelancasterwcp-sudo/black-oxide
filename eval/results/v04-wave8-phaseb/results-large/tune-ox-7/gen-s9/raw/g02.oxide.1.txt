fn main() {
    let words = vec("rotor", "banana", "level", "quiet", "deified", "system", "civic")
    let mut palindrome_count = 0
    for word in words {
        print_str(word)
        print(len(word))
        let reversed_word = reverse_str(word)
        print_str(reversed_word)
        if word == reversed_word {
            print_str("true")
            palindrome_count += 1
        } else {
            print_str("false")
        }
        let vowel_count = count_vowels(word)
        print(vowel_count)
        let unique_char_count = count_unique_chars(word)
        print(unique_char_count)
    }
    print(palindrome_count)
}
fn reverse_str(s: Str) -> Str {
    let chars = chars(s)
    let reversed_chars = reverse(chars)
    join(reversed_chars, "")
}
fn count_vowels(s: Str) -> Int {
    let vowels = "aeiou"
    let mut count = 0
    for c in chars(s) {
        if contains(vowels, c) {
            count += 1
        }
    }
    count
}
fn count_unique_chars(s: Str) -> Int {
    let chars_set = set(chars(s))
    len(chars_set)
}
