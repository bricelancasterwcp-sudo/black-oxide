use std::io::{self, Read};

fn main() {
    let mut buffer = String::new();
    io::stdin().read_to_string(&mut buffer).unwrap();
    let mut vowels = 0;
    for c in buffer.chars() {
        if ["a", "e", "i", "o", "u"].contains(&c.to_lowercase().to_string().as_str()) {
            vowels += 1;
        }
    }
    println!("The number of vowels is {}", vowels);
}
