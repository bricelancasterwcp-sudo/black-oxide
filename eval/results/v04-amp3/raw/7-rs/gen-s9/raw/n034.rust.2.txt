fn main() {
    let list1 = vec![7];
    let list2 = vec![7, 8];
    let list3 = vec![7, 8, 9];

    match second_entry(&list1) {
        Some(val) => println!("{}", val),
        None => println!("None"),
    }

    match second_entry(&list2) {
        Some(val) => println!("{}", val),
        None => println!("None"),
    }

    match second_entry(&list3) {
        Some(val) => println!("{}", val),
        None => println!("None"),
    }
}

fn second_entry<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}
