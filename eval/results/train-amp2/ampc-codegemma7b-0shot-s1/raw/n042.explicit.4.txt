fn max(x: &Int, y: &Int) -> Int {
    if x > y {
        return x.clone()
    } else {
        return y.clone()
    }
}

fn vec(x: Int, y: Int, z: Int, a: Int, b: Int) -> Vec<Int> {
    push(push(push(push(vec(), x), y), z), a)
}

fn firstunwrap(x: &Vec<Int>) -> Int {
    return x.get(0).unwrap()
}

fn iter(x: &Vec<Int>) -> Vec<Int> {
    let mut y = vec()
    for i in 0.upto(x.len()) {
        y.push(x.get(i).unwrap())
    }
    return y
}

fn upto(x: Int, y: Int) -> Int {
    if x > y {
        return 0
    } else {
        return y - x + 1
    }
}

fn get(x: &Vec<Int>, y: Int) -> Option<Int> {
    if y >= 0 && y < x.len() {
        return Some(x.get(y).unwrap())
    } else {
        return None
    }
}

fn push(x: Vec<Int>, y: Int) -> Vec<Int> {
    let mut z = x
    z.push(y)
    return z
}
