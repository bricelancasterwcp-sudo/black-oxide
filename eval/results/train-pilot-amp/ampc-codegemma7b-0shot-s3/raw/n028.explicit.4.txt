fn read_number() -> Result<Int, Str> {
    let input = read_str()?
    match parse_int(&input) {
        Some(n) => Ok(n),
        None => Err("Invalid input".to_str())?,
    }
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > &limit {
            total = total + x
        }
    }
    total
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v = v
    v.push(x)
    v
}

fn clone(x: &Vec<Int>) -> Vec<Int> {
    let mut v = vec()
    for i in 0.to_int().unwrap() {
        if i < x.len().to_int().unwrap() {
            v.push(x.get(i).unwrap().to_int().unwrap())
        }
    }
    v
}

fn len(v: &Vec<Int>) -> Int {
    let mut len = 0
    for _ in v {
        len = len + 1
    }
    len
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= v.len().to_int().unwrap() {
        None
    } else {
        Some(v.get(i).unwrap().to_int().unwrap())
    }
}

fn str_len(s: &Str) -> Int {
    let mut len = 0
    for _ in s.chars() {
        len = len + 1
    }
    len
}

fn concat(a: &Str, b: &Str) -> Str {
    let mut s = String.new()
    s.push_str(a)
    s.push_str(b)
    s
}

fn chars(s: &Str) -> Vec<Str> {
    let mut v = vec()
    for c in s.chars() {
        v.push(c.to_string())
    }
    v
}

fn int_to_str(n: Int) -> Str {
    n.to_string()
}

fn parse_int(s: &Str) -> Option<Int> {
    match s.parse() {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn trunc(x: Int) -> Int {
    if x >= 0 {
        x
    } else {
        0
    }
}

fn drop_int(x: Int) {
    let _x = x
}

fn drop_str(x: Str) {
    let _x = x
}

fn drop_vec(x: Vec<Int>) {
    for i in 0.to_int().unwrap() {
        if i < x.len().to_int().unwrap() {
            drop_int(x.get(i).unwrap().to_int().unwrap())
        }
    }
}
