fn main() {
    let s = "mississippi"
    print(count(s.chars(), |c| c == 's'))
    let chars: Vec<Str> = s.chars().collect()
    let distinct_chars: Vec<Str> = chars.into_iter().fold(vec![], |mut acc, c| {
        if !acc.contains(&c) {
            acc.push(c)
        }
        acc
    })
    print(distinct_chars.len())
}
