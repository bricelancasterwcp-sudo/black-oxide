struct List { values: Vec<Int> }

fn find_position(v: &List, pos: Int) -> Option<Int> {
    v.values.get(pos).cloned()
}

fn main() {
    let list = List { values: vec![10, 20] };
    if let Some(x) = find_position(&list, 1) {
        print(x);
    }

    if let Some(y) = find_position(&list, 3) {
        print(y);
    }
}
