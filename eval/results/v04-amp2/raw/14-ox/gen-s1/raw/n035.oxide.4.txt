fn safe_divide(numerator: Int, denominator: Int) -> Option<Int> {
    if denominator == 0 {
        None
    } else {
        Some(numerator / denominator)
    }
}

fn main() {
    let result = safe_divide(20, 4)?;
    print(result);
    let bad_result = safe_divide(20, 0);
    match bad_result {
        Some(_) => (),
        None => print_str("bad"),
    }
}
