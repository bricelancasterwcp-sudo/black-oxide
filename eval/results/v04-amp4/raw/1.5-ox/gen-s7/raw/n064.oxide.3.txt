```rust
struct List { values: Vec<Int> }

fn main() {
    let list = List { values: push(push(vec(), 10), 20) };
    match get(list.values, 1) {
        Some(n) => print(n),
        None => println("None"),
    }
    if let Some(n) = get(list.values, 3) {
        print(n);
    } else {
        println!("None");
    }
}
```
