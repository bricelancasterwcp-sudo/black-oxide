fn sum_cubes(n: Int) -> Int {
    let mut sum = 0
    for i in range(1, n + 1) {
        sum = push(sum, cube(i))
    }
    sum
}

fn cube(n: Int) -> Int {
    n * n * n
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a.to(b) {
        v = push(v, i)
    }
    v
}

fn sum(v: Vec<Int>) -> Int {
    foldl(0, v, add)
}

fn add(a: Int, b: Int) -> Int {
    a + b
}

fn foldl(init: T, v: Vec<T>, f: F) -> T {
    let mut acc = init
    for x in v {
        acc = f(acc, x)
    }
    acc
}

fn max(v: Vec<Int>) -> Option<Int> {
    foldl(None, v, max2)
}

fn max2(x: Option<Int>, y: Int) -> Option<Int> {
    if x == None || y > x? {
        Some(y)
    } else {
        x
    }
}
