```rust
struct Square { side: Int }
struct Circle { radius: Float }

impl Shape for Square {
    fn area(&self) -> Int {
        self.side * self.side
    }
}

impl Shape for Circle {
    fn area(&self) -> Float {
        std::f64::consts::PI * self.radius * self.radius
    }
}

fn main() {
    let s = Square { side: 4 };
    print_int(s.area());
}
```
