```rust
fn main() {
    let names = vec!["fig", "banana", "kiwi"];
    let mut max_name = "";
    let mut max_length = 0;

    for name in &names {
        if name.len() > max_length {
            max_length = name.len();
            max_name = name;
        }
    }

    println!("The name with the most letters is: {}", max_name);
    println!("Its letter count is: {}", max_length);
}
```
