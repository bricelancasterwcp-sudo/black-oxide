struct Figure {
    size: Int,
}

fn main() {
    let figures = vec![
        Figure { size: 10 },
        Figure { size: 3 },
        Figure { size: 5 },
        Figure { size: 6 },
    ];

    let smallest = figures.iter().min_by(|a, b| a.size.cmp(&b.size)).unwrap().size;
    println(smallest);

    let largest = figures.iter().max_by(|a, b| a.size.cmp(&b.size)).unwrap().size;
    println(largest - smallest);
}
