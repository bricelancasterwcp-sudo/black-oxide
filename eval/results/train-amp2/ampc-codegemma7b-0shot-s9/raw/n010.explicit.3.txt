fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    for i in 2.to_int() < n {
        if n % i == 0 {
            return false
        }
    }
    return true
}

fn main() {
    let number = 91
    if is_prime(number) {
        print("Yes")
    } else {
        print("No")
    }
}
