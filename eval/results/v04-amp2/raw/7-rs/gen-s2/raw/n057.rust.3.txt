fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut max_len = 0;
    let mut longest_name = "";

    for name in names.iter() {
        if name.len() > max_len {
            max_len = name.len();
            longest_name = name;
        }
    }

    println!("{} {}", longest_name, max_len);
}
