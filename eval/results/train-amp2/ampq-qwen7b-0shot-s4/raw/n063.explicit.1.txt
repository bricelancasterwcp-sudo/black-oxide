fn main() {
    let line1 = push(push(vec(), 3), 3)
    let line2 = push(push(vec(), 2), 2)
    let total = push(push(vec(), 3), 2)
    print(sum_big(&line1, 0))
    print(sum_big(&line2, 0))
    print(sum_big(&total, 0))
    drop line1
    drop line2
    drop total
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn clone(x: &Vec<Int>) -> Vec<Int> {
    x.clone()
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    extend(v, x)
}

fn vec() -> Vec<Int> {
    vec()
}

fn print(x: Int) {
    print_int(x)
}

fn print_int(x: Int) {
    print_str(x.to_string().as_str())
}

fn to_string(s: Str) -> Str {
    s.clone()
}

fn to_int(s: Str) -> Option<Int> {
    parse_int(s)
}

fn parse_int(s: Str) -> Option<Int> {
    s.parse().ok()
}

fn len(v: &Vec<Int>) -> Int {
    v.len()
}

fn sum(v: &Vec<Int>) -> Int {
    let total = 0
    for x in &v {
        total = total + x
    }
    total
}

fn main() {
    let line1 = push(push(vec(), 3), 3)
    let line2 = push(push(vec(), 2), 2)
    let total = push(push(vec(), 3), 2)
    print(sum(&line1))
    print(sum(&line2))
    print(sum(&total))
    drop line1
    drop line2
    drop total
}
