fn main() {
    let nums = [12, 5, 20, 8, 17];
    let mut largest = nums[0];
    let mut second_largest = nums[0];

    for num in nums {
        if num > largest {
            second_largest = largest;
            largest = num;
        } else if num > second_largest && num != largest {
            second_largest = num;
        }
    }

    println!("Largest: {}", largest);
    println!("Second Largest: {}", second_largest);
}
