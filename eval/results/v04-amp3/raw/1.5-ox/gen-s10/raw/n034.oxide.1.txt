```rust
fn second(v: Vec<Int>) -> Option<Int> {
    match v.len() {
        0 => None,
        1 => None,
        _ => Some(v[1]),
    }
}

fn main() {
    for i in 7..=9 {
        let result = second(vec![i]);
        match result {
            Some(n) => println!("Second entry: {}", n),
            None => println!("No second entry"),
        }
    }
}
```
