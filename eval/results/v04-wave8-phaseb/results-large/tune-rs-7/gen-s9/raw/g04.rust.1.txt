fn main() {
    let numbers = [360, 97, 1024, 561, 143, 2003];
    let mut primes_count = 0;
    for &num in &numbers {
        println!("{}", num);
        let (distinct_primes, sum, largest) = find_distinct_primes(num);
        println!("{}", distinct_primes);
        println!("{}", sum);
        println!("{}", largest);
        if is_prime(num) {
            println!("true");
            primes_count += 1;
        } else {
            println!("false");
        }
    }
    println!("{}", primes_count);
}
fn find_distinct_primes(mut num: u64) -> (u64, u64, u64) {
    let mut distinct_primes = 0;
    let mut sum = 0;
    let mut largest = 0;
    for i in 2..=num {
        if num % i == 0 {
            distinct_primes += 1;
            sum += i;
            largest = i;
            while num % i == 0 {
                num /= i;
            }
        }
    }
    (distinct_primes, sum, largest)
}
fn is_prime(num: u64) -> bool {
    if num <= 1 {
        return false;
    }
    for i in 2..=((num as f64).sqrt() as u64) {
        if num % i == 0 {
            return false;
        }
    }
    true
}
