fn sum_digits(s: Str) -> Int {
    let mut result = 0;
    for c in s.chars() {
        if c.is_digit(10) {
            result += c.to_digit(10).unwrap();
        }
    }
    result
}

fn count_digits(s: Str) -> Int {
    sum_digits(s)
}

fn main() {
    let a1b22 = "a1b22";
    print(sum_digits(a1b22));  // Output the total of all digit characters
    print(count_digits(a1b22)); // Output how many digit characters there are
}
