fn sum_digits(s: Str) -> Int {
    let mut sum = 0
    for c in chars(s) {
        if parse_int(c).is_some() {
            sum += to_float(parse_int(c).unwrap()).trunc()
        }
    }
    sum
}

fn count_digits(s: Str) -> Int {
    let mut count = 0
    for c in chars(s) {
        if parse_int(c).is_some() {
            count += 1
        }
    }
    count
}

fn main() {
    let input = "a1b22"
    let total = sum_digits(input)
    let digit_count = count_digits(input)
    print(total)
    print(digit_count)
}
