```rust
fn main() {
    let num = 91;
    let mut divisors = Vec::new();
    
    for i in 2..=((num as f64).sqrt() as usize) + 1 {
        if num % i == 0 {
            divisors.push(i);
            if i != (num / i) {
                divisors.push(num / i);
            }
        }
    }
    
    if divisors.is_empty() {
        println!("no");
    } else {
        println!("yes");
    }
}
```
