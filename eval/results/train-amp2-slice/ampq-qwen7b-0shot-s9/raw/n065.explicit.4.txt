fn main() {
    let readings = vec(6, 11, 3, 14, 9)
    let mut count = 0
    let mut max_reading = 0
    for reading in &readings {
        if reading > 5 {
            count = count + 1
        }
        if reading > max_reading {
            max_reading = reading
        }
    }
    print(count)
    print_str(" ")
    print(max_reading)
    drop readings
}
