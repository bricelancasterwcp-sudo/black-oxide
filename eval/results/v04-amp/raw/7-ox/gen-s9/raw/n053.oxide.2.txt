fn reverse(s: Str) -> Str {
    let mut chars = chars(s)
    let mut reversed = vec()
    while !chars.is_empty() {
        reversed = push(reversed, chars.pop())
    }
    concat(reversed, "")
}

fn main() {
    let word = "drum"
    print(reverse(word))  // prints "murd"
    print_str("\n")       // newline
    print(word)         // prints "drum"
}
