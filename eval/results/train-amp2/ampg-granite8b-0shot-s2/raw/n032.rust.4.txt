fn main() {
    let first_point = Point { x: 1, y: 5 };
    let second_point = Point { x: 10, y: 5 };

    println!(
        "The sum of the coordinates of the first point is: {}",
        first_point.x + first_point.y
    );

    println!(
        "The sum of the coordinates of the second point is: {}",
        second_point.x + second_point.y
    );
}

struct Point {
    x: i32,
    y: i32,
}
