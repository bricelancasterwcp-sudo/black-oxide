fn five_larger_than_four(v: &Vec<Int>) -> Int {
    let four = Int(4) // define four
    let mut count = 0
    for x in v {
        if x > &four {
            count = count + 1
        }
    }
    count
}

fn main() {
    let v = vec(1, 2, 3, 5, 7, 8, 10)
    print(five_larger_than_four(&v))
    drop v
}
