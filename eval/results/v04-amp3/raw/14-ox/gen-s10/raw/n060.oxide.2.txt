fn count_digits_and_sum(s: Str) -> (Int, Int) {
    let sum = 0
    let count = 0
    for c in chars(s) {
        if parse_int(c).is_some() {
            sum += parse_int(c).unwrap()
            count += 1
        }
    }
    (sum, count)
}

fn main() {
    let input = "a1b22"
    let (sum, count) = count_digits_and_sum(input)
    print(sum)
    print(count)
}
