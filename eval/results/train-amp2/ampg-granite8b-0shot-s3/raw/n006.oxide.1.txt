struct PowerOfTwo { value: Int }

fn is_power_of_two(x: Int) -> Bool {
    x > 0 && (x && (x - 1)) == 0
}

fn next_bigger_power_of_two(x: Int) -> Option<PowerOfTwo> {
    let mut x = x
    while x < 100 {
        x = x * 2
    }
    if x == 100 {
        Some(PowerOfTwo { value: x })
    } else {
        None
    }
}

fn main() {
    match next_bigger_power_of_two(5) {
        Some(p) => print(p.value),
        None => print_str("none"),
    }
}
