struct Sitting { label: Str, scores: Vec<Int> }
fn count_in_range(v: Vec<Int>, min: Int, max: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= min && x <= max {
            count += 1
        }
    }
    count
}
fn main() {
    let morning = Sitting { label: "Morning", scores: vec(94, 71, 88, 65, 92, 79, 83) }
    let evening = Sitting { label: "Evening", scores: vec(55, 90, 77, 84, 68, 99) }
    for s in vec(morning.clone(), evening.clone()) {
        print_str(s.label)
        let count_90_or_above = count_in_range(s.scores.clone(), 90, 100)
        let count_80_to_89 = count_in_range(s.scores.clone(), 80, 89)
        let count_70_to_79 = count_in_range(s.scores.clone(), 70, 79)
        let count_below_70 = count_in_range(s.scores.clone(), 0, 69)
        print(count_90_or_above)
        print(count_80_to_89)
        print(count_70_to_79)
        print(count_below_70)
        let max_score = match max(s.scores) {
            Some(n) => n,
            None => -1,
        }
        print(max_score)
        let mean_score = sum(s.scores) / len(s.scores)
        print(mean_score)
    }
    let morning_mean = sum(morning.scores) / len(morning.scores)
    let evening_mean = sum(evening.scores) / len(evening.scores)
    if morning_mean > evening_mean {
        print_str("Morning")
    } else {
        print_str("Evening")
    }
}
