fn main() {
    let fig = "fig"
    let banana = "banana"
    let kiwi = "kiwi"

    let lengths = vec()
    push(lengths, str_len(fig))
    let len_banana = str_len(banana)
    push(lengths, len_banana)
    let len_kiwi = str_len(kiwi)
    push(lengths, len_kiwi)

    let max_len = unwrap_or(max(lengths), -1)
    let name = if max_len == str_len(fig) {
        fig
    } else if max_len == len_banana {
        banana
    } else {
        kiwi
    }

    print(name)
    print(max_len)
}
