```rust
fn main() {
    let numbers = vec![5, 12, 3, 18, 9];
    let mut count_below_cutoff = 0;
    let mut count_at_or_over_cutoff = 0;

    for &number in &numbers {
        if number < 10 {
            count_below_cutoff += 1;
        } else {
            count_at_or_over_cutoff += 1;
        }
    }

    println!("Numbers below cutoff: {}", count_below_cutoff);
    println!("Numbers at or over cutoff: {}", count_at_or_over_cutoff);
}
```
