fn main() {
    let text = "order 47 shipped on 2026-03-15, invoice 1290 paid.";
    println!("{}", text.len());
    let mut letters = 0;
    let mut digits = 0;
    let mut spaces = 0;
    let mut others = 0;
    let mut max_digit_streak = 0;
    let mut current_digit_streak = 0;
    for c in text.chars() {
        if c.is_alphabetic() {
            letters += 1;
        } else if c.is_digit(10) {
            digits += 1;
            current_digit_streak += 1;
        } else if c == ' ' {
            spaces += 1;
            current_digit_streak = 0;
        } else {
            others += 1;
            current_digit_streak = 0;
        }
        max_digit_streak = max_digit_streak.max(current_digit_streak);
    }
    println!("{}", letters);
    println!("{}", digits);
    println!("{}", spaces);
    println!("{}", others);
    println!("{}", max_digit_streak);
    let mut digit_string = String::new();
    for c in text.chars() {
        if c.is_digit(10) {
            digit_string.push(c);
        }
    }
    println!("{}", digit_string);
}
