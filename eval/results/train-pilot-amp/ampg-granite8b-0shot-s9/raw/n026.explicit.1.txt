fn int_to_str(number: Int) -> Str {
    let mut text = vec()
    let mut number = number
    let digit: Int = 10
    while number > 0 {
        let digit_value = number % digit
        let digit_text = match digit_value {
            LOWER_0 => "0",
            LOWER_1 => "1",
            LOWER_2 => "2",
            LOWER_3 => "3",
            LOWER_4 => "4",
            LOWER_5 => "5",
            LOWER_6 => "6",
            LOWER_7 => "7",
            LOWER_8 => "8",
            LOWER_9 => "9",
            _ => panic("unknown digit value"),
        }
        text = push(text, digit_text)
        number = number / digit
    }
    concat(&text, "")
}

fn str_len(text: &Str) -> Int {
    len(&chars_in_str)
}

fn chars_in_str(text: &Str) -> Vec<Str> {
    let text_length = str_len(text)
    let mut result = vec()
    for i in 0.to_int(text_length) {
        result = push(result, get_char(text, i))
    }
    result
}

fn get_char(text: &Str, i: Int) -> Str {
    get(&chars_in_str, i)
}

fn panic(message: &Str) -> Never {
    print_str("\npanic: ")
    print(&message)
    abort()
}

    loop()
}
