```rust
fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut count = 0;
    let mut total = 0;

    for text in texts {
        if let Ok(num) = text.parse::<i32>() {
            total += num;
            count += 1;
        }
    }

    println!("Total of numbers: {}", total);
    println!("Numbers not counted: {}", texts.len() - count);
}
```
