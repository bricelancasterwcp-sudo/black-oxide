```rust
fn second(v: Vec<Int>) -> Option<Int> {
    let x = v[1];
    Some(x)
}

fn main() {
    for n in &[7, 8, 9] {
        match second(vec![1, *n]) {
            Some(n) => println!("second entry of list with {} is {}", n, *n),
            None => println!("list with {} does not have a second element", *n),
        }
    }
}
```
