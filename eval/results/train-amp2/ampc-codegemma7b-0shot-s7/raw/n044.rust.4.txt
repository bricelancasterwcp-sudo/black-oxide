fn main() {
    let mut balance = 0;

    deposit(3);
    println!("{}", balance);

    deposit(1);
    println!("{}", balance);

    deposit(4);
    println!("{}", balance);

    deposit(1);
    println!("{}", balance);

    deposit(5);
    println!("{}", balance);
}

fn deposit(amount: i32) {
    let mut current_balance = balance;
    current_balance += amount;
    balance = current_balance;
}
