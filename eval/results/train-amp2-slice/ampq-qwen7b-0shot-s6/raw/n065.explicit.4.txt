    let above_five: Vec<Int> = readings.iter().filter_map(move || if_then_else(reading > 5, Some(reading), None))

fn max(v: &Vec<Int>) -> Int {
    v.into_iter().max().unwrap_or(0)
}

fn filter_map(predicate: &FnMut<Int>) -> Vec<Int> {
    let mut result = VecDeque.new()
    for item in v.iter() {
        if predicate(item) {
            result.push_back(item.clone())
        }
    }
    result.into()
}

fn collect() -> Vec<Int> {
    VecDeque.new().into()
}

fn if_then_else(condition: Bool, then_branch: Option<Int>, else_branch: Option<Int>) -> Option<Int> {
    if condition {
        then_branch
    } else {
        else_branch
    }
}

fn main() {
    let readings = vec(6, 11, 3, 14, 9)
    let above_five: Vec<Int> = readings.iter().filter_map(move || if_then_else(reading > 5, Some(reading), None))
    print(above_five.len())
    print(max(&readings))
    drop readings
}
