fn main() {
    let mut total = Int::zero();
    let mut carry = false;

    for &x in [7, 4, 9, 10].iter().rev() {
        if x % 2 == 1 {
            print!("{}\n", 2 * x);
            carry = true;
        } else {
            if carry {
                total += Int::from(x) + Int::one();
                carry = false;
            }
        }
    }

    if carry {
        total += Int::from(1);
    }

    println!("Total of even entries: {}", total);
}
