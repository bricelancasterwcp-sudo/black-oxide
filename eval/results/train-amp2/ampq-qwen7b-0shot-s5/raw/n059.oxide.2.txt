struct Reading { label: Str, values: Vec<Int> }

fn count_words(s: Str) -> Int {
    let words: Vec<Str> = chars(s).filter_not(s == " ").map(int_to_str).collect()
    words.len()
}

fn count_chars(s: Str) -> Int {
    str_len(s)
}

fn main() {
    let phrase = "when in doubt go home"
    let words_count = count_words(phrase)
    let chars_count = count_chars(phrase)
    print(words_count)
    print(" ")
    print(chars_count)
}
