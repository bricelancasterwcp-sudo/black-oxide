fn main() {
    let r = Rectangle { width: 2, height: 3 }
    let r = Rectangle { width: 5, ..r }
    print(area(&r))
}

struct Rectangle {
    width: Int,
    height: Int,
}

fn area(r: &Rectangle) -> Int {
    r.width * r.height
}

fn main() {
    let r = Rectangle { width: 2, height: 3 }
    let r = Rectangle { width: 5, ..r }
    print(area(&r))
}
