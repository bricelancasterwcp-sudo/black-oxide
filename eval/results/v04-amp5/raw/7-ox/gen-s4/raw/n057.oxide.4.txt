```rust
fn main() {
    let fruits = vec!["fig", "banana", "kiwi"];
    let mut max_len = 0;
    let mut fruit_name = "";

    for fruit in fruits {
        if fruit.len() > max_len {
            max_len = fruit.len();
            fruit_name = fruit;
        }
    }

    print(fruit_name);
    print(max_len.to_string());
}
```
