fn int_to_str(n: Int) -> Str {
    let mut s = Str()
    let mut m = n
    while m > 0 {
        let d = m % 10
        m = m / 10
        let ch = char_from_digit(d)
        s = push(s, ch)
    }
    s
}

fn char_from_digit(d: Int) -> Str {
    match d {
        DIGIT_0 => "0",
        DIGIT_1 => "1",
        DIGIT_2 => "2",
        DIGIT_3 => "3",
        DIGIT_4 => "4",
        DIGIT_5 => "5",
        DIGIT_6 => "6",
        DIGIT_7 => "7",
        DIGIT_8 => "8",
        DIGIT_9 => "9",
        _ => "",
    }
}

fn str_len(s: Str) -> Int {
    let mut len = 0
    for _ in s {
        len = len + 1
    }
    len
}

fn main() {
    let n = 2026
    let s = int_to_str(n)
    print_str(s)
    print(str_len(s))
}
