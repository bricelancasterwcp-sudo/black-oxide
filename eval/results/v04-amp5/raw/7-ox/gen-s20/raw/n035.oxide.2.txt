fn divide(n: Int, d: Int) -> Result<Int, Str> {
    if d == 0 {
        Err("bad".to_str())
    } else {
        Ok(n / d)
    }
}

fn main() {
    let result = match divide(20, 4)? {
        Ok(val) => val,
        Err(msg) => {
            print_str(msg);
            return
        },
    };
    print(result);
}
