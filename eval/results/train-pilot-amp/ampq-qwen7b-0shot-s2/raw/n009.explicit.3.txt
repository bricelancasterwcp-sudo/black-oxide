fn main() {
    let s = "1234"
    let chars = chars(s)
    let reversed_chars: Vec<Str> = chars.iter().rev().cloned().collect()
    let reversed_str = concat(int_to_str(str_len(&reversed_chars)), &reversed_chars)
    print_str(reversed_str)
    drop s
    drop chars
    drop reversed_chars
    drop reversed_str
}
