fn count_vowels_and_consonants(word: &str) {
    let mut vowel_count = 0;
    let mut consonant_count = 0;

    for c in word.chars() {
        if matches!(c, 'a' | 'e' | 'i' | 'o' | 'u') {
            vowel_count += 1;
        } else if !matches!(c, 'A'..='Z') && (c.is_ascii_lowercase() || c.is_ascii_uppercase()) { // Check for lower and upper case vowels
            consonant_count += 1;
        }
    }

    println!("Number of vowels: {}", vowel_count);
    println!("Number of consonants: {}", consonant_count);
}

fn main() {
    let keyboard_word = "keyboard";
    count_vowels_and_consonants(keyboard_word);
}
