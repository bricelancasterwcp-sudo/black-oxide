fn println(x: Int) {
    let mut out = "0"
    loop_while(x > 0, x == 0, x == 0, x == 0, x == 0, x == 0)
    drop out
}

fn loop_while(cond1: Bool, cond2: Bool, cond3: Bool, cond4: Bool, cond5: Bool, cond6: Bool) {
    if cond1 {
        if cond2 {
            if cond3 {
                if cond4 {
                    if cond5 {
                        if cond6 {
                            return
                        } else {
                            drop cond6
                        }
                    } else {
                        drop cond5
                    }
                } else {
                    drop cond4
                }
            } else {
                drop cond3
            }
        } else {
            drop cond2
        }
    } else {
        drop cond1
    }
}

}
