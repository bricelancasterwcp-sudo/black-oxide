struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 5), 3), 9), 1), pushvec: 7 } // Corrected push usage
    print(first_big(r.values))            // field access copies: r stays usable
}
