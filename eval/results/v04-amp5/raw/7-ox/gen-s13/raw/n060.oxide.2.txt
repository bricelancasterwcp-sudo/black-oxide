fn main() {
    let s = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in chars(s) {
        if is_digit(c) {
            sum += parse_int(concat(String::from(c), "")).unwrap();
            count += 1;
        }
    }

    print(sum);
    print(count);
}

fn is_digit(c: Str) -> Bool {
    '0' <= c && c <= '9'
}
