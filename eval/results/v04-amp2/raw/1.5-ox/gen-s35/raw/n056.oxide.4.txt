fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut total = 0;
    let mut count = 0;

    for text in texts {
        match parse_int(text) {
            Some(n) => {
                total += n;
                count += 1;
            }
            None => {}
        }
    }

    println!("Total: {}", total);
    println!("Count: {}", count);
}
