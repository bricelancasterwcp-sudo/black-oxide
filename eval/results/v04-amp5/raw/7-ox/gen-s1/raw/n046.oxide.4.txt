fn count_under_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    count_if(v, |x| x < cutoff)
}

fn count_over_or_equal_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    count_if(v, |x| x >= cutoff)
}

fn main() {
    let mut samples = push(push(push(push(vec(), 5), 12), 3), 18);
    samples.push(9);
    print(count_under_cutoff(samples.clone(), 10));
    print_str(" ");
    print(count_over_or_equal_cutoff(samples, 10));
}
