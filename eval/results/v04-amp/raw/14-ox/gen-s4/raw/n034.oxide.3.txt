fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let list1 = push(push(vec(), 7), 8)
    let list2 = push(push(vec(), 8), 9)
    let list3 = push(vec(), 9)

    match second(list1) {
        Some(n) => print(n),
        None => print_str("none"),
    }

    match second(list2) {
        Some(n) => print(n),
        None => print_str("none"),
    }

    match second(list3) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
