fn main() {
    let north = vec![
        ("bolt", 40, 3),
        ("hinge", 8, 25),
        ("plate", 15, 12),
    ];
    let south = vec![
        ("clamp", 6, 45),
        ("rod", 22, 9),
        ("washer", 100, 1),
        ("gasket", 4, 30),
    ];
    for (i, warehouse) in vec![north, south].iter().enumerate() {
        let name = if i == 0 { "North" } else { "South" };
        let total_value: i32 = warehouse.iter().map(|(_, q, p)| q * p).sum();
        let below_ten_count = warehouse.iter().filter(|&(q, _, _)| *q < 10).count();
        let (max_part_name, _, max_part_value) = warehouse.iter().max_by_key(|&(_, q, p)| q * p).unwrap();
        println!("{}", name);
        println!("{}", total_value);
        println!("{}", below_ten_count);
        println!("{}", max_part_name);
    }
    if north.iter().map(|(_, q, p)| q * p).sum::<i32>() > south.iter().map(|(_, q, p)| q * p).sum::<i32>() {
        println!("North");
    } else {
        println!("South");
    }
}
