fn max(a: &Str, b: &Str) -> Option<String> {
    if a.len() > b.len() {
        Some(a.to_string())
    } else if b.len() > a.len() {
        Some(b.to_string())
    } else {
        None
    }
}

fn main() {
    let fig = "fig".to_string()
    let banana = "banana".to_string()
    let kiwi = "kiwi".to_string()
    let result = max(&fig, &banana)
    match result {
        Some(fruit) => print(fruit),
        None => print("No fruit found"),
    }
    drop fig
    drop banana
    drop kiwi
}
