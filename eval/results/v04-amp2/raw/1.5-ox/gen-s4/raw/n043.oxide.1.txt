```rust
struct Queue {
    items: Vec<Int>,
}

impl Queue {
    fn new() -> Self {
        Queue { items: vec![1, 2, 3, 4, 5] }
    }

    fn exchange_first_last(&mut self) {
        let first = self.items.remove(0);
        let last = self.items.pop().unwrap();
        self.items.insert(0, last);
        self.items.push(first);
    }

    fn print_queue(&self) {
        for &item in &self.items {
            print!("{}", item);
            if item != *self.items.last().unwrap() {
                println!(", ");
            }
        }
        println!();
    }
}

fn main() {
    let mut q = Queue::new();
    q.exchange_first_last();
    q.print_queue();
}
```
