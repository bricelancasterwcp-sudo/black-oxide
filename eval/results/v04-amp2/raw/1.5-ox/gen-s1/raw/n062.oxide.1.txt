```rust
fn main() {
    let mut signal = "green";
    while true {
        match signal {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => panic!("unexpected signal"),
        }
        if signal == "red" {
            break;
        }
        // Wait for the next signal
    }
}
```
