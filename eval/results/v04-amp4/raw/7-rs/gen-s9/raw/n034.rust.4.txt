fn main() {
    let list1 = vec![7];
    let list2 = vec![7, 8];
    let list3 = vec![7, 8, 9];

    fn get_second<T>(vec: &Vec<T>) -> Option<&T> {
        vec.get(1)
    }

    println!("{:?}", get_second(&list1)); // None
    println!("{:?}", get_second(&list2)); // Some(&8)
    println!("{:?}", get_second(&list3)); // Some(&8)
}
