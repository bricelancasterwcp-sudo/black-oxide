fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        false
    } else {
        for i in range(2, trunc(to_float(n) ** 0.5) + 1) {
            if n % i == 0 {
                return false;
            }
        }
        true
    }
}

fn count_primes_below_100() -> Int {
    let mut count = 0;
    for n in range(2, 100) {
        if is_prime(n) {
            count += 1;
        }
    }
    count
}

fn main() {
    print(count_primes_below_100());
}
