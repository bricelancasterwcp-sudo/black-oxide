fn main() {
    let word = "keyboard";
    let vowels = "aeiou";
    let mut vowel_count = 0;
    let mut consonant_count = 0;

    for letter in word.chars() {
        if vowels.contains(letter.to_ascii_lowercase()) {
            vowel_count += 1;
        } else if letter.is_alphabetic() {
            consonant_count += 1;
        }
    }

    println!("Vowels: {}", vowel_count);
    println!("Consonants: {}", consonant_count);
}
