```rust
fn main() {
    let text = "keyboard";

    let mut vowel_count = 0;
    let mut consonant_count = 0;

    for c in text.chars() {
        if ["a", "e", "i", "o", "u"].contains(&c.to_lowercase().as_str()) {
            vowel_count += 1;
        } else {
            consonant_count += 1;
        }
    }

    println!("Vowels: {}", vowel_count);
    println!("Consonants: {}", consonant_count);
}
```
