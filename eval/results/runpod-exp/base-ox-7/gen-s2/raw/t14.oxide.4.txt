fn count_s(text: Str) -> Int {
    let mut count = 0;
    for c in chars(text) {
        if c == "s" {
            count += 1;
        }
    }
    count
}

fn distinct_chars(text: Str) -> Int {
    let mut seen = Vec::new();
    for c in chars(text) {
        if !contains(seen, c) {
            push(seen, c);
        }
    }
    len(seen)
}

fn main() {
    let text = "mississippi";
    print(count_s(text));
    print_str(" ");
    print(distinct_chars(text));
}
