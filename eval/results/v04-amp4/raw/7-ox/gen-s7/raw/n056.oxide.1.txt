fn main() {
    let nums = vec(7, 'x', 9, '?');
    let mut sum = 0;
    let mut count = 0;

    for n in nums {
        match parse_int(n) {
            Some(num) => {
                sum += num;
            },
            None => {
                count += 1;
            }
        }
    }

    print(sum);
    print(count);
}
