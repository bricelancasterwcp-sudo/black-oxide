fn reverse_digits(n: Int) -> Int {
    let mut result = 0
    let mut remaining = n
    while remaining > 0 {
        let digit = remaining % 10
        result = result * 10 + digit
        remaining = remaining / 10
    }
    result
}

fn main() {
    let number = 1234
    print(reverse_digits(number))
}
