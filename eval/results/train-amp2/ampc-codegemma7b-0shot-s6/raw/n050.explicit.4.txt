fn stdcmp(a: &Int, b: &Int) -> Ordering {
    a.cmp(b)
}

enum Ordering {
    Less,
    Greater,
    Equal,
}

fn cmp(a: &Int, b: &Int) -> Ordering {
    if a < b {
        Ordering.less
    } else if a > b {
        Ordering.greater
    } else {
        Ordering.equal
    }
}

fn print(x: Int) {
    let s = int_to_str(x)
    print_str(s)
}

fn print_str(s: Str) {
    let chars = chars(s)
    for c in chars {
        print(c)
    }
}

fn vec(elem: T) -> Vec<T> {
    let mut result = Vec { len: 0, buf: vec() }
    push(result, elem)
    result
}

fn push(mutvec: Vec<T>, elem: T) -> Vec<T> {
    let len = vec.len
    vec.buf.push(elem)
    vec.len = len + 1
    vec
}

fn int_to_str(n: Int) -> Str {
    let mut result = Str { len: 0, buf: vec() }
    let mut x = n
    while x > 0 {
        let digit = x % 10
        let digit_str = int_to_str(digit)
        push(result, digit_str)
        x = trunc(x / 10)
    }
    result
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = Vec { len: 0, buf: vec() }
    let mut index = 0
    while index < s.len {
        let c = s.buf(index)
        let c_str = str_len(c)
        push(result, c_str)
        index = index + 1
    }
    result
}

fn str_len(s: Str) -> Str {
    let mut result = Str { len: 0, buf: vec() }
    let mut index = 0
    while index < s.len {
        index = index + 1
        result.len = result.len + 1
    }
    result
}

fn trunc(x: Int) -> Int {
    if x >= 0 {
        x
    } else {
        x - 1
    }
}

        for j in i + 1.upto(vec.len - 1) {
            let ordering = compare(vec.buf(i), vec.buf(j))
            if ordering == Ordering.greater {
                let temp = vec.buf(i)
                vec.buf(i) == vec.buf(j)
                vec.buf(j) == temp
            }
        }
    }

}
