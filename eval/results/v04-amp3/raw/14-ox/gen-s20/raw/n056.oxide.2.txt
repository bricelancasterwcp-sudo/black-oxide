fn sum_valid_numbers(texts: Vec<Str>) -> Unit {
    let mut sum = 0;
    let mut count_invalid = 0;

    for text in texts {
        match parse_int(text) {
            Some(n) => sum = sum + n,
            None => count_invalid = count_invalid + 1,
        }
    }

    print(sum);
    print(count_invalid);
}

fn main() {
    let texts = vec("7", "x", "9", "?");
    sum_valid_numbers(texts);
}
