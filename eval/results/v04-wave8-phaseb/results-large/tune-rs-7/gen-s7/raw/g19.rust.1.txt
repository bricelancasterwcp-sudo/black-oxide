fn main() {
    let text = "order 47 shipped on 2026-03-15, invoice 1290 paid.";
    println!("{}", text.len());
    let (mut letters, mut digits, mut spaces) = (0, 0, 0);
    for c in text.chars() {
        if c.is_alphabetic() {
            letters += 1;
        } else if c.is_digit(10) {
            digits += 1;
        } else if c == ' ' {
            spaces += 1;
        }
    }
    println!("{}", letters);
    println!("{}", digits);
    println!("{}", spaces);
    let mut other = 0;
    for c in text.chars() {
        if !c.is_alphabetic() && !c.is_digit(10) && !c == ' ' {
            other += 1;
        }
    }
    println!("{}", other);
    let mut longest_streak = 0;
    let mut current_streak = 0;
    for c in text.chars() {
        if c.is_digit(10) {
            current_streak += 1;
            if current_streak > longest_streak {
                longest_streak = current_streak;
            }
        } else {
            current_streak = 0;
        }
    }
    println!("{}", longest_streak);
    let mut digits_only: String = text.chars().filter(|&c| c.is_digit(10)).collect();
    println!("{}", digits_only);
}
