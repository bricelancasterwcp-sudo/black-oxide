struct Queue { items: Vec<Int> }

fn exchange_first_last(q: Queue) -> Queue {
    let len = len(q.items);
    if len > 0 {
        let first = get(q.items, 0)?;
        let q2 = set(q, len - 1, first)?;
        set(q2, 0, get(q2.items, len - 1)?)
    } else {
        q
    }
}

fn main() {
    let q = Queue { items: push(push(push(push(vec(), 1), 2), 3), 4), 5 };
    let q2 = exchange_first_last(q);
    for x in q2.items {
        print(x);
    }
}
