fn readings() -> Vec<Int> {
    vec(6, 11, 3, 14, 9)
}

fn main() {
    let readings = readings()
    let mut count = 0
    let mut max = 0
    for reading in readings {
        if reading > 5 {
            count = count + 1
        }
        if reading > max {
            max = reading
        }
    }
    print(count)
    print(max)
}
