fn main() {
    let mut s = String::from("drum");
    
    // Reverse the string
    for (i, c) in s.chars().enumerate().rev() {
        s.replace_range(i..=i, &c.to_string());
    }
    
    println!("{}", s);  // Print it normally
    
    // Reverse the string again to print it backwards
    for (i, c) in s.chars().enumerate().rev() {
        s.insert_str(0, &c.to_string());
    }
    
    println!("{}", s);
}
