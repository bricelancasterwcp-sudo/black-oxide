struct Entry {
    value: Option<Int>,
    text: Str,
}

fn main() {
    let entries = vec(Entry { value: Some(8), text: "oak" }, Entry { value: None, text: "15" }, Entry { value: Some(15), text: "bad" })
    for entry in entries {
        match entry.value {
            Some(n) => print(n),
            None => print_str("bad"),
        }
    }
}
