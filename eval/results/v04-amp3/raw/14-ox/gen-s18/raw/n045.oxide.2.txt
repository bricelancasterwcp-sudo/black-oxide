struct Log { entries: Vec<Int> }

fn first_entry(log: Log) -> Int {
    get(log.entries, 0)?
}

fn last_entry(log: Log) -> Option<Int> {
    let len = len(log.entries)
    if len == 0 {
        None?
    }
    Some(get(log.entries, len - 1)?)
}

fn main() {
    let log = Log { entries: vec().push(6).push(7).push(8).push(9) }
    print(first_entry(clone(log)))
    match last_entry(log) {
        Some(n) => print(n),
        None => print_str("log is empty"),
    }
}
