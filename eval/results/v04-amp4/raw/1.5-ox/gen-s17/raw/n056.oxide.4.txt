```rust
fn main() {
    let mut count = 0;
    let mut sum = 0;

    for c in "7x9?".chars() {
        if !c.is_digit(10) {
            count += 1;
        } else {
            let num = c.to_digit(10).unwrap();
            sum += num;
        }
    }

    print!("Total: {} ({} not numbers)", sum, count);
}
```
