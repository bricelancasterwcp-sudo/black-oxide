struct Line { items: Vec<Item> }

struct Item { unit_price: Int, quantity: Int }

fn main() {
    let line1 = Line { items: vec![Item { unit_price: 3, quantity: 4 }] };
    let line2 = Line { items: vec![Item { unit_price: 2, quantity: 5 }] };
    
    print(line_value(&line1));
    print(line_value(&line2));
    print(line_value(&Line { items: concat(line1.items, line2.items) }));
}

fn line_value(line: &Line) -> Int {
    sum(line.items.iter().map(|item| item.unit_price * item.quantity))
}
