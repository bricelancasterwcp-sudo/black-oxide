struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn largest(v: Vec<Int>) -> Int {
    let mut largest = -1
    for x in v {
        if x > largest {
            largest = x
        }
    }
    largest
}

fn main() {
    print(largest(vec(5, 3, 9, 1, 7)))
}
