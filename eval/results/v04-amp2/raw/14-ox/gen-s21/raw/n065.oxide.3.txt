struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max.is_none() || x > unwrap_or(max, -1) {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = vec()
    readings.push(6)
    readings.push(11)
    readings.push(3)
    readings.push(14)
    readings.push(9)

    let count = count_exceeding(clone(readings), 5)
    print(count)

    let max = max_reading(readings)
    match max {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
