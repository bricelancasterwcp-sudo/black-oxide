struct Reading { label: Str, values: Vec<Int> }

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 10), 20) }
    match get(r.values, 0)? {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match get(r.values, 3)? {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
