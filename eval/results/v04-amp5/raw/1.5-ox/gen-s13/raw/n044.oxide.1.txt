fn deposit(amount: Int) -> Option<Int> {
    let mut balance = 0;
    balance += amount;
    Some(balance)
}

fn main() {
    let mut deposits = vec![3, 1, 4, 1, 5];
    
    for deposit in deposits {
        match deposit {
            Ok(balance) => println!("Balance after deposit {}: {}", deposit, balance),
            _ => (),
        }
    }
}
