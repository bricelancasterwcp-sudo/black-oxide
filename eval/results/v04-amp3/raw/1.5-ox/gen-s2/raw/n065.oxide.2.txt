struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 5 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    match get(v, 1) {
        Some(n @ 9...14) => Some(n + 1),
        None => None,
    }
}

fn main() {
    let r = Reading { label: "lab", values: vec![6, 11, 3, 14, 9] };
    println!("{}", first_big(r.values));
    println!("{}", second(r.values).unwrap_or(0));
}
