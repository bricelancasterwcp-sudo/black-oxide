```rust
fn main() {
    let nums: [Option<u32>; 4] = [
        Some(7),
        None,
        Some(9),
        None,
    ];

    let total: u32 = nums.iter().filter_map(|&x| x).sum();
    let count: usize = nums.iter().filter(|&x| !x.is_none()).count();

    println!("Total: {}, Count: {}", total, count);
}
```
