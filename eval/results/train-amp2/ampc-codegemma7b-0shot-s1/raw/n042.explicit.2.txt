fn max(x: &Int, y: &Int) -> Int {
    if x > y {
        return x.clone()
    } else {
        return y.clone()
    }
}

fn main() {
    let nums = vec(12, 5, 20, 8, 17)
    let mut max_num = nums.first().unwrap()
    for num in nums.iter() {
        if num > max_num {
            max_num = num
        }
    }
    print(max_num)
    let mut second_max = nums.first().unwrap()
    for num in nums.iter() {
        if num > second_max && num < max_num {
            second_max = num
        }
    }
    print(second_max)
}
