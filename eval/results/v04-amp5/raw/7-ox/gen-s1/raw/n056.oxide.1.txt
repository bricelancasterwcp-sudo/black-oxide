fn main() {
    let numbers = vec![Some(7), Some('x'), Some(9), None];
    let mut sum = 0;
    let mut count_non_numbers = 0;

    for num in numbers {
        match num {
            Some(n @ 0..=9) => sum += n,
            _ => count_non_numbers += 1,
        }
    }

    print(sum);
    print(count_non_numbers);
}
