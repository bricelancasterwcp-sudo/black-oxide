```rust
fn safe_divide(a: Int, b: Int) -> Result<Float, Str> {
    if b == 0 {
        Err("Cannot divide by zero")
    } else {
        Ok(to_float(a) / to_float(b))
    }
}

fn main() {
    let result = safe_divide(20, 4);
    match result {
        Ok(n) => print(n),
        Err(e) => print_str(e),
    }
}
```
