fn main() {
    let names = vec("fig", "banana", "kiwi");
    let mut longest_name = "";
    let mut max_length = 0;

    for name in names {
        if str_len(name) > max_length {
            longest_name = name;
            max_length = str_len(name);
        }
    }

    print_str(longest_name);
    print(max_length);
}
