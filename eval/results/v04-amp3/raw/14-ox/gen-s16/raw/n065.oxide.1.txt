struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn find_highest(v: Vec<Int>) -> Option<Int> {
    if len(v) == 0 {
        return None
    }
    let mut highest = get(v, 0)?
    for x in v {
        if x > highest {
            highest = x
        }
    }
    Some(highest)
}

fn main() {
    let readings = push(push(push(push(push(vec(), 6), 11), 3), 14), 9)
    print(count_exceeding(readings, 5))
    match find_highest(readings) {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
