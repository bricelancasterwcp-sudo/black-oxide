```rust
struct List {
    items: Vec<Int>,
}

impl List {
    fn new(items: Vec<Int>) -> Self {
        List { items }
    }

    fn walk(&self) -> Option<Int> {
        let mut total = 0;
        for &item in self.items.iter() {
            if item % 2 == 1 {
                println!("{}", item * 2);
            } else {
                total += item;
            }
        }
        Some(total)
    }

    fn main() {
        let list = List::new(vec![7, 4, 9, 10]);
        match list.walk() {
            Some(total) => println!("Total of even entries: {}", total),
            None => {}
        }
    }
}
```
