fn get_second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    match get_second(push(vec(), 7)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match get_second(push(push(vec(), 8), 9)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
