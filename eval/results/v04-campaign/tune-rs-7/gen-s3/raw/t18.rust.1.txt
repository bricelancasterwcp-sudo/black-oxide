fn main() {
    let shapes = vec![
        Rectangle { width: 3, height: 4 },
        Triangle { sides: [3, 4, 5] },
        Rectangle { width: 2, height: 5 },
    ];
    for shape in shapes {
        match shape {
            Shape::Rectangle(r) => println!("{}", r.perimeter()),
            Shape::Triangle(t) => println!("{}", t.perimeter()),
        }
    }
}
enum Shape {
    Rectangle(Rectangle),
    Triangle(Triangle),
}
struct Rectangle { width: i32, height: i32 }
impl Rectangle {
    fn perimeter(&self) -> i32 {
        2 * (self.width + self.height)
    }
}
struct Triangle { sides: [i32; 3] }
impl Triangle {
    fn perimeter(&self) -> i32 {
        self.sides.iter().sum()
    }
}
