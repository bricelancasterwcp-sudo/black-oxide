```rust
fn main() {
    let readings = [6, 11, 3, 14, 9];
    let count_over_five = readings.iter().filter(|&&x| x > 5).count();
    println!("{} numbers exceed five", count_over_five);

    let max_reading = readings.into_iter().max().unwrap_or(0);
    println!("Highest reading: {}", max_reading);
}
```
