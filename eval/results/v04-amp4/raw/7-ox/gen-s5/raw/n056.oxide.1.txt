```rust
fn main() {
    let nums = vec![Some(7), None, Some(x), Some(9)];
    let mut count = 0;
    let mut sum = 0;

    for num in nums {
        match num {
            Some(n) => {
                sum += n;
                count += 1;
            }
            None => count += 1,
        }
    }

    print(sum);
    print(count);
}
```
