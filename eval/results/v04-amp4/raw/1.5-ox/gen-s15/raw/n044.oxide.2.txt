```rust
fn deposit(balance: Int, amount: Int) -> Int {
    balance + amount
}

fn withdraw(balance: Int, amount: Int) -> Int {
    if balance >= amount {
        balance - amount
    } else {
        0
    }
}

fn main() {
    let mut balance = 0;
    balance = deposit(balance, 3);
    println!("{}", balance);

    balance = deposit(balance, 1);
    println!("{}", balance);

    balance = withdraw(balance, 4);
    println!("{}", balance);

    balance = deposit(balance, 1);
    println!("{}", balance);

    balance = withdraw(balance, 5);
    println!("{}", balance);
}
```
