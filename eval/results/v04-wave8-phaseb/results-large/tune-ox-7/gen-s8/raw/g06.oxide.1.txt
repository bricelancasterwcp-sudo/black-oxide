struct Sitting { name: Str, scores: Vec<Int> }
fn count_in_range(v: Vec<Int>, min: Int, max: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= min && x <= max {
            count += 1
        }
    }
    count
}
fn main() {
    let morning = Sitting { name: "Morning", scores: vec(94, 71, 88, 65, 92, 79, 83) }
    let evening = Sitting { name: "Evening", scores: vec(55, 90, 77, 84, 68, 99) }
    for sitting in vec(morning, evening) {
        print_str(sitting.name)
        let count_90_or_above = count_in_range(sitting.scores, 90, 100)
        let count_80_to_89 = count_in_range(sitting.scores, 80, 89)
        let count_70_to_79 = count_in_range(sitting.scores, 70, 79)
        let count_below_70 = count_in_range(sitting.scores, 0, 69)
        print(count_90_or_above)
        print(count_80_to_89)
        print(count_70_to_79)
        print(count_below_70)
        let max_score = unwrap_or(max(sitting.scores), -1)
        print(max_score)
        let total: Int = sum(sitting.scores)
        let mean = total / len(sitting.scores)
        print(mean)
    }
    let morning_mean = unwrap_or(div(sum(morning.scores), len(morning.scores)), -1)
    let evening_mean = unwrap_or(div(sum(evening.scores), len(evening.scores)), -1)
    if morning_mean > evening_mean {
        print_str("Morning")
    } else {
        print_str("Evening")
    }
}
