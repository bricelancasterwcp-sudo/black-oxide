fn largest_two(mut_v: Vec<Int>) -> Vec<Int> {
    let mut largest = -1
    let mut second_largest = -1
    for x in mut_v.drain(0) {
        if x > largest {
            second_largest = largest
            largest = x
        } else if x > second_largest && x != largest {
            second_largest = x
        }
    }
    vec(largest, second_largest)
}

fn main() {
    let values = vec(12, 5, 20, 8, 17)
    let largest_two = largest_two(values)
    print(largest_two)
}
