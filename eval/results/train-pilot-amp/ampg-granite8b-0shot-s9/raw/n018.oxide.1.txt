struct List { head: Int, tail: Option<Box<List>> }

fn print_list(l: List) {
    match l.tail {
        Some(refRest) => {
            print_list(refRest.clone())
            print(l.head)
        },
        None => {
            print(l.head)
        },
    }
}

fn main() {
    let l = List { head: 4, tail: Some(Box { head: 3, tail: Some(Box { head: 2, tail: Some(Box { head: 1, tail: None }) }) }) }
    print_list(l)
}
