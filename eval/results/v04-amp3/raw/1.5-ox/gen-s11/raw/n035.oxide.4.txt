fn divide(a: Int, b: Int) -> Option<Int> {
    if b == 0 {
        None
    } else {
        Some(a / b)
    }
}

fn main() {
    match divide(20, 4) {
        Some(result) => print_str(&format!("{} divided by 4 is {}", result)),
        None => print_str("Cannot divide by zero"),
    }
}
