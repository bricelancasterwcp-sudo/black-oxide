struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding_five(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 5 {
            count += 1
        }
    }
    count
}

fn find_highest(v: Vec<Int>) -> Option<Int> {
    if len(v) == 0 {
        return None
    }
    let mut max = get(v, 0)?
    for x in v {
        if x > max {
            max = x
        }
    }
    Some(max)
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14)
    let readings = push(readings, 9)
    let count = count_exceeding_five(readings)
    print(count)
    let highest = find_highest(readings)
    match highest {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
