fn main() {
    let v = vec()
    v = push(push(v, 10), 20)
    match get(v, 1) {
        Some(pos1) => print(pos1),
        None => print_str("None")
    }
    v = push(push(vec(), 10), 20) // Reinitialize v
    match get(v, 4) {
        Some(pos4) => print(pos4),
        None => print_str("None")
    }
}
