```rust
fn is_prime(n: u32) -> bool {
    if n <= 1 {
        return false;
    }
    for i in 2..=((n as f64).sqrt() as u32) {
        if n % i == 0 {
            return false;
        }
    }
    true
}

fn main() {
    let count = (0..100).filter(|&x| is_prime(x)).count();
    println!("There are {} prime numbers strictly less than 100.", count);
}
```
