fn mean(values: Vec<Int>) -> Float {
    let mut sum = 0
    for x in values {
        sum = sum + x
    }
    to_float(sum) / values.len()
}

fn main() {
    let readings = push(push(push(push(vec(), 4), 8), 15), push(push(vec(), 16), push(push(vec(), 23), 42)))
    let m = mean(readings)
    print_str(int_to_str(trunc(m)))
}
