```rust
fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut max_name = &names[0];
    let mut max_length = max_name.len();

    for name in &names[1..] {
        if name.len() > max_length {
            max_name = name;
            max_length = name.len();
        }
    }

    println!("{} {}", max_name, max_length);
}
```
