struct Queue { values: Vec<Int> }

fn main() {
    let q = Queue { values: vec(1, 2, 3, 4, 5) }
    let first = get(q.values, 0)?
    let last = get(q.values, len(q.values) - 1)?
    let q2 = Queue { values: concat(last, concat(drop_first(drop_last(q.values)))) }
    for x in q2.values {
        print(x)
    }
}

fn drop_first(v: Vec<Int>) -> Vec<Int> {
    v.slice(1, len(v))
}

fn drop_last(v: Vec<Int>) -> Vec<Int> {
    v.slice(0, len(v) - 1)
}

fn slice(v: Vec<Int>, start: Int, end: Int) -> Vec<Int> {
