fn main() {
    let pi = 3.1415926
    let square = Square { side: 4.0 }
    let circle = Circle { radius: 3.0 }
    print(area(&square, pi))
    print(area(&circle, pi))
}

fn area(shape: &Shape, pi: Float) -> Float {
    match shape {
        Shape(square) => {
            let side = square.side
            pi * side * side
        },
        Shape(circle) => {
            let radius = circle.radius
            pi * radius * radius
        },
    }
}

    radius: Float,
}

struct Square {
    side: Float,
}

    Square(Square),
    Circle(Circle),
}
