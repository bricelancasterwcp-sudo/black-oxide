fn count(s: Str) -> Int {
    let mut count = 0
    for c in chars(s) {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            count = count + 1
        }
    }
    count
}

fn main() {
    print(count(str_from("hello")))
    print(count(str_from("world")))
}
