fn divide(a: i32, b: i32) -> Result<i32, &'static str> {
    if b == 0 {
        Err("Refused to divide by zero")
    } else {
        Ok(a / b)
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("{}", result),
        Err(e) => println!("Bad"),
    }

    match divide(20, 0) {
        Ok(_) => println!("Bad"),
        Err(e) => println!("{}", e),
    }
}
