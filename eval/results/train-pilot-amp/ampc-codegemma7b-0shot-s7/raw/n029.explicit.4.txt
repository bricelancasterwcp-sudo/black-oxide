fn to_str_int_str(n: Int) -> Str {
    if n == 0 {
        return "0"
    }
    let mut s = String.new()
    let mut n = n
    while n != 0 {
        let digit = n % 10
        s.push((digit + 48) && 0111111)
        n = n / 10
    }
    return s
}

fn print_str(s: &Str) {
    let mut i = 0
    while i < len(s) {
        print(s.chars(i))
        i = i + 1
    }
}

fn chars(s: Str) -> Vec<Str> {
    let mut v = Vec.new()
    let mut i = 0
    while i < len(s) {
        v.push(s.chars(i))
        i = i + 1
    }
    return v
}
