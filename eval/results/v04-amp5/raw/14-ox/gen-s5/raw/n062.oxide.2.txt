```rust
struct Signal { color: Str }

fn print_signal_meaning(s: Signal) {
    match s.color {
        "green" => print_str("go"),
        "red" => print_str("stop"),
        "yellow" => print_str("wait"),
        _ => print_str("unknown signal"),
    }
}

fn main() {
    let signals = vec!(Signal { color: "green".to_str() }, Signal { color: "red".to_str() }, Signal { color: "yellow".to_str() });
    for s in signals {
        print_signal_meaning(s);
    }
}
```
