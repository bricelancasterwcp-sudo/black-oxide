fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > &limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn main() {
    let nums = push(push(vec(), 5), 40)
    let bigger = extend(clone(&nums), 70)
    print(sum_big(&nums, 10))
    drop nums
    print(sum_big(&bigger, 10))
    drop bigger
    let cutoff = 10
    let mut under = 0
    let mut at = 0
    for num in &nums {
        if num < cutoff {
            under = under + 1
        } else {
            at = at + 1
        }
    }
    print(under)
    print(at)
    drop nums
    drop bigger
}
