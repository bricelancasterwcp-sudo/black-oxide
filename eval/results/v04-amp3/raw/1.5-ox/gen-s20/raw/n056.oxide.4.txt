```rust
fn main() {
    let numbers: Vec<Option<Int>> = vec![Some(7), None, Some(x), None, Some(9), None];
    let mut total = 0;
    let mut count = 0;

    for number in &numbers {
        match number {
            Some(n) => {
                total += n;
                count += 1;
            },
            None => (),
        }
    }

    println!("Total: {}", total);
    println!("Count of ones: {}", count);
}
```
