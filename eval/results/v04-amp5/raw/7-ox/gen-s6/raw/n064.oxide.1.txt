fn get_value(v: Vec<Int>, i: Int) -> Option<Int> {
    get(v, i)
}

fn main() {
    let v = vec(10, 20);
    match get_value(v, 0) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match get_value(v, 3) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
