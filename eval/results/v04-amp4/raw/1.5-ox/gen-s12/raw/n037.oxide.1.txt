```rust
struct Rectangle {
    width: Int,
    height: Int,
}

fn set_width(r: &mut Rectangle, new_width: Int) -> Result<(), Box<dyn std::error::Error>> {
    if r.width <= 0 {
        return Err("Width must be positive".into());
    }
    r.width = new_width;
    Ok(())
}

fn area(r: &Rectangle) -> Int {
    r.width * r.height
}

fn main() {
    let mut r = Rectangle { width: 2, height: 3 };
    set_width(&mut r, 5)?;
    print(area(&r));
}
```
