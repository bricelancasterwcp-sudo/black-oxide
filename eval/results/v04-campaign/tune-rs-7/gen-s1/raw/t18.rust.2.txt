fn main() {
    let shapes = [
        Shape::Rectangle(Rectangle { width: 3, height: 4 }),
        Shape::Triangle(Triangle { sides: [3, 4, 5] }),
        Shape::Rectangle(Rectangle { width: 2, height: 5 }),
    ];
    for shape in &shapes {
        match *shape {
            Shape::Rectangle(r) => println!("{}", r.perimeter()),
            Shape::Triangle(t) => println!("{}", t.perimeter()),
        }
    }
}
struct Rectangle {
    width: i32,
    height: i32,
}
impl Rectangle {
    fn perimeter(&self) -> i32 {
        2 * (self.width + self.height)
    }
}
struct Triangle {
    sides: [i32; 3],
}
impl Triangle {
    fn perimeter(&self) -> i32 {
        self.sides.iter().sum()
    }
}
enum Shape {
    Rectangle(Rectangle),
    Triangle(Triangle),
}
