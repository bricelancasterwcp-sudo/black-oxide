fn int_to_str(n: Int) -> Str {
    "0123456789abcdef"
    let mut digits = vec()
    let mut n2 = n
    while n2 != 0 {
        digits.push(to_char(n2 % 16))

fn to_char(n: Int) -> Str {
    if n < 10 {
        return Str { len: 1, data: vec_char("0123456789".chars().nth(n)) }
    }
    Str
    let mut digits = vec_char("abcdefghijklmnopqrstuvwxyz".chars().nth(n - 10))
    digits.push(Str { len: 1, data: vec_char(".") })
    let mut s = Str { len: digits.len(), data: digits }
    reverse(s)
    s
}

fn reverse(mutmuts: Str) -> Str {
    let mut d = 0
    let mut h = muts.len - 1
    while d < h {
        swap(muts.data, d, h)
        d = d + 1
        h = h - 1
    }
    muts
}

fn swap(v: Vec<Str>, i: Int, j: Int) {
    let tmp = v.get(i)
    v.set(i, v.get(j))
    v.set(j, tmp)
}

fn vec_str(s: Str) -> Vec<Str> {
    let mut v = vec()
