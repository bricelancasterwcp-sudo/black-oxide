fn main() {
    let p1 = Point { x: 1, y: 5 }
    let p2 = Point { x: 10, y: &p1 }
    print(&p2)
}

struct Point {
    x: Int,
    y: Int,
}

fn print(p: &Point) {
    let sum = p.x + p.y
    print_str(&sum)
}

fn print_str(s: &Int) {
    let s = int_to_str(s)
    print(&s)
}

fn int_to_str(n: &Int) -> Str {
    let s = str_len(n)
    let mut result = vec()
    for i in range(0, s) {
        let digit = to_float(trunc(n / pow(10, i)))
        let digit = int_to_str(&digit)
        push(result, &digit)
    }
    let result = concat(result, vec())
    result
}

fn pow(base: &Int, exp: &Int) -> Int {
    let mut result = 1
    for _ in range(0, exp) {
        result = result * base
    }
    result
}

        push(result, i)
    }
    result
}
