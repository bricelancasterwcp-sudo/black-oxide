fn sum_first_four(v: &Vec<Int>) -> Int {
    let total = 0
    for x in &v {
        if x % 4 == 0 {
            total = total + x
        }
        if total >= 24 {
            return total
        }
    }
    total
}

fn main() {
    let nums = vec(4, 8, 12, 16, 20, 24, 28, 32, 36, 40)
    print(sum_first_four(&nums))
    drop nums
}
