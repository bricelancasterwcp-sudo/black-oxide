fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut total = 0;
    let mut count_non_numbers = 0;

    for text in texts {
        if let Ok(num) = text.parse::<i32>() {
            total += num;
        } else {
            count_non_numbers += 1;
        }
    }

    println!("Total: {}", total);
    println!("Non-numbers: {}", count_non_numbers);
}
