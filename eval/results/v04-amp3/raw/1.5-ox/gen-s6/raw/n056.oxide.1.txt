fn count_numbers(vec: Vec<Option<i32>>) -> (i32, i32) {
    let mut sum = 0;
    let mut non_sum = 0;

    for opt in vec {
        if let Some(num) = opt {
            sum += num;
        } else {
            non_sum += 1;
        }
    }

    (sum, non_sum)
}

fn main() {
    let texts: Vec<Option<i32>> = vec![Some(7), None, Some(9), Some(?), None];
    let result = count_numbers(texts);
    print!("Sum of numbers: {}, Non-sum numbers: {}", result.0, result.1);
}
