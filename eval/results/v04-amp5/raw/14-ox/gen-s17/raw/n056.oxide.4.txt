struct Reading { label: Str, values: Vec<Int> }

fn sum_of_numbers(v: Vec<Str>) -> Option<Int> {
    let mut sum = 0
    for s in v {
        if let Some(x) = parse_int(s) {
            sum += x
        }
    }
    Some(sum)
}

fn count_non_numbers(v: Vec<Str>) -> Int {
    let mut count = 0
    for s in v {
        if parse_int(s).is_none() {
            count += 1
        }
    }
    count
}

fn main() {
    let readings = vec("7", "x", "9", "?")
    match sum_of_numbers(readings.clone()) {
        Some(total) => print(total),
        None => print(0),
    }
    let non_number_count = count_non_numbers(readings)
    print(non_number_count)
}
