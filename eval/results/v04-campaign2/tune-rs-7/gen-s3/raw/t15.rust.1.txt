fn main() {
    let inputs = ["12", "x", "30"];
    let mut sum = 0;
    let mut failed_count = 0;
    for input in inputs.iter() {
        match input.parse::<i32>() {
            Ok(num) => sum += num,
            Err(_) => failed_count += 1,
        }
    }
    println!("{}", sum);
    println!("{}", failed_count);
}
