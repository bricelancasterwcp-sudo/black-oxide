fn main() {
    let num = 1234;
    let mut digits: Vec<i32> = vec![];

    // Extract digits into a vector
    while num > 0 {
        digits.insert(0, num % 10);
        num /= 10;
    }

    // Print digits in reverse order
    for digit in digits {
        print!("{}", digit);
    }
    println!();
}
