fn is_vowel(c: Str) -> Bool {
    match c {
        Str(v) => match v {
            Some(x) => match x {
                _ => false,
                _ => true,
            },
        },
    }
}

fn main() {
    let word = "keyboard"
    let mut vowels = 0
    let mut not_vowels = 0
    for c in word.chars() {
        if is_vowel(Str(Some(c))) {
            vowels = vowels + 1
        } else {
            not_vowels = not_vowels + 1
        }
    }
    print(vowels)
    print(not_vowels)
}
