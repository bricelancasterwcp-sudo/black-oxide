fn find_positions(nums: &Vec<Int>, target: Int) -> Vec<Int> {
    let mut positions = vec()
    for i in 0.to(nums.len()) {
        if nums.get(i) == Some(&target) {
            push(positions, i)
        }
    }
    positions
}

fn main() {
    let nums = vec(4, 7, 4, 9, 4)
    let positions = find_positions(&nums, 4)
    for pos in &positions {
        print_str(pos.to_str())
        print_str("\n")
    }
    drop nums
    drop positions
}
