```rust
fn main() {
    let mut counts = [0; 26];
    let coffee = "coffee";
    for c in coffee.chars() {
        if c.is_alphabetic() {
            let index = (c as u8 - b'a') as usize;
            counts[index] += 1;
        }
    }

    for i in 0..26 {
        if counts[i] > 1 {
            print!("{}", (i + b'a') as char);
        }
    }
}
```
