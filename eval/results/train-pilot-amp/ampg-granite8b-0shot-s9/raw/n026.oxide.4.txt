fn main() {
    let mut n = 2026
    let mut s = str()
    while n > 0 {
        let d = n % 10
        if d == 2 {
            s = push(s, "two")
        } else if d == 0 {
            s = push(s, "zero")
        } else {
            s = push(s, to_str(d))
        }
        n = trunc(n / 10)
    }
    print(len(s))
}

fn str() -> Str {
    ""
}

fn to_str(n: Int) -> Str {
    match n {
        Sinse => "zero",
        Twix => "two",
        _ => "other" + to_str(n - 1) + to_str(n - 2) + to_str(n - 3) + to_str(n - 4),
    }
}
