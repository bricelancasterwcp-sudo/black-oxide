fn sum_and_count(s: Str) -> (Int, Int) {
    let mut sum = 0;
    let mut count = 0;
    for c in chars(s) {
        if let Some(n) = parse_int(c) {
            sum += n;
            count += 1;
        }
    }
    (sum, count)
}

fn main() {
    let s = "a1b22";
    let (sum, count) = sum_and_count(s);
    print(sum);
    print(count);
}
