fn main() {
    let mut sum = 0;
    let mut count = 0;

    for s in ["12", "x", "30"].iter() {
        match parse_int(s) {
            Some(n) => {
                sum += n;
                count += 1;
            }
            None => {
                println!("Failed to parse: {}", s);
            }
        }
    }

    println!("Sum of parsed integers: {}", sum);
    println!("Count of failed parses: {}", count);
}
