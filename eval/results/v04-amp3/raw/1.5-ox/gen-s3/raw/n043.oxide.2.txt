struct Queue {
    items: Vec<Int>,
}

impl Queue {
    fn new() -> Self {
        Queue { items: vec![] }
    }

    fn push(&mut self, item: Int) {
        self.items.push(item);
    }

    fn pop(&mut self) -> Option<Int> {
        if let Some(first) = self.items.pop() {
            Some(first)
        } else {
            None
        }
    }

    fn peek(&self) -> &Int {
        self.items.first().unwrap()
    }
}

fn main() {
    let mut queue = Queue::new();
    for i in 1..=5 {
        queue.push(i);
    }

    if let Some(first) = queue.pop() {
        let last = queue.peek().clone(); // Clone to avoid ownership issues
        queue.items.remove(0); // Remove the first item without consuming it
        queue.items.insert(0, last); // Insert the cloned last item at the beginning
    }

    for item in &queue.items {
        print!("{}", item);
        println!();
    }
}
