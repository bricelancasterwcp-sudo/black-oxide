fn word_keyboard(word: Str) -> Str {
    let mut vowels = 0
    let mut consonants = 0
    for c in word.chars() {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            vowels = vowels + 1
        } else {
            consonants = consonants + 1
        }
    }
    let vowel_str = int_to_str(vowels)
    let cons_str = int_to_str(consonants)
    concat(vowel_str, cons_str)
}

fn main() {
    print(word_keyboard("word"))
}
