```rust
fn main() {
    let mut result = 100;
    for divisor in &[5, 0, 4] {
        if divisor == 0 {
            println!("undefined");
        } else {
            let quotient = result / divisor;
            println!("{}", quotient);
        }
        result /= divisor; // Reassign to avoid loop-carried move
    }
}
```
