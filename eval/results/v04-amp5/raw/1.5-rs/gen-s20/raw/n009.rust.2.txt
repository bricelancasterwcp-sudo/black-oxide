fn main() {
    let num: u32 = 1234;
    
    // Reverse the number by converting it to a string, reversing it using slicing, and then converting back to an integer
    let reversed_num: u32 = i32::from_str_radix(&num.to_string()[::-1], 10).unwrap();
    
    println!("Reversed number: {}", reversed_num);
}
