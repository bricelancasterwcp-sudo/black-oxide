```rust
struct Rectangle {
    width: Int,
    height: Int,
}

impl Rectangle {
    fn new(width: Int, height: Int) -> Self {
        Rectangle { width, height }
    }

    fn area(&self) -> Int {
        self.width * self.height
    }

    fn perimeter(&self) -> Int {
        2 * (self.width + self.height)
    }
}

fn main() {
    let rect = Rectangle::new(9, 3);
    print_int(rect.area());            // field access copies: r stays usable
    print_int(rect.perimeter());
}
```
