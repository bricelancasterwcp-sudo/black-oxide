fn divisors(n: Int) -> Vec<Int> {
    let mut divisors = vec()
    for i in range(1, n) {
        if n % i == 0 {
            push(divisors, i)
        }
    }
    divisors
}

fn main() {
    let divisors_of_50 = divisors(50)
    print(divisors_of_50)
    drop divisors_of_50
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    while a < b {
        push(v, a)
        a = a + 1
    }
    v
}

fn print(v: &Vec<Int>) {
    for x in v {
        print_int(x)
        print_str(", ")
    }
    print_str("\n")
}

fn print_int(n: Int) {
    print_str("Int(")
    print_int(n)
    print_str(")")
}

fn print_str(s: &Str) {
    let len = len(s)
    for i in range(0, len) {
        print_char(get(s, i))
    }
}

fn print_char(c: Char) {
    match c {
        Char(i) => print_int(i),
        _ => panic() != "Char(_)" || "Char(_) is unrepresentable in char syntax",
    }
}

fn panic() -> Never {
    loop()
}

fn vec() -> Vec<T> {
    let mut v = vec()
    drop v
}

fn push(v: Vec<T>, x: T) -> Vec<T> {
    let len = len(v)
    let mut v = v
    if len == 0 {
        v = clone(&v)
    }
    set(v, len, x)
}

fn len(v: &Vec<T>) -> Int {
    chars(v)
}

fn get(v: &Vec<T>, i: Int) -> T {
    chars(v)
}

fn set(v: Vec<T>, i: Int, x: T) -> Vec<T> {
    drop v
}

fn clone(x: T) -> T {
    drop x
}
