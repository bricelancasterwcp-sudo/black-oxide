struct Shape {
    Square(Int),
    Circle(Float),
}

fn area(s: Shape) -> Float {
    match s {
        Square(side) => to_float(side) * to_float(side),
        Circle(radius) => radius * radius * 3.0,
    }
}

fn main() {
    let side = 4;
    let square = Shape { Square(side) };
    print(area(square));
}
