fn main() {
    let names = vec!["fig", "banana", "kiwi"];
    let longest_name = names.iter().fold(&names[0], |a, b| if a.len() > b.len() { a } else { b });
    println!("The name with the most letters is: {}", longest_name);
    println!("Its letter count is: {}", longest_name.chars().count());
}
