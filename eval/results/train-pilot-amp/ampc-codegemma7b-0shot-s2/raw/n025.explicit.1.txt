fn parse_int(s: Str) -> Option<Int> {
    let mut digits: Vec<Int> = vec()
    for c in chars(s) {
        match c {
            Str(n) => digits.push(n - 48),
            _ => returnNone(),
        }
    }
    let mut result = 0
    for n in digits {
        result = result * 10 + n
    }
    Some(result)
}

fn chars(s: Str) -> Vec<Str> {
    let mut result: Vec<Str> = vec()
    for i in 0.range(str_len(s)) {
        result.push(Str(get(s, i).unwrap()))
    }
    result
}

fn str_len(s: Str) -> Int {
    let mut result = 0
    for _ in chars(s) {
        result = result + 1
    }
    result
}
