struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(first_big(r.values))            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}

fn read_num() -> Option<Int> {
    let mut buf = ""
    loop
    match read_line(buf) {
        None => returnNone,
        Some(line) => {
            buf = line
            match parse_int(buf) {
                None => print_str("The text does not hold a number. Try again.\n"),
                Some(num) => returnSome(num) + breakOut,
            }
        },
    }
}

fn main() {
    match read_num() {
        Some(num) => print(num),
        None => print_str("Too many lines. Try again.\n"),
    }
}
