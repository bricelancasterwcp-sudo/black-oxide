```rust
fn main() {
    let fruits = ["fig", "banana", "kiwi"];
    let mut max_name = "";
    let mut max_length = 0;

    for fruit in fruits.iter() {
        if fruit.len() > max_length {
            max_length = fruit.len();
            max_name = fruit;
        }
    }

    println!("The name with the most letters is: {} ({})", max_name, max_length);
}
```
