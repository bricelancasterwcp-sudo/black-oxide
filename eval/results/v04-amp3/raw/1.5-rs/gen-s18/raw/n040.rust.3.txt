```rust
fn main() {
    let s = Square { side: 4 };
    println!("{}", s.area());

    let r = Circle { radius: 3.0 };
    println!("{}", r.area());
}
struct Square {
    side: u32,
}

impl Square {
    fn area(&self) -> u32 {
        self.side * self.side
    }
}

struct Circle {
    radius: f64,
}

impl Circle {
    fn area(&self) -> f64 {
        std::f64::consts::PI * self.radius * self.radius
    }
}
```
