```rust
fn main() {
    let numbers = vec![5, 12, 3, 18, 9];
    let cutoff = 10;

    let below_cutoff_count = numbers.iter().filter(|&&num| num < cutoff).count();
    let above_or_equal_cutoff_count = numbers.iter().filter(|&&num| num >= cutoff).count();

    println!("Numbers below the cutoff: {}", below_cutoff_count);
    println!("Numbers at or over the cutoff: {}", above_or_equal_cutoff_count);
}
```
