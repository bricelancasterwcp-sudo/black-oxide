```rust
fn main() {
    let samples = [5, 12, 3, 18, 9];
    let cutoff = 10;
    let below = samples.iter().filter(|&&x| x < cutoff).count();
    let above_or_equal = samples.iter().filter(|&&x| x >= cutoff).count();
    println!("Below {}: {}", cutoff, below);
    println!("At or over {}: {}", cutoff, above_or_equal);
}
```
