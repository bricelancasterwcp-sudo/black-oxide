```rust
use std::io::{stdin, stdout, Write};

fn main() {
    let mut total = 0;
    let mut not_total = 0;

    for _ in 0..4 {
        print!("Enter a text (7, x, 9 or ?): ");
        stdout().flush().expect("Failed to flush stdout");

        let mut input = String::new();
        stdin().read_line(&mut input).expect("Failed to read line");
        input = input.trim().to_lowercase();

        match input.as_str() {
            "7" => total += 7,
            "x" => not_total += 1,
            "9" => total += 9,
            "?" => not_total += 1,
            _ => panic!("Invalid input"),
        }
    }

    println!("Total of ones: {}", total);
    println!("Total of not ones: {}", not_total);
}
```
