struct List { values: Vec<Str> }

fn get_value(list: Vec<Str>, index: Int) -> Option<Str> {
    get(list, index)?
}

fn main() {
    let list = List { values: push(push(vec(), "10"), "20") }
    match get_value(list.values.clone(), 1) {
        Some(value) => print_str(value),
        None => print_str("none"),
    }
    match get_value(list.values.clone(), 4) {
        Some(value) => print_str(value),
        None => print_str("none"),
    }
}
