fn main() {
    let shapes = vec![
        Shape::Rectangle(3, 4),
        Shape::Triangle(3, 4, 5),
        Shape::Rectangle(2, 5)
    ];
    for shape in &shapes {
        println!("{}", shape.perimeter());
    }
}
enum Shape {
    Rectangle(i32, i32),
    Triangle(i32, i32, i32)
}
impl Shape {
    fn perimeter(&self) -> i32 {
        match self {
            Shape::Rectangle(w, h) => 2 * (w + h),
            Shape::Triangle(a, b, c) => a + b + c
        }
    }
}
