fn main() {
    let mut digits = vec(1, 2, 3, 4)
    while len(&digits) > 0 {
        let d = get(&digits, 0).unwrap()
        print_str(str_len(&d) + " ")
        digits = drop_front(&digits)
    }
}

fn drop_front(v: Vec<Int>) -> Vec<Int> {
    for i in 1.to(len(&v)) {
        get(&v, i).unwrap()
    }
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < len(&v) {
        Some(clone(&v).get(i).unwrap())
    } else {
        None
    }
}

fn len(v: &Vec<Int>) -> Int {
    clone(&v).len()
}

fn print_str(s: Str) {
    print(clone(&s))
}

fn print(i: Int) {
    let s = int_to_str(i)
    push(s, 10)
    print_str(&s)
    drop s
}
