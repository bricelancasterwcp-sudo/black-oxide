fn smallest(v: Vec<Int>) -> Int {
    let min_value = get(v, 0)?
    for x in v {
        if x < min_value {
            min_value = x
        }
    }
    Some(min_value)?
}

fn largest(v: Vec<Int>) -> Int {
    let max_value = get(v, 0)?
    for x in v {
        if x > max_value {
            max_value = x
        }
    }
    Some(max_value)?
}

fn gap(v: Vec<Int>) -> Int {
    let min_value = smallest(v)?
    let max_value = largest(v)?
    max_value - min_value
}
