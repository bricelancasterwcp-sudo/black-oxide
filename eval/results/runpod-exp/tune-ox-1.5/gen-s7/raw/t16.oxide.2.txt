fn divide(x: Int, y: Int) -> Option<Int> {
    if y == 0 {
        None
    } else {
        Some(x / y)
    }
}
fn main() {
    match divide(100, 5) {
        Some(q) => print(q),
        None => print_str("undefined"),
    }
    match divide(100, 0) {
        Some(q) => print(q),
        None => print_str("undefined"),
    }
    match divide(100, 4) {
        Some(q) => print(q),
        None => print_str("undefined"),
    }
}
