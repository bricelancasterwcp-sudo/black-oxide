fn main() {
    let word = "banana"
    let chars = chars(word)
    for (i, c) in enumerate(chars) {
        if c == "n" {
            print(i)
        }
    }
}

fn enumerate(v: Vec<Str>) -> Vec<(Int, Str)> {
    let mut acc = vec()
    for (i, x) in range(0, len(v)) {
        acc = push(acc, (i, get(v, i).unwrap_or("")))
    }
    acc
}
