fn reverse_number(n: Int) -> Str {
    let s = int_to_str(n)
    let chars = chars(s)
    let reversed_chars: Vec<Str> = chars.iter().rev().map(|c| c.clone()).collect()
    concat("", reversed_chars)
}

fn main() {
    let number = 1234
    let reversed = reverse_number(number)
    print(reversed)
}
