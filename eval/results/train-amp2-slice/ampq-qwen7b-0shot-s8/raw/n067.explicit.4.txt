fn walk_and_print(v: &Vec<Int>) -> Int {
    let mut total = 0
    for x in v {
        if x % 2 == 1 {
            print_str((x * 2).to_string())
        } else {
            total = total + x
        }
    }
    total
}

fn main() {
    let nums = vec(7, 4, 9, 10)
    let total = walk_and_print(&nums)
    print_str(total.to_string())
    drop nums
}
