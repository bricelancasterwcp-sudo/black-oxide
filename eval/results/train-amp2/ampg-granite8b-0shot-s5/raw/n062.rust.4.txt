const SIGNALS: [&str; 3] = ["green", "red", "yellow"];

fn main() {
    for signal in SIGNALS {
        match signal {
            "green" => println!("go for {}", signal),
            "red" => println!("stop for {}", signal),
            "yellow" => println!("wait for {}", signal),
            _ => unreachable!(),
        }
    }
}
