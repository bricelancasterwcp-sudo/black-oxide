```rust
fn second_entry<T>(list: &[T]) -> Option<&T> {
    if list.len() > 1 {
        list.get(1)
    } else {
        None
    }
}

fn main() {
    let numbers = vec![7];
    match second_entry(&numbers) {
        Some(second) => println!("The second entry is: {}", second),
        None => println!("There is no second entry"),
    }
}
```
