```rust
struct Measurement { value: Int }

fn main() {
    let measurements = vec![Measurement { value: 9 }, Measurement { value: 2 }, Measurement { value: 7 }, Measurement { value: 4 }];
    
    for (i, &m1) in measurements.iter().enumerate() {
        let mut smallest = m1.clone();
        for &m2 in measurements[i+1..].iter() {
            if m2.value < smallest.value {
                smallest = m2.clone();
            }
        }
        print(smallest.value);
    }

}
```
