fn main() {
    let readings = vec(6, 11, 3, 14, 9)
    let count = filter(readings, is_big).len()
    let max = reduce(readings, max)
    print(count)
    print(max)
}

fn is_big(n: Int) -> Bool {
    n > 5
}

fn filter(v: Vec<Int>, p: Int) -> Vec<Int> {
    let result = vec()
    for x in v {
        if p(x) {
            result = push(result, x)
        }
    }
    result
}

fn reduce(v: Vec<Int>, f: Int, a: Int) -> Int {
    let result = a
    for x in v {
        result = f(result, x)
    }
    result
}

fn max(a: Int, b: Int) -> Int {
    if a > b {
        a
    } else {
        b
    }
}

struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(first_big(r.values))            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}
