fn int_to_str(n: Int) -> Str {
    let mut acc = Str("")
    let mut x = n
    while x >= 10 {
        let y = x % 10
        acc = concat(int_to_str(trunc(x / 10)), char_to_str(y + 48))
        x = trunc(x / 10)
    }
    acc = concat(acc, char_to_str(x + 48))
    acc
}
