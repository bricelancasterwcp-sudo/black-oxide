fn main() {
    let seven = maybe_number(7).map(Some).unwrap_or(0)
    let nine = maybe_number(9).map(Some).unwrap_or(0)
    let x = maybe_number("x").map(Some).unwrap_or(0)
    let question = maybe_number("?").map(Some).unwrap_or(0)
    println(seven + x + nine + question)
    println(4 - (seven + x + nine + question))
}

fn maybe_number(s: Str) -> Option<Int> {
    match s.parse() {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

    Some(i)

fn maybe_number(c: Char) -> Option<Int> {
    if c.is_numeric() {
        Some(c.to_digit().unwrap() - 48)
    } else {
        None
    }
}

    if s.chars().all(is_number) {
        Some(s.parse().unwrap())
    } else {
        None
    }
}
