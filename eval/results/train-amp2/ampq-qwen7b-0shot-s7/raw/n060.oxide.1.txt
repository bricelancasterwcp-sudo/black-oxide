fn total_and_count(s: Str) -> Int {
    let mut total = 0
    let mut count = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            total = total + parse_int(c)?
            count = count + 1
        }
    }
    total
}

fn main() {
    let s = "a1b22"
    let total = total_and_count(s)
    let count = total_and_count(s)
    print(total)
    print_str(" ")
    print(count)
}
