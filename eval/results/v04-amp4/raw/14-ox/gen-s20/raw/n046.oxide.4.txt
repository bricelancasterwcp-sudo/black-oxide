fn count_below_and_above(v: Vec<Int>, cutoff: Int) -> (Int, Int) {
    let below = 0
    let above = 0
    for x in v {
        if x < cutoff {
            below += 1
        } else {
            above += 1
        }
    }
    (below, above)
}

fn main() {
    let samples = vec()
    samples = push(samples, 5)
    samples = push(samples, 12)
    samples = push(samples, 3)
    samples = push(samples, 18)
    samples = push(samples, 9)
    let cutoff = 10
    let (below, above) = count_below_and_above(samples, cutoff)
    print(below)
    print(above)
}
