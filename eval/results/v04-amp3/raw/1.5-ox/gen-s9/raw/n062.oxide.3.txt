```rust
fn main() {
    let mut signal = "yellow";

    while signal != "stop" {
        match signal {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => {}
        }
        if signal == "green" { signal = "red"; } else if signal == "red" { signal = "yellow"; } else if signal == "green" { signal = "red"; }
    }

    println!("stopped");
}
```
