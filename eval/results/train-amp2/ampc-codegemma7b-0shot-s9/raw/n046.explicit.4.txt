fn count_below_cutoff(numbers: &Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for number in numbers {
        if number < &cutoff {
            count = count + 1
        }
    }
    count
}

fn main() {
    let numbers = vec(5, 12, 3, 18, 9)
    let cutoff = 10
    let below_cutoff = count_below_cutoff(&numbers, cutoff)
    let at_or_above_cutoff = numbers.len() - below_cutoff
    print(below_cutoff)
    print(at_or_above_cutoff)
    drop numbers
}
