struct Log { entries: Vec<Int> }

fn start_entry(log: Log) -> Int {
    match get(log.entries, 0) {
        Some(n) => n,
        None => -1,
    }
}

fn end_entry(log: Log) -> Int {
    match get(log.entries, len(log.entries) - 1) {
        Some(n) => n,
        None => -1,
    }
}

fn main() {
    let log = Log { entries: vec![6, 7, 8, 9] }
    print(start_entry(log))
    let log2 = Log { entries: log.entries }
    print(end_entry(log2))
}
