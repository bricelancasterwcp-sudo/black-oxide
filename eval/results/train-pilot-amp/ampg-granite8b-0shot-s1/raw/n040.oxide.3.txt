struct Shape {
    kind: ShapeKind,
}

enum ShapeKind {
    Square(Float),
    Circle(Float),
}

fn area(s: Shape) -> Float {
    match s.kind {
        ShapeKind(kind) => {
            match kind {
                ShapeKind(kind) => 4.0 * kind * kind,
                ShapeKind(kind) => 3.1416 * kind * kind,
            }
        },
    }
}

fn main() {
    print(area(Shape { kind: ShapeKind(Square(4.0)) }))
}
