struct CountDivisors { count: Int }

fn count_divisors(n: Int) -> CountDivisors {
    let mut count = 0
    for i in range(1, n + 1) {
        if n % i == 0 {
            count = count + 1
        }
    }
    CountDivisors(count)
}

fn main() {
    let divisors = count_divisors(36)
    print(divisors.count)
}
