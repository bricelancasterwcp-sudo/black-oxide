fn main() {
    let mut a = 1;
    let mut b = 1;

    for _ in 1..9 {
        let temp = b; // Temporary variable to hold b's value before it is updated
        b = a + b;
        a = temp; // Reverse the order of terms by assigning the previous b back to a
    }
