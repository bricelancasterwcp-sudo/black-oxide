fn main() {
    let mut v = [9, 2, 7, 4];
    for i in 0..v.len() - 1 { // adjust loop range to avoid out of bounds error
        for j in (i + 1)..v.len() {
            if v[i] > v[j] {
                v.swap(i, j);
            }
        }
    }

    for x in v.iter() {
        print!("{}", x);
        println!();
    }
}
