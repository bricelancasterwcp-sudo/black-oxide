```rust
struct Figure {
    value: Int,
}

fn main() {
    let figures = vec![
        Figure { value: 8 },
        Figure { value: 3 },
        Figure { value: 11 },
        Figure { value: 6 },
    ];

    let min_value = figures.iter().min_by(|a, b| a.value.cmp(&b.value)).unwrap().value;
    let max_value = figures.iter().max_by(|a, b| a.value.cmp(&b.value)).unwrap().value;

    print(min_value);
    println(max_value - min_value);
}
```
