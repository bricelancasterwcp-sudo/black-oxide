fn main() {
    let fig = "fig";
    let banana = "banana";
    let kiwi = "kiwi";

    let longest_name = if fig.len() > banana.len() && fig.len() > kiwi.len() {
        fig
    } else if banana.len() > kiwi.len() {
        banana
    } else {
        kiwi
    };

    println!("The longest name is: {}", longest_name);
    println!("It has {} letters.", longest_name.len());
}
