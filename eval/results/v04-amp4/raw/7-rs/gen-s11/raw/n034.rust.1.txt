```rust
fn main() {
    let v1 = vec![7];
    match second_entry(&v1) {
        Some(val) => println!("{}", val),
        None => println!("None"),
    }

    let v2 = vec![7, 8];
    match second_entry(&v2) {
        Some(val) => println!("{}", val),
        None => println!("None"),
    }

    let v3 = vec![7, 8, 9];
    match second_entry(&v3) {
        Some(val) => println!("{}", val),
        None => println!("None"),
    }
}

fn second_entry<T>(v: &[T]) -> Option<&T> {
    v.get(1)
}
```
