```rust
fn main() {
    let v: Vec<Int> = vec![7, 8, 9];
    match second(v) {
        Some(n) => println!("{}", n),
        None => println!("no second entry"),
    }
}

// Helper function to get the second element of a vector
fn second(v: Vec<Int>) -> Option<Int> {
    if v.len() > 1 {
        Some(v[1])
    } else {
        None
    }
}
```
