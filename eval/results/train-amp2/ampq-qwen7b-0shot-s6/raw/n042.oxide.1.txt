struct Reading { label: Str, values: Vec<Int> }

fn second_largest(v: Vec<Int>) -> Option<Int> {
    let mut largest = -1
    let mut second_largest = -1
    for x in v {
        if x > largest {
            second_largest = largest
            largest = x
        } else if x > second_largest && x != largest {
            second_largest = x
        }
    }
    if second_largest == -1 {
        None
    } else {
        Some(second_largest)
    }
}
