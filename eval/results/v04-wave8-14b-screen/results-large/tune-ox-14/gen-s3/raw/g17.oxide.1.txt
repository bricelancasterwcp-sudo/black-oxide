fn merge_sorted(a: Vec<Int>, b: Vec<Int>) -> Vec<Int> {
    let mut result = vec()
    let mut i = 0
    let mut j = 0
    while i < len(a) && j < len(b) {
        if a[i] <= b[j] {
            result = push(result, a[i])
            i += 1
        } else {
            result = push(result, b[j])
            j += 1
        }
    }
    while i < len(a) {
        result = push(result, a[i])
        i += 1
    }
    while j < len(b) {
        result = push(result, b[j])
        j += 1
    }
    result
}
fn main() {
    let list1 = vec(2, 5, 9, 14, 21, 30)
    let list2 = vec(1, 5, 8, 15, 22)
    let merged = merge_sorted(list1, list2)
    print(len(merged)) // how many numbers the merged list holds
    if len(merged) > 0 {
        print(merged[0]) // its first number
        print(merged[len(merged) - 1]) // its last number
    }
    let sum = 0
    for x in merged {
        sum += x
    }
    print(sum) // the sum of all its numbers
    let mut is_sorted = true
    for i in range(1, len(merged)) {
        if merged[i] < merged[i - 1] {
            is_sorted = false
            break
        }
    }
    print(is_sorted) // whether it really is in non-decreasing order
    let mut duplicates = 0
    for i in range(1, len(merged)) {
        if merged[i] == merged[i - 1] {
            duplicates += 1
        }
    }
    print(duplicates) // how many times a number is immediately followed by an equal number
}
