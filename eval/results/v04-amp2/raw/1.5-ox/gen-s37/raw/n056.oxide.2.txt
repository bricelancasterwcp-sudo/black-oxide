```rust
fn main() {
    let mut sum = 0;
    let mut count = 0;

    for c in "7x9?".chars() {
        match c {
            '0'..='9' => sum += c.to_digit(10).unwrap(),
            _ => {}
        }
        if c != '?' {
            count += 1;
        }
    }

    println!("Sum: {}", sum);
    println!("Count: {}", count);
}
```
