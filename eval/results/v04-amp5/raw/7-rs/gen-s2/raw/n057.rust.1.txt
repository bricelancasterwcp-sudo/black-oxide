```rust
fn main() {
    let mut max_length = 0;
    let mut word_with_max_length = "";

    for &word in ["fig", "banana", "kiwi"].iter() {
        if word.len() > max_length {
            max_length = word.len();
            word_with_max_length = word;
        }
    }

    println!("{} {}", word_with_max_length, max_length);
}
```
