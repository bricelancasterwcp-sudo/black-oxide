fn main() {
    let numbers = vec("7", "x", "9", "?");
    let mut sum = 0;
    let mut count_non_numbers = 0;

    for s in numbers {
        match parse_int(s) {
            Some(n) => sum = sum + n,
            None => count_non_numbers = count_non_numbers + 1,
        }
    }

    print(sum);
    print(count_non_numbers);
}
