fn compare(x: &Str, y: &Str) -> Int {
    if x.len() > y.len() {
        1
    } else if x.len() < y.len() {
        -1
    } else {
        0
    }
}

fn main() {
    let fig = "fig".to_string()
    let banana = "banana".to_string()
    let kiwi = "kiwi".to_string()
    let largest = compare(&fig, &banana)
    if largest > 0 {
        print(fig)
        print(fig.len())
    } else if largest < 0 {
        print(banana)
        print(banana.len())
    } else {
        print(kiwi)
        print(kiwi.len())
    }
}
