struct Line { units: Int, price: Int }

fn line_value(line: Line) -> Int {
    line.units * line.price
}

fn main() {
    let line1 = Line { units: 4, price: 3 }
    let line2 = Line { units: 5, price: 2 }
    print(line_value(line1))
    print(line_value(line2))
    let total = line_value(line1) + line_value(line2)
    print(total)
}
