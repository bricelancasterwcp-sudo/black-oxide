fn main() {
    let list = [7, 4, 9, 10];

    let mut odd_sum = 0;

    for &item in list.iter() {
        if item % 2 != 0 {
            println!("{} doubled: {}", item, item * 2);
            odd_sum += item;
        }
    }

    // This line should print the total of the even entries
    println!("Total of even entries: {}", odd_sum);
}
