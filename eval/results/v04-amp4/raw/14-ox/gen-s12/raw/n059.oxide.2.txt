fn main() {
    let phrase = "when in doubt go home"
    let words: Vec<Str> = vec()
    for word in phrase {
        if char(word) != ' ' {
            words = push(words, concat(word))
        } else {
            words = push(words, "")
        }
    }
    let word_count = len(words)
    let char_count = str_len(phrase)
    print(word_count)
    print(char_count)
}
