struct Number { value: Option<Int> }

fn sum_numbers(numbers: Vec<Number>) -> Int {
    let mut sum = 0
    for n in numbers {
        if let Some(v) = n.value {
            sum = sum + v
        }
    }
    sum
}

fn count_non_numbers(numbers: Vec<Number>) -> Int {
    let mut count = 0
    for n in numbers {
        if n.value == None {
            count = count + 1
        }
    }
    count
}

fn main() {
    let nums = vec![
        Number { value: Some(7) },
        Number { value: None },
        Number { value: Some(9) },
        Number { value: None }
    ]
    print(sum_numbers(nums))
    print(count_non_numbers(nums))
}
