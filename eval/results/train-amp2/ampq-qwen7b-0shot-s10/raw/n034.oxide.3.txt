fn second(v: Vec<Int>) -> Option<Int> {
    get(v, 1)?
}

fn main() {
    match second(push(push(vec(), 7), 8)) {
        Some(n) => print(n),
        None => print_str("None"),
    }
    match second(push(vec(), 9)) {
        Some(n) => print(n),
        None => print_str("None"),
    }
    match second(push(push(vec(), 10), 11)) {
        Some(n) => print(n),
        None => print_str("None"),
    }
}
