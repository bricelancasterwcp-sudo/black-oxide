fn main() {
    let mut sum = 0;
    let mut total = 0;
    for num in [7, 4, 9, 10] {
        if num % 2 == 1 {
            println!("{}", num * 2);
        } else {
            sum += num;
            total += 1;
        }
    }
    println!("Sum of even entries: {}", if total == 0 { 0 } else { sum / total as f64 });
}
