```rust
fn main() {
    let figures = vec![8, 3, 11, 6];
    let min_figure = figures.iter().cloned().min().unwrap();
    let max_figure = figures.iter().cloned().max().unwrap();
    let gap = max_figure - min_figure;
    println!("Smallest figure: {}", min_figure);
    println!("Gap between largest and smallest: {}", gap);
}
```
