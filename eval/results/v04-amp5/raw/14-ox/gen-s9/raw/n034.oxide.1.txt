fn second_entry(v: Vec<Int>) -> Option<Int> {
    match get(v, 1) {
        Some(x) => Some(x + 1),
        None => None,
    }
}

fn main() {
    print(second_entry(push(push(vec(), 7), 8)))  // Should print "9"
    print(second_entry(push(push(vec(), 8), 9)))  // Should print "10"
    print(second_entry(push(vec(), 9)))           // Should print "none"
}
