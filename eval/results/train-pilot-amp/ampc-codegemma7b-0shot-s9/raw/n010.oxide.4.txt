fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    for i in range(2, trunc(n / 2) + 1) {
        if n % i == 0 {
            return false
        }
    }
    return true
}

fn main() {
    let n = 91
    if is_prime(n) {
        print_str("Yes")
    } else {
        print_str("No")
    }
}
