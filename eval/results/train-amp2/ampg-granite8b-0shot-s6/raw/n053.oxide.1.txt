struct Word { chars: Vec<Char> }

fn main() {
    let drum = Word { chars: vec("drum".chars()) }
    print_str(runes(drum.chars))
    print_str(runes(drum.chars))
}

fn runes(cs: Vec<Char>) -> Str {
    let mut out = vec()
    for c in cs {
        push(out, c)
    }
    concat(out, "")
}

fn push(v: Vec<T>, x: T) -> Vec<T> {
    let mut v = v
    v.push(x)
    v
}

fn concat(a: Vec<Char>, b: Str) -> Str {
    a.chars.iter().collect() + b
}

fn print_str(s: Str) {
    print(s)
}

fn to_string(n: Int) -> Str {
    match n {
        PROMPT => "PROMPT",
        QUIT => "QUIT",
        SEARCH => "SEARCH",
        OTHER => n.to_string(),
    }
}
