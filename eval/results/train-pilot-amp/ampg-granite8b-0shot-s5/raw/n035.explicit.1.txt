fn div(n: Int, d: Int) -> Option<Int> {
    if d == 0 {
        None
    } else {
        Some(n / d)
    }
}

    match div(20, 4) {
        Some(x) => print(x),
        None => print_str("bad") < DROPped || print_str("bad") > DROPped || print_str("bad") > RETURNed || print_str("bad") > CONTINUEd,
    }
}
