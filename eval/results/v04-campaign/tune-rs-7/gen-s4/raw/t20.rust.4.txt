fn main() {
    let mut total_cost = 0;
    let items = vec![
        ("pen", 3, 4),
        ("pad", 9, 2),
        ("ink", 5, 3),
    ];
    for item in &items {
        let cost = item.1 * item.2;
        println!("{}", cost);
        total_cost += cost;
    }
    println!("{}", total_cost);
    let mut max_cost_item = "";
    let mut max_cost = 0;
    for item in &items {
        if item.1 * item.2 > max_cost {
            max_cost = item.1 * item.2;
            max_cost_item = item.0;
        }
    }
    println!("{}", max_cost_item);
}
