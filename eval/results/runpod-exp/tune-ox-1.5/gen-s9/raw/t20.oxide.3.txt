struct Item { unit_price: Int, quantity: Int }
fn main() {
    let pen = Item { unit_price: 3, quantity: 4 }
    let pad = Item { unit_price: 9, quantity: 2 }
    let ink = Item { unit_price: 5, quantity: 3 }
    let items = vec(pen, pad, ink)
    let total_cost = items.iter().map(|i| i.unit_price * i.quantity).sum()
    let max_cost_item = items
        .iter()
        .max_by_key(|i| i.unit_price * i.quantity)
        .unwrap()
    print(total_cost)
    print(max_cost_item.name())
}
