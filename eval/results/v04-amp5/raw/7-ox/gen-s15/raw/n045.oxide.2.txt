```rust
struct Log { entries: Vec<Int> }

fn start_and_end(log: Log) {
    let start = log.entries[0];
    let end = match log.entries.last() {
        Some(e) => *e,
        None => -1,
    };
    print(start);
    print_str(" ");
    print(end);
}

fn main() {
    let log = Log { entries: vec![6, 7, 8, 9] };
    start_and_end(log);
}
```
