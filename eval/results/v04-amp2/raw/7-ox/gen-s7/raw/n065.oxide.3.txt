fn count_above_five(v: Vec<Int>) -> Int {
    let mut count = 0;
    for x in v {
        if x > 5 {
            count += 1;
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max_val = None;
    for &x in v.iter() {
        match max_val {
            Some(m) => if x > m { max_val = Some(x); },
            None => max_val = Some(x),
        }
    }
    max_val
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14);
    let readings = push(readings, 9);
    print(count_above_five(readings.clone()));
    match max_reading(&readings) {
        Some(n) => print(n),
        None => print_str("empty"),
    }
}
