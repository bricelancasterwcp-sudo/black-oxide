fn digit_sum(n: Int) -> Int {
    let mut sum = 0;
    let mut num = n; // Add a mutable variable to carry the current value of n
    while num > 0 {
        sum += num % 10;
        num /= 10; // Use num instead of n
    }
    sum
}

fn main() {
    let number = 9875;
    print(digit_sum(number));
}
