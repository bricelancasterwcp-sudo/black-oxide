```rust
struct Square { side: Int }
struct Circle { radius: Float }

impl Square {
    fn area(&self) -> Int {
        self.side * self.side
    }
}

impl Circle {
    fn area(&self) -> Float {
        self.radius * self.radius * 3.14
    }
}

fn main() {
    let sq = Square { side: 4 };
    println!("{}", sq.area());
}
```
