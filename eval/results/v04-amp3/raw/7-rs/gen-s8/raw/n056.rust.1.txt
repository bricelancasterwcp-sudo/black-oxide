```rust
fn main() {
    let values = vec!["7", "x", "9", "?"];
    let mut sum = 0;
    let mut count_non_numbers = 0;

    for value in values {
        if let Ok(num) = value.parse::<i32>() {
            sum += num;
        } else {
            count_non_numbers += 1;
        }
    }

    println!("Total of numbers: {}", sum);
    println!("Count of non-numbers: {}", count_non_numbers);
}
```
