struct Reading { label: Str, values: Vec<Int> }

fn lookup(v: Vec<Int>, pos: Int) -> Option<Int> {
    get(v, pos)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 10), 20) }
    match lookup(r.values, 0)? {
        Some(n) => print(n),
        None => print_str("Position empty"),
    }
    match lookup(r.values, 3)? {
        Some(n) => print(n),
        None => print_str("Position empty"),
    }
}
