fn main() {
    let text = "coffee";
    let mut char_count: [usize; 26] = [0; 26];
    let alphabet: [char; 26] = [
        'a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'i', 'j', 
        'k', 'l', 'm', 'n', 'o', 'p', 'q', 'r', 's', 't', 
        'u', 'v', 'w', 'x', 'y', 'z'
    ];

    for &c in text.chars() {
        let index = c as usize - 'a' as usize;
        char_count[index] += 1;
    }

    for i in 0..26 {
        if char_count[i] > 1 {
            println!("{}", alphabet[i]);
        }
    }
}
