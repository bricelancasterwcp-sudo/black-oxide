```rust
fn main() {
    let mut count = 0;
    for n in 2..100 {
        if is_prime(n) {
            count += 1;
        }
    }
    print(count);
}

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false;
    }
    for i in 2..=((n as f64).sqrt() as Int) + 1 {
        if n % i == 0 {
            return false;
        }
    }
    true
}
```
