fn main() {
    let numbers = vec![5, 12, 3, 18, 9];
    let below_cutoff_count = numbers.iter().filter(|&&x| x < 10).count();
    let at_or_over_cutoff_count = numbers.iter().filter(|&&x| x >= 10).count();

    println!("Numbers below the cutoff: {}", below_cutoff_count);
    println!("Numbers at or over the cutoff: {}", at_or_over_cutoff_count);
}
