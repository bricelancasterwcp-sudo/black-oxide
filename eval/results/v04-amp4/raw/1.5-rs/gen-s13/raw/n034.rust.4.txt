```rust
fn main() {
    let numbers = vec![7, 8, 9];
    
    match get_second_entry(&numbers) {
        Some(second) => println!("Second entry: {}", second),
        None => println!("No second entry"),
    }
}

fn get_second_entry(numbers: &[i32]) -> Option<i32> {
    numbers.get(1).copied()
}
```
