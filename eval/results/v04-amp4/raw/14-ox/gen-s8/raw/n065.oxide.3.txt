struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max? < x || max == None {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14)
    readings = push(readings, 9)
    
    let r = Reading { label: "lab", values: readings.clone() }
    
    print(count_exceeding(r.values.clone(), 5))            // prints the count of readings exceeding 5
    match find_max(r.values.clone()) {
        Some(n) => print(n),                       // prints the highest reading
        None => print_str("No readings"),
    }
}
