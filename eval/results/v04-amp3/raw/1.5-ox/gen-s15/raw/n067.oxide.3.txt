fn walk(v: Vec<Int>) -> Int {
    let mut total = 0;
    for i in 0..v.len() { // Use &x to avoid copying
        if i % 2 != 0 && v[i] % 2 != 0 {
            println!("{}", v[i] * 2);
            total += v[i]; // Update total before printing
        }
    }
    return total;
}

fn main() {
    let v = vec![7, 4, 9, 10];
    let result = walk(v);
    println!("Total of even entries: {}", result);
}
