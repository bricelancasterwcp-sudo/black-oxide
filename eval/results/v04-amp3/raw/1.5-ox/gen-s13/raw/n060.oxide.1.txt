fn main() {
    let mut total = 0;
    let count = 0;
    for c in a1b22.chars() {
        if c.is_digit(10) {
            total += c.to_digit(10).unwrap();
            count += 1;
        }
    }
    println!("Total sum of digits: {}", total);
    println!("Number of digit characters: {}", count);
}
