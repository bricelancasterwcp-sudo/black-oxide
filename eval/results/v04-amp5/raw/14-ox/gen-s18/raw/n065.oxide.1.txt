fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let count = 0
    for x in v {
        if x > threshold {
            count = count + 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        max = match max {
            Some(m) => {
                if x > m {
                    Some(x)
                } else {
                    Some(m)
                }
            },
            None => Some(x),
        }
    }
    max
}

fn main() {
    let readings = push(push(push(push(push(vec(), 6), 11), 3), 14), 9)
    let count = count_exceeding(readings, 5)
    print(count)
    let max = max_reading(readings)
    match max {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
