```rust
fn main() {
    let words = vec!["coffee", "hello", "world"];
    
    for word in words {
        let mut seen_letters: std::collections::HashSet<char> = std::collections::HashSet::new();
        
        for letter in word.chars() {
            if !seen_letters.contains(&letter) {
                println!("{}", letter);
                seen_letters.insert(letter);
            }
        }
    }
}
```
