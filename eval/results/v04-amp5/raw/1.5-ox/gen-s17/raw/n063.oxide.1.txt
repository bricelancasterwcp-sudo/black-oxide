```rust
struct Line {
    name: Str,
    price: Int,
    quantity: Int,
}

fn main() {
    let first_line = Line { name: "Four units at $3 each", price: 3, quantity: 4 };
    let second_line = Line { name: "Five units at $2 each", price: 2, quantity: 5 };

    print(first_line.price * first_line.quantity);
    print(second_line.price * second_line.quantity);

    println((first_line.price * first_line.quantity) + (second_line.price * second_line.quantity));
}
```
