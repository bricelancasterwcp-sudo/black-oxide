```rust
fn main() {
    let numbers = [5, 12, 3, 18, 9];
    let cutoff = 10;
    let (under, over) = numbers.iter().fold((0, 0), |(under, over), &num| {
        if num < cutoff {
            (under + 1, over)
        } else {
            (under, over + 1)
        }
    });
    println!("Under cutoff: {}, Over cutoff: {}", under, over);
}
```
