struct Queue { values: Vec<Int> }

fn swap_first_last(q: Queue) -> Queue {
    let len = len(q.values)
    if len < 2 {
        return q
    }
    let first = get(q.values, 0)?
    let last = get(q.values, len - 1)?
    let middle = range(1, len - 1)
    let new_values = push(push(push(vec(), last), middle...), first)
    Queue { values: new_values }
}

fn main() {
    let q = Queue { values: vec![1, 2, 3, 4, 5] }
    let swapped_q = swap_first_last(q)
    for x in swapped_q.values {
        print(x)
    }
}
