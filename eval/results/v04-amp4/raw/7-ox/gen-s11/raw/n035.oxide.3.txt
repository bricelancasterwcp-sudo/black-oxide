fn divide(x: Int, y: Int) -> Result<Float> {
    if y == 0 {
        Err("bad".to_string())
    } else {
        Ok(to_float(x) / to_float(y))
    }
}

fn main() {
    let result = divide(20, 4);
    match result {
        Ok(result) => print(result),
        Err(error) => print_str(&error),
    }
}
