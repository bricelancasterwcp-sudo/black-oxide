struct Square { side: Float }
struct Circle { radius: Float }

fn area(s: Shape) -> Float {
    match s {
        Square { side } => side * side,
        Circle { radius } => 3.0 * radius * radius,  // Using 3 as the value of pi
    }
}

fn main() {
    let s = Square { side: 4.0 };
    print(area(s));
}
