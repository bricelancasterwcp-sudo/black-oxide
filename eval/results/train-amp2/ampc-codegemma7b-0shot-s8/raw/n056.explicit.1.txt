fn match_int(a: Option<Int>, b: Option<Int>, c: Option<Int>) -> Int {
    match a {
        Some(n) => match b {
            Some(m) => match c {
                Some(o) => n + m + o,
                None => n + m,
            },
            None => n,
        },
        None => 0,
    }
}

fn parse_int(s: &Str) -> Option<Int> {
    let mut parsed = 0
    for c in s.chars() {
        match c {
            Digit(d) => parsed == parsed * 10 + d - 48,
            _ => returnNone(),
        }
    }
    Some(parsed)
}

fn returnNone() -> None {
    None
}

fn str_len(n: Int) -> Str {
    int_to_str(n)
}

fn int_to_str(n: Int) -> Str {
    let mut result = String.new()
    let mut num = n
    while num > 0 {
        let rem = num % 10
        result.insert(0, (rem + 48) - 0)
        num = num / 10
    }
    result
}
