```rust
fn main() {
    let mut count = 0;
    let sum = 0;
    for text in vec!["7", "x", "9", "?"] {
        if text.parse::<i32>().is_ok() {
            sum += text.parse::<i32>().unwrap();
            count += 1;
        }
    }
    println!("Sum: {}", sum);
    println!("Count: {}", count);
}
```
