fn main() {
    let values = vec(12, 5, 20, 8, 17)
    let mut largest = 0
    let mut second_largest = 0
    for x in values {
        if x > largest {
            second_largest = largest
            largest = x
        } else if x > second_largest && x != largest {
            second_largest = x
        }
    }
    print(largest)
    print(second_largest)
}
