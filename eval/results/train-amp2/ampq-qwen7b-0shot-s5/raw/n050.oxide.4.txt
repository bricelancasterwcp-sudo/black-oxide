fn main() {
    let mut scores = vec(5, 9, 2)
    scores.sort_by_int_desc()
    for score in scores {
        print_str(score.to_string())
        print_str("\n")
    }
}

fn sort_by_int_desc(v: Vec<Int>) -> Vec<Int> {
    let mut v = v
    v.sort()
    v.reverse()
    v
}
