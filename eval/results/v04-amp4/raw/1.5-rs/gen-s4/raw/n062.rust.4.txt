fn main() {
    loop {
        let signal = get_signal(); // Replace with actual logic to get the signal
        match signal {
            "green" => println!("Go"),
            "red" => println!("Stop"),
            "yellow" => println!("Wait"),
            _ => break, // Break the loop if an unexpected signal is received
        }
    }

    fn get_signal() -> &'static str {
        rand::random::<bool>()  // Importing random function from rand crate

        // Use a random number generator to simulate getting signals
        let green = rand::random::<bool>();
        if green { "green" } else { "yellow" }
    }
}
