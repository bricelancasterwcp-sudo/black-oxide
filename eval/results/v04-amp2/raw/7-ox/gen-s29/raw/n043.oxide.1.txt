struct Queue { items: Vec<Int> }

fn enqueue(q: Queue, x: Int) -> Queue {
    Queue { items: push(q.items, x) }
}

fn dequeue(q: Queue) -> (Option<Int>, Queue) {
    let (head, tail) = split_at(q.items, 1);
    (head.get(0), Queue { items: tail })
}

fn main() {
    let mut q = enqueue(enqueue(enqueue(enqueue(Queue { items: vec() }, 1), 2), 3), 4);
    q = enqueue(q, 5);

    let (first, q) = dequeue(q);
    let (last, _) = dequeue(dequeue(dequeue(dequeue(q))));

    q = enqueue(enqueue(enqueue(Queue { items: vec() }, last.unwrap()), first.unwrap()), 2);
    q = enqueue(enqueue(Queue { items: vec() }, 3));
    q = enqueue(enqueue(Queue { items: vec() }, 4));
    q = enqueue(q, 5);

    while let Some(x) = dequeue(q).0 {
        print(x);
    }
}
