```rust
fn main() {
    let word = "keyboard";
    let mut vowels = 0;
    let mut not_vowels = 0;

    for letter in word.chars() {
        match letter {
            'a' | 'e' | 'i' | 'o' | 'u' => vowels += 1,
            _ => not_vowels += 1,
        }
    }

    println!("Vowels: {}", vowels);
    println!("Not vowels: {}", not_vowels);
}
```
