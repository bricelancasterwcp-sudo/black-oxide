```rust
fn main() {
    let input = "a1b22";

    let mut total = 0;
    let mut digit_count = 0;

    for c in input.chars() {
        if c.is_digit(10) {
            total += c.to_digit(10).unwrap();
            digit_count += 1;
        }
    }

    println!("Total: {}", total);
    println!("Digit count: {}", digit_count);
}
```
