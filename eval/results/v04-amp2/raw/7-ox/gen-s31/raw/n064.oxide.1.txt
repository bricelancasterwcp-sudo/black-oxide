struct List { values: Vec<Int> }

fn get_value(v: Vec<Int>, i: Int) -> Option<Int> {
    let x = get(v, i)?
    Some(x)
}

fn main() {
    let l = List { values: push(push(vec(), 10), 20) }
    match get_value(l.values, 1) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match get_value(l.values, 4) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
