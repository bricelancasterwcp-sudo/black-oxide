```rust
fn get_second_item<T>(list: &[T]) -> Option<&T> {
    if list.len() > 1 {
        Some(&list[1])
    } else {
        None
    }
}

fn main() {
    let list7 = [1, 2, 3, 4, 5, 6, 7];
    let list8 = [10, 20];
    let list9 = [];

    println!("{:?}", get_second_item(&list7)); // Some(2)
    println!("{:?}", get_second_item(&list8)); // Some(20)
    println!("{:?}", get_second_item(&list9)); // None
}
```
