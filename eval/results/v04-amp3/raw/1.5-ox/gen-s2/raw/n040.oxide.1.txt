```rust
struct Square { side: Int }
struct Circle { radius: Float }

fn main() {
    let s = Square { side: 4 };
    println!("{}", area_of_shape(s)); // prints `16.0`
}

fn area_of_shape(shape: Shape) -> Float {
    match shape {
        Square { side } => side * side,
        Circle { radius } => (radius * radius).pi(),
    }
}
```
