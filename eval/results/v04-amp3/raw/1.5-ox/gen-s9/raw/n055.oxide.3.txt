```rust
fn main() {
    let oak = Some(3);
    match oak {
        Some(n) => println!("{} is an oak", n),
        None => println!("bad"),
    }

    let nine: i32 = 9;
    match nine > 10 {
        true => println!("{} is bad", nine),
        false => println!("{}", nine),
    }

    let fifteen: Option<i32> = Some(15);
    match fifteen % 3 == 0 {
        true => println!("{} is divisible by 3", fifteen.unwrap()),
        false => println!("bad"),
    }
}
```
