fn count_positives(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count += 1
        }
    }
    count
}

fn max_value(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if let Some(m) = max {
            if x > m {
                max = Some(x)
            }
        } else {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let values = push(push(push(push(vec(), 3), 8), -2), 12)
    let values = push(values, 7)
    print(count_positives(values))
    match max_value(values) {
        Some(n) => print(n),
        None => print_str("no values"),
    }
}
