fn divide(a: Int, b: Int) -> Option<Int> {
    if b == 0 {
        None
    } else {
        Some(a / b)
    }
}

fn main() {
    let result = divide(20, 4).unwrap_or(0)
    print(result)
    drop result
    match divide(20, 0) {
        Some(bad) => print(bad),
        None => print_str("bad"),
    }
    drop bad
}
