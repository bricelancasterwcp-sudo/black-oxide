fn main() {
    let nums: [i32; 4] = [7, 10, 9, 8];
    let mut sum: i32 = 0;
    let mut count: u32 = 0;

    for &num in nums.iter() {
        if num != None {
            sum += num;
            count += 1;
        }
    }

    println!("Sum of valid numbers: {}", sum);
    println!("Count of invalid numbers: {}", count);
}
