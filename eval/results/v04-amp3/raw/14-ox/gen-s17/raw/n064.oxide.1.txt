struct Reading { label: Str, values: Vec<Int> }

fn lookup(v: Vec<Int>, i: Int) -> Option<Int> {
    get(v, i)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 10), 20) }
    match lookup(r.values, 1) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match lookup(r.values, 4) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
