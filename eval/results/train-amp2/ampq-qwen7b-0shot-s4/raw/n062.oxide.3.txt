struct Signal { name: Str }

fn main() {
    let signals = vec(Signal { name: "green" }, Signal { name: "red" }, Signal { name: "yellow" })
    for signal in signals {
        match signal.name.as_str() {
            Str(x) => match x.as_str() {
                Str(x) => {
                    if x == "green" {
                        print_str("go for green")
                    } else if x == "red" {
                        print_str("stop for red")
                    } else if x == "yellow" {
                        print_str("wait for yellow")
                    }
                },
                _ => print_str("unknown signal"),
            },
            _ => print_str("unknown signal"),
        }
    }
}
