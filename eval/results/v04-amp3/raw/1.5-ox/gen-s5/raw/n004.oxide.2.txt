```rust
struct Sequence {
    first: Int,
    second: Int,
}

impl Sequence {
    fn new(first: Int) -> Self {
        Sequence { first, second }
    }

    fn next(&mut self) -> Int {
        let third = self.first + self.second;
        self.first = self.second;
        self.second = third;
        third
    }

    fn nth(&self, n: Int) -> Option<Int> {
        if n <= 0 {
            return None;
        }
        let mut sequence = Sequence::new(1);
        for _ in 1..n {
            sequence.next();
        }
        Some(sequence.second)
    }
}

fn main() {
    match Sequence::nth(10) {
        Some(n) => print_int(n),
        None => println_str("Sequence not defined"),
    }
}
```
