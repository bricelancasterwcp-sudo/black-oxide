struct Reading { label: Str, values: Vec<Str> }

fn sum_of_numbers(v: Vec<Str>) -> Int {
    let mut sum = 0;
    for x in v {
        match parse_int(x) {
            Some(n) => sum += n,
            None => continue,
        }
    }
    sum
}

fn count_non_numbers(v: Vec<Str>) -> Int {
    let mut count = 0;
    for x in v {
        if parse_int(x).is_none() {
            count += 1;
        }
    }
    count
}

fn main() {
    let readings = Reading { label: "data", values: vec("7", "x", "9", "?") };
    print(sum_of_numbers(readings.values));
    print(count_non_numbers(readings.values));
}
