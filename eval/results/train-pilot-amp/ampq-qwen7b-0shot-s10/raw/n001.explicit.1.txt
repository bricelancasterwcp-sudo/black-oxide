fn cube(x: Int) -> Int {
    x * x * x
}

fn sum_cubes(v: &Vec<Int>) -> Int {
    let total = 0
    for x in v {
        total = total + cube(x)
    }
    total
}

fn main() {
    let nums = vec()
    for i in range(1, 6) {
        push(nums, i)
    }
    let total = sum_cubes(&nums)
    print(total)
    drop nums
}
