struct Reading { label: Str, values: Vec<Int> }

fn lookup(v: Vec<Int>, i: Int) -> Option<Int> {
    get(v, i)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 10), 20) }
    match lookup(r.values.clone(), 0) { // Clone the values to avoid moving
        Some(n) => print(n),
        None => print_str("position 0 is empty"),
    }
    match lookup(r.values.clone(), 3) { // Clone the values to avoid moving
        Some(n) => print(n),
        None => print_str("position 3 is empty"),
    }
}
