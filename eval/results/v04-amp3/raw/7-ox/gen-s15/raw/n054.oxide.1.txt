```rust
fn main() {
    let mut vowels = 0;
    let mut consonants = 0;

    for c in "keyboard".chars() {
        if matches!(c, 'a' | 'e' | 'i' | 'o' | 'u' | 'A' | 'E' | 'I' | 'O' | 'U') {
            vowels += 1;
        } else {
            consonants += 1;
        }
    }

    print(vowels);
    print_str(" ");
    print(consonants);
}
```
