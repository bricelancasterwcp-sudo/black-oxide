fn main() {
    let item1_price = 3;
    let item2_price = 2;

    // Calculate the total price for four units of each type
    let total_value_item1 = 4 * item1_price;
    let total_value_item2 = 5 * item2_price;

    // Print the value of each line
    println!("Value of the first line: {}", total_value_item1);
    println!("Value of the second line: {}", total_value_item2);

    // Calculate the combined total value of both lines
    let combined_total_value = total_value_item1 + total_value_item2;

    // Print the combined total value
    println!("Total value: {}", combined_total_value);
}
