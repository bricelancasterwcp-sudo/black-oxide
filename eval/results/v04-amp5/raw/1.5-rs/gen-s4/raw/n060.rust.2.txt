fn main() {
    let text = "a1b22";
    let mut sum_digits: u32 = 0;
    let mut count_digits: usize = 0;

    for c in text.chars() {
        if c.is_digit(10) {
            let digit: u32 = c.to_digit(10).unwrap();
            sum_digits += digit;
            count_digits += 1;
        }
    }

    println!("Sum of digits: {}", sum_digits);
    println!("Count of digits: {}", count_digits);
}
