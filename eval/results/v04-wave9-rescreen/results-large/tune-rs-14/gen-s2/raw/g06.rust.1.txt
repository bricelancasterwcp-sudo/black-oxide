fn main() {
    let morning = vec![94, 71, 88, 65, 92, 79, 83];
    let evening = vec![55, 90, 77, 84, 68, 99];
    let mut results = Vec::new();
    for (name, scores) in [("Morning", &morning), ("Evening", &evening)] {
        let mut count_90_plus = 0;
        let mut count_80_to_89 = 0;
        let mut count_70_to_79 = 0;
        let mut count_below_70 = 0;
        let mut total = 0;
        let mut max_score = 0;
        for &score in scores {
            if score >= 90 {
                count_90_plus += 1;
            } else if score >= 80 {
                count_80_to_89 += 1;
            } else if score >= 70 {
                count_70_to_79 += 1;
            } else {
                count_below_70 += 1;
            }
            total += score;
            if score > max_score {
                max_score = score;
            }
        }
        let mean = total / scores.len() as i32;
        results.push((name, count_90_plus, count_80_to_89, count_70_to_79, count_below_70, max_score, mean));
    }
    for (name, count_90_plus, count_80_to_89, count_70_to_79, count_below_70, max_score, mean) in results {
        println!("{}", name);
        println!("{}", count_90_plus);
        println!("{}", count_80_to_89);
        println!("{}", count_70_to_79);
        println!("{}", count_below_70);
        println!("{}", max_score);
        println!("{}", mean);
    }
    if results[0].6 > results[1].6 {
        println!("Morning");
    } else {
        println!("Evening");
    }
}
