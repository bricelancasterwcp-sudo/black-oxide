```rust
fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut longest_name = &names[0];
    let mut max_length = names[0].len();

    for name in names.iter() {
        if name.len() > max_length {
            longest_name = name;
            max_length = name.len();
        }
    }

    println!("{} has the most letters, with {} letters.", longest_name, max_length);
}
```
