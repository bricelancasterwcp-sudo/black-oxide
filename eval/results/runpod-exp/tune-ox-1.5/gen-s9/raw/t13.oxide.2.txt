fn is_palindrome(s: Str) -> Bool {
    let chars = chars(s)
    for i in range(0, len(chars) / 2) {
        if chars[i] != chars[len(chars) - 1 - i] {
            return false
        }
    }
    true
}
fn main() {
    print(is_palindrome("level"))
    print(is_palindrome("stereo"))
    print(is_palindrome("rotor"))
}
