struct List { first: Option<Int>, second: Option<Int> }

fn get_at(list: List, pos: Int) -> Option<Int> {
    if pos == 1 {
        list.first
    } else if pos == 4 {
        list.second
    } else {
        None
    }
}

fn main() {
    let list = List { first: Some(10), second: Some(20) }
    match get_at(list, 1) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match get_at(list, 4) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
