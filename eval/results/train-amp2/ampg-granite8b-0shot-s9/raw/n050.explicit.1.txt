fn main() {
    let scores = vec(2, 9, 5)
    let sorted_scores = sort(scores)
    for score in sorted_scores {
        print(score)
    }
}

fn sort(vec: Vec<Int>) -> Vec<Int> {
    if len(&vec) <= 1 {
        vec
    } else {
        let mut left = vec()
        let mut right = vec()
        for i in 0.to_i64(len(&vec) - 1) {
            let score = get(&vec, i)
            if score < 5 {
                left = push(left, score)
            } else {
                right = push(right, score)
            }
        }
        concat(sort(left), sort(right))
    }
}

    let n = vec.len()
    let mut v = vec
    v = v.resize(n + 1)
