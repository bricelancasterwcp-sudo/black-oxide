```rust
fn sum_of_digits(n: Int) -> Int {
    let mut sum = 0
    let mut remaining = n

    while remaining > 0 {
        let digit = remaining % 10
        sum = sum + digit
        remaining = remaining / 10
    }

    sum
}

fn main() {
    let number = 9875
    print(sum_of_digits(number))
}
```
