struct Queue { items: Vec<Int> }

fn push_back(q: Queue, x: Int) -> Queue {
    let mut q = q;
    q.items.push(x);
    q
}

fn swap_first_last(mut q: Queue) -> Queue {
    if q.items.len() > 1 {
        let n = q.items.len();
        q.items.swap(0, n - 1);
    }
    q
}

fn print_queue(q: Queue) {
    for item in &q.items {
        print(item);
        print_str("\n");
    }
}

fn main() {
    let mut q = Queue { items: vec![1, 2, 3, 4, 5] };
    q = swap_first_last(q);
    print_queue(q);
}
