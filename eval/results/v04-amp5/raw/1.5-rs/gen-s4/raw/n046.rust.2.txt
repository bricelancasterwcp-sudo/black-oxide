fn main() {
    let sample = vec![5, 12, 3, 18, 9];
    let cutoff = 10;

    let mut under_cutoff = 0;
    let mut over_cutoff = 0;

    for &number in &sample {
        if number < cutoff {
            under_cutoff += 1;
        } else {
            over_cutoff += 1;
        }
    }

    println!("Under {}: {}", cutoff, under_cutoff);
    println!("Over {}: {}", cutoff, over_cutoff);
}
