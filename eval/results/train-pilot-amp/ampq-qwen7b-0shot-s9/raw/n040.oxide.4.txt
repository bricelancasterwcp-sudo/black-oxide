struct Square { side: Int }

struct Circle { radius: Float }

enum Shape {
    Square(Square),
    Circle(Circle),
}

fn area(shape: Shape) -> Float {
    match shape {
        ShapeSquare(side) => 3.0 * to_float(side) * side,
        ShapeCircle(radius) => 3.0 * radius * radius,
    }
}

fn main() {
    let s = Square { side: 4 }
    print(area(ShapeSquare(s)))
}
