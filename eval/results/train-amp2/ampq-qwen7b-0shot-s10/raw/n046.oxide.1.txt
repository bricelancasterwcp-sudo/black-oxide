fn main() {
    let samples = vec(5, 12, 3, 18, 9)
    let cutoff = 10
    let mut under = 0
    let mut over = 0
    for x in samples {
        if x < cutoff {
            under = push(under, 1)
        } else {
            over = push(over, 1)
        }
    }
    print(under)
    print_str(" ")
    print(over)
}
