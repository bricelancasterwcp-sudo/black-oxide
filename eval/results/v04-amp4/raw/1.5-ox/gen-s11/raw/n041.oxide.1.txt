struct Measurement { value: Int }

fn main() {
    let readings = push(push(push(vec(), 4), 7), 9);
    let mut sorted_readings = vec![];
    for reading in readings.iter().rev() {
        // Move out of the loop if we've already added this measurement
        if !sorted_readings.contains(&reading) {
            sorted_readings.push(reading);
        }
    }

    for reading in sorted_readings {
        print_str(int_to_str(reading.value));
    }
}
