fn is_prime(num: Int) -> Bool {
    if num <= 1 {
        return false
    }
    for i in 2.to_int() + 1.to_int() {
        if num % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let num = 91
    if is_prime(num) {
        print("Yes")
    } else {
        print("No")
    }
}
