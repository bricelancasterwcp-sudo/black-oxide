fn main() {
    let product = multiply(multiply(multiply(multiply(multiply(1, 2), 3), 4), 5), 6)
    print(product)
    drop product
}

fn multiply(a: Int, b: Int) -> Int {
    a * b
}

fn print_str(s: Str) {
    println(s)
}

fn println(x: Str) {
    print_str(x)
    print_str("\n")
}
