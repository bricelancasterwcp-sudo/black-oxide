```rust
use std::fmt::{self};

fn main() {
    let input = ["8", "oak", "15"].map(|s| s.to_string());

    for (i, item) in input.iter().enumerate() {
        if item.parse::<u32>().is_ok() {
            print!("{} is {}", item, if i == 0 { "the first" } else if i == 1 { "the second" } else { "the third" });
        } else {
            println!("bad");
        }
    }
}
```
