fn main() {
    let list = [7, 4, 9, 10];
    let mut even_total = 0;

    for &item in list.iter() {
        if item % 2 == 0 {
            even_total += item;
        } else {
            println!("{}", item * 2);
        }
    }

    println!("Total of even entries: {}", even_total);
}
