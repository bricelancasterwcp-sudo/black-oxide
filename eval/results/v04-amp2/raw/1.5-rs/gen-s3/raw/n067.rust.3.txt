```rust
fn main() {
    let mut list = vec![7, 4, 9, 10];
    let mut sum_of_evens = 0;

    for &num in &list {
        if num % 2 != 0 {
            println!("{}", num * 2);
        } else {
            sum_of_evens += num;
        }
    }

    println!("Sum of even entries: {}", sum_of_evens);
}
```
