fn push(v: Vec<Str>, x: Str) -> Vec<Str> {
    let mut vec = v
    vec.push(x)
    vec

fn str(s: &Vec<Str>) -> Str {
    let mut str = String.new()
    for x in s {
        str.push_str(x)
    }
    str
}

fn new() -> Vec<Str> {
    let mut vec = Vec.new()
    vec
}

fn length(s: &Vec<Str>) -> Int {
    let mut length = 0
    for _ in s {
        length = length + 1
    }
    length
}

fn get(v: &Vec<Str>, i: Int) -> Option<Str> {
    if i >= length(v) {
        None
    } else {
        Some(v.get(i).unwrap())
    }
}
