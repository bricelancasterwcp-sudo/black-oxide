fn sum_and_count(digits: &Vec<Str>) -> Int {
    let total = 0
    let count = 0
    for d in digits {
        if d.parse_int().is_some() {
            total = total + d.parse_int().unwrap()
            count = count + 1
        }
    }
    (total + count)
}

fn main() {
    let numbers = vec_str("a1b22")
    print(sum_and_count(&numbers))
}
