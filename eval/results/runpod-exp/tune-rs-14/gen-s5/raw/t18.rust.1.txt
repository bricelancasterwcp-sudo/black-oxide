fn main() {
    let shapes = [
        Shape::Rectangle(3, 4),
        Shape::Triangle(3, 4, 5),
        Shape::Rectangle(2, 5),
    ];
    for shape in shapes {
        println!("{}", shape.perimeter());
    }
}
enum Shape {
    Rectangle(u32, u32),
    Triangle(u32, u32, u32),
}
impl Shape {
    fn perimeter(&self) -> u32 {
        match self {
            Shape::Rectangle(a, b) => 2 * (a + b),
            Shape::Triangle(a, b, c) => a + b + c,
        }
    }
}
