fn main() {
    let readings = vec(4, 8, 15, 16, 23, 42)
    let total = sum(&readings)
    let count = readings.len()
    let mean = total / count
    print(mean)
    drop readings
}

fn sum(v: &Vec<Int>) -> Int {
    let mut total = 0
    for x in v {
        total = total + x
    }
    total
}
