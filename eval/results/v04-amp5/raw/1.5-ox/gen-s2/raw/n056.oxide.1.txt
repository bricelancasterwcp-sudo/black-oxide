```rust
fn main() {
    let nums: Vec<Option<Int>> = vec![Some(7), None, Some(x), None];
    let mut total: Int = 0;
    let mut non_numbers: usize = 0;

    for n in &nums {
        if let Some(num) = n {
            total += num;
        } else {
            non_numbers += 1;
        }
    }

    print!("Total sum of numbers: {}", total);
    println!("Number of non-numbers: {}", non_numbers);
}
```
