fn first_half(list: &Vec<Int>) -> Int {
    let mut sum = 0
    let mut i = 0
    for x in &list {
        if i < len(&list) / 2 {
            sum = sum + x
        }
        i = i + 1
    }
    sum
}

fn main() {
    let list = push(push(vec(), 3), 2)
    let list = push(list, 1)
    print(first_half(&list))
    drop list
}
