fn main() {
    let mut sum = 0;
    let mut fail_count = 0;

    for &s in ["12", "x", "30"].iter() {
        match s.parse::<i32>() {
            Ok(num) => {
                sum += num;
            }
            Err(_) => {
                fail_count += 1;
            }
        }
    }

    println!("Sum: {}", sum);
    println!("Failed count: {}", fail_count);
}
