```rust
fn main() {
    let text = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in text.chars() {
        if c.is_digit(10) {
            sum += c.to_digit(10).unwrap();
            count += 1;
        }
    }

    println!("Sum of digits: {}", sum);
    println!("Count of digits: {}", count);
}
```
