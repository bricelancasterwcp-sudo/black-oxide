```rust
fn main() {
    let numbers = vec![5, 12, 3, 18, 9];
    let cutoff = 10;

    let below_cutoff_count: Int = numbers.into_iter().filter(|&x| x < cutoff).count();
    let above_or_equal_cutoff_count: Int = numbers.into_iter().filter(|&x| x >= cutoff).count();

    print(below_cutoff_count);
    print_str("\n");
    print(above_or_equal_cutoff_count);
}
```
