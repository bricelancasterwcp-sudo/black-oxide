fn main() {
    let x = parse_int(text_input()?).unwrap_or(0)
    match x {
        Some(n) => print(n),
        None => print_str("bad"),
    }
    let s = parse_str(text_input()?).unwrap_or("")
    match s {
        Some(s) => print_str(s),
        None => print_str("bad"),
    }
    let y = parse_int(text_input()?).unwrap_or(0)
    match y {
        Some(n) => print(n),
        None => print_str("bad"),
    }
}
