```rust
fn main() {
    let mut sum = 0;
    let mut count = 0;

    for text in vec!["7", "x", "9", "?"] {
        match parse_int(text) {
            Some(n) => {
                sum += n;
                count += 1;
            }
            None => {}
        }
    }

    println!("Total: {}", sum);
    println!("Count: {}", count);
}
```
