```rust
fn main() {
    let sample = vec![5, 12, 3, 18, 9];
    let cutoff = 10;
    
    let mut count_below_cutoff = 0;
    let mut count_over_cutoff = 0;

    for &num in &sample {
        if num < cutoff {
            count_below_cutoff += 1;
        } else {
            count_over_cutoff += 1;
        }
    }

    println!("Numbers below the cutoff: {}", count_below_cutoff);
    println!("Numbers over or equal to the cutoff: {}", count_over_cutoff);
}
```
