fn main() {
    let samples = vec(5, 12, 3, 18, 9)
    let cutoff = 10
    let mut under = 0
    let mut over = 0
    for sample in samples {
        if sample < cutoff {
            under = under + 1
        } else {
            over = over + 1
        }
    }
    print_str("Under cutoff: ")
    print(under)
    print_str(" Over cutoff: ")
    print(over)
}
