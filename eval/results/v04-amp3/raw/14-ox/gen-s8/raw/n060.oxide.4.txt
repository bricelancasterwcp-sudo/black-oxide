fn main() {
    let input = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in chars(input) {
        if parse_int(c).is_some() {
            let digit = parse_int(c).unwrap();
            sum += digit;
            count += 1;
        }
    }

    print(sum);
    print(count);
}
