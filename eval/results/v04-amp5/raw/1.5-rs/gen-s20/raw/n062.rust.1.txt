fn main() {
    loop {
        match std::thread::sleep(std::time::Duration::from_secs(1)) {
            Ok(_) => match std::thread::sleep(std::time::Duration::from_secs(1)) {
                Ok(_) => {
                    println!("green");
                    std::thread::sleep(std::time::Duration::from_secs(1));
                    println!("yellow");
                }
                Err(_e) => break, // Error occurred while sleeping
            },
            Err(_e) => break, // Error occurred while sleeping
        }
    }
}
