fn main() {
    let s = "a1b22";
    let mut num_sum: u32 = 0;
    let mut digit_count: usize = 0;

    for c in s.chars() {
        if c.is_digit(10) {
            let d = c.to_digit(10).unwrap();
            num_sum += d;
            digit_count += 1;
        }
    }

    println!("Sum of digits: {}", num_sum);
    println!("Number of digits: {}", digit_count);
}
