fn reverse_digits(n: Int) -> Vec<Int> {
    let mut v = vec()
    let mut x = n
    while x > 0 {
        let digit = x % 10
        v = push(v, digit)
        x = trunc(x / 10)
    }
    v
}

fn main() {
    let number = 1234
    let reversed_digits = reverse_digits(number)
    for d in reversed_digits {
        print(d)
    }
}
