#![no_std]

use core::panic;

fn divide(x: i32, y: i32) -> i32 {
    if y == 0 {
        panic!("Cannot divide by zero");
    }
    x / y
}

fn main() {
    use core::fmt::Write;
    let mut stdout = unsafe { core::ptr::NonNull::new_unchecked(0x100000) };
    let _result = divide(20, 4);
    write!(stdout, "Result: {}", _result).unwrap();
    let _result = divide(20, 0); // This will panic
}
