struct PrimeChecker {
    number: Int,
}

fn main() {
    let mut pc = PrimeChecker { number: 91 }
    let mut is_prime = true
    while is_prime {
        let d = first_big(push(vec(), 2), pc.number)
        if d * d > pc.number {
            is_prime = false
        } else if pc.number % d == 0 {
            is_prime = false
        } else {
            pc.number = pc.number / d
        }
    }
    if pc.number == 1 {
        print_str("yes")
    } else {
        print_str("no")
    }
}

fn first_big(v: Vec<Int>, n: Int) -> Int {
    let found = -1
    for x in v {
        if x > n {
            found = x
            break
        }
    }
    found
}
