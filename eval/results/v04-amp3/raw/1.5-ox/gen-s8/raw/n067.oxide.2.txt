```rust
fn main() {
    let list = vec![7, 4, 9, 10];
    let mut total = 0;
    
    for &value in list.iter() {
        if value % 2 != 0 {
            print!("{} {}\n", value * 2, "\n");
        } else {
            total += value;
        }
    }
    
    println!("Total of even entries: {}", total);
}
```
