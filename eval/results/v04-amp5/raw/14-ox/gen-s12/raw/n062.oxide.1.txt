```rust
enum Signal {
    Green,
    Red,
    Yellow,
}

fn interpret(signal: Signal) -> Str {
    match signal {
        Green => "go".to_str(),
        Red => "stop".to_str(),
        Yellow => "wait".to_str(),
    }
}

fn main() {
    let signals = vec![Signal::Green, Signal::Red, Signal::Yellow];
    for s in signals {
        print(interpret(s));
    }
}
```
