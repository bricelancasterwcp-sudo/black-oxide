fn main() {
    let mut arr = vec(9, 2, 7, 4)
    let mut i = 0
    let mut j = 1
    while i < 4 {
        if arr(i) < arr(j) {
            i = i + 1
        } else {
            j = j + 1
        }
        if j == 4 {
            print(arr(i))
            drop arr
            i = i + 1
            j = i + 1
        }
    }
}
