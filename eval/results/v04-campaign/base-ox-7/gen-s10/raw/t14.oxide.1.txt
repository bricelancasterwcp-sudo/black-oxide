fn main() {
    let s = "mississippi"
    let count_s = s.chars().filter(|&c| c == 's').count()
    let unique_chars: Vec<Str> = s.chars().collect::<Vec<_>>().into_iter().collect::<std::collections::HashSet<_>>().iter().map(|&&c| c.to_string()).collect()
    print(count_s)
    print(unique_chars.len())
}
