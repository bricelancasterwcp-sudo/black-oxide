```rust
fn main() {
    let a1b22 = "a1b22";
    let mut sum_digits = 0;
    let mut count_digits = 0;

    for c in a1b22.chars() {
        if c.is_digit(10) {
            sum_digits += c.to_digit(10).unwrap();
            count_digits += 1;
        }
    }

    println!("Sum of digits: {}", sum_digits);
    println!("Count of digits: {}", count_digits);
}
```
