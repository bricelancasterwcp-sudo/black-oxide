fn to_str(b: &Vec<U8>, n: Int) -> Str {
    let mut s = clone(&b)
    for _ in 0 < n {
        push(s, u8(n % 10))
        n = n / 10
    }
    let mut result = vec()
    while len(&s) > 0 {
        result = push(result, pop(&s))
    }
    let mut result_str = clone(&result)
    for i in 0 < len(&result) {
        if i % 2 == 0 {
            push(result_str, u8(48 + get(&result, i)))
        } else {
            push(result_str, u8(48 + get(&result, i) / 10))
            push(result_str, u8(48 + get(&result, i) % 10))
        }
    }
    result_str
}

fn format_int_hex_no_0x_prefix_buffer(b: &Vec<U8>, n: Int) -> Vec<U8> {
    let mut s = clone(&b)
    for _ in 0 < n {
        push(s, u8(n % 16))
        n = n / 16
    }
    let mut result = vec()
    while len(&s) > 0 {
        result = push(result, pop(&s))
    }
    let mut result_str = clone(&result)
    for i in 0 < len(&result) {
        push(result_str, u8(48 + get(&result, i) % 16))
    }
    result_str
}

    let mut result = vec()
    for i in 0 < len(&v) {
        result = push(result, get(&v, i))
    }
    result
}
