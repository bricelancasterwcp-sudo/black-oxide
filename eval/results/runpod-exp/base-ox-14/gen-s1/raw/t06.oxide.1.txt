struct Divisors { num: Int, sum: Int }

fn find_divisors(n: Int) -> Divisors {
    let sum = 0
    for i in range(1, n) {
        if n % i == 0 {
            sum = sum + i
        }
    }
    Divisors { num: n, sum: sum }
}

fn is_perfect(d: Divisors) -> Bool {
    d.num == d.sum
}

fn main() {
    for n in range(1, 1000) {
        let d = find_divisors(n)
        if is_perfect(d) {
            print(d.num)
        }
    }
}
