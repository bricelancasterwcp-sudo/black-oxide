struct Reading { label: Str, values: Vec<Int> }

fn min(v: Vec<Int>) -> Int {
    let mut min = v.get(0).unwrap()
    for x in v {
        if x < min {
            min = x
        }
    }
    min
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 5), 3), 9), 1) }
    let r = Reading { label: "lab2", ..r }
    r.values = push(r.values, 7)
    print(min(r.values))
}
