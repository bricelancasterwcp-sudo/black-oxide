fn count_vowels(words: &Vec<Str>) -> Int {
    let mut count = 0
    for word in words {
        for c in chars(word) {
            if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" || c == "A" || c == "E" || c == "I" || c == "O" || c == "U" {
                count = count + 1
            }
        }
    }
    count
}

fn main() {
    let words = push(push(push(vec(), "hello"), "world"), "programming")
    let vowel_count = count_vowels(&words)
    print(vowel_count)
    drop words
}
