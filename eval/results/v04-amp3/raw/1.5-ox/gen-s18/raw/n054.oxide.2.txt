```rust
fn main() {
    let str = "abcdefghijklmnopqrstuvwxyz";
    let mut vowel_count = 0;
    let mut non_vowel_count = 0;

    for ch in str.chars() {
        if ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u' {
            vowel_count += 1;
        } else {
            non_vowel_count += 1;
        }
    }

    println!("Vowels: {}", vowel_count);
    println!("Non-vowels: {}", non_vowel_count);
}
```
