fn main() {
    let numbers = vec(360, 97, 1024, 561, 143, 2003)
    let mut primes_count = 0
    for n in numbers {
        print(n)
        let (divisors, sum, max) = prime_divisors(n)
        print(divisors)
        print(sum)
        print(max)
        if is_prime(n) {
            primes_count += 1
        }
    }
    print(primes_count)
}
fn prime_divisors(n: Int) -> Tuple<Int, Int, Option<Int>> {
    let mut sum = 0
    let mut max = None
    for i in range(2, n + 1) {
        if n % i == 0 && is_prime(i) {
            sum += i
            max = Some(i)
        }
    }
    (sum, max.unwrap_or(-1), max)
}
fn is_prime(n: Int) -> Bool {
    if n <= 1 { return false }
    for i in range(2, n + 1) {
        if n % i == 0 && i != n { return false }
    }
    true
}
