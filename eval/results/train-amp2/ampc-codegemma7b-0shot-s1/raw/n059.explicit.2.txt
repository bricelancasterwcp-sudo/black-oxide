fn len(v: &Vec<Str>) -> Int {
    let mut total = 0
    for x in &v {
        total = total + x.len()
    }
    total
}

fn chars(s: &Str) -> Vec<Str> {
    let mut v = vec()
    for c in s.chars() {
        v.push(str_len(c.to_string()))
    }
    v
}

    let text = push(push(vec(), "when"), "in")

    let text = push(text, "go")

    print(len(&text))
    let chars = chars(&text)
