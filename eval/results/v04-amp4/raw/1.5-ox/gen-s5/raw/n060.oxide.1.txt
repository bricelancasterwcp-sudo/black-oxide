```rust
fn main() {
    let text = "a1b22";
    let mut sum = 0;
    let mut digits_count = 0;

    for c in text.chars() {
        if c.is_digit(10) {
            sum += c.to_digit(10).unwrap();
            digits_count += 1;
        }
    }

    println!("Total of all digit characters: {}", sum);
    println!("Number of digit characters: {}", digits_count);
}
```
