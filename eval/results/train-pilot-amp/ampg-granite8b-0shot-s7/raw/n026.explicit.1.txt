fn length(s: Str) -> Int {
    let mut i = 0
    while get(s, i) != Option(None) {
        i = i + 1
    }
    i
}

fn to_str(n: Int) -> Str {
    if n == 0 {
        "zero"
    } else if n == 1 {
        "one"
    } else if n == 2 {
        "two"
    } else {
        concat(to_str(n - 1), " " + to_str(n - 2))
    }
}
