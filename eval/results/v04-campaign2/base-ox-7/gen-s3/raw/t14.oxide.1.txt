fn main() {
    let s = "mississippi"
    print(count(s.chars(), "s"))
    let chars: Vec<Str> = s.chars().collect()
    let distinct_chars: Vec<Str> = chars.into_iter().fold(vec![], |mut acc, c| {
        if !contains(acc, c) {
            push(acc, c)
        }
        acc
    })
    print(length(distinct_chars))
}
