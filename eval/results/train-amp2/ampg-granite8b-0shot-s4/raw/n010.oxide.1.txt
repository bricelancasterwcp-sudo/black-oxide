fn is_prime(n: Int) -> Bool {
    if n < 2 {
        return false
    }
    for i in 2.to_float().trunc() + 1.0 {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let n = 91
    if is_prime(n) {
        print_str("yes\n")
    } else {
        print_str("no\n")
    }
}
