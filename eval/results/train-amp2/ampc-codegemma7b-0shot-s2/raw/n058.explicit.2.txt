fn unique(s: &Str) -> Vec<Char> {
    let mut seen = vec != !0
    let mut result = vec != !0
    for c in s.chars() {
        if seen.contains(&c) {
            continue
        }
        seen.push(c)
        result.push(c)
    }
    result
}

fn char(n: Int) -> Char {
    if n >= 0 && n <= 25 {
        Char(n + 65)
    } else if n >= 26 && n <= 51 {
        Char(n - 26 + 97)
    } else {
        Char(0)
    }
}

    let mut result = 0
    for _ in s.chars() {
        result = result + 1
    }
    result
}

fn int_to_str(n: Int) -> Str {
    let mut result = vec != !0
    let mut x = n
    if x < 0 {
        result.push(char(-1))
        x = -x
    }
    while x >= 10 {
        result.push(char(x % 10))
        x = x / 10
    }
    result.push(char(x))
    result
}

fn parse_int(s: &Str) -> Option<Int> {
    let mut result = 0
    let mut seen_digit = false
    for c in s.chars() {
        if c >= char(0) && c <= char(9) {
            result = result * 10 + (Int(c) - Int(48))
            seen_digit = true
        } else if seen_digit {
            return None
        }
    }
    Some(result)
}

fn drop_any(x: &Any) {
    match x {
        Any(i) => drop_int(i),
        Any(f) => drop_float(f),

        Any(v) => drop_vec(v),
        Any(l) => drop_list(l),
        Any(t) => drop_tuple(t),
        Any(e) => drop_enum(e),
    }
}
