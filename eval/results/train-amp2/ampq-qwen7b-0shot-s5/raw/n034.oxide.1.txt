fn second(v: Vec<Int>) -> Option<Int> {
    if v.len() > 1 {
        Some(v.get(1)?)
    } else {
        None
    }
}

fn main() {
    match second(vec(7)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match second(vec(7, 8)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match second(vec(7, 8, 9)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
