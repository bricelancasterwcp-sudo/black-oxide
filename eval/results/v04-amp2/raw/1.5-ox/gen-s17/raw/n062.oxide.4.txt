```rust
fn main() {
    let mut state = "green";
    loop {
        match state {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => panic!("unexpected state: {}", state),
        }
        if state == "red" {
            break;
        }
        state = match state {
            "green" => "yellow",
            "yellow" => "red",
            _ => unreachable!(),
        };
    }
}
```
