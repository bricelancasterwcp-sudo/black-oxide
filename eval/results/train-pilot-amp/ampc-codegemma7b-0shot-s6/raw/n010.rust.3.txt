use std::io;

fn main() {
    let mut num = 91;

    let mut is_prime = true;

    for i in 2..(num / 2) + 1 { // Corrected loop range
        if num % i == 0 {
            is_prime = false;
            break;
        }
    }

    if is_prime {
        println!("Yes");
    } else {
        println!("No");
    }
}
