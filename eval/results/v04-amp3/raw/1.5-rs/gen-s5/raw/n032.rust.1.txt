fn main() {
    // Create a Point struct for the first point with coordinates (1, 5)
    let point1 = Point { x: 1, y: 5 };

    // Create a Point struct for the second point with coordinates (10, 5)
    let point2 = Point { x: 10, y: 5 };

    // Add the x and y coordinates of the two points
    let sum_of_coordinates = point1.x + point1.y + point2.x + point2.y;

    // Print the result of the addition
    println!("{}", sum_of_coordinates);
}

struct Point {
    x: isize,
    y: isize,
}
