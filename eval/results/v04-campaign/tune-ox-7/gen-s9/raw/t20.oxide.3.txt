struct Item { name: Str, unit_price: Int, quantity: Int }
fn main() {
    let items = vec(
        Item { name: "pen", unit_price: 3, quantity: 4 },
        Item { name: "pad", unit_price: 9, quantity: 2 },
        Item { name: "ink", unit_price: 5, quantity: 3 },
    )
    let mut total_cost = 0
    for item in items {
        let cost = item.unit_price * item.quantity
        print(cost)
        total_cost += cost
    }
    print(total_cost)
    let max_item = match items {
        [] => None,
        [first, ..] => Some(first),
    }
    for item in items {
        if let Some(max) = max_item {
            if item.unit_price * item.quantity > max.unit_price * max.quantity {
                max_item = Some(item)
            }
        }
    }
    print(max_item.unwrap().name)
}
