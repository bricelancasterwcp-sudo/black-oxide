fn print_even_backwards(n: Int) {
    let mut numbers = vec()
    for i in 0.to_int().unwrap() {
        if (2 * i + 1) <= n {
            numbers = push(numbers, 2 * i + 1)
        }
    }
    let mut idx = len(numbers) - 1
    while idx >= 0 {
        print(numbers.get(idx).unwrap())
        idx = idx - 1
    }
    drop numbers
}

fn main() {
    print_even_backwards(10)
}
