fn main() {
    let numbers = vec![10, 20];

    if let Some(&number) = numbers.get(0) {
        println!("Position one: {}", number);
    } else {
        println!("Position one: none");
    }

    if let Some(&number) = numbers.get(4) {
        println!("Position four: {}", number);
    } else {
        println!("Position four: none");
    }
}
