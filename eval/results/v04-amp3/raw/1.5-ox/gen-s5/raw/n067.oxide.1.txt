```rust
use std::io;

struct List { elements: Vec<Int> }

impl List {
    fn new(elements: Vec<Int>) -> List {
        List { elements }
    }

    fn walk(self) {
        let mut total = 0;
        for &x in self.elements.iter() {
            if x % 2 != 0 {
                println!("{}", x * 2);
            } else {
                total += x;
            }
        }
        println!("Total of even entries: {}", total);
    }
}

fn main() {
    let list = List::new(vec![7, 4, 9, 10]);
    list.walk();
}
```
