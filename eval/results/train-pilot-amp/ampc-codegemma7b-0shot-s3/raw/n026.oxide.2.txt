fn int_to_str(n: Int) -> Str {
    if n < 10 {
        match n {
            _ => "0",
            _ => "1",
            _ => "2",
            _ => "3",
            _ => "4",
            _ => "5",
            _ => "6",
            _ => "7",
            _ => "8",
            _ => "9",
        }
    } else if n < 100 {
        match (n / 10) {
            _ => "10",
            _ => "20",
            _ => "30",
            _ => "40",
            _ => "50",
            _ => "60",
            _ => "70",
            _ => "80",
            _ => "90",
            _ => "?",
        }
    } else if n < 1000 {
        match (n / 100) {
            _ => "100",
            _ => "200",
            _ => "300",
            _ => "400",
            _ => "500",
            _ => "600",
            _ => "700",
            _ => "800",
            _ => "900",
            _ => "?",
        }
    } else {
        "many"
    }
}

fn main() {
    let number = 2026
    let number_as_text = int_to_str(number)
    let number_length = str_len(number_as_text)
    print(number_length)
}
