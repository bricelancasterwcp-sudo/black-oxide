struct Fruit { name: Str, length: Int }

fn longest_name(fruits: Vec<Fruit>) -> Fruit {
    let mut longest = fruits.get(0).unwrap_or(Fruit { name: "", length: 0 })
    for fruit in fruits {
        if fruit.length > longest.length {
            longest = fruit
        }
    }
    longest
}

    let fruits = vec(
        Fruit { name: "fig", length: str_len("fig") },
        Fruit { name: "banana", length: str_len("banana") },
        Fruit { name: "kiwi", length: str_len("kiwi") }
    )
    let longest = longest_name(fruits)
    print(longest.name)
    print(longest.length)
}
