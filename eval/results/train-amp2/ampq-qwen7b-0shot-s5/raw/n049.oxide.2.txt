fn main() {
    let figures = vec(8, 3, 11, 6).sort().clone().unwrap()
    print(figures.first().unwrap())
    print_str(" ")
    print(trunc(sub(figures.last().unwrap(), figures.first().unwrap())))
}

    a - b
