```rust
fn main() {
    let mut list = vec![7, 4, 9, 10];
    let mut even_total: Int = 0;

    for &value in list.iter_mut() {
        if value % 2 != 0 {
            println!("{}", value * 2);
        } else {
            even_total += value;
        }
    }

    println!("Even total: {}", even_total);
}
```
