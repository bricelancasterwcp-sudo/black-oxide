fn main() {
    let readings = [6, 11, 3, 14, 9];
    let count_above_five = readings.iter().filter(|&&x| x > 5).count();
    println!("{} reading(s) exceed five", count_above_five);

    let highest_reading = readings.iter().max().unwrap_or(&0);
    println!("The highest reading is {}", highest_reading);
}
