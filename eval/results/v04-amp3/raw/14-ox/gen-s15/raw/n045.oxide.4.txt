fn main() {
    let log = vec();
    log = push(log, 6);
    log = push(log, 7);
    log = push(log, 8);
    log = push(log, 9);
    
    let first_entry = get(log, 0)?;
    print(first_entry);
    
    let last_index = len(log) - 1;
    let last_entry = get(log, last_index)?;
    print(last_entry);
}
