```rust
enum Shape {
    Square { side: f64 },
    Circle { radius: f64 },
}

impl Shape {
    fn area(&self) -> f64 {
        match self {
            Shape::Square { side } => side * side,
            Shape::Circle { radius } => 3.0 * radius * radius,
        }
    }
}

fn main() {
    let square = Shape::Square { side: 3.0 };
    println!("The area of the square is {}", square.area());
}
```
