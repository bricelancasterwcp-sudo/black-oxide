fn change_area(mutrectangle: &Vec<Int>) -> Vec<Int> {
    mutrectangle = push(mutrectangle, 5)
    mutrectangle
}

fn area(rectangle: &Vec<Int>) -> Int {
    return (rectangle.get(0).unwrap() * rectangle.get(1).unwrap())
}

fn main() {
    let mut rectangle = vec(2, 3)
    rectangle = change_area(&mutrectangle)
    print(area(&rectangle))
    drop rectangle
}
