```rust
fn main() {
    let s = "keyboard";
    let mut count_vowels = 0;
    let mut count_not_vowels = 0;

    for c in s.chars() {
        match c.to_ascii_lowercase() {
            'a' | 'e' | 'i' | 'o' | 'u' => count_vowels += 1,
            _ => count_not_vowels += 1,
        }
    }

    println!("Vowel count: {}", count_vowels);
    println!("Consonant count: {}", count_not_vowels);
}
```
