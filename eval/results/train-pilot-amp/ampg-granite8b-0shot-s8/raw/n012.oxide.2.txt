struct Counter { value: Int }

fn make_counter(start: Int) -> Counter {
    Counter(start)
}

fn main() {
    let mut counter = make_counter(1)
    let mut items = vec()
    for i in range(0, 15) {
        print(counter.value)
        items = push(items, counter.value)
        counter = Counter(counter.value + 1)
    }
    print(len(items))
}
