fn main() {
    let figures = vec![(7, 4), (5, 0), (3, 0), (2, 9)];
    let mut total_area = 0.0;
    for figure in &figures {
        match figure {
            (w, 0) if *w > 0 => {
                let area = w as f64 * w as f64;
                println!("rect {}", area);
                total_area += area;
            }
            (s, 0) if *s > 0 => {
                let area = s as f64 * s as f64;
                println!("square {}", area);
                total_area += area;
            }
            (r, 0) if *r > 0 => {
                let area = 3.0 * r as f64 * r as f64;
                println!("circle {}", area);
                total_area += area;
            }
            _ => {}
        }
    }
    println!("{}", total_area);
    let mut max_area = 0.0;
    let mut max_type = "";
    for figure in &figures {
        match figure {
            (w, 0) if *w > 0 => {
                let area = w as f64 * w as f64;
                if area > max_area {
                    max_area = area;
                    max_type = "rect";
                }
            }
            (s, 0) if *s > 0 => {
                let area = s as f64 * s as f64;
                if area > max_area {
                    max_area = area;
                    max_type = "square";
                }
            }
            (r, 0) if *r > 0 => {
                let area = 3.0 * r as f64 * r as f64;
                if area > max_area {
                    max_area = area;
                    max_type = "circle";
                }
            }
            _ => {}
        }
    }
    println!("{}", max_type);
}
