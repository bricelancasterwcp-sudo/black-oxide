struct Square { side: Int }
struct Circle { radius: Float }

fn area(s: Shape) -> Float {
    match s {
        Shape::Square(Square { side }) => 3 * to_float(side) * to_float(side),
        Shape::Circle(Circle { radius }) => 3 * radius * radius,
    }
}

enum Shape { Square(Square), Circle(Circle) }

fn main() {
    let square = Square { side: 4 };
    print(area(Shape::Square(square)))
}
