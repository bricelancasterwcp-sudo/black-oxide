}

fn first_small(mutv: Vec<Int>) -> Int {
    let mut least = v.first().unwrap()
    for x in v {
        if x < least {
            least = x
        }
    }
    least
}

fn print(x: Int) {
    let s = int_to_str(x)
    for c in chars(s) {
        print_str(c)

fn print_str(s: Str) {
    for c in chars(s) {
        putchar(c)
    }
}

fn putchar(c: Str) {
    let mut buffer = Vec { len: 1 }
    buffer.push(c)
    print_char(buffer)
}

fn int_to_str(n: Int) -> Str {
    let mut result = Vec { len: 0 }
    let mut x = n
    while x > 0 {
        let rem = x % 10
        let digit = int_to_char(rem)
        result.push(digit)
        x = x / 10
    }
    if result.len() == 0 {
        result.push(int_to_char(0))
    }
    result.reverse()
    str_from_chars(result)
}

fn int_to_char(n: Int) -> Str {
    if n < 10 {
        return int_to_str(n)
    } else {
        return char_from_int(n + 48)
    }
}

fn char_from_int(n: Int) -> Str {
    let mut buffer = Vec { len: 1 }
    buffer.push(char(n))
    str_from_chars(buffer)
}

fn str_from_chars(v: Vec<Str>) -> Str {
    let mut result = Str { len: v.len() }
    let mut i = 0
    for c in v {
        result.chars.push(c.chars.get(0).unwrap())
    }
    result
}

fn str_len(s: Str) -> Int {
    s.chars.len()
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = Vec { len: s.chars.len() }
    for c in s.chars {
        result.push(Str { len: 1, chars: vec(c) })
    }
    result
}

fn print_mem(position: Int) {
    print_char(read_mem(position))
}

fn read_mem(position: Int) -> Str {
    let mut buffer = Vec { len: 1 }
    read_mem_bytes(position, buffer)
    buffer.get(0).unwrap()
}

fn read_mem_bytes(position: Int, v: Vec<Str>) {
    for i in 0.0.to(v.len() - 1) {
        read_mem_byte(position + i, v.get(i).unwrap())
    }
}

fn read_mem_byte(position: Int, v: Str) {
    let mut buffer = Vec { len: 1 }
    read_mem_byte_bytes(position, buffer)
    if buffer.get(0).unwrap() != v {
        print_mem(position)
        print_str(" ")
        print_str(v)
        print_str("\n")
    }
}
