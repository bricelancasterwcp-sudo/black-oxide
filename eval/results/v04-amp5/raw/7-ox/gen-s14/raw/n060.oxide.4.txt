```rust
fn main() {
    let s = "a1b22";
    let mut sum = 0;
    let mut count = 0;
    
    for c in s.chars() {
        match c.to_digit(10) {
            Some(digit) => {
                sum += digit as usize;
                count += 1;
            },
            None => {}
        }
    }
    
    print(sum);
    print_str("\n");
    print(count);
}
```
