fn count_digits(s: Str) -> Unit {
    let mut sum = 0
    let mut count = 0
    for ch in chars(s) {
        match parse_int(ch) {
            Some(n) => {
                sum += n
                count += 1
            }
            None => {}
        }
    }
    print(sum)
    print(count)
}

fn main() -> Unit {
    let input = "a1b22"
    count_digits(input)
}
