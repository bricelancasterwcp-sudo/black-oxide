fn read_int(text: &Str) -> Option<Int> {
    let mut chars = chars(text)
    let neg = if chars.len() > 0 && chars.get(0) == Some(&chr_sub) { 1 } else { 0 }
    let mut res = 0
    for c in chars {
        if c < chr_0 || c > chr_9 {
            return None
        }
        res = res * 10 + (c - chr_0)
    }
    if neg == 1 {
        Some(-res)
    } else {
        Some(res)
    }
}

fn add_one(x: Int) -> Int {
    x + 1
}

fn print_int(x: Int) {
    print(int_to_str(x))
}

        },
    }
}
