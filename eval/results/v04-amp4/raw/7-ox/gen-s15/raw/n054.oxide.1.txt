fn main() {
    let vowels = "aeiou";
    let mut count_vowels = 0;
    let mut count_consonants = 0;

    for c in concat("abcdfghjklmnpqrstvwxyz", "AEIOU") {
        if contains(vowels, c) {
            count_vowels += 1;
        } else {
            count_consonants += 1;
        }
    }

    print(count_vowels);
    print_str(" ");
    print(count_consonants);
}
