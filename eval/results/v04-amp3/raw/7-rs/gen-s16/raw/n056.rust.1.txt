```rust
fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut sum = 0;
    let mut count = 0;

    for text in texts {
        if let Ok(num) = text.parse::<i32>() {
            sum += num;
        } else {
            count += 1;
        }
    }

    println!("Total of the numbers: {}", sum);
    println!("Count of non-numbers: {}", count);
}
```
