fn sum_of_digits(n: Int) -> Int {
    let mut sum = 0
    let s = int_to_str(n)
    for ch in chars(s) {
        let digit = parse_int(ch)?
        sum += digit
    }
    Some(sum)
}

fn main() {
    let number = 9875
    match sum_of_digits(number) {
        Some(sum) => print(sum),
        None => print_str("Error parsing digits"),
    }
}
