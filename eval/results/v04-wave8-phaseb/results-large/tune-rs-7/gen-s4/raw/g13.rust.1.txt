fn main() {
    let mut c = 9;
    let mut r = 0;
    let mut h = 0;
    let mut a = 0;
    let mut t = 0;
    let mut l = 0;
    let mut n = 1;
    let mut s = 0;
    let mut b = 0;
    while r < 30 {
        if c % 2 == 0 {
            c /= 2;
            h += 1;
        } else if c % 3 == 0 {
            c += 5;
            a += 1;
        } else {
            c = (c * 3 + 1) % 1000;
            t += 1;
        }
        s += c;
        if c > l {
            l = c;
            n = r + 1;
        }
        if c < b || b == 0 {
            b = c;
        }
        if c < 10 {
            b += 1;
        }
        r += 1;
    }
    println!("{}", c);
    println!("{}", h);
    println!("{}", a);
    println!("{}", t);
    println!("{}", l);
    println!("{}", n);
    println!("{}", b);
    println!("{}", s);
    println!("{}", if c > 100 { "high" } else { "low" });
}
