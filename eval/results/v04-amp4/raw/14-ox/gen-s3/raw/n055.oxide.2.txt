```rust
fn parse_int(s: Str) -> Option<Int> {
    match s {
        "8" => Some(8),
        "15" => Some(15),
        _ => None,
    }
}

fn main() {
    let entries = vec("8", "oak", "15");

    for entry in entries {
        match parse_int(entry) {
            Some(n) => print(n),
            None => print_str("bad"),
        }
    }
}
```
