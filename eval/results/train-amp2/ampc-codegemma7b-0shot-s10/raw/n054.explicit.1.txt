fn is_vowel(c: &Char) -> Bool {
    match c {
        Char(a) => {
            if a == b || a == e || a == i || a == o || a == u {
                return true
            } else {
                return false
            }
        },
        _ => false,
    }
}

fn main() {
    let mut vowels = 0
    let mut not_vowels = 0
    for c in "keyboard" {
        if is_vowel(&c) {
            vowels = vowels + 1
        } else {
            not_vowels = not_vowels + 1
        }
    }
    print(vowels)
    print(not_vowels)
}
