struct Queue { values: Vec<Int> }

fn swap_first_last(q: Queue) -> Queue {
    let len = len(q.values);
    if len > 0 {
        let first = get(q.values, 0)?;
        let last = get(q.values, len - 1)?;
        let mut new_values = q.values;
        new_values[0] = last;
        new_values[len - 1] = first;
        Queue { values: new_values }
    } else {
        q
    }
}

fn main() {
    let q = Queue { values: push(push(push(push(vec(), 1), 2), 3), 4, 5) };
    let q_swapped = swap_first_last(q);
    for x in q_swapped.values {
        print(x);
        print_str("\n");
    }
}
