fn main() {
    let s = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in chars(s) {
        if let Some(_) = parse_int(c) {
            match parse_int(c) {
                Some(n) => {
                    sum += n;
                    count += 1;
                },
                None => ()
            }
        }
    }

    print(sum);
    print(count);
}
