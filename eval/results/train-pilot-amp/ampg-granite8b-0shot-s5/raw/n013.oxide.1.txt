fn main() {
    orphan(readings_vec(vec(5, 3, 9, 1, 7)))
}

fn readings_vec(r: Vec<Int>) -> Vec<Reading> {
    let mut results = vec()
    for i in 0 < r.len() {
        results = push(results, Reading { label: int_to_str(i), values: vec(r.get(i)?) })
    }
    results
}

fn orphan(r: Vec<Reading>) -> Option<Int> {
    let mut largest: Option<Int> = None
    for reading in r {
        let largest_reading_value = first_big(reading.values)
        if largest.is_none() || largest_reading_value > largest.get()? {
            largest = Some(largest_reading_value)
        }
    }
    largest
}

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}
