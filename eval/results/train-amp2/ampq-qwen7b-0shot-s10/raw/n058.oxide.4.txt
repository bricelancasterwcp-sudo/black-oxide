fn main() {
    let word = "coffee"
    let mut seen = Vec()
    for c in chars(word) {
        if contains(seen, c) {
            continue
        }
        if contains(word, c, seen.len()) {
            print_str(c)
            push(seen, c)
        }
    }
}

    contains(s, c, 0)

fn contains(s: Str, c: Str, start: Int) -> Bool {
    for i in range(start, str_len(s)) {
        if get(s, i) == Some(c) {
            return True
        }
    }
    False
}
