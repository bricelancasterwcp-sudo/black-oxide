fn main() {
    let v = vec(4, 4, 4, 7, 7, 2, 9, 9, 9, 9, 1, 1, 5)
    print(len(v))
    let longest_streak_value = -1
    let mut longest_streak_length = 0
    let mut current_streak_value = v[0]
    let mut current_streak_length = 1
    for i in range(1, len(v)) {
        if v[i] == current_streak_value {
            current_streak_length += 1
        } else {
            if current_streak_length > longest_streak_length {
                longest_streak_length = current_streak_length
                longest_streak_value = current_streak_value
            }
            current_streak_value = v[i]
            current_streak_length = 1
        }
    }
    print(longest_streak_length)
    print(longest_streak_value)
    let unique_values = count_unique(v)
    print(unique_values)
    let first_repetition_index = find_first_repetition(v)
    print(first_repetition_index)
    let never_decreases = check_never_decreases(v)
    print(never_decreases)
}
fn count_unique(v: Vec<Int>) -> Int {
    let mut seen = vec()
    for x in v {
        if !contains(seen, x) {
            push(seen, x)
        }
    }
    len(seen)
}
fn find_first_repetition(v: Vec<Int>) -> Option<Int> {
    let mut seen = vec()
    for i in range(0, len(v)) {
        if contains(seen, v[i]) {
            return Some(i)
        } else {
            push(seen, v[i])
        }
    }
    None
}
fn check_never_decreases(v: Vec<Int>) -> Bool {
    for i in range(1, len(v)) {
        if v[i] < v[i - 1] {
            return false
        }
    }
    true
}
