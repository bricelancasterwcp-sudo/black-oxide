struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max.is_none() || x > max.unwrap_or(0) {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = push(push(push(push(push(vec(), 6), 11), 3), 14), 9)
    let count = count_exceeding(readings.clone(), 5)
    print(count)
    print_str("\n")
    match find_max(readings) {
        Some(n) => print(n),
        None => print_str("no readings"),
    }
}
