fn main(){
    let mut input = String::new();
    std::io::stdin().read_line(&mut input).unwrap();
    let input: Vec<String> = input.trim().split(' ').map(|s| s.to_string()).collect();

    for x in &input {
        match x.parse::<i32>() {
            Ok(num) => println!("{}", num),
            Err(_) => println!("bad")
        }
    }
}
