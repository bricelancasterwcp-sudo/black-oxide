```rust
fn main() {
    let mut max_power_of_two = 0;
    for i in 0..32 { // 32 is the maximum number of bits in a u32
        if (1 << i) <= 100 {
            max_power_of_two = 1 << i;
        } else {
            break;
        }
    }
    println!("The biggest power of two that does not exceed 100 is: {}", max_power_of_two);
}
```
