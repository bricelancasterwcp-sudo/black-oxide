fn main() {
    let samples = [5, 12, 3, 18, 9];
    let cutoff = 10;
    let (under, over_or_equal) = samples.iter().fold((0, 0), |(under, over_or_equal), &sample| {
        if sample < cutoff {
            (under + 1, over_or_equal)
        } else {
            (under, over_or_equal + 1)
        }
    });
    println!("Under cutoff: {}", under);
    println!("Over or equal to cutoff: {}", over_or_equal);
}
