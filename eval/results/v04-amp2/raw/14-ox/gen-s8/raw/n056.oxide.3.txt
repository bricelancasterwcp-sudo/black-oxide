struct Number { value: Int }

fn parse_number(s: Str) -> Option<Number> {
    match parse_int(s) {
        Some(n) => Some(Number { value: n }),
        None => None,
    }
}

fn sum_valid_numbers(numbers: Vec<Str>) -> (Int, Int) {
    let mut sum = 0
    let mut invalid_count = 0
    for s in numbers {
        match parse_number(s) {
            Some(n) => sum += n.value,
            None => invalid_count += 1,
        }
    }
    (sum, invalid_count)
}

fn main() {
    let inputs = vec("7", "x", "9", "?")
    let (total, count) = sum_valid_numbers(inputs)
    print(total)
    print(count)
}
