fn divide(n: Int, d: Int) -> Option<Int> {
    if d == 0 {
        None
    } else {
        Some(n / d)
    }
}
fn main() {
    let result = divide(100, 5)
    if result.is_some() {
        print(result.unwrap())
    } else {
        print_str("undefined")
    }
    let result = divide(100, 0)
    if result.is_some() {
        print(result.unwrap())
    } else {
        print_str("undefined")
    }
    let result = divide(100, 4)
    if result.is_some() {
        print(result.unwrap())
    } else {
        print_str("undefined")
    }
}
