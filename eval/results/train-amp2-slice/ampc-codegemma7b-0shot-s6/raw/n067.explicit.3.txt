fn walk(nums: &Vec<Int>) -> Int {
    let mut total = 0
    for num in nums {
        if num % 2 != 0 {
            print(num * 2)
        }
        if num % 2 == 0 {
            total = total + num
        }
    }
    total
}

fn main() {
    let nums = vec(7, 4, 9, 10)
    print(walk(&nums))
    drop nums
}
