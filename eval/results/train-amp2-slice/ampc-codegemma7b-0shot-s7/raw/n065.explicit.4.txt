fn main() {
    let readings = vec(6, 11, 3, 14, 9)
    let mut count = 0
    let mut max = 0
    for reading in &readings {
        if reading > &mut && 5 {
            count = count + 1
            if reading > &mut && max {
                max = reading
            }
        }
    }
    print(count)
    print(max)
}
