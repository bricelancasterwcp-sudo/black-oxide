```rust
use std::f64;

enum Shape {
    Square(f64),
    Circle(f64),
}

impl Shape {
    fn area(&self) -> f64 {
        match self {
            Shape::Square(side) => side * side,
            Shape::Circle(radius) => radius * radius * 3.0, // Use 3.0 instead of PI
        }
    }
}

fn main() {
    let square = Shape::Square(4.0);
    println!("Area of the square: {}", square.area());
}
```
