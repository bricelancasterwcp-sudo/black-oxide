fn main() {
    let queue = vec(1, 2, 3, 4, 5)
    let len = len(queue)
    
    if len > 1 {
        let first = get(queue, 0)?
        let last = get(queue, len - 1)?

        let queue = set(set(queue, 0, last), len - 1, first)

        for x in queue {
            print(x)
        }
    } else {
        print_str("Queue too short")
    }
}
