fn main() {
    let (numerator, denominator) = (20, 4);
    
    match divide(numerator, denominator) {
        Ok(result) => println!("Result: {}", result),
        Err(e) => println!("Bad: {}", e),
    }
}

fn divide(x: i32, y: i32) -> Result<f64, &'static str> {
    if y == 0 {
        Err("Cannot divide by zero")
    } else {
        Ok((x as f64 / y as f64))
    }
}
