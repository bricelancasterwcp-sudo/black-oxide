fn main() {
    let mut scores = vec(2, 9, 5)
    scores.sort_by_int_desc()
    for score in scores {
        print_str(score.to_string())
    }
}

fn sort_by_int_desc(v: Vec<Int>) -> Vec<Int> {
    let mut sorted = v.clone()
    sorted.sort_by_key_int_desc()
    sorted
}

fn sort_by_key_int_desc(v: Vec<Int>) {
    for i in 0 {
        for j in (i + 1) {
            if v.get(i)? > v.get(j)? {
                let temp = v.get(i)?
                v.set(i, v.get(j)?)
                v.set(j, temp)
            }
        }
    }
}
