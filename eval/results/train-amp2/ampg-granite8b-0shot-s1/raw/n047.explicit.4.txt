fn collect_even(limit: Int) -> Vec<Int> {
    let result = vec()
    let mut i = 1
    while i <= limit {
        if i % 2 == 0 {
            result = push(result, i)
        }
        i = i + 1
    }
    result
}

fn print_in_reverse(v: Vec<Int>) {
    let mut i = len(&v)
    while i > 0 {
        i = i - 1
        print(&get && v, i)
    }
}

fn main() {
    let evens = collect_even(10)
    print_in_reverse(evens)
    drop evens
}
