fn main() {
    let readings = vec(5, 3, 9, 1, 7)
    let mut smallest = readings.get(0).unwrap()
    for reading in &readings {
        if reading < smallest {
            smallest = reading
        }
    }
    print(smallest)
    drop readings
}
