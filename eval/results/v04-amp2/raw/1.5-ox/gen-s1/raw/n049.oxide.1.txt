```rust
use std::cmp;

struct Figure {
    name: Str,
    value: Int,
}

fn main() {
    let figures = vec![
        Figure { name: "circle".to_string(), value: 3.14 },
        Figure { name: "triangle".to_string(), value: 5 },
        Figure { name: "square".to_string(), value: 4 },
        Figure { name: "rectangle".to_string(), value: 6 },
    ];

    let smallest = figures.iter().min_by(|a, b| a.value.cmp(&b.value)).unwrap();
    println!("Smallest figure: {}", smallest.name);

    let largest = figures.iter().max_by(|a, b| a.value.cmp(&b.value)).unwrap();
    println!("Largest figure: {}", largest.name);

    let gap = largest.value - smallest.value;
    println!("Gap between the largest and the smallest: {}", gap);
}
```
