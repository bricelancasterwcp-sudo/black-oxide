fn main() {
    let samples = vec![5, 12, 3, 18, 9];
    let cutoff = 10;
    let (under, over_or_equal) = samples.iter().fold((0, 0), |(u, o), &x| if x < cutoff { (u + 1, o) } else { (u, o + 1) });
    println!("Under cutoff: {}", under);
    println!("Over or equal to cutoff: {}", over_or_equal);
}
