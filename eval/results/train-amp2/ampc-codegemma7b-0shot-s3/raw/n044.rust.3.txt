use std::collections::HashMap;

fn main() {
    let mut accounts: HashMap<u32, u32> = HashMap::new();

    accounts.insert(3, 0);
    accounts.insert(1, 0);
    accounts.insert(4, 0);
    accounts.insert(1, 0);
    accounts.insert(5, 0);

    for (account, deposit) in accounts.iter_mut() {
        *deposit += account;
        println!("{}", accounts.get(account).unwrap());
    }
}
