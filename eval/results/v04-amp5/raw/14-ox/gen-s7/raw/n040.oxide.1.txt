struct Square { side: Int }
struct Circle { radius: Float }

const PI: Float = 3.0

fn area(shape: Square) -> Int {
    shape.side * shape.side
}

fn area(shape: Circle) -> Float {
    PI * shape.radius * shape.radius
}

fn main() {
    let square = Square { side: 4 }
    print(area(square))
}
