fn main() {
    let fig = "fig";
    let banana = "banana";
    let kiwi = "kiwi";

    let names = vec![fig, banana, kiwi];
    let mut max_name = "";
    let mut max_length = 0;

    for name in names {
        if name.len() > max_length {
            max_name = name;
            max_length = name.len();
        }
    }

    println!("{} has the most letters with a count of {}", max_name, max_length);
}
