fn is_power_of_two(x: Int) -> Bool {
    if x <= 0 {
        return false
    }
    (x && (x - 1)) == 0
}

fn main() {
    let mut i = 1
    loop     // loops until it finds the biggest power 
    if i * 2 > 100 {
        breaki
    }
    i = i * 2
    if is_power_of_two(i) {
        print(i)
        breaki
    }
}
