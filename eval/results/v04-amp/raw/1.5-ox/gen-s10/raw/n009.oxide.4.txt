```rust
fn reverse_digits(n: Int) -> Vec<Int> {
    let mut result = vec![];
    while n > 0 {
        result.push(n % 10);
        n /= 10;
    }
    result.reverse();
    result
}

fn main() {
    let num = 1234;
    let reversed_digits = reverse_digits(num);
    for digit in reversed_digits {
        print!("{}", digit);
    }
}
```
