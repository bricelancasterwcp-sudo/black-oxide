fn count_vowels(s: &Str) -> Int {
    let vowels = "aeiouAEIOU".to_string().chars().collect_vec()
    let mut count = 0
    for c in s.chars() {
        if vowels.contains(&c) {
            count = count + 1
        }
    }
    count
}

fn main() {
    let s = "hello world".to_string()
    print(count_vowels(&s))
    drop s
}
