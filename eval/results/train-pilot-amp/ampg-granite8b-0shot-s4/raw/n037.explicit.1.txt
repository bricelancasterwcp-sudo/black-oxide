    let rect = Rectangle { width: 2, height: 3 }
    let rect = rect.with_width(5)
    print(rect.area())
}

struct Rectangle {
    width: Int,
    height: Int,
}

fn with_width(self: &Rectangle, new_width: Int) -> Rectangle {
    Rectangle(width == new_width, height == self.height)
}

fn area(self: &Rectangle) -> Int {
    self.width * self.height
}
