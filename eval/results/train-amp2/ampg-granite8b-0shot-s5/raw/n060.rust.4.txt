fn main() {
    let mut total = 0;
    let mut count = 0;
    for c in "a1b22".chars() {
        match c {
            c @ '0'..='9' => {
                total += c.to_digit(10).unwrap() as usize;
                count += 1;
            }
            _ => continue,
        }
    }
    println!("total = {}", total);
    println!("count = {}", count);
}
