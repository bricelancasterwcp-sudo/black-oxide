fn main() {
    let nums = vec(3, 8, -2, 12, 7)
    let mut pos_count = 0
    for n in nums {
        if n > 0 {
            pos_count = pos_count + 1
        }
    }
    print(pos_count)
    let mut max_val = nums[0]
    for n in nums {
        if n > max_val {
            max_val = n
        }
    }
    print(max_val)
}
