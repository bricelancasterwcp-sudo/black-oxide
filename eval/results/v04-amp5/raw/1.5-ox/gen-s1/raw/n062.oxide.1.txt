fn main() {
    let signals = vec![Str::from("green"), Str::from("red"), Str::from("yellow")];
    for signal in signals {
        match signal.as_str() {
            "green" => print_str("go for green"),
            "red" => print_str("stop for red"),
            "yellow" => print_str("wait for yellow"),
            _ => print_str("invalid signal"),
        }
    }
}
