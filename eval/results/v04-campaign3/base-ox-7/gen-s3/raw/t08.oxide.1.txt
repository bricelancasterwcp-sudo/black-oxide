fn count_positive(v: Vec<Int>) -> Int {
    let mut count = 0;
    for x in v {
        if x > 0 {
            count += 1;
        }
    }
    count
}

fn max_value(v: Vec<Int>) -> Option<Int> {
    let mut max = None;
    for x in v {
        match max {
            Some(m) => if x > m { max = Some(x); },
            None => max = Some(x),
        }
    }
    max
}

fn main() {
    let values = push(push(push(push(vec(), 3), 8), -2), 12);
    values.push(7);

    let positive_count = count_positive(values.clone());
    print(positive_count);

    match max_value(values) {
        Some(m) => print(m),
        None => print_str("empty"),
    }
}
