fn main() {
    let mut r = Rectangle { width: 2, height: 3 }
    r.change_width(5)
    print(r.area())
}

struct Rectangle {
    width: Int,
    height: Int,
}

fn change_width(mutself: Rectangle, new_width: Int) -> Rectangle {
    mutself.width = new_width
    mutself
}

fn area(self: Rectangle) -> Int {
    self.width * self.height
}

fn main() {
    let mut r = Rectangle { width: 2, height: 3 }
    r.change_width(5)
    print(r.area())
}
