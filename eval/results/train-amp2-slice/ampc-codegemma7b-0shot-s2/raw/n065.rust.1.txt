fn main() {
    let readings = [6, 11, 3, 14, 9];
    let mut count = 0;
    let mut max = 0;

    for reading in readings {
        if reading > 5 {
            count += 1;
        }
        if reading > max {
            max = reading;
        }
    }

    println!("{} readings exceed five", count);
    println!("The highest reading is {}", max);
}
