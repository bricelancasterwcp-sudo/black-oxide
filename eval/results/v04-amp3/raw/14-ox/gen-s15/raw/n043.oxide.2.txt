struct Queue { elements: Vec<Int> }

fn swap_first_last(q: Queue) -> Queue {
    let len = len(q.elements)
    if len < 2 {
        return q
    }
    let first = get(q.elements, 0)?
    let last = get(q.elements, len - 1)?
    let middle = range(1, len - 1)
    let new_elements = push(push(push(vec(), last), middle), first)
    Queue { elements: new_elements }
}

fn print_queue(q: Queue) {
    for x in q.elements {
        print(x)
    }
}

fn main() {
    let q = Queue { elements: range(1, 6) }
    let swapped_q = swap_first_last(q)
    print_queue(swapped_q)
}
