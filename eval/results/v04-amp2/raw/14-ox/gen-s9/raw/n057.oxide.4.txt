fn main() {
    let fig = "fig"
    let banana = "banana"
    let kiwi = "kiwi"

    let names = vec(fig, banana, kiwi)
    let mut longest_name = ""
    let mut max_length = 0

    for name in names {
        let length = str_len(name)
        if length > max_length {
            longest_name = name
            max_length = length
        }
    }

    print(longest_name)
    print(max_length)
}
