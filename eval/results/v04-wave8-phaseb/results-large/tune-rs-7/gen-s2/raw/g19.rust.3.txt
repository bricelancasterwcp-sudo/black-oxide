fn main() {
    let text = "order 47 shipped on 2026-03-15, invoice 1290 paid.";
    println!("{}", text.len());
    println!("{}", text.chars().filter(|c| c.is_alphabetic()).count());
    println!("{}", text.chars().filter(|c| c.is_digit(10)).count());
    println!("{}", text.chars().filter(|c| *c == ' ').count());
    println!("{}", text.chars().filter(|c| !c.is_alphanumeric() && *c != ' ').count());
    let mut longest_streak = 0;
    let mut current_streak = 0;
    for c in text.chars() {
        if c.is_digit(10) {
            current_streak += 1;
            if current_streak > longest_streak {
                longest_streak = current_streak;
            }
        } else {
            current_streak = 0;
        }
    }
    println!("{}", longest_streak);
    let digits: String = text.chars().filter(|c| c.is_digit(10)).collect();
    println!("{}", digits);
}
