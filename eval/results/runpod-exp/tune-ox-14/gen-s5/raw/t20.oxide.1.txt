struct Item { name: Str, price: Int, quantity: Int }
fn main() {
    let pen = Item { name: "pen", price: 3, quantity: 4 }
    let pad = Item { name: "pad", price: 9, quantity: 2 }
    let ink = Item { name: "ink", price: 5, quantity: 3 }
    let total_cost = pen.price * pen.quantity + pad.price * pad.quantity + ink.price * ink.quantity
    print(total_cost)
    let max_cost = max(pen.price * pen.quantity, max(pad.price * pad.quantity, ink.price * ink.quantity))
    if max_cost == pen.price * pen.quantity {
        print_str("pen")
    } else if max_cost == pad.price * pad.quantity {
        print_str("pad")
    } else {

    }
}
