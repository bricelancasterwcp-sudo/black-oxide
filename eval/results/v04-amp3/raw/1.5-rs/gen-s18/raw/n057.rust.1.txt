```rust
fn main() {
    let words = ["fig", "banana", "kiwi"];
    let mut max_length_word = "";
    let mut max_length = 0;

    for word in words {
        if word.len() > max_length {
            max_length_word = word;
            max_length = word.len();
        }
    }

    println!("The word with the most letters is '{}'", max_length_word);
    println!("It has {} letters.", max_length);
}
```
