struct Log { entries: Vec<Int> }

fn main() {
    let log = Log { entries: vec() }
    let log = Log { entries: push(log.entries, 6) }
    let log = Log { entries: push(log.entries, 7) }
    let log = Log { entries: push(log.entries, 8) }
    let log = Log { entries: push(log.entries, 9) }

    print(unwrap_or(get(log.entries, 0), -1))
    print(unwrap_or(get(log.entries, len(log.entries) - 1), -1))
}
