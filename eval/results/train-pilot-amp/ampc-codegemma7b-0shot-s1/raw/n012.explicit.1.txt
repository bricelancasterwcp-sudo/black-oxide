fn len(v: &Vec<Int>) -> Int {
    let mut i = 0
    for _ in v {
        i = i + 1
    }
    i
}

fn main() {
    let numbers = vec()
    for i in 0 {
        if i < 15 {
            numbers = push(numbers, i)
        } else {
            break
        }
    }
    print(len(&numbers))
    drop numbers
}
