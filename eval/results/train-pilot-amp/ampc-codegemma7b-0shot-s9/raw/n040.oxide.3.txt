fn area(s: Shape) -> Float {
    match s {
        Shape(square) => square.side * square.side,
        Shape(circle) => circle.pi * circle.radius * circle.radius,
    }
}

fn main() {
    let sq = Shape(Square { side: 4.0 })
    let circle = Shape(Circle { pi: 3.14159, radius: 2.0 })
    print(area(sq))
    print(area(circle))
}

fn parse_int(s: Str) -> Option<Int> {
    let mut n = 0
    for c in s {
        let digit = digit_from_char(c, 10)?
        n = n * 10 + digit
    }
    Some(n)
}

fn int_to_digits(mutn: Int) -> Vec<Int> {
    let mut digits = vec()
    while mutn != 0 {
        digits.push(mutn % 10)
        mutn = mutn / 10
    }
    digits
}

fn char_from_digit(mutn: Int, base: Int) -> Option<Char> {
    if mutn >= 0 && mutn < base {
        Some(char_from_u32(mutn + char_from_u32("0", 10)))
    } else {
        None
    }
}

fn digit_from_char(c: Char, base: Int) -> Option<Int> {
    if c >= char_from_u32("0", 10) && c <= char_from_u32((base - 1).to_string().chars().nth(0).unwrap(), 10) {
        Some((c - char_from_u32("0", 10)).to_int())
    } else {
        None
    }
}

fn char_from_u32(u: U32) -> Char {
    unsafe
    return std.mem.transmute(u)
}

fn digit_from_u32(u: U32) -> Int {
    u.to_int()
}

fn to_u32(i: Int) -> U32 {
    i.to_uint()
}

fn int_to_bytes(i: Int) -> Vec<U8> {
    let mut bytes = vec()
    for b in to_u32(i).to_be_bytes() {
        bytes.push(b)
    }
    bytes
}

fn bytes_to_int(bytes: Vec<U8>) -> Int {
    let mut n = 0
    for b in bytes {
        n = n * 256 + b
    }
    n
}

fn string_to_bytes(s: Str) -> Vec<U8> {
    let mut bytes = vec()
    for c in s {
        bytes.push(c.to_u32())
    }
    bytes
}

fn bytes_to_string(bytes: Vec<U8>) -> Str {
    let mut s = String.new()
    for b in bytes {
        s.push(char_from_u32(b))
    }
    s
}

fn u32_to_string(u: U32) -> Str {
    let mut s = String.new()
    for d in u32_to_digits(u) {
        s.push(char_from_digit(d, 10))
    }
    s
}

fn u32_to_digits(mutu: U32) -> Vec<U32> {
    let mut digits = vec()
    while mutu != 0 {
        digits.push(mutu % 10)
        mutu = mutu / 10
    }
    digits
}
