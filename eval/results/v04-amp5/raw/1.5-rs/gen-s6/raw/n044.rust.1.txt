```rust
fn deposit(balance: f64, amount: f64) -> f64 {
    balance + amount
}

fn main() {
    let mut balance = 0.0;
    let deposits = vec![3.0, 1.0, 4.0, 1.0, 5.0];

    for deposit in deposits {
        balance = deposit(balance);
        println!("Balance after deposit: {:.2}", balance);
    }
}
```
