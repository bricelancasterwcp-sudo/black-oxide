fn find_n_positions(s: Str) -> Vec<Int> {
    let positions = vec()
    let i = 0
    while i < str_len(s) {
        if get(s, i) == Some('n') {
            positions = push(positions, i)
        }
        i = i + 1
    }
    positions
}

fn main() {
    let s = "banana"
    let positions = find_n_positions(s.clone())
    for pos in positions {
        print(pos)
    }
}
