```rust
fn main() {
    let readings = [6, 11, 3, 14, 9];
    let count_exceed_five = readings.iter().filter(|&r| r > 5).count();
    println!("Number of readings exceeding five: {}", count_exceed_five);

    let max_reading = *readings.iter().max().unwrap_or(&0);
    println!("Highest reading: {}", max_reading);
}
```
