struct Reading { label: Str, values: Vec<Int> }

fn digit_total_and_count(s: Str) -> (Int, Int) {
    let mut total = 0
    let mut count = 0
    for c in chars(s) {
        if parse_int(c).is_some() {
            total += to_float(parse_int(c).unwrap()).trunc()
            count += 1
        }
    }
    (total, count)
}

fn main() {
    let s = "a1b22"
    let (total, count) = digit_total_and_count(s)
    print(total)
    print(count)
}
