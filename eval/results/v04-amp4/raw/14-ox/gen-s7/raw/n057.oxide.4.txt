fn main() {
    let fig = "fig"
    let banana = "banana"
    let kiwi = "kiwi"

    let lengths = vec()
    lengths = push(lengths, (fig, str_len(fig)))
    lengths = push(lengths, (banana, str_len(banana)))
    lengths = push(lengths, (kiwi, str_len(kiwi)))

    let mut max_name = ""
    let mut max_length = 0

    for (name, length) in lengths {
        if length > max_length {
            max_name = name
            max_length = length
        }
    }

    print_str(max_name)
    print(max_length)
}
