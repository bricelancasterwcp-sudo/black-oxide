fn digit_sum(s: Str) -> Int {
    let mut sum = 0
    let mut digits = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            sum = sum + c.parse_int()?.unwrap()
            digits = digits + 1
        }
    }
    sum
}

fn digit_count(s: Str) -> Int {
    let mut digits = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            digits = digits + 1
        }
    }
    digits
}

fn main() {
    print(digit_sum("a1b22"))
    print(digit_count("a1b22"))
}
