struct Signal { color: Str }

fn interpret(s: Vec<Signal>) {
    for sig in s {
        match sig.color {
            "green" => print_str("go"),
            "red" => print_str("stop"),
            "yellow" => print_str("wait"),
            _ => print_str("unknown signal"),
        }
    }
}

fn main() {
    let signals = vec()
    signals = push(signals, Signal { color: "green" })
    signals = push(signals, Signal { color: "red" })
    signals = push(signals, Signal { color: "yellow" })
    interpret(signals)
}
