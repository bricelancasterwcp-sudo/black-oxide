struct Queue { items: Vec<Int> }

fn main() {
    let mut q = Queue { items: vec().push(1).push(2).push(3).push(4).push(5) }
    if len(q.items) > 1 {
        let first = get(q.items, 0).unwrap()
        let last = get(q.items, len(q.items) - 1).unwrap()
        q.items = set(set(q.items, 0, last), len(q.items) - 1, first)
    }
    for item in q.items {
        print(item)
    }
}
