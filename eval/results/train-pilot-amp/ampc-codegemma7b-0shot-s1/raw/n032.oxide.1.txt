struct Point { x: Int, y: Int }

fn add(a: Point, b: Point) -> Point {
    Point(a.x + b.x, a.y + b.y)
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < 0 || i >= v.len() {
        None
    } else {
        Some(v.get(i).unwrap())
    }
}

    if cond != 0 {
        true_branch

        false_branch

    let mut accumulator = 0
    while cond != 0 {
        accumulator = body
        cond = cond - 1
    }
    accumulator
}
