struct Reading { label: String, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 5 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    match get(v, 2)? {
        x => Some(x + 1),
    }
}

fn main() {
    let r = Reading { label: String::from("lab"), values: vec![3, 42] };
    print(first_big(r.values));
    if let Some(n) = second(r.values) {
        println!("{}", n);
    } else {
        println!("too short");
    }
}
