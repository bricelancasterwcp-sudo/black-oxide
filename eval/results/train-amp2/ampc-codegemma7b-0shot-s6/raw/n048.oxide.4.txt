fn main() {
    let v = vec(4, 7, 4, 9, 4)
    let mut acc = 0
    for x in v {
        if x == 4 {
            print(acc)
            acc = acc + 1
        }
    }
    print(acc)
}

fn vec(x: Int, xs: Vec<Int>) -> Vec<Int> {
    push(xs, x)
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v2 = v
    v2.push(x)
    v2
}

fn print(x: Int) {
    print_str(int_to_str(x))
}

fn print_str(s: Str) {
    let stdout = std_io_stdout()
    write(stdout, s)
}

fn int_to_str(n: Int) -> Str {
    format("%d", n)
}

fn write(stdout: StdOut, s: Str) {
    stdout.write_str(s)
}
