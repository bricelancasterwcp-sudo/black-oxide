fn main() {
    let readings = [6, 11, 3, 14, 9];
    let mut count_exceeding_five = 0;
    let mut highest_reading = readings[0];

    for &reading in &readings {
        if reading > 5 {
            count_exceeding_five += 1;
        }
        if reading > highest_reading {
            highest_reading = reading;
        }
    }

    println!("{} of the readings exceed five.", count_exceeding_five);
    println!("The highest reading is {}", highest_reading);
}
