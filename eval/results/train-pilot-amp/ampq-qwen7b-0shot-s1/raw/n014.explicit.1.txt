fn find_min(a: Int, b: Int, c: Int, d: Int, e: Int) -> Int {
    let min = a
    if b < min {
        min = b
    }
    if c < min {
        min = c
    }
    if d < min {
        min = d
    }
    if e < min {
        min = e
    }
    min
}

fn main() {
    let readings = find_min(5, 3, 9, 1, 7)
    print(readings)
}
