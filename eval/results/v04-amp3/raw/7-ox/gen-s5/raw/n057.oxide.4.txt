fn main() {
    let mut max_len = 0;
    let mut longest_name = "";

    fn check_name(name: &Str) -> (&Str, Int) {
        let len = str_len(name);
        if len > max_len {
            (name, len)
        } else {
            ("", 0)
        }
    }

    let names = vec!["fig", "banana", "kiwi"];
    for name in names.iter() {
        let (name, len) = check_name(name);
        if len > max_len {
            longest_name = name;
            max_len = len;
        }
    }

    print_str(longest_name);
    print(max_len);
}
