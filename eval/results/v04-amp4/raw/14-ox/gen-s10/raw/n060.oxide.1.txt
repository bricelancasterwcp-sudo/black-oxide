fn count_digits_and_sum(s: Str) -> (Int, Int) {
    let sum = 0
    let count = 0
    for c in chars(s) {
        if parse_int(c).is_some() {
            sum += to_float(parse_int(c).unwrap()).trunc()
            count += 1
        }
    }
    (sum, count)
}

fn main() {
    let input = "a1b22"
    let (total_sum, digit_count) = count_digits_and_sum(input)
    print(total_sum)
    print(digit_count)
}
