fn main() {
    loop {
        let signal = get_signal(); // Replace with actual logic to get the signal
        match signal {
            "green" => println!("Go"),
            "red" => println!("Stop"),
            "yellow" => println!("Wait"),
            _ => break, // Break the loop if an unexpected signal is received
        }
    }
}

fn get_signal() -> &'static str {
    rand::thread_rng().gen_bool(1.0) // Simulate a random choice
        .then(|| "green")
        .or_else(|| rand::thread_rng().gen_bool(0.5)
            .then(|| "yellow"))
        .or_else(|| rand::thread_rng().gen_bool(0.5)
            .then(|| "red"));
}
