```rust
fn main() {
    let arr = [5, 12, 3, 18, 9];
    let cutoff = 10;

    let mut count_under_cutoff = 0;
    let mut count_over_or_equal = 0;

    for num in arr.iter() {
        if num < cutoff {
            count_under_cutoff += 1;
        } else {
            count_over_or_equal += 1;
        }
    }

    println!("Numbers under cutoff: {}", count_under_cutoff);
    println!("Numbers over or equal to cutoff: {}", count_over_or_equal);
}
```
