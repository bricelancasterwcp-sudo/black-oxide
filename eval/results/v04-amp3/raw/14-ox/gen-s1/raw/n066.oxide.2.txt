fn filter_and_count_under_ten(values: Vec<Int>) -> (Vec<Int>, Int) {
    let mut kept = vec()
    let mut count = 0
    for x in values {
        if x < 10 {
            kept = push(kept, x)
            count = count + 1
        }
    }
    (kept, count)
}

fn main() {
    let values = vec(9, 2, 12, 5, 16)
    let (kept, count) = filter_and_count_under_ten(values)
    for x in kept {
        print(x)
    }
    print(count)
}
