```rust
fn main() {
    let mut vowels = 0;
    let mut consonants = 0;

    for c in "abcdefghijklmnopqrstuvwxyz".chars() {
        match c.to_lowercase().next() {
            Some('a' | 'e' | 'i' | 'o' | 'u') => vowels += 1,
            _ => consonants += 1,
        }
    }

    print(vowels);
    print(consonants);
}
```
