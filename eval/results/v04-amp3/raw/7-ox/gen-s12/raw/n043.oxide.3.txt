struct Queue { items: Vec<Int> }

fn enqueue(q: Queue, x: Int) -> Queue {
    let mut q = q;
    q.items.push(x);
    q
}

fn dequeue(q: Queue) -> (Queue, Option<Int>) {
    let mut q = q;
    let item = get(q.items, 0)?;
    q.items = push(vec(), ..q.items[1..]);
    (q, Some(item))
}

fn main() {
    let mut q = enqueue(enqueue(enqueue(enqueue(Queue { items: vec() }, 1), 2), 3), 4);
    q = enqueue(q, 5);
    let q = dequeue(q).0;
    let q = enqueue(dequeue(q).0, dequeue(q).1.unwrap());
    for x in q.items {
        print(x);
        print("\n");
    }
}
