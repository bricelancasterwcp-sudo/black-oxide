    let square = Shape { side: 4.0, kind: Kind { tag: Tag { variant: Square } } }
    print(area(&square))
}

fn area(shape: &Shape) -> Float {
    match shape.kind {
        Kind(refTag) => match refTag.variant {
            Square => side_area(shape.side),
            Circle => circle_area(shape.radius),
        },
    }
}

fn side_area(s: Float) -> Float {
    s * s
}

fn circle_area(r: Float) -> Float {
    3.14 * r * r
}

struct Shape {
    side: Float,
    kind: Kind,
}

struct Kind {
    tag: Tag,
}

struct Tag {
    variant: Variant,
}

enum Variant {
    Square,
    Circle,
}

    print_str("x + y = ")

struct Point {
    x: Float,
    y: Float,
}

struct Rect {
    top_left: Point,
    bottom_right: Point,
}

fn area(r: &Rect) -> Float {
    (r.bottom_right.x - r.top_left.x) * (r.bottom_right.y - r.top_left.y)
}

fn main() {
    let r = Rect { top_left: Point { x: 1.0, y: 2.0 }, bottom_right: Point { x: 3.0, y: 4.0 } }
    print(area(&r))
}
