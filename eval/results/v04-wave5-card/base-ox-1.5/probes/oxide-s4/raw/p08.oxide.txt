fn size(o: Option<Vec<Int>>) -> Int {
    match o {
        Some(w) => len(w),
        None => 0,
    }
}

fn main() {
    let v = push(push(vec(), 1), 2);
    let o = Some(v.clone()); // Clone the vector before passing it to size
    print(len(v)); // Now you can use v again
    print(size(o));
}
