```rust
fn main() {
    let numbers = vec![7, 10, 9, 23];
    let mut count = 0;
    for &num in &numbers {
        if num.is_integer() {
            println!("{} is a number", num);
            count += 1;
        } else {
            println!("{} is not a number", num);
        }
    }
    println!("Total numbers: {}", count);
    println!("Numbers that are not numbers: {}", numbers.len() - count);
}
```
