fn walk_list(v: &Vec<Int>) -> Int {
    let mut total = 0
    for x in v {
        if x % 2 == 0 {
            total = total + x
        } else {
            print(x * 2)
        }
    }
    total
}

fn main() {
    let nums = push(push(push(vec(), 7), 4), push(vec(), 9))
    let total = walk_list(&nums)
    print(total)
    drop nums
}
