fn main() {
    let nums = vec("7", "x", "9", "?");
    let mut sum = 0;
    let mut count_non_nums = 0;

    for s in nums {
        match parse_int(s) {
            Some(n) => sum += n,
            None => count_non_nums += 1
        }
    }

    print(sum);
    print(count_non_nums);
}
