fn int_to_str(mutn: Int) -> Str {
    let mut digits = vec()
    while mutn > 0 {
        digits.push((mutn % 10).to_int())
        mutn = trunc(mutn / 10)
    }
    let mut result = Str("")
    for d in digits {
        result = concat(result, int_to_str(d))
    }
    result
}
