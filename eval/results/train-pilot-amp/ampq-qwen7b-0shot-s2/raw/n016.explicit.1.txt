fn filter_and_sum(v: &Vec<Int>, divisor: Int) -> Int {
    let mut total = 0
    for x in &v {
        if x % divisor == 0 {
            total = total + x
        }
    }
    total
}

fn main() {
    let nums = push(push(vec(), 3), 6)
    let filtered = push(push(vec(), 9), 12)
    print(filter_and_sum(&nums, 3))
    drop nums
    print(filter_and_sum(&filtered, 3))
    drop filtered
}
