```rust
fn main() {
    let s = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in s.chars() {
        if let Some(digit) = c.to_digit(10) {
            sum += digit;
            count += 1;
        }
    }

    println!("Sum of digits: {}", sum);
    println!("Number of digit characters: {}", count);
}
```
