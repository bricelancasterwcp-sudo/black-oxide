```rust
fn main() {
    let mut count_vowels = 0;
    let mut count_not_vowels = 0;

    for c in "abcdefghijklmnopqrstuvwxyz".chars() {
        if is_vowel(c) {
            count_vowels += 1;
        } else {
            count_not_vowels += 1;
        }
    }

    println!("Vowels: {}", count_vowels);
    println!("Not Vowels: {}", count_not_vowels);
}

fn is_vowel(c: char) -> bool {
    matches!(c, 'a' | 'e' | 'i' | 'o' | 'u')
}
```
