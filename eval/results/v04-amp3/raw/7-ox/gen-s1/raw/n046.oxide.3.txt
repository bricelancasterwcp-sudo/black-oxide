fn count_under_over(v: Vec<Int>, cutoff: Int) -> (Int, Int) {
    let mut under = 0;
    let mut over = 0;
    for x in v {
        if x < cutoff {
            under += 1;
        } else {
            over += 1;
        }
    }
    (under, over)
}

fn main() {
    let nums = push(push(push(push(vec(), 5), 12), 3), 18);
    let nums = push(nums, 9);
    let (under, over) = count_under_over(nums.clone(), 10);
    print_str("Under: ");
    print(under);
    print_str("\nOver: ");
    print(over);
}
