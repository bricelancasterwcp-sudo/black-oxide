```rust
fn main() {
    let s = "keyboard";
    let mut count_vowels = 0;
    let mut count_consonants = 0;

    for c in s.chars() {
        if c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' || 
           c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U' {
            count_vowels += 1;
        } else if c.is_alphabetic() {
            count_consonants += 1;
        }
    }

    print(count_vowels);
    print_str(" ");
    print(count_consonants);
}
```
