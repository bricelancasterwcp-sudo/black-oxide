fn sort_descending(v: Vec<Int>) -> Vec<Int> {
    let mut sorted = v;
    for i in range(0, len(sorted)) {
        for j in range(i + 1, len(sorted)) {
            if get(sorted, i)? < get(sorted, j)? {
                let temp = get(sorted, i)?;
                set(sorted, i, get(sorted, j)?)?;
                set(sorted, j, temp)?;
            }
        }
    }
    Some(sorted)
}

fn main() {
    let scores = vec();
    push(scores, 2);
    push(scores, 9);
    push(scores, 5);

    let sorted_scores = unwrap_or(sort_descending(scores)?, vec());
    for score in sorted_scores {
        print(score);
    }
}
