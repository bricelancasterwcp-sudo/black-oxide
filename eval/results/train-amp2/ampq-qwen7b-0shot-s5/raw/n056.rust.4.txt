fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut total = 0;
    let mut missing = 0;

    for text in texts {
        match text.parse::<i32>() {
            Ok(num) => total += num,
            Err(_) => missing += 1,
        }
    }

    println!("Total of the numbers: {}", total);
    println!("Numbers missing: {}", missing);
}
