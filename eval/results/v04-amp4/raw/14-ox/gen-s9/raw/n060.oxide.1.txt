fn main() {
    let s = "a1b22";
    let mut sum = 0;
    let mut count = 0;
    for c in chars(s) {
        if len(c) == 1 && get(c, 0)? >= '0' && get(c, 0)? <= '9' {
            sum += to_float(get(c, 0)?)? as Int;
            count += 1;
        }
    }
    print(sum);
    print(count);
}
