fn main() {
    let mut queue = push(push(push(push(push(vec(), 1), 2), 3), 4), 5)
    let len = len(queue)
    if len > 1 {
        let first = unwrap_or(get(queue, 0), -1)
        let last = unwrap_or(get(queue, len - 1), -1)
        queue = set(queue, 0, last)
        queue = set(queue, len - 1, first)
    }
    for i in range(0, len) {
        print(unwrap_or(get(queue, i), -1))
    }
}
