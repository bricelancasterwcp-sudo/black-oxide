```rust
fn main() {
    let readings = [6, 11, 3, 14, 9];
    let exceeding_five_count: i32 = readings.iter().filter(|&&read| read > 5).count();
    println!("The number of readings that exceed five is: {}", exceeding_five_count);

    let highest_reading = readings.iter().max().unwrap();
    println!("The highest reading is: {}", highest_reading);
}
```
