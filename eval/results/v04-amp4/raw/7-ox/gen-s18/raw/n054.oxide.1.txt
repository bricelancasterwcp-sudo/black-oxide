fn main() {
    let vowels = "aeiouAEIOU";
    let mut vowel_count = 0;
    let mut non_vowel_count = 0;

    for c in 'a'..='z' {
        if vowels.contains(c) {
            vowel_count += 1;
        } else {
            non_vowel_count += 1;
        }
    }

    print(vowel_count);
    print(non_vowel_count);
}
