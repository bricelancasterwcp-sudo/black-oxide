fn first_odd(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x % 2 != 0 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    match get(v, 1)? {
        x => Some(x + 1),
    }
}

fn main() {
    let v = vec![7, 4, 9, 10];
    let total = first_odd(v).map(|x| x * 2);
    let total_even = v.iter().filter(|&x| x % 2 == 0).sum::<Int>();
    println!("{:?}", total);
    println!("Total of even entries: {}", total_even);
}
