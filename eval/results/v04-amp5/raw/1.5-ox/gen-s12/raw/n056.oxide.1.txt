```rust
fn main() {
    let mut count = 0;
    let mut total = 0;
    for text in vec!["7", "x", "9", "?"] {
        match parse_int(text) {
            Some(n) => {
                total += n;
                count += 1;
            }
            None => {}
        }
    }
    println!("Total: {}", total);
    println!("Count: {}", count);
}
```
