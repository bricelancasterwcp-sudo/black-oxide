struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    count_if(v, |x| x > threshold)
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    match v {
        [] => None,
        [..xs] => Some(max(xs)?),
    }
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14)
    let count = count_exceeding(readings, 5)
    print(count)
    
    let max_reading = find_max(readings)
    match max_reading {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
