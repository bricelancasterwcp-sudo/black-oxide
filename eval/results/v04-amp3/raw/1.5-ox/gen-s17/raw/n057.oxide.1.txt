```rust
fn main() {
    let fruits = ["fig", "banana", "kiwi"];
    let mut max_length = 0;
    let mut longest_fruit = "";

    for fruit in fruits {
        if fruit.len() > max_length {
            max_length = fruit.len();
            longest_fruit = fruit;
        }
    }

    print!("The longest fruit is {} with {} letters.\n", longest_fruit, max_length);
}
```
