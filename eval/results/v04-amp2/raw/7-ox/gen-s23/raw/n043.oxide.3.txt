struct Queue { items: Vec<Int> }

fn exchange_first_last(q: Queue) -> Queue {
    let len = len(q.items);
    if len > 0 {
        let first = get(q.items, 0)?;
        let last = get(q.items, len - 1)?;
        let mut new_items = vec![last];
        for i in 1..len - 1 {
            push(new_items, get(q.items, i)?);
        }
        push(new_items, first);
        Queue { items: new_items }
    } else {
        q
    }
}

fn main() {
    let q = Queue { items: vec![1, 2, 3, 4, 5] };
    let q_exchanged = exchange_first_last(q);
    for x in q_exchanged.items {
        print(x);
        print_str("\n");
    }
}
