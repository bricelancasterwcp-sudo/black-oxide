struct LetterCount { letter: Str, count: Int }

fn main() {
    let word = "coffee";
    let mut counts = vec();
    let mut seen = vec();

    for ch in chars(word) {
        let existing = None;
        for count in counts.clone() {
            if count.letter == ch {
                existing = Some(count);
                break
            }
        }

        match existing {
            Some(c) => {
                let new_count = LetterCount { letter: c.letter, count: c.count + 1 };
                counts = set(counts, index_of(counts, c), new_count);
            },
            None => {
                let new_count = LetterCount { letter: ch, count: 1 };
                push(seen, ch);
                push(counts, new_count);
            }
        };
    }

    for s in seen {
        for c in counts.clone() {
            if c.letter == s && c.count > 1 {
                print(s);
                break
            }
        }
    }
}

fn index_of(v: Vec<LetterCount>, c: LetterCount) -> Int {
    for (i, count) in enumerate(v) {
        if count == c {
            return i;
        }
    }
    -1
}
