fn main() {
    let v: Vec<Int> = vec![7, 8, 9];
    match second(v) {
        Some(n) => println!("{}", n),
        None => println!("None"),
    }
}

fn second(mut v: Vec<Int>) -> Option<Int> {
    let first = v.pop();
    match first {
        Some(first) => {
            for x in v.iter() {
                if x > first {
                    return Some(x);
                }
            }
            None
        },
        _ => None,
    }
}
