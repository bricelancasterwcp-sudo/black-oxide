struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 5 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(first_big(r.values))            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }

    let readings = vec![6, 11, 3, 14, 9];
    let count_exceeds_five = readings.iter().filter(|&&x| x > 5).count();
    println!("count_exceeds_five: {}", count_exceeds_five);

    let max_reading = readings.into_iter().max().unwrap_or(0);
    println!("max_reading: {}", max_reading);
}
