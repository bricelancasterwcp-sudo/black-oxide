struct Queue {
    items: Vec<Int>,
}

impl Queue {
    fn new() -> Self {
        Queue { items: vec![] }
    }

    fn push(&mut self, item: Int) {
        self.items.push(item);
    }

    fn pop(&mut self) -> Option<Int> {
        if let Some(first) = self.items.remove(0) {
            Some(first)
        } else {
            None
        }
    }

    fn peek(&self) -> &Int {
        &self.items[0]
    }

    fn exchange(&mut self) {
        if let (Some(first), Some(last)) = (&self.items[0], &self.items[self.items.len() - 1]) {
            self.items.swap(0, self.items.len() - 1);
        }
    }

    fn print_queue(&self) {
        for item in &self.items {
            println!("{}", item);
        }
    }
}

fn main() {
    let mut queue = Queue::new();
    queue.push(1);
    queue.push(2);
    queue.push(3);
    queue.push(4);
    queue.push(5);

    queue.exchange();

    queue.print_queue();
}
