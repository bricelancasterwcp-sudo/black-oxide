fn swap_first_last(q: Vec<Int>) -> Vec<Int> {
    let len = len(&q)
    if len > 0 {
        let first = get(&q, 0)
        let last = get(&q, len - 1)
        if letSome(first) == first {
            if letSome(last) == last {
                let mut new_q = q.clone()
                new_q.set(0, last)
                new_q.set(len - 1, first)
                return new_q
            }
        }
    }
    q
}

fn main() {
    let mut q = vec(1, 2, 3, 4, 5)
    q = swap_first_last(q)
    for x in &q {
        print(x)
        println()
    }
    drop q
}
