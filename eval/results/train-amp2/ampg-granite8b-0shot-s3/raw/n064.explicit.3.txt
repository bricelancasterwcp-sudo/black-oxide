fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i < 0 || i >= len(&v) {
        return None
    }
    Some(get_at(&v, i))
}

fn get_at(v: &Vec<Int>, i: Int) -> Int {
    v.get(i).unwrap()
}

fn main() {
    let v = push(push(push(vec(), 10), 20), 30)
    let x = get(&v, 0)
    let y = get(&v, 2)
    print_str("x = ")
    print(x)
    print_str(", y = ")
    print(y)
}
