fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if match max {
            Some(m) => x > m,
            None => true,
        } {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = vec()
    readings.push(6)
    readings.push(11)
    readings.push(3)
    readings.push(14)
    readings.push(9)
    
    let count = count_exceeding(readings, 5)
    print(count)
    
    let max_reading = find_max(readings)
    match max_reading {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
