fn merge_sorted_lists(a: Vec<Int>, b: Vec<Int>) -> Vec<Int> {
    let mut merged = vec()
    let mut i = 0
    let mut j = 0
    while i < len(a) && j < len(b) {
        if a[i] <= b[j] {
            merged = push(merged, a[i])
            i += 1
        } else {
            merged = push(merged, b[j])
            j += 1
        }
    }
    while i < len(a) {
        merged = push(merged, a[i])
        i += 1
    }
    while j < len(b) {
        merged = push(merged, b[j])
        j += 1
    }
    merged
}
fn main() {
    let list1 = vec(2, 5, 9, 14, 21, 30)
    let list2 = vec(1, 5, 8, 15, 22)
    let merged_list = merge_sorted_lists(list1, list2)
    print(len(merged_list)) // how many numbers the merged list holds
    if len(merged_list) > 0 {
        print(merged_list[0]) // its first number
        print(merged_list[len(merged_list) - 1]) // its last number
    } else {
        print(-1)
        print(-1)
    }
    let sum = 0
    for x in merged_list {
        sum += x
    }
    print(sum) // the sum of all its numbers
    let mut is_non_decreasing = true
    for i in range(1, len(merged_list)) {
        if merged_list[i] < merged_list[i - 1] {
            is_non_decreasing = false
            break
        }
    }
    print(is_non_decreasing) // whether it really is in non-decreasing order
    let mut consecutive_duplicates = 0
    for i in range(1, len(merged_list)) {
        if merged_list[i] == merged_list[i - 1] {
            consecutive_duplicates += 1
        }
    }
    print(consecutive_duplicates) // how many times a number is immediately followed by an equal number
}
