```rust
fn main() {
    let arr = [5, 12, 3, 18, 9];
    let mut count_below_cutoff = 0;
    let mut count_above_or_equal_cutoff = 0;

    for &x in arr.iter() {
        if x < 10 {
            count_below_cutoff += 1;
        } else {
            count_above_or_equal_cutoff += 1;
        }
    }

    println!("Count below cutoff: {}", count_below_cutoff);
    println!("Count above or equal to cutoff: {}", count_above_or_equal_cutoff);
}
```
