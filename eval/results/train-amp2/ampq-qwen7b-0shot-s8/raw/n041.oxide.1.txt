struct Reading { label: Str, values: Vec<Int> }

fn sort_ascending(v: Vec<Int>) -> Vec<Int> {
    let mut sorted = v
    for i in 0 < sorted.len() {
        for j in i + 1 < sorted.len() {
            if sorted.get(i) > sorted.get(j) {
                let temp = sorted.get(i)
                sorted.set(i, sorted.get(j))
                sorted.set(j, temp)
            }
        }
    }
    sorted
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(vec(), 9), 2), push(vec(), 7), 4) }
    let sorted_values = sort_ascending(r.values)
    for x in sorted_values {
        print(x)
    }
}
