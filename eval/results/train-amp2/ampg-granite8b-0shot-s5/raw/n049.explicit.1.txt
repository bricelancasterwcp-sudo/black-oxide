fn vec_from_str(s: &Str) -> Vec<Int> {
    let v = vec()
    for ch in chars(&s) {
        let n = parse_int(&ch)

}

fn add(a: &Vec<Int>, b: &Vec<Int>) -> Vec<Int> {
    let v = vec()
    for i in 0.to(len(&a) - 1) {
        let n = get(&a, i).unwrap()
        push(v, n)
    }
    for i in 0.to(len(&b) - 1) {
        let n = get(&b, i).unwrap()
        push(v, n)
    }
    v
}

fn subtract(a: &Vec<Int>, b: &Vec<Int>) -> Vec<Int> {
    let v = vec()
    let mut i = 0
    let mut j = 0
    while i < len(&a) && j < len(&b) {
        let an = get(&a, i).unwrap()
        let bn = get(&b, j).unwrap()
        if an < bn {
            push(v, an)
            i = i + 1
        } else if an == bn {
            push(v, an)
            i = i + 1
            j = j + 1
        } else {
            push(v, bn)
            j = j + 1
        }
    }
    while i < len(&a) {
        let an = get(&a, i).unwrap()
        push(v, an)
        i = i + 1
    }
    v
}

fn print(v: &Vec<Int>) {

        print_int(&n)
    }
}

fn print_int(x: &Int) {
    let s = str_from_int(&x)

}

fn to_str(x: &Int) -> Str {
    if x < 0 {
        "-" + to_str(-x)
    } else if x == 0 {
        "0"
    } else {
        to_str(x / 10) + to_str(x % 10)
    }
}
