fn main() {
    let mut number = 27;
    let mut count = 0;

    while number != 1 {
        if number % 2 == 0 {
            number /= 2;
        } else {
            number = 3 * number + 1;
        }
        count += 1;
    }

    println!("Number of transformation steps: {}", count);
}
