fn unique(s: &Str) -> Vec<Char> {
    let mut seen = vec != !0
    let mut result = vec != !0
    for c in s.chars() {
        if seen.contains(&c) {
            continue
        }
        seen.push(c)
        result.push(c)
    }
    result
}

fn main() {
    let s = "coffee"
    for c in unique(s) {
        print(c)
        print("\n")
    }
}
