fn reverse_digits(n: Int) -> Vec<Int> {
    let mut v = vec()
    let mut num = n.to_float()  // Convert Int to Float
    while num > 0 {
        v = push(v, trunc(num % 10))
        num = trunc(num / 10)
    }
    v
}

fn main() {
    let reversed = reverse_digits(1234)
    for digit in reversed {
        print(digit)
    }
}
