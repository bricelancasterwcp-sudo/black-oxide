fn main() {
    let morning = vec![94, 71, 88, 65, 92, 79, 83];
    let evening = vec![55, 90, 77, 84, 68, 99];
    let sittings = [("Morning", &morning), ("Evening", &evening)];
    for (name, marks) in sittings {
        println!("{}", name);
        let mut above_90 = 0;
        let mut between_80_and_89 = 0;
        let mut between_70_and_79 = 0;
        let mut below_70 = 0;
        let mut max_mark = i32::MIN;
        for &mark in marks {
            if mark >= 90 {
                above_90 += 1;
            } else if mark >= 80 {
                between_80_and_89 += 1;
            } else if mark >= 70 {
                between_70_and_79 += 1;
            } else {
                below_70 += 1;
            }
            max_mark = max_mark.max(mark);
        }
        let mean_mark = marks.iter().sum::<i32>() / marks.len() as i32;
        println!("{}", above_90);
        println!("{}", between_80_and_89);
        println!("{}", between_70_and_79);
        println!("{}", below_70);
        println!("{}", max_mark);
        println!("{}", mean_mark);
    }
    let (name1, marks1) = sittings[0];
    let (name2, marks2) = sittings[1];
    if marks1.iter().sum::<i32>() / marks1.len() as i32 > marks2.iter().sum::<i32>() / marks2.len() as i32 {
        println!("{}", name1);
    } else {
        println!("{}", name2);
    }
}
