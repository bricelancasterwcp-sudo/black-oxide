```rust
fn main() {
    let s = "a1b22";
    let mut total = 0;
    let mut count = 0;

    for c in s.chars() {
        if let Some(digit) = c.to_digit(10) {
            total += digit;
            count += 1;
        }
    }

    println!("Total: {}", total);
    println!("Count: {}", count);
}
```
