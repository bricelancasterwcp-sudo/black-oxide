fn is_vowel(c: Str) -> Bool {
    match c {
        Str(a) => match a.to_ascii_lowercase() {
            Str(b) => b == "a" || b == "e" || b == "i" || b == "o" || b == "u",
            _ => false,
        },
        _ => false,
    }
}

fn count_letters(s: Str) {
    let mut vowels = 0
    let mut consonants = 0
    for c in chars(s) {
        if is_vowel(c) {
            vowels = vowels + 1
        } else {
            consonants = consonants + 1
        }
    }
    print(vowels)
    print_str(" ")
    print(consonants)
}

fn main() {
    let s = "keyboard"
    count_letters(s)
}
