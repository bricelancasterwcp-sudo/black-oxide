```rust
fn main() {
    let s = "keyboard";
    let mut vowel_count = 0;
    let mut non_vowel_count = 0;

    for c in s.chars() {
        if ["a", "e", "i", "o", "u"].contains(&c.to_ascii_lowercase()) {
            vowel_count += 1;
        } else {
            non_vowel_count += 1;
        }
    }

    println!("Number of vowels: {}", vowel_count);
    println!("Number of non-vowels: {}", non_vowel_count);
}
```
