```rust
struct List { values: Vec<Int> }

fn main() {
    let list = List { values: vec![10, 20] };
    match first(&list) {
        Some(x) => print(x),
        None => print_str("Position one is empty"),
    }
    match second(&list) {
        Some(x) => print(x),
        None => print_str("Position four is empty"),
    }
}

fn first(l: &List) -> Option<&Int> {
    l.values.get(0)
}

fn second(l: &List) -> Option<&Int> {
    l.values.get(3)
}
```
