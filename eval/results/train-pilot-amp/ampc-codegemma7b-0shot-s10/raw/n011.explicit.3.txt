    a + b
}

fn push(v: Vec<Float>, x: Float) -> Vec<Float> {
    let mut new_vec = v
    new_vec = new_vec.push(x)
    new_vec
}

fn vec() -> Vec<Float> {
    let mut new_vec = Vec.new()
    new_vec
}

fn len(v: &Vec<Float>) -> Int {
    let len = v.len()
    len
}

fn iter(v: &Vec<Float>) -> Vec<Float> {
    let mut new_vec = vec()
    for i in 0.0.to_int().to_float() + 0.0 {
        if i < v.len().to_float() {
            new_vec = new_vec.push(v.get(i).unwrap())
        } else {
            break
        }
    }
    new_vec
}

fn fold(v: &Vec<Float>, init: Float, f: FnMutTwoArgs<Float, Float, Float>) -> Float {
    let mut accumulator = init
    for i in 0.0.to_int().to_float() + 0.0 {
        if i < v.len().to_float() {
            accumulator = f(accumulator, v.get(i).unwrap())
        } else {
            break
        }
    }
    accumulator
}

fn unwrap(o: &Option<Float>) -> Float {
    let float = o.unwrap()
    float
}

fn drop_all(v: Vec<Float>) {
    let mut new_vec = v
    while len(&new_vec) > 0.0.to_int().to_float() {
        drop new_vec
        new_vec = vec()
    }
    drop new_vec
}

fn drop_all(v: &Vec<Float>) {
    let mut new_vec = v.clone()
    while len(&new_vec) > 0.0.to_int().to_float() {

}

fn drop_all(v: &Vec<Float>, f: FnMutOneArg<Float>) {
    let mut new_vec = v.clone()
    while len(&new_vec) > 0.0.to_int().to_float() {
        f(new_vec.get(0).unwrap())
        new_vec = vec()
    }
    drop new_vec
}
