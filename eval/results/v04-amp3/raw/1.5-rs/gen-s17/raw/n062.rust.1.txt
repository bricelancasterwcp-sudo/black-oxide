fn main() {
    loop {
        let signal = read_signal();
        match signal {
            "green" => println!("go for green"),
            "red" => println!("stop for red"),
            "yellow" => println!("wait for yellow"),
            _ => println!("invalid signal"),
        }
    }
}

    // Simulate reading the signal (replace with actual input logic)
