fn main() {
    let values = vec(5, 12, 3, 18, 9);
    let cutoff = 10;
    let below = count_below(values, cutoff);
    let above_or_equal = len(values) - below;
    print(below);
    print(above_or_equal);
}

fn count_below(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0;
    for x in v {
        if x < cutoff {
            count = count + 1;
        }
    }
    count
}
