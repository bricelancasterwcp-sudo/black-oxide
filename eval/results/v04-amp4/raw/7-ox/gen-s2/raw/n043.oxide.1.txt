struct Queue { items: Vec<Int> }

fn exchange_first_last(q: Queue) -> Queue {
    let len = len(q.items);
    if len > 0 {
        let first = get(q.items, 0)?;
        set(set(q, 0, get(q.items, len - 1)?), len - 1, first)?
    }
    q
}

fn main() {
    let mut q = Queue { items: push(push(push(push(vec(), 1), 2), 3), 4), 5) };
    q = exchange_first_last(q);
    for x in q.items {
        print(x);
    }
}
