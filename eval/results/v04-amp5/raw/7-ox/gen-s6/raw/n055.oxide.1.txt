struct Entry { value: Option<Int> }

fn parse_entry(s: Str) -> Entry {
    let v = parse_int(s)?
    Entry { value: Some(v) }
}

fn main() {
    for entry in vec("8", "oak", "15") {
        match parse_entry(entry).value {
            Some(n) => print(n),
            None => print_str("bad"),
        }
    }
}
