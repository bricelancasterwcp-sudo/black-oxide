fn main() {
    let p = Point { x: 1, y: 5 }
    let p2 = Point { x: p.x + 10, y: p.y }
    print(p2.x + p2.y)
}

struct Point { x: Int, y: Int }

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v2 = v
    v2.push(x)
    v2
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < len(v) {
        Some(v.get(i).unwrap())
    } else {
        None
    }
}

fn len(v: Vec<Int>) -> Int {
    let mut i = 0
    for _ in v {
        i = i + 1
    }
    i
}

fn print(x: Int) {
    print_str(to_str(x))
}

fn print_str(s: Str) {
    print(s)
}

fn to_str(n: Int) -> Str {
    let mut s = Str { chars: vec() }
    let mut n2 = n
    while n2 > 0 {
        let r = n2 % 10
        s = push(s.chars, char(r + 48))
        n2 = n2 / 10
    }
    s
}

fn char(n: Int) -> Str {
    let mut s = Str { chars: vec() }
    s.chars.push(n)
    s
}

fn int_to_str(n: Int) -> Str {
    if n >= 0 {
        to_str(n)
    } else {
        let s = to_str(-n)
        push(Str { chars: vec(45) }, s)
    }
}

fn parse_int(s: Str) -> Option<Int> {
    let mut n = 0
    let mut neg = false
    for c in chars(s) {
        if c == char(45) {
            neg = true
        } else if c >= char(48) && c <= char(57) {
            n = n * 10 + int(c) - int(char(48))
        } else {
            return None
        }
    }
    if neg {
        Some(-n)
    } else {
        Some(n)
    }
}

fn int(c: Str) -> Int {
    let mut n = 0
    for i in 0.upto(str_len(c)) {
        n = n * 256 + c.get(i).unwrap()
    }
    n
}

fn str_len(s: Str) -> Int {
    let mut i = 0
    for _ in chars(s) {
        i = i + 1
    }
    i
}

fn chars(s: Str) -> Vec<Str> {
    let mut v = vec()
    for i in 0.upto(str_len(s)) {
        v.push(Str { chars: vec(s.get(i).unwrap()) })
    }
    v
}

fn clone(x: Int) -> Int {
    let mut y = x
    y
}

    Float(n)
}

fn trunc(x: Float) -> Int {
    Int(x.n)
}
