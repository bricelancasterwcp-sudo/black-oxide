struct Reading { label: Str, values: Vec<Int> }

fn max_value(v: Vec<Int>) -> Int {
    let mut max = -1
    for x in v {
        if x > max {
            max = x
        }
    }
    max
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 5), 3), 9), 1), values: push(vec(), 7) } // Corrected push sequence
    print(max_value(r.values))
}
