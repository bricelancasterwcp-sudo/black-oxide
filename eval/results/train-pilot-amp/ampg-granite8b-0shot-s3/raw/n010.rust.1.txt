```rust
fn main() {
    let n = 91;
    let mut is_prime = true;

    if n <= 1 {
        is_prime = false;
    } else {
        for i in 2..(n as f64).sqrt() as usize + 1 {
            if n % i == 0 {
                is_prime = false;
                break;
            }
        }
    }

    println!("{}", if is_prime { "yes" } else { "no" });
}
```
