fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    print_str(match second(push(push(vec(), 7), 8)) {
        Some(n) => int_to_str(n),
        None => "none".clone(),
    });
    print_str(match second(push(push(vec(), 8), 9)) {
        Some(n) => int_to_str(n),
        None => "none".clone(),
    });
    print_str(match second(vec()) {
        Some(n) => int_to_str(n),

    });
}
