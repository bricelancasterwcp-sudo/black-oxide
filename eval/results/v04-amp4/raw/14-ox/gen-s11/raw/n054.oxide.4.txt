struct LetterCount { vowels: Int, consonants: Int }

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    let mut count = LetterCount { vowels: 0, consonants: 0 }
    
    for c in chars(keyboard) {
        if is_vowel(c.clone()) {
            count.vowels += 1
        } else {
            count.consonants += 1
        }
    }
    
    print(count.vowels)
    print(count.consonants)
}

fn is_vowel(c: Str) -> Bool {
    match c {
        "a" => true,
        "e" => true,
        "i" => true,
        "o" => true,
        "u" => true,
        _ => false
    }
}
