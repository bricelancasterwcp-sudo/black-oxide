```rust
struct Device {
    sensor: Sensor,
}

struct Sensor {
    reading: Int,
}

fn set_reading(device: Device, value: Int) -> Device {
    let new_device = Device {
        sensor: Sensor { reading: value },
    };
    new_device
}

fn main() {
    let initial_device = Device {
        sensor: Sensor { reading: 0 },
    };
    let updated_device = set_reading(initial_device, 7);
    print(updated_device.sensor.reading);
}
```
