fn main() {
    let values = vec![Some(7), Some('x'), Some(9), None];
    let mut sum = 0;
    let mut count_not_num = 0;

    for v in values {
        match v {
            Some(n @ (7 | 8 | 9)) => sum += n,
            Some(_) => count_not_num += 1,
            None => count_not_num += 1,
        }
    }

    print(sum);
    print(count_not_num);
}
