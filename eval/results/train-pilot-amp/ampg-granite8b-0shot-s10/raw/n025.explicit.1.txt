fn parse_int(s: Str) -> Option<Int> {
    let mut it = chars(s).iter()
    let negate = it.next() == Some("-")
    let digits = it.take_while(is_digit).collect()
    if digits.is_empty() {
        None
    } else {
        Some(if negate { -int_from_str(&digits) } else { int_from_str(&digits) })
    }
}

fn is_digit(c: Str) -> Bool {
    c >= "0" && c <= "9"
}

fn int_from_str(s: Str) -> Int {
    s.chars().map(int_from_char).fold(0, add)
}

fn int_from_char(c: Str) -> Int {
    c.to_byte() - b0
}

fn add(a: Int, b: Int) -> Int {
    a + b
