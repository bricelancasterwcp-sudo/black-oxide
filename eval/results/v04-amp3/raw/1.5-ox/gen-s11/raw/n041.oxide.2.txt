```rust
struct Measurement { value: Int }

fn main() {
    let readings = vec![
        Measurement { value: 9 },
        Measurement { value: 2 },
        Measurement { value: 7 },
        Measurement { value: 4 }
    ];

    for reading in readings.sort_by_key(|m| m.value) {
        println!("{}", reading.value);
    }
}
```
