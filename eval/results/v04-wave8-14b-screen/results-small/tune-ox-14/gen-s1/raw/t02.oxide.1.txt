fn collatz(n: Int) -> Int {
    let mut count = 0
    let mut current = n
    while current != 1 {
        if current % 2 == 0 {
            current = current / 2
        } else {
            current = current * 3 + 1
        }
        count += 1
    }
    count
}
fn main() {
    print(collatz(27))
}
