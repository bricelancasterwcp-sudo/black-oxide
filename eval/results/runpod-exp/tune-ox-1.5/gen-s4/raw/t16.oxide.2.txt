fn div(x: Int, y: Int) -> Result<Int, Str> {
    if y == 0 {
        Err("division by zero")
    } else {
        Ok(x / y)
    }
}
fn main() {
    match div(100, 5) {
        Ok(q) => print(q),
        Err(e) => print_str(e),
    }
    match div(100, 0) {
        Ok(q) => print(q),
        Err(e) => print_str(e),
    }
    match div(100, 4) {
        Ok(q) => print(q),
        Err(e) => print_str(e),
    }
}
