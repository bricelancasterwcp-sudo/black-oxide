fn main() {
    let words = ["rotor", "banana", "level", "quiet", "deified", "system", "civic"];
    let mut palindromes = 0;
    for &word in words.iter() {
        println!("{}", word);
        println!("{}", word.len());
        let reversed: String = word.chars().rev().collect();
        println!("{}", reversed);
        let is_palindrome = word == reversed;
        println!("{}", is_palindrome);
        let vowel_count = word.chars().filter(|&c| "aeiou".contains(c)).count();
        println!("{}", vowel_count);
        let unique_chars: std::collections::HashSet<char> = word.chars().collect();
        println!("{}", unique_chars.len());
        if is_palindrome {
            palindromes += 1;
        }
    }
    println!("{}", palindromes);
}
