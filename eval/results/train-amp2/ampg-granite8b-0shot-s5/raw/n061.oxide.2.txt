struct Rectangle { length: Int, width: Int }

fn area(r: Rectangle) -> Int {
    r.length * r.width
}

fn circumference(r: Rectangle) -> Int {
    r.length * 2 + r.width * 2
}

fn main() {
    let rectangle = Rectangle { length: 9, width: 3 }
    print(area(rectangle))
    print(circumference(rectangle))
}
