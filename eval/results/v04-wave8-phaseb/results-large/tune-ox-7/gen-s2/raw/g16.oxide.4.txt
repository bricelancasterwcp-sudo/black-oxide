fn main() {
    let mut n = 44
    let mut seen = vec()
    let mut max_val = -1
    let mut min_val = -1
    let mut sum_vals = 0
    let mut even_rounds = 0
    let mut digit_sum = 0
    let mut reached_one = false
    for i in range(0, 50) {
        seen = push(seen, n)
        if max_val == -1 || n > max_val { max_val = n }
        if min_val == -1 || n < min_val { min_val = n }
        sum_vals += n
        let mut next_n = 0
        for c in chars(int_to_str(n)) {
            let d = parse_int(c).unwrap_or(-1)
            digit_sum += d
            next_n += d * d
        }
        if next_n == 1 {
            reached_one = true
            break
        }
        n = next_n
    }
    print(len(seen))
    print(n)
    let last_val = get(seen, len(seen) - 2).unwrap_or(-1)
    print(last_val)
    print(max_val)
    print(min_val)
    print(sum_vals)
    for x in seen {
        if x % 2 == 0 { even_rounds += 1 }
    }
    print(even_rounds)
    print(digit_sum)
    print(reached_one)
}
