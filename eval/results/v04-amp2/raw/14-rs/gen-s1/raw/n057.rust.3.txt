```rust
fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut max_name = "";
    let mut max_length = 0;

    for name in names {
        if name.len() > max_length {
            max_length = name.len();
            max_name = name;
        }
    }

    println!("{} has the most letters: {}", max_name, max_length);
}
```
