fn is_palindrome(s: Str) -> Bool {
    let len = str_len(s)
    for i in range(0, len / 2) {
        if get(s, i) != get(s, len - i - 1) {
            return false
        }
    }
    true
}

fn main() {
    print(is_palindrome("level"))   // true
    print(is_palindrome("stereo"))  // false
    print(is_palindrome("rotor"))   // true
}
