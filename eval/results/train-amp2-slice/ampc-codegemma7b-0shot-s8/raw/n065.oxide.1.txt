fn main() {
    let readings = vec(6, 11, 3, 14, 9)
    let mut above_five = 0
    let mut highest = 0
    for reading in readings {
        if reading > 5 {
            above_five = above_five + 1
        }
        if reading > highest {
            highest = reading
        }
    }
    print(above_five)
    print(highest)
}
