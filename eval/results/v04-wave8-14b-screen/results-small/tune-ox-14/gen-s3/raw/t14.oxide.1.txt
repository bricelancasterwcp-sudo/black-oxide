fn count_s(s: Str) -> Int {
    let mut count = 0
    for c in chars(s) {
        if c == "s" {
            count += 1
        }
    }
    count
}
fn distinct_chars(s: Str) -> Int {
    let mut seen = vec()
    for c in chars(s) {
        if !contains(seen, c) {
            seen = push(seen, c)
        }
    }
    len(seen)
}
fn main() {
    let text = "mississippi"
    print(count_s(text))
    print(distinct_chars(text))
}
