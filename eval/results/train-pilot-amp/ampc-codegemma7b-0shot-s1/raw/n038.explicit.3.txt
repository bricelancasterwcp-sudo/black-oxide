fn print(x: Int) {
    print_str(int_to_str(x))
}

fn int_to_str(n: Int) -> Str {
    if n == 0 {
        return "0"
    }
    let mut digits = vec()
    let mut n = n
    while n != 0 {
        digits = push(digits, n % 10)
        n = trunc(n / 10)
    }
    let mut result = ""
    for d in &digits {
        result = concat(result, int_to_str(d))
    }
    result
}

fn print_str(s: Str) {
    for c in &chars {
        print(c)
    }
}

fn chars(s: Str) -> Vec<Str> {
    if s == "" {
        return vec()
    }
    let mut result = vec(s.chars().next().unwrap())
    for c in s.chars().skip(1) {
        result = push(result, c)
    }
    result
}

fn push(v: Vec<T>, x: T) -> Vec<T> {
    let mut v = v
    v.push(x)
    v
}

fn vec() -> Vec<T> {
    Vec()
}

fn trunc(x: Int) -> Int {
    if x >= 0 {
        return x
    }
    return x + 1
}

fn concat(a: Str, b: Str) -> Str {
    let mut result = a
    for c in &chars {
        result = concat(result, c)
    }
    result
}
