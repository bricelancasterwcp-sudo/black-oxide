```rust
struct Device {
    sensor: Sensor,
}

struct Sensor {
    reading: Int,
}

fn main() {
    let device = Device { sensor: Sensor { reading: 0 } };
    let updated_device = Device {
        sensor: Sensor {
            reading: 7,
            ..device.sensor
        },
    };
    print(updated_device.sensor.reading)
}
```
