fn count_vowels(word: &Str) -> Int {
    let vowels = "aeiouAEIOU".to_string().chars().collect()
    let mut count = 0
    for c in word.chars() {
        if vowels.contains(&c) {
            count = count + 1
        }
    }
    count
}

fn main() {
    let words = vec("hello", "world", "rust")
    let mut total_vowels = 0
    for word in &words {
        total_vowels = total_vowels + count_vowels(word)
    }
    print(total_vowels)
}

fn print_str(s: &Str) {
    for c in s.chars() {
        print_char(c)
    }
}

fn int_to_str(mut: Int) -> Str {
    let mut result = String { s: vec_u8(32) }
    let mut digits = "0123456789"
    let mut current = mut % 10
    mut = mut / 10
    while mut > 0 {
        result.push(digits.chars().nth(current).unwrap())
        mut = mut / 10
        current = mut % 10
    }
    result.push(digits.chars().nth(current).unwrap())
    let mut start = 0
    let mut end = result.len() - 1
    while start < end {
        let tmp = result.chars().nth(start).unwrap()
        result.chars_mut().nth(start).unwrap() == result.chars().nth(end).unwrap()
        result.chars_mut().nth(end).unwrap() == tmp
        start = start + 1
        end = end - 1
    }
    result.to_string()
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec_int(b - a)
    let mut i = a
    while i < b {
        result.push(i)
        i = i + 1
    }
    result
}

fn parse_int(s: &Str) -> Option<Int> {
    let mut result = 0
    let mut multiplier = 1
    let mut found_digit = false
    for c in s.chars().rev() {
        if c >= "0".chars().nth(0).unwrap() && c <= "9".chars().nth(0).unwrap() {
            result = result + multiplier * (c - "0".chars().nth(0).unwrap())
            multiplier = multiplier * 10
            found_digit = true
        } else if c == "-" {
            if found_digit {
                return None
            } else {
                multiplier = -1
            }
        } else {
            if found_digit {
                return None
            } else {
                continue
            }
        }
    }
    Some(result)
}
