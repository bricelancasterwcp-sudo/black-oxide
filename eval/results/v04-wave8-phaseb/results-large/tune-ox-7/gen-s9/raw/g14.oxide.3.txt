fn main() {
    let v = vec(4, 4, 4, 7, 7, 2, 9, 9, 9, 9, 1, 1, 5)
    let n = len(v)
    print(n)
    let mut longest_streak = 0
    let mut current_value = -1
    let mut current_streak = 0
    for x in v {
        if x == current_value {
            current_streak += 1
        } else {
            if current_streak > longest_streak {
                longest_streak = current_streak
            }
            current_value = x
            current_streak = 1
        }
    }
    print(longest_streak)
    print(current_value)
    let unique_values = count_unique(v)
    print(unique_values)
    let mut first_repetition = -1
    for i in range(n) {
        if contains(v, v[i]) && i != find(v, v[i]).unwrap() {
            first_repetition = i
            break
        }
    }
    print(first_repetition)
    let mut never_decreases = true
    for i in range(1, n) {
        if v[i] < v[i - 1] {
            never_decreases = false
            break
        }
    }
    print(never_decreases)
}
