struct Square { side: Int }
struct Circle { radius: Float }

fn area_square(s: Square) -> Float {
    s.side.to_float() * s.side.to_float()
}

fn area_circle(c: Circle) -> Float {
    3.0 * c.radius * c.radius
}

fn main() {
    let square = Square { side: 4 };
    print(area_square(square));
}
