```rust
fn main() {
    let numbers = vec![7, 10, 9, ?];
    let mut sum = 0;
    let mut count = 0;

    for &num in &numbers {
        if num.is_integer() && num >= 0 {
            sum += num as i32;
            count += 1;
        }
    }

    println!("Sum of numbers: {}", sum);
    println!("Count of numbers: {}", count);
}
```
