fn reverse_digits(n: Int) -> Vec<Int> {
    let mut v = vec()
    let mut num = n
    while num > 0 {
        let digit = num % 10
        v = push(v, digit)
        num = trunc(num / 10)
    }
    v
}

fn main() {
    let number = 1234
    let reversed_digits = reverse_digits(number)
    for d in reversed_digits {
        print(d)
    }
}
