fn main() {
    let vowels = "aeiouAEIOU".chars().collect_vec().into_iter().fold(0, vowel_count(&vowels, _))
    let consonants = 26 - vowels
    print(vowels)
    print(consonants)
}

fn vowel_count(vowels: &Vec<Char>, c: Char) -> Int {
    if vowels.contains(&c) {
        1
    } else {
        0
    }
}

    let mut vec = Vec()
    vec.push_back(c)
    vec
}
