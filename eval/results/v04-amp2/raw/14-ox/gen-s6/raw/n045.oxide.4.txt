struct Log { entries: Vec<Int> }

fn first_entry(log: Log) -> Option<Int> {
    let first = get(log.entries, 0)?
    Some(first)
}

fn last_entry(log: Log) -> Option<Int> {
    let len = len(log.entries)
    if len == 0 {
        return None
    }
    let last = get(log.entries, len - 1)?
    Some(last)
}

fn main() {
    let log = Log { entries: vec(6, 7, 8, 9) }
    print(first_entry(log.clone())?)
    let log2 = Log { entries: vec(6, 7, 8, 9) }
    print(last_entry(log2.clone())?)
}
