```rust
struct Rectangle {
    width: Int,
    height: Int,
}

impl Rectangle {
    fn area(&self) -> Int {
        self.width * self.height
    }

    fn perimeter(&self) -> Int {
        2 * (self.width + self.height)
    }
}

fn main() {
    let rect = Rectangle { width: 9, height: 3 };
    print_int(rect.area());
    print_int(rect.perimeter());
}
```
