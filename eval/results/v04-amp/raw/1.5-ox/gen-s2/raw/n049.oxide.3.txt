```rust
struct Figure {
    name: Str,
    size: Int,
}

fn main() {
    let figures = vec![
        Figure { name: "circle", size: 5 },
        Figure { name: "square", size: 10 },
        Figure { name: "triangle", size: 7 },
        Figure { name: "rectangle", size: 8 },
    ];

    let smallest = figures.iter().min_by_key(|f| f.size).unwrap().size;
    let largest = figures.iter().max_by_key(|f| f.size).unwrap().size;

    print(smallest);
    print_str(" ");
    print(largest - smallest);
}
```
