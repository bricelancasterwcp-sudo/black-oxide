struct Reading { label: Str, values: Vec<Int> }

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    let found = loop_find(r.values, 10)
    print(found)
}

fn loop_find(v: Vec<Int>, target: Int) -> Int {
    let mut found = -1
    for x in v {
        if x > target {
            break
        }
        if x % 3 == 0 || x % 5 == 0 {
            found = x
        }
    }
    found
}
