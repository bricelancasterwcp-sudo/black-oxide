fn main() {
    let s = Shape { side: 4.0, variant: Variant(Nullary) }
    match s {
        Square(side) => {
            let pi = 3.0
            print(pi * side * side)
        },
        _ => {
            print(0.0)
        },
    }
}

enum Variant {
    Nullary,
    Variant(Type),
}

enum Shape {
    Square(Float),
    Circle(Float),
}

fn area(s: &Shape) -> Float {
    match s {
        Square(side) => {
            let pi = 3.0
            pi * side * side
        },
        Circle(radius) => {
            let pi = 3.0
            pi * radius * radius
        },
    }
}

fn print(x: Float) {
    drop x
}

fn main() {
    let s = Square { side: 4.0 }
    print(area(&s))
    drop s
}

struct Square {
    side: Float,
    variant: Variant,
}

enum Variant {
    Nullary,
    Variant(Type),
}

fn area(s: &Square) -> Float {
    let pi = 3.0
    pi * s.side * s.side
}
