```rust
fn main() {
    let input = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in input.chars() {
        if let Some(digit) = c.to_digit(10) {
            sum += digit as u32; // Ensure sum is of type u32
            count += 1;
        }
    }

    println!("Total of all digit characters: {}", sum);
    println!("Number of digit characters: {}", count);
}
```
