fn int_to_str(n: Int) -> Str {
    let mut digits = vec()
    let mut num = n
    while num > 0 {
        digits.push(num % 10)
        num = num / 10
    }
    let mut result = str()
    for digit in digits {
        result = concat(result, int_to_str(digit))
    }
    result
}

fn concat(a: Str, b: Str) -> Str {
    let mut result = str()
    for c in a {
        result = concat(result, c)
    }
    for c in b {
        result = concat(result, c)
    }
    result
}
