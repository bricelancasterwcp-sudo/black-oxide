struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max_value = None
    for x in v {
        if max_value? < x {
            max_value = Some(x)
        }
    }
    max_value
}

fn main() {
    let readings = push(push(push(push(push(vec(), 6), 11), 3), 14), 9)
    let count = count_exceeding(readings, 5)
    print(count)
    let max_value = max_reading(readings)?
    print(max_value)
}
