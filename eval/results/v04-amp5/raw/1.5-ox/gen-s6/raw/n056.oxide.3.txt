fn count_numbers(vec: Vec<Option<i32>>) -> (i32, i32) {
    let mut sum = 0;
    let mut non_counted = 0;
    for v in vec {
        match v {
            Some(n) => {
                sum += n;
            }
            None => {
                non_counted += 1;
            }
        }
    }
    (sum, non_counted)
}

fn main() {
    let vec: Vec<Option<i32>> = vec![Some(7), Some(42), Some(9), Some(?)];
    let (total, count) = count_numbers(vec);
    println!("Total: {}", total);
    println!("Count of numbers: {}", count);
}
