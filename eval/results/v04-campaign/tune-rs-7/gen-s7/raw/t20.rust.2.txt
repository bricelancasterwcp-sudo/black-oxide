fn main() {
    let mut total_cost = 0;
    let items = vec![
        ("pen", 3, 4),
        ("pad", 9, 2),
        ("ink", 5, 3),
    ];
    for item in &items {
        let cost = item.1 * item.2;
        total_cost += cost;
        println!("{}: {}", item.0, cost);
    }
    println!("Total: {}", total_cost);
    let mut max_cost_item = "";
    let mut max_cost = 0;
    for item in &items {
        if item.1 * item.2 > max_cost {
            max_cost_item = item.0;
            max_cost = item.1 * item.2;
        }
    }
    println!("Highest cost item: {}", max_cost_item);
}
