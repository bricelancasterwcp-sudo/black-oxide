fn main() {
    let measurements = vec(9, 2, 7, 4).sort_by_key(measure_temparature)
    for m in measurements {
        print(m)
    }
}

fn measure_temparature(x: Int) -> Int {
    x * 3 / 2
}

fn sort_by_key(v: Vec<Int>, f: Int) -> Vec<Int> {
    push(v, f)
}

fn print(x: Int) {
    print_str(str_cat(str_cat("Temparature ", int_to_str(x)), " degrees.\n"))
}

fn str_cat(a: Str, b: Str) -> Str {
    concat(a, b)
}

fn int_to_str(n: Int) -> Str {
    trunc(to_float(n))
}

fn to_float(n: Int) -> Float {
    n
}

fn trunc(x: Float) -> Int {
    x
}

}
