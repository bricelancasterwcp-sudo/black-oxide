```
struct Line { units: Int, price: Int }

fn calculate_value(line: Line) -> Int {
    line.units * line.price
}

fn main() {
    let line1 = Line { units: 4, price: 3 }
    let line2 = Line { units: 5, price: 2 }
    
    let value1 = calculate_value(line1)
    let value2 = calculate_value(line2)
    
    print(value1)  // prints: 12
    print(value2)  // prints: 10
    
    let total_value = value1 + value2
    print(total_value)  // prints: 22
}
```
