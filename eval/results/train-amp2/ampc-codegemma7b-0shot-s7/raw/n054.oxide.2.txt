fn is_vowel(c: Str) -> Bool {
    match c {
        Str(c) => match c {
            _ => false,
            _ => true,
        },
        _ => false,
    }
}

fn main() {
    let s = "keyboardletter".chars().collect()
    let mut vowels = 0
    let mut not_vowels = 0
    for c in s {
        if is_vowel(c) {
            vowels = vowels + 1
        } else {
            not_vowels = not_vowels + 1
        }
    }
    print(vowels)
    print(not_vowels)
}
