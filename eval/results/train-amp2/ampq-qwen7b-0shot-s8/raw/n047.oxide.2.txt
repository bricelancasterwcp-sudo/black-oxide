struct List { values: Vec<Int> }

fn push_back(list: List, value: Int) -> List {
    List(values == push(list.values, value))
}

fn main() {
    let mut list = List { values: vec() }
    for i in range(1, 11) {
        if i % 2 == 0 {
            list = push_back(list, i)
        }
    }
    for i in range(len(list.values) - 1, -1, -1) {
        print(get(list.values, i))
    }
}
