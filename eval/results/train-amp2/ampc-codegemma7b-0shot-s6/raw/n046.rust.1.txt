use std::collections::HashSet;

fn main() {
    let samples = vec![5, 12, 3, 18, 9];
    let cutoff = 10;

    let (under_cutoff, at_or_over_cutoff) = samples
        .iter()
        .fold((0, 0), |(mut under, mut at_or_over), &sample| {
            if sample < cutoff {
                under += 1;
            } else {
                at_or_over += 1;
            }
            (under, at_or_over)
        });

    println!("Under cutoff: {}", under_cutoff);
    println!("At or over cutoff: {}", at_or_over_cutoff);
}
