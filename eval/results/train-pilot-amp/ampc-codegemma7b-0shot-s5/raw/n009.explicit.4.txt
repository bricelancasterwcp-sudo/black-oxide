fn print_reverse(x: Int) {
    if x >= 1000 {
        print((x / 1000) % 10)
        print((x / 100) % 10)
        print((x / 10) % 10)
        print(x % 10)
    } else if x >= 100 {
        print((x / 100) % 10)
        print((x / 10) % 10)
        print(x % 10)
    } else if x >= 10 {
        print((x / 10) % 10)
        print(x % 10)
    } else {
        print(x)
    }
    drop x // drop should be the last use of the last
}

fn main() {
    let num = 1234
    print_reverse(num)
    drop num // drop should be the last use of the last
}
