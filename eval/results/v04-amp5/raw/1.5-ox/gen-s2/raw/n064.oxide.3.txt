fn main() {
    let list = vec![10, 20];

    if let Some(pos1) = get(list, 0) {
        println!("{}", pos1);
    }

    if let Some(pos4) = get(list, 3) {
        println!("{}", pos4);
    }
}

fn get<T>(vec: Vec<T>, index: Int) -> Option<T> {
    let length = vec.len();
    if index < length {
        Some(vec[index])
    } else {
        None
    }
}
