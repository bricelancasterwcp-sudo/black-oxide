fn read_and_print() {
    if let_ok(num) == parse_int("abc") {
        print(num)
    } else {
        print("bad")
    }
}

fn parse_int(s: Str) -> Result<Int, Str> {
    match s.parse() {
        Ok(num) => Ok(num),
        Err(_) => Err(s),
    }
}

fn print(x: Int) {
    println(x)
}

fn print_str(s: Str) {
    println(s)
}

fn main() {
    read_and_print()
}
