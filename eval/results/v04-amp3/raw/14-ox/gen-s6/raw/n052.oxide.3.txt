fn main() {
    let word = "banana"
    let chars = chars(word)
    for i in range(0, len(chars)) {
        let c = get(chars, i).unwrap_or("".to_string())
        if c == "n" {
            print(i)
        }
    }
}

fn enumerate(v: Vec<Str>) -> Vec<(Int, Str)> {
    let mut acc = vec()
    for i in range(0, len(v)) {
        acc = push(acc, (i, get(v, i).unwrap_or("".to_string())))
    }
    acc
}
