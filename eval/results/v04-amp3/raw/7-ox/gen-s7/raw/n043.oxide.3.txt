struct Queue { items: Vec<Int> }

fn new_queue() -> Queue {
    Queue { items: vec![1, 2, 3, 4, 5] }
}

fn swap_first_last(q: Queue) -> Queue {
    let mut q = q;
    if let Some(first) = get(q.items.clone(), 0) {
        if let Some(last_index) = len(q.items).checked_sub(1) {
            if let Some(last) = get(q.items.clone(), last_index) {
                std::mem::swap(&mut q.items[0], &mut q.items[last_index]);
            }
        }
    }
    q
}

fn print_queue(q: Queue) {
    for item in q.items {
        print(item);
        print_str("\n");
    }
}

fn main() {
    let queue = new_queue();
    let swapped_queue = swap_first_last(queue);
    print_queue(swapped_queue);
}
