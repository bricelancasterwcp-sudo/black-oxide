struct Warehouse { name: Str, parts: Vec<Part> }
struct Part { name: Str, quantity: Int, price: Int }
fn total_value(parts: Vec<Part>) -> Int {
    let mut sum = 0
    for p in parts {
        sum += p.quantity * p.price
    }
    sum
}
fn count_below_ten(parts: Vec<Part>) -> Int {
    let mut count = 0
    for p in parts {
        if p.quantity < 10 {
            count += 1
        }
    }
    count
}
fn max_value_part(parts: Vec<Part>) -> Option<Str> {
    let mut max_val = -1
    let mut max_name = ""
    for p in parts {
        let value = p.quantity * p.price
        if value > max_val {
            max_val = value
            max_name = p.name
        }
    }
    Some(max_name)
}
fn main() {
    let north_parts = push(push(push(vec(), Part { name: "bolt", quantity: 40, price: 3 }), Part { name: "hinge", quantity: 8, price: 25 }), Part { name: "plate", quantity: 15, price: 12 })
    let south_parts = push(push(push(push(push(vec(), Part { name: "clamp", quantity: 6, price: 45 }), Part { name: "rod", quantity: 22, price: 9 }), Part { name: "washer", quantity: 100, price: 1 }), Part { name: "gasket", quantity: 4, price: 30 }))
    let north = Warehouse { name: "North", parts: north_parts }
    let south = Warehouse { name: "South", parts: south_parts }
    print(north.name)
    print(total_value(north.parts))
    print(count_below_ten(north.parts))
    match max_value_part(north.parts) {
        Some(name) => print_str(name),
        None => print_str(""),
    }
    print(south.name)
    print(total_value(south.parts))
    print(count_below_ten(south.parts))
    match max_value_part(south.parts) {
        Some(name) => print_str(name),
        None => print_str(""),
    }
    if total_value(north.parts) > total_value(south.parts) {
        print_str(north.name)
    } else {
        print_str(south.name)
    }
}
