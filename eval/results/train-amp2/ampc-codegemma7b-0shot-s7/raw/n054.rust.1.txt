```rust
fn is_vowel(c: char) -> bool {
    match c {
        'a' | 'e' | 'i' | 'o' | 'u' => true,
        _ => false,
    }
}

fn main() {
    let word = "keyboard";

    let mut vowels = 0;
    let mut consonants = 0;

    for c in word.chars() {
        if is_vowel(c) {
            vowels += 1;
        } else {
            consonants += 1;
        }
    }

    println!("Vowels: {}", vowels);
    println!("Consonants: {}", consonants);
}
```
