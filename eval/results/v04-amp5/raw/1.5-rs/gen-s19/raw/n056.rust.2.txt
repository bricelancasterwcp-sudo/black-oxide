```rust
fn main() {
    let text = "7, x, 9, ?";
    
    let mut sum_of_numbers: u32 = 0;
    let mut count_of_numbers: usize = 0;

    for c in text.chars() {
        if !c.is_numeric() || c == '?' {
            continue;
        }

        let num = c.to_digit(10).unwrap();
        sum_of_numbers += num;
        count_of_numbers += 1;
    }

    println!("Sum of numbers: {}", sum_of_numbers);
    println!("Count of numbers: {}", count_of_numbers);
}
```
