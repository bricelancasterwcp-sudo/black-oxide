fn main() {
    let word = "keyboard";
    let mut vowel_count = 0;
    let mut consonant_count = 0;

    for c in word.chars() {
        if is_vowel(c) {
            vowel_count += 1;
        } else if c.is_alphabetic() {
            consonant_count += 1;
        }
    }

    println!("Vowels: {}", vowel_count);
    println!("Consonants: {}", consonant_count);
}

fn is_vowel(c: char) -> bool {
    matches!(c.to_ascii_lowercase(), 'a' | 'e' | 'i' | 'o' | 'u')
}
