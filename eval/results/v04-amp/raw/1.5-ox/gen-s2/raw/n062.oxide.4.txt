```rust
fn main() {
    let mut signal = "green";
    while true {
        match signal {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => panic!("unexpected signal"),
        }
        // Simulate receiving a new signal
        signal = match rand::random::<u8>() % 3 {
            0 => "green",
            1 => "red",
            2 => "yellow",
            _ => unreachable!(),
        };
    }
}
```
