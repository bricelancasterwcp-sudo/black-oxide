struct Log { entries: Vec<Int> }

fn first_entry(log: Log) -> Int {
    get(log.entries, 0).unwrap_or(-1)
}

fn last_entry(log: Log) -> Int {
    let len = len(log.entries)
    if len == 0 {
        -1
    } else {
        get(log.entries, len - 1).unwrap_or(-1)
    }
}

fn main() {
    let log = Log { entries: push(push(push(push(vec(), 6), 7), 8), 9) }
    print(first_entry(log))
    let log2 = Log { ..log }
    print(last_entry(log2))
}
