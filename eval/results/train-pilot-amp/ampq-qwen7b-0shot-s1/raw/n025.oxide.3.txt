struct Reading { label: Str, value: Str }

fn parse_int(s: Str) -> Option<Int> {
    match s.parse_int() {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn main() {
    let r = Reading { label: "num", value: "417" }
    match parse_int(r.value) {
        Some(n) => print(n + 1),
        None => print_str("bad"),
    }
}
