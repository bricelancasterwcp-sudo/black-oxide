struct Point { x: Int, y: Int }

fn is_even(n: Int) -> Bool {
    n % 2 == 0
}

fn main() {
    let mut x = 7
    while x > 0 {
        print(x)
        print("\n")
        x = x - 2
    }
    let sum = for_each(range(1, 11), add(0))
    print(sum)
}

fn for_each(v: Vec<Int>, f: Int, acc: Int) -> Int {
    match v {
        Vec(e, rest) => {
            let acc2 = f(acc, e)
            for_each(rest, f, acc2)
        },
        Nullary => acc,
    }
}

fn add(a: Int, b: Int) -> Int {
    a + b
}

fn abs(n: Int) -> Int {
    if n > 0 {
        n
    } else {
        -n
    }
}

fn main() {
    print(add(1, 2))
    print(abs(-3))
    let x = Point { x: 1, y: 2 }
    print(x.x + x.y)
}
