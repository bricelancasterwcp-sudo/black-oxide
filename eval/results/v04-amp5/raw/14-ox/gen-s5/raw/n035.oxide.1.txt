struct DivisionResult {
    label: Str,
    result: Option<Float>
}

fn divide(a: Int, b: Int) -> DivisionResult {
    if b == 0 {
        DivisionResult { label: "Error", result: None }
    } else {
        let res = to_float(a) / to_float(b)
        DivisionResult { label: "Success", result: Some(res) }
    }
}

fn main() {
    let twenty = 20
    let four = 4
    let div_result = divide(twenty, four)
    
    match div_result.result {
        Some(res) => print(res),
        None => print_str("bad")
    }
}
