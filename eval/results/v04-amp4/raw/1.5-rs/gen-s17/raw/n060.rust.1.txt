```rust
fn main() {
    let input = "a1b22";
    let mut digit_count = 0;
    let mut sum = 0;

    for c in input.chars() {
        if c.is_digit(10) {
            digit_count += 1;
            sum += c.to_digit(10).unwrap();
        }
    }

    println!("Digit count: {}", digit_count);
    println!("Sum of digits: {}", sum);
}
```
