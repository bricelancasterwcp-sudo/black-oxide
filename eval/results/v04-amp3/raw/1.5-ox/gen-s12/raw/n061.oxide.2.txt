```rust
struct Rectangle {
    width: Int,
    height: Int,
}

fn calculate_area(rectangle: Rectangle) -> Int {
    rectangle.width * rectangle.height
}

fn calculate_perimeter(rectangle: Rectangle) -> Int {
    2 * (rectangle.width + rectangle.height)
}

fn main() {
    let rect = Rectangle { width: 9, height: 3 };
    print(calculate_area(rect));            // field access copies: rect stays usable
    print(calculate_perimeter(rect));
}
```
