```rust
fn main() {
    let batch = vec![9, 2, 12, 5, 16];
    let mut kept: Vec<i32> = Vec::new();

    for &entry in &batch {
        if entry < 10 {
            kept.push(entry);
        }
    }

    for &entry in &kept {
        println!("{}", entry);
    }

    println!("Kept {} entries.", kept.len());
}
```
