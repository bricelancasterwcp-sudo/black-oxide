```rust
fn main() {
    let numbers = vec![7, 8, 9];
    if let Some(second) = second_entry(&numbers) {
        println!("The second entry is: {}", second);
    } else {
        println!("None");
    }
}

fn second_entry<T>(list: &[T]) -> Option<&T> {
    if list.len() > 1 {
        Some(&list[1])
    } else {
        None
    }
}
```
