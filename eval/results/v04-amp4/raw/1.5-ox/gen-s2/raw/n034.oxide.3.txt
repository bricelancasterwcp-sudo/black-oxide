fn second(v: Vec<Int>) -> Option<Int> {
    let x = v[1];
    Some(x)
}

fn main() {
    for i in vec![7, 8, 9] {
        match second(vec![(i - 1) as Int, i as Int, (i + 1) as Int]) {
            Some(n) => print_str(&n.to_string()),
            None => print_str("none"),
        }
    }
}
