fn is_prime(num: Int) -> Bool {
    if num <= 1 {
        return false
    }
    for i in range(2, num - 1) {
        if num % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let result = is_prime(91)
    if result {
        print("yes")
    } else {
        print("no")
    }
}
