```rust
fn main() {
    let nums = vec![5, 12, 3, 18, 9];
    let cutoff = 10;

    let mut count_below_cutoff = 0;
    let mut count_at_or_over_cutoff = 0;

    for &num in &nums {
        if num < cutoff {
            count_below_cutoff += 1;
        } else {
            count_at_or_over_cutoff += 1;
        }
    }

    println!("Numbers below the cutoff: {}", count_below_cutoff);
    println!("Numbers at or over the cutoff: {}", count_at_or_over_cutoff);
}
```
