```rust
fn main() {
    let fruits = ["fig", "banana", "kiwi"];
    let mut max_len = 0;
    let mut longest_fruit = "";

    for fruit in &fruits {
        if fruit.len() > max_len {
            max_len = fruit.len();
            longest_fruit = fruit;
        }
    }

    println!("The name with the most letters is: {}", longest_fruit);
    println!("Its letter count is: {}", max_len);
}
```
