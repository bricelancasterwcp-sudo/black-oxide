fn count_under_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x < cutoff {
            count += 1
        }
    }
    count
}

fn count_at_or_over_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= cutoff {
            count += 1
        }
    }
    count
}

fn main() {
    let values = vec!(5, 12, 3, 18, 9)
    let cutoff = 10
    print(count_under_cutoff(clone(values), cutoff))
    print(count_at_or_over_cutoff(clone(values), cutoff))
}
