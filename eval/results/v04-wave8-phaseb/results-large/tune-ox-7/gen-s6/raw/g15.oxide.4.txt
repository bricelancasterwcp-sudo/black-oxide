fn main() {
    let record = "name=widget;qty=17;price=245;tag=blue"
    let pairs = count(split(record, ";"))
    print(pairs)
    for pair in split(record, ";") {
        let parts = split(pair, "=")
        let key = get(parts, 0).unwrap_or("")
        let value = get(parts, 1).unwrap_or("")
        print_str(key)
        print_str(value)
    }
    let qty = parse_int(get(split(record, ";"), 1).unwrap_or("0")).unwrap_or(0)
    let price = parse_int(get(split(record, ";"), 2).unwrap_or("0")).unwrap_or(0)
    print(qty * price)
}
