struct List { data: Int, next: Option<Box<List>> }

fn collect_evens() -> Option<Box<List>> {
    let mut head: Option<Box<List>> = None
    for i in range(1, 11) {
        if i % 2 == 0 {
            head = Some(Box { data: i, next: head })
        }
    }
    head
}

fn print_list(head: Option<Box<List>>) {
    for mut in inhead {
        print(inn.data)
        inn = inn.next?
        print("\n")
    }
}

fn main() {
    let list = collect_evens()
    print_list(list)
}
