fn main() {
    let nums = push(push(vec(), 5), 40)
    let bigger = extend(clone(&nums), 70)
    print(sum_big(&nums, 10))
    drop nums
    print(sum_big(&bigger, 10))
    drop bigger
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn vec() -> Vec<T> {
    Vec
}

fn push(v: Vec<T>, x: T) -> Vec<T> {
    v.push(x)
}

fn clone(x: T) -> T {
    x
}

fn print(x: T) {
    println(x)
}

fn print_str(s: Str) {
    println(s)
}

fn len(v: Vec<T>) -> Int {
    v.len()
}

fn get(v: Vec<T>, i: Int) -> Option<T> {
    v.get(i)
}

fn range(a: Int, b: Int) -> Vec<Int> {
    a.range_to(b - 1).collect()
}

fn str_len(s: Str) -> Int {
    s.len()
}

fn concat(a: Str, b: Str) -> Str {
    a + b
}

fn chars(s: Str) -> Vec<Str> {
    s.chars().map(String.from_char).collect()
}

fn int_to_str(n: Int) -> Str {
    n.to_string()
}

fn parse_int(s: Str) -> Option<Int> {
    s.parse().ok()
}

fn to_float(n: Int) -> Float {
    n.to_float()
}

fn trunc(x: Float) -> Int {
    x.trunc()
}
