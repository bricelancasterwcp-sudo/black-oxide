struct Number { value: Int }

fn sum_numbers(numbers: Vec<Number>) -> Int {
    let mut sum = 0
    for n in numbers {
        sum += n.value
    }
    sum
}

fn count_non_numbers(texts: Vec<Str>) -> Int {
    let mut count = 0
    for t in texts {
        if parse_int(t).is_none() {
            count += 1
        }
    }
    count
}

fn main() {
    let texts = vec("7", "x", "9", "?")
    let numbers: Vec<Number> = texts.iter().filter_map(|t| parse_int(*t).map(|n| Number { value: n })).collect()
    let non_numbers: Vec<Str> = texts.iter().filter(|t| parse_int(*t).is_none()).cloned().collect()

    print(sum_numbers(numbers))
    print(count_non_numbers(non_numbers))
}
