fn is_vowel(c: Str) -> Bool {
    match c {
        Str(c) => match c {
            _ => false,
            _ => match c {
                _ => match c {
                    _ => false,
                    _ => true,
                },
            },
        },
    }
}

fn main() {
    let keyboard = str("abcdefghijklmnopqrstuvwxyz")
    let mut vowels = 0
    let mut not_vowels = 0
    for c in keyboard.chars() {
        if is_vowel(Str(c)) {
            vowels = vowels + 1
        } else {
            not_vowels = not_vowels + 1
        }
    }
    print(vowels)
    print(not_vowels)
}
