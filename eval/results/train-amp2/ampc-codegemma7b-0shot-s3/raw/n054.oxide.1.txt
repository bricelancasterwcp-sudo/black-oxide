fn main() {
    let word = "keyboard".to_lowercase().chars().collect()
    let mut vowels = 0
    let mut not_vowels = 0
    for c in word {
        if matches(c, "aeiou".chars()) {
            vowels = vowels + 1
        } else {
            not_vowels = not_vowels + 1
        }
    }
    print(vowels)
    print(not_vowels)
}
