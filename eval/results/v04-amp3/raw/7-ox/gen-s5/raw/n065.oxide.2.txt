fn count_above_five(readings: Vec<Int>) -> Int {
    let mut count = 0;
    for r in readings.clone() {
        if r > 5 {
            count += 1;
        }
    }
    count
}

fn max_reading(readings: Vec<Int>) -> Option<Int> {
    let mut max = None;
    for r in readings {
        match &mut max {
            Some(m) => if r > *m { *m = r; },
            None => max = Some(r),
        }
    }
    max
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14), 9);
    print(count_above_five(readings.clone()));
    match max_reading(readings) {
        Some(m) => print(m),
        None => print_str("no readings"),
    }
}
