struct Line { items: Vec<(Int, Int)> }

fn total(line: &Line) -> Int {
    sum(map(line.items, |(price, count)| price * count))
}

fn main() {
    let line1 = Line { items: vec![(3, 4), (2, 5)] };
    print(total(&line1));
    print_str("\n");
    let line2 = Line { items: vec![(3, 4), (2, 5)] };
    print(total(&line2));
    print_str("\n");
    print(total(&Line { items: concat(line1.items.clone(), line2.items) }));
}
