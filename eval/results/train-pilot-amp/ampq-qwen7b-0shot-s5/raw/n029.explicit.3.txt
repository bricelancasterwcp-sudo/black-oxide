fn main() {
    let num1 = 1
    let num2 = 2
    let num3 = 3
    let text = format_str("One{}", num1).concat(format_str("Two{}", num2)).concat(format_str("Three{}", num3))
    print_str(&text)
    drop num1
    drop num2
    drop num3
    drop text
}

fn format_str(s: &Str, n: Int) -> Str {
    concat(s, int_to_str(n))
}

fn concat(a: Str, b: Str) -> Str {
    let mut result = a
    let chars = chars(b)
    for ch in chars {
        result = push(result, ch)
    }
    result
}

fn int_to_str(n: Int) -> Str {
    let mut chars = vec()
    let mut num = n
    if num == 0 {
        push(chars, push(vec(), 48))
    } else {
        while num != 0 {
            let digit = num % 10
            num = trunc(num / 10)
            push(chars, push(vec(), 48 + digit))
        }
    }
    let mut result = vec()
    let chars = chars.chars()
    for ch in chars {
        push(result, ch)
    }
    concat("".to_string(), chars)
}

fn print_str(s: &Str) {
    for ch in chars(s) {
        print(&ch)
    }
}

fn print(ch: &Char) {
    let mut chars = vec()
    push(chars, ch)
    let mut result = ""
    for ch in chars {
        result = push(result, ch)
    }
    println(result)
}

fn chars(s: &Str) -> Vec<Str> {
    let mut result = vec()
    for ch in s.chars() {
        push(result, push(vec(), ch))
    }
    result
}

fn push(v: Vec<Str>, x: Str) -> Vec<Str> {
    concat(v, push(vec(), x))
}

    }

}

fn println(s: &Str) {
    for ch in chars(s) {
        print(&ch)
    }
    print("\n")
}

fn str_len(s: &Str) -> Int {
    chars(s).len()
}

fn clone(x: &Str) -> Str {
    let mut result = vec()
    let chars = chars(x)
    for ch in chars {
        push(result, ch)
    }
    result
}

fn trunc(x: Float) -> Int {
    x.trunc() - x.fract()
}

fn int_to_float(n: Int) -> Float {
    n.to_float()
}

fn to_float(n: Float) -> Float {
    n
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec()
    for i in a + 1 {
        if i < b {
            push(result, i)
        }
    }
    result
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {

}

}

    let mut chars = vec()

    if num == 0 {
        push(chars, push(vec(), 48))
    } else {
        while num != 0 {
            let digit = num % 10

            push(chars, push(vec(), 48 + digit))
        }
