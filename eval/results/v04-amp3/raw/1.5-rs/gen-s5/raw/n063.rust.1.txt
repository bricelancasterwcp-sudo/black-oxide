fn main() {
    // Value of first line (four units at 3 each)
    let first_line_value = 4 * 3;
    
    // Value of second line (five units at 2 each)
    let second_line_value = 5 * 2;
    
    // Total value of both lines
    let total_value = first_line_value + second_line_value;
    
    // Print the values
    println!("First line: {}", first_line_value);
    println!("Second line: {}", second_line_value);
    println!("Total value: {}", total_value);
}
