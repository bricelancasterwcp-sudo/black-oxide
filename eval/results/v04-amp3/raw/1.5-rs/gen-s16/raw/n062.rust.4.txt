```rust
fn main() {
    loop {
        let signal = get_signal();
        match signal {
            "green" => println!("go for green"),
            "red" => println!("stop for red"),
            "yellow" => println!("wait for yellow"),
            _ => panic!("unexpected signal: {}", signal),
        }
    }
}

fn get_signal() -> &'static str {
    use rand::Rng;

    let mut rng = rand::thread_rng();
    if rng.gen_bool(0.3) {
        return "green";
    } else if rng.gen_bool(0.3) {
        return "red";
    } else {
        return "yellow";
    }
}
```
