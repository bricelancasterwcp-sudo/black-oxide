```rust
fn main() {
    let sample_values = [5, 12, 3, 18, 9];
    let cutoff = 10;

    let mut below_cutoff_count = 0;
    let mut above_or_equal_cutoff_count = 0;

    for &value in &sample_values {
        if value < cutoff {
            below_cutoff_count += 1;
        } else {
            above_or_equal_cutoff_count += 1;
        }
    }

    println!("Below cutoff: {}", below_cutoff_count);
    println!("Above or equal to cutoff: {}", above_or_equal_cutoff_count);
}
```
