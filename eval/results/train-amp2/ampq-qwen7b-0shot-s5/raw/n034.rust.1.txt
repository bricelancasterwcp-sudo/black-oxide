```rust
fn second_element<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}

fn main() {
    let vec7 = vec![1, 2, 3, 4, 5, 6, 7];
    let vec8 = vec![1, 2];
    let vec9: Vec<i32> = Vec::new();

    println!("{:?}", second_element(&vec7)); // Some(2)
    println!("{:?}", second_element(&vec8)); // Some(2)
    println!("{:?}", second_element(&vec9)); // None
}
```
