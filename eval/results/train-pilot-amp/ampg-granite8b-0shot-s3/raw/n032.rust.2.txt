fn main() {
    let p1 = Point { x: 1, y: 5 };
    let p2 = Point { x: 10, y: 5 };
    println!("{:?}", p1 + p2);
}

#[derive(Debug)]
struct Point {
    x: i32,
    y: i32,
}

impl std::ops::Add for Point {
    type Output = Point;

    fn add(self, rhs: Point) -> Self::Output {
        Point { x: self.x + rhs.x, y: self.y + rhs.y }
    }
}
