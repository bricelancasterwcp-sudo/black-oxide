fn main() {
    let numbers = vec![7];
    match get_second(&numbers) {
        Some(second) => println!("Second element: {}", second),
        None => println!("No second element"),
    }

    let numbers = vec![8];
    match get_second(&numbers) {
        Some(second) => println!("Second element: {}", second),
        None => println!("No second element"),
    }

    let numbers = vec![9];
    match get_second(&numbers) {
        Some(second) => println!("Second element: {}", second),
        None => println!("No second element"),
    }
}

fn get_second<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}
