```rust
fn main() {
    let (seven, x, nine, _) = (7, 10, 9, ?);

    let mut sum = 0;
    let mut count = 0;

    if seven.is_some() { sum += seven.unwrap(); count += 1; }
    if x.is_some() { sum += x.unwrap(); count += 1; }
    if nine.is_some() { sum += nine.unwrap(); count += 1; }

    println!("Sum: {}, Count: {}", sum, count);
}
```
