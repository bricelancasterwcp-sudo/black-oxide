struct Queue { items: Vec<Int> }

fn exchange_first_last(q: Queue) -> Option<Queue> {
    let len = len(q.items)
    if len < 2 {
        return Some(q)
    }
    let first = get(q.items, 0)?
    let last = get(q.items, len - 1)?
    let mut new_items = vec()
    push(new_items, last)
    for i in range(1, len - 1) {
        push(new_items, get(q.items, i)?)
    }
    push(new_items, first)
    Some(Queue { items: new_items })
}

fn main() {
    let q = Queue { items: push(push(push(push(push(vec(), 1), 2), 3), 4), 5) }
    match exchange_first_last(q) {
        Some(exchanged_q) => {
            for item in exchanged_q.items {
                print(item)
            }
        },
        None => print_str("Queue too short"),
    }
}
