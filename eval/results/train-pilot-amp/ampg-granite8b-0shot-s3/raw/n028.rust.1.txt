```rust
fn main() {
    let input = std::io::stdin();
    let mut input = input.lock();
    let mut buffer = String::new();
    match input.read_line(&mut buffer) {
        Ok(_) => {}
        Err(_) => {
            println!("bad");
            return;
        }
    }
    if let Ok(num) = buffer.trim().parse::<f64>() {
        println!("{:?}", num);
    } else {
        println!("bad");
    }
}
```
