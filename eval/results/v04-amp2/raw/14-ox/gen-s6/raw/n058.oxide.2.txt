struct Reading { label: Str, values: Vec<Str> }

fn find_duplicates(s: Str) -> Vec<Str> {
    let mut seen = vec()
    let mut duplicates = vec()
    for c in chars(s) {
        if contains(seen, c) {
            if !contains(duplicates, c) {
                push(duplicates, c)
            }
        } else {
            push(seen, c)
        }
    }
    duplicates
}
