fn sort(mut v: Vec<Int>) -> Vec<Int> {
    let n = len(v)
    for i in range(0, n) {
        for j in range(i + 1, n) {
            let a = get(v, i)?
            let b = get(v, j)?
            if a > b {
                v = swap(v, i, j)
            }
        }
    }
    v
}

fn swap(mut v: Vec<Int>, i: Int, j: Int) -> Vec<Int> {
    let a = get(v, i)?
    let b = get(v, j)?
    v = set(v, i, b)
    v = set(v, j, a)
    v
}

fn set(mut v: Vec<Int>, i: Int, x: Int) -> Vec<Int> {
    v = push(vec(), x)
    for j in range(0, i) {
        let y = get(v, 0)?
        v = push(v, y)
    }
    for j in range(i + 1, len(v)) {
        let y = get(v, j)?
        v = push(v, y)
    }
    v
}
