fn reverse_num(n: Int) -> Str {
    let s = int_to_str(n)
    let chars = chars(s)
    let reversed_chars: Vec<Str> = chars.rev().cloned().collect()
    let reversed_str = concat(vec(), reversed_chars)
    reversed_str
}

fn main() {
    let num = 1234
    let reversed_num = reverse_num(num)
    print_str(reversed_num)
}
