fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut max_name = "";
    let mut max_length = 0;

    for name in names.iter() {
        if name.len() > max_length {
            max_name = name;
            max_length = name.len();
        }
    }

    println!("{} {}", max_name, max_length);
}
