struct Number {
    value: Option<Int>
}

fn sum_and_count_non_numbers(numbers: Vec<Number>) -> (Int, Int) {
    let mut sum = 0
    let mut count = 0
    for num in numbers {
        match num.value {
            Some(v) => sum += v,
            None => count += 1,
        }
    }
    (sum, count)
}

fn main() {
    let nums = vec![
        Number { value: Some(7) },
        Number { value: None },
        Number { value: Some(9) },
        Number { value: None },
    ]
    let (total, non_numbers_count) = sum_and_count_non_numbers(nums)
    print(total)
    print(non_numbers_count)
}
