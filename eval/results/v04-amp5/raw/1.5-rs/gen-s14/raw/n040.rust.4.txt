fn main() {
    let square_side = 4;
    let shape_area_square: f64 = square_side as f64 * square_side as f64;

    let circle_radius = 3.0;
    const PI: f64 = std::f64::consts::PI; // Use the std::f64::consts::PI constant
    let shape_area_circle: f64 = PI * circle_radius * circle_radius;

    println!("The area of the square is {:.2}", shape_area_square);
    println!("The area of the circle is {:.2}", shape_area_circle);
}
