struct Reading { label: Str, temps: Vec<Int> }
fn main() {
    let alpha = Reading { label: "alpha", temps: vec(-5, 12, 30, 22, 0, 35, 18) }
    let beta = Reading { label: "beta", temps: vec(8, -12, 5, 41, 27) }
    for r in vec(alpha, beta) {
        print(r.label)
        let below_zero = count_if(r.temps, |t| t <= 0)
        let above_thirty = count_if(r.temps, |t| t >= 30)
        let max_fahrenheit = match max(r.temps) { Some(t) => (t * 9 / 5 + 32), None => -1 }
        let min_fahrenheit = match min(r.temps) { Some(t) => (t * 9 / 5 + 32), None => -1 }
        let mean_celsius = if len(r.temps) > 0 {
            let sum = sum(r.temps)
            sum / len(r.temps)
        } else {
            -1
        }
        print(below_zero)
        print(above_thirty)
        print(max_fahrenheit)
        print(min_fahrenheit)
        print(mean_celsius)
    }
    let alpha_mean = if len(alpha.temps) > 0 { sum(alpha.temps) / len(alpha.temps) } else { -1 }
    let beta_mean = if len(beta.temps) > 0 { sum(beta.temps) / len(beta.temps) } else { -1 }
    if alpha_mean >= beta_mean {
        print_str("alpha")
    } else {
        print_str("beta")
    }
}
