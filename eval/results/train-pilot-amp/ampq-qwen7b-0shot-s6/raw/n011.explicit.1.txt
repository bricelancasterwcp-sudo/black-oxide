fn main() {
    let mut multiples = vec()
    for i in range(1, 7) {
        let x = i * 4
        multiples = push(multiples, x)
    }
    let total = sum(multiples)
    print(total)
    drop multiples
}

fn sum(v: Vec<Int>) -> Int {
    let total = 0
    for x in &v {
        total = total + x
    }
    total
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec()
    for i in a.range(b) {
        result = push(result, i)
    }
    result
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut new = vec()
    for y in &v {
        new = push(new, y)
    }
    new = push(new, x)
    new
}

fn print(x: Int) {
    print_int(x)
}

fn print_int(x: Int) {
    print_str(to_string(x))
}

fn to_string(x: Int) -> Str {
    let mut result = ""
    let mut n = x
    while n > 0 {
        result = concat(to_char(n % 10), result)
        n = trunc(n / 10)
    }
    result
}

fn to_char(n: Int) -> Str {
    let mut result = ""
    if n == 0 {
        result = concat(result, "0")
    } else if n == 1 {
        result = concat(result, "1")
    } else if n == 2 {
        result = concat(result, "2")
    } else {
        result = concat(result, "3")
    }
    result
}

fn concat(a: Str, b: Str) -> Str {
    let mut result = ""
    for c in chars(a) {
        result = concat(result, c)
    }
    for c in chars(b) {
        result = concat(result, c)
    }
    result
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = vec()
    for i in range(0, str_len(s)) {
        result = push(result, to_char(get(s, i)))
    }
    result
}

fn str_len(s: Str) -> Int {
    let mut result = 0
    for _ in chars(s) {
        result = result + 1
    }
    result
}

fn get(v: Str, i: Int) -> Str {
    let mut result = ""
    for c in chars(v) {
        if i == 0 {
            result = concat(result, c)
            break
        }
        i = i - 1
    }
    result
}
