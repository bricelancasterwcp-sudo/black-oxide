struct Count { vowels: Int, consonants: Int }

fn is_vowel(c: Str) -> Bool {
    match c {
        "a" => true, "e" => true, "i" => true, "o" => true, "u" => true,
        "A" => true, "E" => true, "I" => true, "O" => true, "U" => true,
        _ => false,
    }
}

fn count_letters(s: Str) -> Count {
    let vowels = 0
    let consonants = 0
    for c in chars(s) {
        if is_vowel(c) {
            vowels = vowels + 1
        } else if c >= "a" && c <= "z" || c >= "A" && c <= "Z" {
            consonants = consonants + 1
        }
    }
    Count { vowels: vowels, consonants: consonants }
}

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    let count = count_letters(keyboard)
    print(count.vowels)
    print(count.consonants)
}
