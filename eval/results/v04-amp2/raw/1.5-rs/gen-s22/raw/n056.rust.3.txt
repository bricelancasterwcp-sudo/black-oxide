```rust
fn main() {
    let mut count = 0;
    let mut sum = 0;

    let texts = vec![7, 10, 9, '?'];

    for &text in &texts {
        if text.is_digit(10) {
            sum += text.to_digit(10).unwrap();
            count += 1;
        }
    }

    println!("Total of the ones that do: {}", sum);
    println!("Number of ones that do not: {}", texts.len() - count);
}
```
