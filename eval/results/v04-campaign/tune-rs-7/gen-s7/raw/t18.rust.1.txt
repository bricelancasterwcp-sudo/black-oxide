fn main() {
    let shapes = vec![
        Rectangle { width: 3, height: 4 },
        Triangle { sides: [3, 4, 5] },
        Rectangle { width: 2, height: 5 },
    ];
    for shape in &shapes {
        println!("{}", shape.perimeter());
    }
}
struct Rectangle {
    width: i64,
    height: i64,
}
impl Rectangle {
    fn perimeter(&self) -> i64 {
        2 * (self.width + self.height)
    }
}
struct Triangle {
    sides: [i64; 3],
}
impl Triangle {
    fn perimeter(&self) -> i64 {
        self.sides.iter().sum()
    }
}
