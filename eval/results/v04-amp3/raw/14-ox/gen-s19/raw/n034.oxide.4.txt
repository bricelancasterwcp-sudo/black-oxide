fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x)
}

fn main() {
    let list1 = push(push(push(vec(), 7), 8), 9)
    match second(list1) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
    
    let list2 = push(push(vec(), 7), 8)
    match second(list2) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
    
    let list3 = push(vec(), 7)
    match second(list3) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}
