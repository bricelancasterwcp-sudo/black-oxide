struct List { head: Int, tail: Option<Box<List>> }

fn walk_list(l: Option<Box<List>>, acc_even: Int) -> (Int, Int) {
    match l {
        None => (acc_even, 0),
        Some(boxed_l) => {
            let List { head, tail } = *boxed_l;
            if head % 2 != 0 {
                print(head * 2);
                print_str("\n");
            }
            let (new_acc_even, acc_odd) = walk_list(tail, acc_even + if head % 2 == 0 { head } else { 0 });
            (new_acc_even, acc_odd)
        }
    }
}

fn main() {
    let list = Box::new(List { head: 7, tail: Some(Box::new(List { head: 4, tail: Some(Box::new(List { head: 9, tail: Some(Box::new(List { head: 10, tail: None })) })) })) }));
    let (total_even, _) = walk_list(Some(list), 0);
    print(total_even);
}
