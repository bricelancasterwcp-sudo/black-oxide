```rust
fn divide(numerator: Int, denominator: Int) -> Option<Int> {
    if denominator == 0 {
        None
    } else {
        Some(numerator / denominator)
    }
}

fn main() {
    let answer = divide(20, 4);
    match answer {
        Some(result) => print(result),
        None => println("bad"),
    }
}
```
