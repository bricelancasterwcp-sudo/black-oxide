```rust
fn main() {
    let names = vec!["fig", "banana", "kiwi"];
    let mut max_len = 0;
    let mut longest_name = "";

    for name in names {
        if str_len(name) > max_len {
            max_len = str_len(name);
            longest_name = name;
        }
    }

    print_str(longest_name);
    print(max_len);
}
```
