```rust
fn main() {
    let numbers = [8, 3, 11, 6];
    let mut min = numbers[0];
    let mut max = numbers[0];

    for &num in &numbers {
        if num < min {
            min = num;
        }
        if num > max {
            max = num;
        }
    }

    println!("Smallest figure: {}", min);
    println!("Gap between largest and smallest: {}", max - min);
}
```
