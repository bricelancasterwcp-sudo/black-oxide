fn main() {
    let s = "abcdefghijklmnopqrstuvwxyz";
    let mut vowels = 0;
    let mut consonants = 0;

    for c in chars(s) {
        if c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' ||
           c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U' {
            vowels += 1;
        } else if (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') {
            consonants += 1;
        }
    }

    print(vowels);
    print(consonants);
}
