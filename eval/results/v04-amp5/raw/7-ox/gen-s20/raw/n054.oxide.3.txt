fn main() {
    let mut vowel_count = 0;
    let mut consonant_count = 0;
    let keyboard = "qwertyuiopasdfghjklzxcvbnm";

    for letter in keyboard.chars() {
        if is_vowel(letter) {
            vowel_count += 1;
        } else {
            consonant_count += 1;
        }
    }

    print(vowel_count);
    print(consonant_count);
}

fn is_vowel(c: Str) -> Bool {
    match c.to_lowercase().as_str() {
        "a" | "e" | "i" | "o" | "u" => true,
        _ => false,
    }
}
