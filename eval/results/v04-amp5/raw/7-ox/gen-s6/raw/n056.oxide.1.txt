fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut sum = 0;
    let mut count_not_number = 0;

    for text in texts {
        match parse_int(text) {
            Some(num) => sum += num,
            None => count_not_number += 1,
        }
    }

    print(sum);
    print(count_not_number);
}
