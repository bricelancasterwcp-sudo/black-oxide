fn main() {
    let v = vec(4, 4, 4, 7, 7, 2, 9, 9, 9, 9, 1, 1, 5)
    print(len(v))
    let longest_stretch = 0
    let current_value = -1
    let current_stretch = 0
    for x in v {
        if x == current_value {
            current_stretch += 1
        } else {
            if current_stretch > longest_stretch {
                longest_stretch = current_stretch
            }
            current_value = x
            current_stretch = 1
        }
    }
    print(longest_stretch)
    print(current_value)
    let unique_values = count_unique(v)
    print(unique_values)
    let first_repetition = -1
    for i in range(len(v)) {
        if contains(v, v[i]) && i != find(v, v[i]).unwrap_or(-1) {
            first_repetition = i
            break
        }
    }
    print(first_repetition)
    let never_decrease = true
    for i in range(1, len(v)) {
        if v[i] < v[i - 1] {
            never_decrease = false
            break
        }
    }
    print(never_decrease)
}
