fn main() {
    let morning = [94, 71, 88, 65, 92, 79, 83];
    let evening = [55, 90, 77, 84, 68, 99];
    let mut higher_mean_sitting: &[i32] = &[];
    for sitting in &[&morning, &evening] {
        let (mut above_90, mut between_80_and_89, mut between_70_and_79, mut below_70) = (0, 0, 0, 0);
        let mut highest = i32::MIN;
        let sum: i32 = sitting.iter().map(|&x| {
            if x >= 90 { above_90 += 1; }
            else if x >= 80 { between_80_and_89 += 1; }
            else if x >= 70 { between_70_and_79 += 1; }
            else { below_70 += 1; }
            highest = highest.max(x);
            x
        }).sum();
        let mean = sum / sitting.len() as i32;
        println!("{}", if sitting == &morning { "Morning" } else { "Evening" });
        println!("{}", above_90);
        println!("{}", between_80_and_89);
        println!("{}", between_70_and_79);
        println!("{}", below_70);
        println!("{}", highest);
        println!("{}", mean);
        if higher_mean_sitting.is_empty() || mean > higher_mean_sitting.iter().sum::<i32>() / sitting.len() as i32 {
            higher_mean_sitting = sitting;
        }
    }
    println!("{}", if higher_mean_sitting == &morning { "Morning" } else { "Evening" });
}
