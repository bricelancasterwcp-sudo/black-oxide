```rust
struct Rectangle { width: Int, height: Int }

fn area(r: Rectangle) -> Int {
    r.width * r.height
}

fn perimeter(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn main() {
    let rect = Rectangle { width: 9, height: 3 }
    print(area(rect))          // prints the area: 27
    let rect2 = Rectangle { width: 9, height: 3 } // reassign for perimeter calculation
    print(perimeter(rect2))   // prints the perimeter: 24
}
```
