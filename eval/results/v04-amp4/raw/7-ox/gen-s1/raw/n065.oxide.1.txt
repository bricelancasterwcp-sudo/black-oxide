fn count_above_five(v: Vec<Int>) -> Int {
    count_if(v, |x| x > 5)
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    max(v)
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14);
    print(count_above_five(readings)) // Output the count of readings above five
    print(max_reading(readings)) // Output the maximum reading
}
