struct Reading { label: Str, values: Vec<Int> }

fn count_above_five(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 5 {
            count = count + 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Int {
    let mut max = -1
    for x in v {
        if x > max {
            max = x
        }
    }
    max
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 6), 11), 3), 14), pushvec: 9 }  // Note: Corrected the pushvec to push
    print(count_above_five(r.values))  // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    print(max_reading(r2.values))
}
