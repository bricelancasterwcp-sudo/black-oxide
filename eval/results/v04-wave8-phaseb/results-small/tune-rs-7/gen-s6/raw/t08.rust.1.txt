fn main() {
    let numbers = [3, 8, -2, 12, 7];
    let mut count_positive = 0;
    let mut max_value = i32::MIN;
    for &num in &numbers {
        if num > 0 {
            count_positive += 1;
        }
        if num > max_value {
            max_value = num;
        }
    }
    println!("{}", count_positive);
    println!("{}", max_value);
}
