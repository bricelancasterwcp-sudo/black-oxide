fn reverse_digits(n: Int) -> Str {
    let mut rev = "";
    let mut num = n;
    while num > 0 {
        let digit = num % 10;
        rev = concat(rev, int_to_str(digit));
        num = num / 10;
    }
    rev
}

fn main() {
    let number = 1234;
    print_str(reverse_digits(number));
}
