fn divide(a: Float, b: Float) -> Result<Float, String> {
    if b == 0.0 {
        Err("Cannot divide by zero.".to_string())
    } else {
        Ok(a / b)
    }
}

fn main() {
    let result = divide(20.0, 4.0).unwrap_or_else(print_err)
    print(result)
}

fn print_err(e: String) {
    print(e)
}

fn print(s: Float) {
    print_str(to_str(s))
}

fn to_str(n: Float) -> Str {
    int_to_str(trunc(n)) + "." + int_to_str(trunc((n - trunc(n)) * 10))
}

fn print_str(s: Str) {
    for c in chars(s) {
        print_char(c)
    }
}

fn print_char(c: Str) {
    if c == "0" {
        print_str("0")
    } else if c == "1" {
        print_str("1")
    } else if c == "2" {
        print_str("2")
    } else {
        print_str("3")
    }
}

fn int_to_str(n: Int) -> Str {
    let mut s = str_len(n)
    while s == 0 {
        s = s + 1
        n = n / 10
    }
    chars(n)
}

fn str_len(n: Int) -> Int {
    let mut s = 0
    let mut n = n
    while n > 0 {
        s = s + 1
        n = n / 10
    }
    s
}

fn chars(s: Int) -> Str {
    let mut v = vec()
    let mut s = s
    while s > 0 {
        let d = s % 10
        v = push(v, d)
        s = s / 10
    }
    v
}

fn push(mutv: Vec<Int>, x: Int) -> Vec<Int> {
    v = v + x
    v
}

fn vec() -> Vec<Int> {
    Vec()
}

fn len(v: Vec<Int>) -> Int {
    let mut s = 0
    for _ in v {
        s = s + 1
    }
    s
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= len(v) {
        None
    } else {
        Some(v)
    }
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a + 1 {
        if i >= b {
            break
        }
        v = push(v, i)
    }
    v
}

fn clone(x: Int) -> Int {
    x
}

fn to_float(n: Int) -> Float {
    n
}

fn trunc(x: Float) -> Int {
    x
}
