use std::io;

fn main() {
    let mut input = String::new();

    loop {
        match io::stdin().read_line(&mut input) {
            Ok(_) => break,
            Err(e) => println!("Error reading input: {}", e),
        }
    }

    match input.trim().parse::<i32>() {
        Ok(_) => println!("Not a number"),
        Err(_) => println!("Bad"),
    }
}
