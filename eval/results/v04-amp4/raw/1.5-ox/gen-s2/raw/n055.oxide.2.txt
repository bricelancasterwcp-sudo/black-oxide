fn main() {
    let x = read_str("Enter a number: ");
    match x {
        "8" => println!("Eight"),
        "oak" => println!("Oak"),
        "15" => println!("Fifteen"),
        _ => println!("Bad")
    }
}

fn read_str(s: &str) -> String {
    let mut input = String::new();
    print!("{}", s);
    io::stdin().read_line(&mut input).expect("Failed to read line");
    input.trim_end_matches('\n').to_string()
}
