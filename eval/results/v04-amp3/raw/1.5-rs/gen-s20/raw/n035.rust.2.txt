fn divide_numbers(a: f64, b: f64) -> Result<f64, &'static str> {
    if b == 0.0 {
        Err("Cannot divide by zero")
    } else {
        Ok(a / b)
    }
}

fn main() {
    match divide_numbers(20.0, 4.0) {
        Ok(result) => println!("The division result is {}", result),
        Err(error) => println!("Error: {}", error),
    }
}
