struct Reading { label: Str, values: Vec<Int> }

fn count_digits(s: Str) -> Int {
    let mut total = 0
    let mut count = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            total += parse_int(c)?
            count += 1
        }
    }
    Some(total)
}

fn main() {
    let s = "a1b22"
    let digits = count_digits(s)?
    print(digits)
    let count = count_digits(s)?
    print(count)
}
