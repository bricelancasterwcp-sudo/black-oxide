fn first_letter(s: Str) -> Option<Str> {
    let chars = chars(s)
    let first = get(chars, 0)?
    Some(first)
}

fn last_letter(s: Str) -> Option<Str> {
    let chars = chars(s)
    let len = len(chars)
    let last = get(chars, len - 1)?
    Some(last)
}

fn main() {
    let word = "granite"
    let first = first_letter(word.clone())?
    let last = last_letter(word.clone())?
    print_str(first)
    print_str(last)
}
