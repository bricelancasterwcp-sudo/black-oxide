fn is_prime(n: Int) -> Bool {
    if n < 2 {
        return false;
    }
    for i in range(2, n) {
        if n % i == 0 {
            return false;
        }
    }
    true
}

fn count_primes_below(limit: Int) -> Int {
    let mut count = 0;
    for n in range(2, limit) {
        if is_prime(n) {
            count = count + 1;
        }
    }
    count
}

fn main() {
    let primes_count = count_primes_below(100);
    print(primes_count);
}
