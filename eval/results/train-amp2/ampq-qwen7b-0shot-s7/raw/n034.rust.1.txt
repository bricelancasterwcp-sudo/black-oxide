```rust
fn second_entry(list: &[i32]) -> Option<&i32> {
    list.get(1)
}

fn main() {
    let list7 = vec![1, 2];
    let list8 = vec![1, 2, 3];
    let list9 = vec![1];

    println!("{:?}", second_entry(&list7)); // Some(2)
    println!("{:?}", second_entry(&list8)); // Some(3)
    println!("{:?}", second_entry(&list9)); // None
}
```
