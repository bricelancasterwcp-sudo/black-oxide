```rust
fn divide(a: i32, b: i32) -> Result<f64, String> {
    if b == 0 {
        Err("Error: Dividing by zero".to_string())
    } else {
        Ok(a as f64 / b as f64)
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("Result is: {}", result),
        Err(e) => println!("{}", e),
    }
}
```
