fn main() {
    let mut sum = 0
    let mut failures = 0
    for s in vec("12", "x", "30") {
        match parse_int(s) {
            Some(n) => {
                sum = sum + n
            },
            None => {
                failures = failures + 1
            },
        }
    }
    print(sum)
    print(failures)
}
