fn area(w: Int, h: Int) -> Int {
    w * h
}

fn perimeter(w: Int, h: Int) -> Int {
    2 * (w + h)
}

fn main() {
    let width = 9.0.to_int().clone()
    let height = 3.0.to_int().clone()
    print(area(width, height))
    drop width
    drop height
    print(perimeter(width, height))
    drop width
    drop height
}
