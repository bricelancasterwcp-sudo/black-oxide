fn divisors(n: Int) -> Vec<Int> {
    let mut divisors = vec()
    for i in range(1, n) {
        if n % i == 0 {
            divisors = push(divisors, i)
        }
    }
    divisors
}

fn main() {
    print(len(&divisors))
    drop divisors
}
