fn count_words_and_chars(s: Str) -> (Int, Int) {
    let mut word_count = 0;
    let mut char_count = 0;

    for c in s.chars() {
        if c == ' ' {
            word_count += 1;
        }
        char_count += 1;
    }

    // Add the last word count
    word_count += 1;

    (word_count, char_count)
}

fn main() {
    let phrase = "when in doubt go home";
    let (words, chars) = count_words_and_chars(phrase);
    println!("Words: {}", words);
    println!("Characters including spaces: {}", chars);
}
