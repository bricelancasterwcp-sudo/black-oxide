struct Number { value: Option<Int> }

fn sum_valid(numbers: Vec<Number>) -> Int {
    let mut sum = 0;
    for n in numbers {
        match n.value {
            Some(v) => sum += v,
            None => (),
        }
    }
    sum
}

fn count_invalid(numbers: Vec<Number>) -> Int {
    let mut count = 0;
    for n in numbers {
        if n.value == None {
            count += 1;
        }
    }
    count
}

fn main() {
    let nums = vec![
        Number { value: Some(7) },
        Number { value: None },
        Number { value: Some(9) },
        Number { value: None },
    ];
    print(sum_valid(nums));
    print(count_invalid(nums));
}
