```rust
use std::collections::VecDeque;

fn main() {
    let mut queue: VecDeque<i32> = VecDeque::from([1, 2, 3, 4, 5]);
    
    if let Some(first) = queue.pop_front() {
        if let Some(last) = queue.pop_back() {
            queue.push_back(first);
            queue.push_front(last);
        }
    }

    for entry in &queue {
        println!("{}", entry);
    }
}
```
