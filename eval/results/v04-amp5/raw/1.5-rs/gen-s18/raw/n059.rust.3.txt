```rust
fn main() {
    let sentence = "when in doubt go home";
    let mut word_count = 0;
    let mut char_count = 0;

    for c in sentence.chars() {
        if c == ' ' {
            word_count += 1;
        }
        char_count += c.len_utf8();
    }

    println!("Word count: {}", word_count);
    println!("Character count (including spaces): {}", char_count);
}
```
