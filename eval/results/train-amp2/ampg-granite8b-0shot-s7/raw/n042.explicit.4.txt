fn main() {
    let nums = vec(12, 5, 20, 8, 17)
    let mut second = None
    let mut largest = nums.get(0)
    for i in range(1, nums.len()) {
        let num = nums.get(i)
        largest = if num > largest { Some(&num) } else { largest }
        second = if num > second { Some(&num) } else { second }
    }
    print_str("The second largest number is ")
    print(second.unwrap())
    print_str("\n")
    print_str("The largest number is ")
    print(largest.unwrap())
}
