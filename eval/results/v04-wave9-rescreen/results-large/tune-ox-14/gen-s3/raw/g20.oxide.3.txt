fn main() {
    let mut primes = vec()
    for i in range(2, 101) {
        primes = push(primes, i)
    }
    let mut i = 0
    while i < len(primes) {
        let p = primes[i]
        let mut j = i + 1
        while j < len(primes) {
            if primes[j] % p == 0 {
                primes = concat(primes, range(j + 1, len(primes)))
                break
            }
            j += 1
        }
        i += 1
    }
    print(len(primes))
    print(primes[len(primes) - 1])
    let sum = 0
    for p in primes {
        sum += p
    }
    print(sum)
    let mut count = 0
    for i in range(0, len(primes) - 1) {
        if primes[i + 1] - primes[i] == 2 {
            count += 1
        }
    }
    print(count)
    print(primes[9])
}
