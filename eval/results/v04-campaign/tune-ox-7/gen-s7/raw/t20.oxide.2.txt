struct Item {
    name: String,
    unit_price: Int,
    quantity: Int,
}
fn main() {
    let items = vec(
        Item { name: "pen", unit_price: 3, quantity: 4 },
        Item { name: "pad", unit_price: 9, quantity: 2 },
        Item { name: "ink", unit_price: 5, quantity: 3 },
    )
    let total_cost = sum(items.map(|item| item.unit_price * item.quantity))
    print(total_cost)
    let max_item = items.max_by_key(|item| item.unit_price * item.quantity)
    match max_item {
        Some(item) => print(item.name),
        None => print("No items"),
    }
}
