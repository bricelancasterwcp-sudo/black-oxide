fn rver(word: Str) -> Str {
    let mut word = word
    let mut i = 0
    while i < len(word) {
        let c = get(word, i)?
        let j = len(word) - 1 - i
        let last = get(word, j)?
        set(word, i, last)
        set(word, j, c)
        i = i + 1
    }
    word
}

fn set(word: Str, i: Int, c: Char) {
    let s = vec_from_str(str_from_char(c))
    let word = functional_update(word, i, s)
    print(word)
}

fn functional_update(word: Str, index: Int, value: Vec<Char>) -> Str {
    let mut result = vec()
    let mut i = 0
    while i < len(word) {
        if i == index {
            result = concat(result, value)
        } else {
            let c = get(word, i)?
            let s = vec_from_str(str_from_char(c))
            result = push(result, s)
        }
        i = i + 1
    }
    str_from_vec(result)
}

fn vec_from_str(s: Str) -> Vec<Char> {

    let mut i = 0
    while i < len(s) {
        let c = get(s, i)?
        result = push(result, c)
