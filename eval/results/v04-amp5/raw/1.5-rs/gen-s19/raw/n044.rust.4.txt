```rust
fn main() {
    let mut balance = 0;
    println!("Initial balance: {}", balance);

    // Walk through deposits 3, 1, 4, 1 and 5 in order
    let deposits = vec![3, 1, 4, 1, 5];
    for deposit in deposits {
        // Update the balance by adding the current deposit amount
        balance += deposit;
        println!("Current balance after depositing {}: {}", deposit, balance);
    }
}
```
