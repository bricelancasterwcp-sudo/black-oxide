fn main() {
    let morning = vec![94, 71, 88, 65, 92, 79, 83];
    let evening = vec![55, 90, 77, 84, 68, 99];
    let mut higher_mean_sitting = "";
    let mut highest_mean = 0;
    for sitting in &[&morning, &evening] {
        let (mut above_90, mut between_80_and_89, mut between_70_and_79, mut below_70) = (0, 0, 0, 0);
        let mut highest = 0;
        let sum: i32 = sitting.iter().map(|&x| {
            if x >= 90 { above_90 += 1; }
            else if x >= 80 { between_80_and_89 += 1; }
            else if x >= 70 { between_70_and_79 += 1; }
            else { below_70 += 1; }
            highest = std::cmp::max(highest, x);
            x
        }).sum();
        let mean = sum / sitting.len() as i32;
        println!("{}", if *sitting == morning { "Morning" } else { "Evening" });
        println!("{}", above_90);
        println!("{}", between_80_and_89);
        println!("{}", between_70_and_79);
        println!("{}", below_70);
        println!("{}", highest);
        println!("{}", mean);
        if mean > highest_mean {
            higher_mean_sitting = if *sitting == morning { "Morning" } else { "Evening" };
            highest_mean = mean;
        }
    }
    println!("{}", higher_mean_sitting);
}
