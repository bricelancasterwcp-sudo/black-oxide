struct Counter {
    vowels: Int,
    consonants: Int,
}

fn count_vowels(s: Str) -> Counter {
    let vowels = 0
    let consonants = 0
    for c in s.chars() {
        match c {
            CharVowel => {
                let vowels = vowels + 1
                Counter(vowels, consonants)
            },
            _ => {
                let consonants = consonants + 1
                Counter(vowels, consonants)
            },
        }
    }
}

fn main() {
    let s = "the quick brown fox jumps over the lazy dog"
    let c = count_vowels(s)
    print(c.consonants)
}
