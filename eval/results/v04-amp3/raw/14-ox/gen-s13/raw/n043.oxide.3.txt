struct Queue { entries: Vec<Int> }

fn exchange_first_last(q: Queue) -> Option<Queue> {
    let len = len(q.entries)
    if len < 2 {
        return Some(q)
    }
    let first = get(q.entries, 0)?
    let last = get(q.entries, len - 1)?
    let mut new_entries = push(vec(), last)
    for i in range(1, len - 1) {
        new_entries = push(new_entries, get(q.entries, i)?)
    }
    new_entries = push(new_entries, first)
    Some(Queue { entries: new_entries })
}

fn print_queue(q: Queue) {
    for x in q.entries {
        print(x)
    }
}

fn main() {
    let q = Queue { entries: vec(1, 2, 3, 4, 5) }
    match exchange_first_last(q) {
        Some(exchanged_q) => print_queue(exchanged_q),
        None => print_str("Queue too short"),
    }
}
