```rust
struct Square { side: Int }
struct Circle { radius: Float }

enum Shape { Square(Square), Circle(Circle) }

fn area(shape: Shape) -> Float {
    match shape {
        Shape::Square(square) => (square.side * square.side).to_float(),
        Shape::Circle(circle) => 3.0 * circle.radius * circle.radius,
    }
}

fn main() {
    let s = Shape::Square(Square { side: 4 })
    print(area(s))
}
```
