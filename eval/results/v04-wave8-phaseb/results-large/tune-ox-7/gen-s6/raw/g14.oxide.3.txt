struct Reading { label: Str, values: Vec<Int> }
fn main() {
    let v = vec(4, 4, 4, 7, 7, 2, 9, 9, 9, 9, 1, 1, 5)
    let n = len(v)
    print(n)
    let (longest_stretch, value) = longest_consecutive(v)
    print(longest_stretch)
    print(value)
    let unique_values = count_unique(v)
    print(unique_values)
    let first_repetition_position = find_first_repetition(v)
    match first_repetition_position {
        Some(pos) => print(pos),
        None => print_str("No repetition"),
    }
    let never_decrease = check_never_decrease(v)
    print(never_decrease)
}
fn longest_consecutive(values: Vec<Int>) -> (Int, Int) {
    let mut max_stretch = 0
    let mut current_stretch = 1
    let mut value = -1
    for i in range(1, len(values)) {
        if values[i] == values[i - 1] {
            current_stretch += 1
        } else {
            if current_stretch > max_stretch {
                max_stretch = current_stretch
                value = values[i - 1]
            }
            current_stretch = 1
        }
    }
    if current_stretch > max_stretch {
        max_stretch = current_stretch
        value = values[len(values) - 1]
    }
    (max_stretch, value)
}
fn count_unique(values: Vec<Int>) -> Int {
    let mut seen = vec()
    for x in values {
        if !contains(seen, x) {
            push(seen, x)
        }
    }
    len(seen)
}
fn find_first_repetition(values: Vec<Int>) -> Option<Int> {
    let mut seen = vec()
    for i in range(0, len(values)) {
        if contains(seen, values[i]) {
            return Some(i)
        } else {
            push(seen, values[i])
        }
    }
    None
}
fn check_never_decrease(values: Vec<Int>) -> Bool {
    for i in range(1, len(values)) {
        if values[i] < values[i - 1] {
            return false
        }
    }
    true
}
