fn count_under_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x < cutoff {
            count += 1
        }
    }
    count
}

fn count_over_or_at_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= cutoff {
            count += 1
        }
    }
    count
}

fn main() {
    let samples = vec()
    samples = push(samples, 5)
    samples = push(samples, 12)
    samples = push(samples, 3)
    samples = push(samples, 18)
    samples = push(samples, 9)
    let cutoff = 10
    print(count_under_cutoff(samples, cutoff))
    print(count_over_or_at_cutoff(samples, cutoff))
}
