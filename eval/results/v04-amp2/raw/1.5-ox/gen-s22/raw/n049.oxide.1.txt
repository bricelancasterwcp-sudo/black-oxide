```rust
struct Figure { value: Int }

fn main() {
    let figs = vec![
        Figure { value: 8 },
        Figure { value: 3 },
        Figure { value: 11 },
        Figure { value: 6 }
    ];

    let min_fig = *figs.iter().min().unwrap();
    let max_fig = *figs.iter().max().unwrap();

    print(min_fig.value);
    println(max_fig.value - min_fig.value);
}
```
