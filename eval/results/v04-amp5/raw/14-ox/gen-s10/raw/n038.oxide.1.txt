struct Device { sensor: Sensor }
struct Sensor { reading: Int }

fn set_reading(d: Device, r: Int) -> Device {
    let s = Sensor { reading: r }
    Device { sensor: s }
}

fn main() {
    let d = Device { sensor: Sensor { reading: 0 } }
    let d2 = set_reading(d, 7)
    print(d2.sensor.reading)
}
