fn second(v: Vec<Int>) -> Option<Int> {
    v.get(1)
}

fn main() {
    match second(vec(7)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match second(vec(7, 8)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match second(vec(7, 8, 9)) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
