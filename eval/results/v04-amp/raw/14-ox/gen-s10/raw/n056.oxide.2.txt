struct Number { value: Int }

fn sum_numbers(numbers: Vec<Number>) -> Int {
    let mut sum = 0
    for n in numbers {
        sum = sum + n.value
    }
    sum
}

fn count_non_numbers(texts: Vec<Str>) -> Int {
    let mut count = 0
    for t in texts {
        if parse_int(t).is_none() {
            count = count + 1
        }
    }
    count
}

fn main() {
    let texts = vec("7", "x", "9", "?")
    let numbers = texts.map(|t| parse_int(t)).filter_map(|o| o)
    let total = sum_numbers(numbers.map(|n| Number { value: n }))
    print(total)
    let non_numbers_count = count_non_numbers(texts)
    print(non_numbers_count)
}
