fn count_marks(v: Vec<Int>, lo: Int, hi: Int) -> Int {
    let mut c = 0
    for x in v {
        if x >= lo && x <= hi {
            c += 1
        }
    }
    c
}
fn analyze_sitting(name: Str, marks: Vec<Int>) {
    print_str(name)
    print(count_marks(marks.clone(), 90, 100))
    print(count_marks(marks.clone(), 80, 89))
    print(count_marks(marks.clone(), 70, 79))
    print(count_marks(marks.clone(), 0, 69))
    let max_mark = unwrap_or(max(marks.clone()), -1)
    print(max_mark)
    let total = sum(marks.clone())
    let mean = total / len(marks)
    print(mean)
}
fn main() {
    let morning_marks = vec(94, 71, 88, 65, 92, 79, 83)
    let evening_marks = vec(55, 90, 77, 84, 68, 99)
    analyze_sitting("Morning", morning_marks)
    analyze_sitting("Evening", evening_marks)
    let morning_mean = sum(morning_marks) / len(morning_marks)
    let evening_mean = sum(evening_marks) / len(evening_marks)
    if morning_mean > evening_mean {
        print_str("Morning")
    } else {
        print_str("Evening")
    }
}
