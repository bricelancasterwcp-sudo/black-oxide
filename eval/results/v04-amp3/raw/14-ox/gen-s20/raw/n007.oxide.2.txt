fn sum_digits(n: Int) -> Option<Int> {
    let s = int_to_str(n)
    let chars = chars(s)
    let mut sum = 0
    for c in chars {
        let digit = parse_int(c)?
        sum = sum + digit
    }
    Some(sum)
}

fn main() {
    let num = 9875
    match sum_digits(num) {
        Some(sum) => print(sum),
        None => print_str("Error parsing digit"),
    }
}
