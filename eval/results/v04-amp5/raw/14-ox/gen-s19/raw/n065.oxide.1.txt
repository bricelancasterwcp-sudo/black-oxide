struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>) -> Int {
    count_if(v, |x| x > 5)
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    match max(v) {
        Some(n) => Some(n),
        None => None,
    }
}

fn main() {
    let readings = push(push(push(push(push(vec(), 6), 11), 3), 14), 9)
    let count = count_exceeding(readings)
    print(count)
    let max_reading = find_max(clone(readings))
    match max_reading {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
