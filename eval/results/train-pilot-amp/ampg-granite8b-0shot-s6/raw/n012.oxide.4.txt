struct List {
    first: Int,
    rest: Option<Box<List>>,
}

fn make_list(n: Int) -> List {
    if n <= 0 {
        List(0, None)
    } else {
        List(n, Some(Box(make_list(n - 1))))
    }
}

fn main() {
    let list = make_list(15)
    print(list.first)
}
