```rust
fn main() {
    let fruits = ["fig", "banana", "kiwi"];
    let mut longest_fruit = "";
    let mut max_length = 0;

    for &fruit in fruits.iter() {
        if fruit.len() > max_length {
            longest_fruit = fruit;
            max_length = fruit.len();
        }
    }

    println!("The fruit with the most letters is: {}", longest_fruit);
    println!("It has {} letters.", max_length);
}
```
