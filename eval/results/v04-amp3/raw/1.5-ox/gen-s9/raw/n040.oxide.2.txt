```rust
struct Square {
    side: Int,
}

struct Circle {
    radius: Float,
}

fn area(s: &Shape) -> Float {
    match s {
        Shape::Square { side } => side * side,
        Shape::Circle { radius } => 3.14 * radius * radius,
    }
}

fn main() {
    let shape = Shape::Square { side: 4 };
    print(area(&shape));
}
```
