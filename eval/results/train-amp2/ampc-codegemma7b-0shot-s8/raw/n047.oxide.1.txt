fn main() {
    let mut evens = vec_int()
    for i in range(1, 11) {
        if i % 2 == 0 {
            evens = push(evens, i)
        }
    }
    for i in range(len(evens) - 1, -1, -1) {
        print(get(evens, i)?)
        if i != 0 {
            print_str(", ")
        }
    }
    print_str("\n")
}
