```rust
fn main() {
    let mut sum = 0;
    let mut failures = 0;

    for s in ["12", "x", "30"] {
        match s.parse::<i32>() {
            Ok(num) => {
                sum += num;
            }
            Err(_) => {
                failures += 1;
            }
        }
    }

    println!("Sum: {}", sum);
    println!("Failures: {}", failures);
}
```
