struct Name { first: Str, last: Str }

fn main() {
    let n = Name { first: "fig", last: "banana" }
    match n.last {
        Str(s) => {
            let len = str_len(s)
            if len > 5 {
                print_str("The name has ")
                print(len)
                print_str(" letters")
            } else {
                print_str("The name is only ")
                print(len)
                print_str(" letters")
            }
        },
        _ => print_str("The name has a non-string last name"),
    }
}
