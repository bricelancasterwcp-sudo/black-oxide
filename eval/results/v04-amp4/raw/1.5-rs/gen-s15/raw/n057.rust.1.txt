```rust
fn main() {
    let mut max_length = 0;
    let mut longest_name = "";

    for &name in ["fig", "banana", "kiwi"].iter() {
        if name.len() > max_length {
            max_length = name.len();
            longest_name = name;
        }
    }

    println!("The name with the most letters is {} with a length of {}.", longest_name, max_length);
}
```
