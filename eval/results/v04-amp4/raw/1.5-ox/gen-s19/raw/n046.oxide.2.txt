fn main() {
    let mut vec = vec![5, 12, 3, 18, 9];
    let count_under_cutoff = vec.iter().filter(|&&x| x < 10).count();
    let count_over_or_equal_cutoff = vec.iter().filter(|&&x| x >= 10).count();

    print_str(format!("Count of values under cutoff: {}", count_under_cutoff));
    println_str(format!("Count of values over or equal to cutoff: {}", count_over_or_equal_cutoff));
}
