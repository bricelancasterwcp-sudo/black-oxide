```rust

fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut longest_name = "";
    let mut max_length = 0;

    for name in names {
        if name.len() > max_length {
            longest_name = name;
            max_length = name.len();
        }
    }

    println!("The name with the most letters is '{}'", longest_name);
    println!("It has {} letters.", max_length);
}
```
