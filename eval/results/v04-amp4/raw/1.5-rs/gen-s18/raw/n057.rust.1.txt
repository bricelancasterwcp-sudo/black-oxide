```rust
fn main() {
    let words = ["fig", "banana", "kiwi"];
    let mut max_word: &str = &words[0];
    let mut max_count = words[0].len();

    for word in words.iter() {
        if word.len() > max_count {
            max_word = word;
            max_count = word.len();
        }
    }

    println!("The name with the most letters is: {} ({})", max_word, max_count);
}
```
