fn divide(x: Int, y: Int) -> Result<Int, String> {
    if y == 0 {
        Err("Cannot divide by zero".to_string())
    } else {
        Ok(x / y)
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => print(result),
        Err(error) => print("Error: {}", &error),
    }
    match divide(20, 0) {
        Ok(result) => print(result),
        Err(error) => print("Error: {}", &error),
    }
}
