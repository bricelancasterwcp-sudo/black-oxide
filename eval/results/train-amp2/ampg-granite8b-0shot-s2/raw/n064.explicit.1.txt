fn lookup_at(list: &Vec<Int>, pos: Int) -> Option<Int> {
    match pos {
        INT(i) => {
            match i {
                INT(i) => {
                    match i {
                        INT(i) => {
                            match i {
                                INT(i) => match i {
                                    INT(i) => match i {
                                        INT(i) => match_at()?.get()?,
                                        _ => None,
                                    },
                                    _ => None,
                                },
                                _ => None,
                            }
                        },
                        _ => None,
                    }
                },
                _ => None,
            }
        },
        _ => None,
    }
}

fn print_or_none(opt: Option<Int>) {
    match opt {
        SOME(x) => print(x),
        _ => print(0),
    }
}

    let list = push(push(push(push(vec(), 10), 20), 30), 40)
    print_or_none(lookup_at(&list, 1))
    drop list

}
