fn longest_name(names: Vec<Str>) -> (Str, Int) {
    let mut longest = names[0];
    for name in &names {  // Use references instead of iter() to avoid moving elements
        if name.len() > longest.len() {
            longest = name.clone();
        }
    }
    (longest, longest.len())
}

fn main() {
    let names = vec!["fig", "banana", "kiwi"];
    let (longest_name, count) = longest_name(names);
    print_str(&format!("The longest name is {} with {} letters.", longest_name, count));
}
