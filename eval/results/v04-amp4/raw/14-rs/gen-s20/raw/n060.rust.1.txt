```rust
fn main() {
    let input = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in input.chars() {
        if c.is_digit(10) {
            let digit: u32 = c.to_digit(10).unwrap();
            sum += digit;
            count += 1;
        }
    }

    println!("{}", sum);
    println!("{}", count);
}
```
