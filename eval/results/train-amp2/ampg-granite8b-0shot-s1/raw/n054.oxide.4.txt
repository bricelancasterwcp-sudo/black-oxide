struct Keyboard {
    letters: Vec<Char>,
    vowels: Int,
}

fn count_vowels(k: Keyboard) -> Int {
    let vowels = 0
    for c in k.letters {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            let vowels = vowels + 1
        }
    }
    vowels
}

fn count_consonants(k: Keyboard) -> Int {
    let consonants = 0
    for c in k.letters {
        if c != "a" && c != "e" && c != "i" && c != "o" && c != "u" {
            let consonants = consonants + 1
        }
    }
    consonants
}

fn main() {
    let k = Keyboard { letters: "abcde".chars().collect(), vowels: 0 }
    let vowels = count_vowels(k)
    let consonants = count_consonants(k)
    print_str("vowels: ")
    print(vowels)
    print_str(", consonants: ")
    print(consonants)
}
