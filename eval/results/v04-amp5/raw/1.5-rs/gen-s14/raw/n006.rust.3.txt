fn main() {
    let mut largest_power = 1;
    while largest_power < 200 { // Change the condition to ensure it covers up to but not exceeding 100
        if largest_power * 2 > 100 {
            break;
        }
        largest_power *= 2;
    }
    println!("The biggest power of two not exceeding 100 is: {}", largest_power);
}
