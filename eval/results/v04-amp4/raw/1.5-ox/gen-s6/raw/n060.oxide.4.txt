fn sum_digits(s: Str) -> Int {
    s.chars()
        .filter(|c| c.is_digit(10))
        .map(|c| c.to_digit(10).unwrap())
        .sum()
}

fn count_digits(s: Str) -> Int {
    s.chars()
        .filter(|c| c.is_digit(10))
        .count()
}

fn main() {
    let a1b22 = "a1b22";
    println!("{}", sum_digits(a1b22)); // 3
    println!("{}", count_digits(a1b22)); // 2
}
