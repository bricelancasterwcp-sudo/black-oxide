```rust
fn safe_divide(dividend: i32, divisor: i32) -> Result<i32, &'static str> {
    if divisor == 0 {
        Err("bad")
    } else {
        Ok(dividend / divisor)
    }
}

fn main() {
    match safe_divide(20, 4) {
        Ok(result) => println!("{}", result),
        Err(e) => println!("{}", e),
    }
}
```
