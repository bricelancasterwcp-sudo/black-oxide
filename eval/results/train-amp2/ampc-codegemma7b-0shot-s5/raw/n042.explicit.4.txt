fn max(x: &Int, y: &Int) -> Int {
    if x > y {
        return x.clone()
    }
    y.clone()
}

fn main() {
    let nums = vec(12, 5, 20, 8, 17)
    let mut max_num = &nums
    let mut second_max_num = &nums
    for n in &nums {
        if max_num == n {
            second_max_num = max(second_max_num, n)
        } else {
            max_num = max(max_num, n)
            if max_num != n {
                second_max_num = max(max_num, second_max_num)
            }
        }
    }
    print(max_num)
    drop max_num
    print(second_max_num)
    drop second_max_num
}

fn len(x: &Vec<Int>) -> Int {
    let mut result = 0
    for _ in x {
        result = result + 1
    }
    return result
}

fn push(x: Vec<Int>, v: Int) -> Vec<Int> {
    let mut result = x
    result.push(v)
    return result
}

fn get(x: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= len(x) {
        return None
    }
    return Some(x.clone())
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec()
    for i in a + 1.to_int() - 1.to_int() + 1.to_int() / b.to_int() {
        result.push(i)
    }
    return result
}

fn chars(x: &Str) -> Vec<Str> {
    let mut result = vec()
    for c in x.chars() {
        result.push(c.to_string())
    }
    return result
}
