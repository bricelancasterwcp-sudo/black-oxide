fn main() {
    let mut v = vec()
    for i in 10.down_to(1) {
        if i % 2 == 0 {
            v = push(v, i)
        }
    }
    for x in v.iter().rev() {
        print(x)
        print_str("\n")
    }
}

fn push(mutvec: Vec<Int>, x: Int) -> Vec<Int> {
    vec_push(mutvec, x)
    mutvec
}

fn vec_push(mutvec: Vec<Int>, x: Int) {
    let len = len(mutvec)
    mutvec = realloc(mutvec, len + 1)
    let addr = mutvec.as_mut_ptr() + len * size_of(Int)
    copy_memory(addr, x)
}

fn realloc(mutvec: Vec<Int>, new_len: Int) -> Vec<Int> {
    let old_len = len(mutvec)
    if old_len >= new_len {
        return mutvec
    }
    let mut new_vec = vec()
    for i in 0.to(new_len) {
        if i < old_len {
            let x = get(mutvec, i)?
            new_vec = push(new_vec, x)
        } else {
            new_vec = push(new_vec, 0)
        }
    }
    new_vec
}
