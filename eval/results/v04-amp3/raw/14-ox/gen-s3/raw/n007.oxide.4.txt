fn digit_sum(n: Int) -> Option<Int> {
    let s = int_to_str(n)
    let mut sum = 0
    for ch in chars(s) {
        let digit = parse_int(ch)?
        sum += digit
    }
    Some(sum)
}

fn main() {
    let number = 9875
    let sum = match digit_sum(number) {
        Some(result) => result,
        None => 0,  
    }
    print(sum)
}
