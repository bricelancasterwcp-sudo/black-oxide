struct Number { digits: Vec<Int> }

fn sum_of_digits(n: Int) -> Int {
    let s = int_to_str(n)
    let chars = chars(s)
    let mut sum = 0
    for c in chars {
        let d = parse_int(c)?
        sum = sum + d
    }
    sum
}

fn main() {
    let num = Number { digits: vec() }
    let digits = push(push(push(push(vec(), 5), 7), 8), 9)
    for d in digits {
        num.digits = push(num.digits, d)
    }
    let sum = sum_of_digits(9875)
    print(sum)
}
