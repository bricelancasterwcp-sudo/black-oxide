fn main() {
    let inputs = vec("7", "x", "9", "?");
    let mut sum = 0;
    let mut non_numbers = 0;

    for input in inputs {
        match parse_int(input) {
            Some(n) => sum += n,
            None => non_numbers += 1,
        }
    }

    print(sum);
    print(non_numbers);
}
