struct Queue { items: Vec<Int> }

fn swap_first_last(q: Queue) -> Queue {
    let len = len(q.items);
    if len > 1 {
        let first = get(q.items, 0)?;
        let last = get(q.items, len - 1)?;
        let mut new_items = q.items;
        set(new_items, 0, last);
        set(new_items, len - 1, first);
        Queue { items: new_items }
    } else {
        q
    }
}

fn print_queue(q: Queue) {
    for x in q.items {
        print(x);
        print_str("\n");
    }
}

fn main() {
    let q = Queue { items: push(push(push(push(vec(), 1), 2), 3), 4), 5) };
    let swapped_q = swap_first_last(q);
    print_queue(swapped_q);
}
