struct Rectangle { width: Int, height: Int }
struct Triangle { side1: Int, side2: Int, side3: Int }

fn rectangle_perimeter(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn triangle_perimeter(t: Triangle) -> Int {
    t.side1 + t.side2 + t.side3
}

fn main() {
    let shapes = vec(
        Rectangle { width: 3, height: 4 },
        Triangle { side1: 3, side2: 4, side3: 5 },
        Rectangle { width: 2, height: 5 }
    )
    for shape in shapes {
        match shape {
            Rectangle(r) => print(rectangle_perimeter(r)),
            Triangle(t) => print(triangle_perimeter(t)),
        }
    }
}
