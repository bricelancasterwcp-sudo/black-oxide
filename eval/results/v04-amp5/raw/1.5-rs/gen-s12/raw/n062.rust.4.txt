```rust
fn main() {
    let signal = [1, 2, 3];

    for (index, &value) in signal.iter().enumerate() {
        match value {
            0 => println!("Go for green"),
            1 => println!("Stop for red"),
            2 => println!("Wait for yellow"),
            _ => panic!("Invalid signal: {}", value),
        }
    }
}
```
