fn main() {
    let group_a = [120, 95, 143, 87, 210, 156, 99];
    let group_b = [175, 132, 88, 240, 119, 167];
    for (label, group) in [("A", &group_a), ("B", &group_b)] {
        let sum: i32 = group.iter().sum();
        let count = group.len();
        let average = sum / count as i32;
        let above_avg_count = group.iter().filter(|&&x| x > average).count();
        let min_max_diff = group.iter().max().unwrap() - group.iter().min().unwrap();
        println!("{}", label);
        println!("{}", count);
        println!("{}", average);
        println!("{}", above_avg_count);
        println!("{}", min_max_diff);
    }
    if group_a.iter().sum::<i32>() > group_b.iter().sum::<i32>() {
        println!("A");
    } else {
        println!("B");
    }
}
