```rust
fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut sum = 0;
    let mut count_non_numbers = 0;

    for text in texts {
        match parse_int(text) {
            Some(n) => sum += n,
            None => count_non_numbers += 1,
        }
    }

    print(sum);
    print_str(" ");
    print(count_non_numbers);
}
```
