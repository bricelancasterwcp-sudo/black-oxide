struct Line { items: Vec<Int>, price: Int }

fn main() {
    let line1 = Line { items: push(push(vec(), 4), 4), price: 3 }
    let line2 = Line { items: push(push(vec(), 5), 5), price: 2 }
    print(line1.items.len() * line1.price)
    print(line2.items.len() * line2.price)
    print((line1.items.len() * line1.price) + (line2.items.len() * line2.price))
}
