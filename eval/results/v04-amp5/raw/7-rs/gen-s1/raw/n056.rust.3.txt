```rust
fn main() {
    let mut sum = 0;
    let mut count_not_numbers = 0;

    let texts = ["7", "x", "9", "?"];

    for text in texts.iter() {
        match text.parse::<i32>() {
            Ok(num) => sum += num,
            Err(_) => count_not_numbers += 1,
        }
    }

    println!("Total of numbers: {}", sum);
    println!("Count of non-numbers: {}", count_not_numbers);
}
```
