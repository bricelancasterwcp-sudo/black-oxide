fn main() {
    let morning = vec![94, 71, 88, 65, 92, 79, 83];
    let evening = vec![55, 90, 77, 84, 68, 99];
    let mut sittings = vec![("Morning", &morning), ("Evening", &evening)];
    sittings.sort_by_key(|&(_, marks)| -(marks.iter().sum::<i32>() / marks.len() as i32));
    for (name, marks) in sittings {
        println!("{}", name);
        let mut count_90 = 0;
        let mut count_80 = 0;
        let mut count_70 = 0;
        let mut count_below_70 = 0;
        let mut max_mark = i32::MIN;
        for &mark in marks {
            if mark >= 90 {
                count_90 += 1;
            } else if mark >= 80 {
                count_80 += 1;
            } else if mark >= 70 {
                count_70 += 1;
            } else {
                count_below_70 += 1;
            }
            if mark > max_mark {
                max_mark = mark;
            }
        }
        println!("{}", count_90);
        println!("{}", count_80);
        println!("{}", count_70);
        println!("{}", count_below_70);
        println!("{}", max_mark);
        println!("{}", marks.iter().sum::<i32>() / marks.len() as i32);
    }
}
