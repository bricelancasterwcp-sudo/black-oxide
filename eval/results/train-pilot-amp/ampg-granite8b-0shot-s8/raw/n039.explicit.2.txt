fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < len(&v) {
        Some(v.get(i.trunc()))
    } else {
        None
    }
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v = v
    v.push(x)
    v
}

fn vec() -> Vec<Int> {
    let mut v = vec()
    v
}

fn len(v: &Vec<Int>) -> Int {
    v.len()
}

fn print(x: Option<Int>) {
    match x {
        Some(x) => print_str(str_cat("Some(", int_to_str(x), ")")),

}
