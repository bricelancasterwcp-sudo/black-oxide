fn main() {
    let mut counter = 9
    let mut halved_count = 0
    let mut added_five_count = 0
    let mut tripled_count = 0
    let mut max_value = 0
    let mut max_round = 0
    let mut min_value = 1_000_000
    let mut running_total = 0
    let mut below_ten_count = 0
    for i in range(1, 31) {
        if counter % 2 == 0 {
            counter /= 2
            halved_count += 1
        } else if counter % 3 == 0 {
            counter += 5
            added_five_count += 1
        } else {
            counter = (counter * 3 + 1) % 1_000
            tripled_count += 1
        }
        running_total += counter
        if counter > max_value {
            max_value = counter
            max_round = i
        }
        if counter < min_value {
            min_value = counter
        }
        if counter < 10 {
            below_ten_count += 1
        }
    }
    print(max_value)
    print(halved_count)
    print(added_five_count)
    print(tripled_count)
    print(max_round)
    print(min_value)
    print(running_total)
    print(below_ten_count)
    if counter > 100 {
        print_str("high")
    } else {
        print_str("low")
    }
}
