fn main() {
    let numbers = [360, 97, 1024, 561, 143, 2003];
    let mut primes_count = 0;
    for &num in &numbers {
        println!("{}", num);
        let (divisors, sum, max) = find_divisors(num);
        println!("{}", divisors.len());
        println!("{}", sum);
        println!("{}", max);
        if is_prime(num) {
            println!("true");
            primes_count += 1;
        } else {
            println!("false");
        }
    }
    println!("{}", primes_count);
}
fn find_divisors(mut num: u32) -> (Vec<u32>, u32, u32) {
    let mut divisors = Vec::new();
    let mut sum = 0;
    let mut max = 0;
    for i in 2..=num {
        if num % i == 0 {
            divisors.push(i);
            sum += i;
            max = i;
            while num % i == 0 {
                num /= i;
            }
        }
    }
    (divisors, sum, max)
}
fn is_prime(num: u32) -> bool {
    if num <= 1 {
        return false;
    }
    for i in 2..=((num as f64).sqrt() as u32) {
        if num % i == 0 {
            return false;
        }
    }
    true
}
