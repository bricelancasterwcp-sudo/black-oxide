struct Reading { label: Str, values: Vec<Int> }
fn analyze_sensor(r: Reading) -> Option<Reading> {
    let mut increases = 0
    let mut decreases = 0
    let mut max_increase = 0
    let mut longest_streak = 0
    let mut current_streak = 0
    for i in range(1, len(r.values)) {
        let prev = get(r.values, i - 1)?
        let curr = get(r.values, i)?
        if curr > prev {
            increases += 1
            max_increase = max(max_increase, curr - prev)
            current_streak += 1
        } else if curr < prev {
            decreases += 1
            current_streak = 0
        }
        longest_streak = max(longest_streak, current_streak)
    }
    Some(Reading { label: r.label, values: vec(increases, decreases, max_increase, longest_streak) })
}
fn main() {
    let north_values = push(push(push(push(push(push(push(vec(), 12), 15), 19), 14), 14), 20), 31), 30)
    let south_values = push(push(push(push(push(vec(), 40), 38), 41), 45), 50), 44)
    let north_result = analyze_sensor(Reading { label: "north", values: north_values })
    let south_result = analyze_sensor(Reading { label: "south", values: south_values })
    match (north_result, south_result) {
        (Some(north), Some(south)) => {
            print_str(north.label)

    }
}
