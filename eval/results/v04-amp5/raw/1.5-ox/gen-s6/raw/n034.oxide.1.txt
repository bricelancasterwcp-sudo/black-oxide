```rust
fn helper(v: Vec<Int>) -> Option<Int> {
    let mut second = None;
    for x in v {
        if second.is_none() {
            second = Some(x);
        } else if x > second.unwrap() {
            second = Some(x);
        }
    }
    second
}

fn main() {
    println!("{:?}", helper(vec![7, 8, 9])); // prints: Some(8)
}
```
