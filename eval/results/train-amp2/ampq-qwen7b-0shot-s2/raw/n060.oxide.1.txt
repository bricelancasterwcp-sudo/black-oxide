fn sum_and_count_digits(s: Str) -> Int {
    let mut sum = 0
    let mut count = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            sum = sum + parse_int(c).unwrap()
            count = count + 1
        }
    }
    sum
}

fn main() {
    let s = "a1b22"
    let total = sum_and_count_digits(s)
    print(total)
    print_str("\n")
    print(total)
}
