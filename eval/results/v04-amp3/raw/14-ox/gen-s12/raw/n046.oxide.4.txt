fn main() {
    let values = push(push(push(push(vec(), 5), 12), 3), 18)
    let cutoff = 10
    let under = 0
    let over = 0

    for x in values {
        if x < cutoff {
            under = under + 1
        } else {
            over = over + 1
        }
    }

    print(under)
    print(over)
}
