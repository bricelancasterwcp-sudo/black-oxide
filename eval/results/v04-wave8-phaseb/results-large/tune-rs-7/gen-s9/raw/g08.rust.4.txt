fn main() {
    let north = vec![12, 15, 19, 14, 14, 20, 31, 30];
    let south = vec![40, 38, 41, 45, 50, 44];
    fn analyze(sensor: &Vec<i32>) -> (i32, i32, i32, i32) {
        let mut increased = 0;
        let mut decreased = 0;
        let mut max_increase = 0;
        let mut longest_streak = 0;
        let mut current_streak = 0;
        for i in 1..sensor.len() {
            if sensor[i] > sensor[i - 1] {
                increased += 1;
                current_streak += 1;
                max_increase = max_increase.max(sensor[i] - sensor[i - 1]);
            } else if sensor[i] < sensor[i - 1] {
                decreased += 1;
                current_streak = 0;
            }
            longest_streak = longest_streak.max(current_streak);
        }
        (increased, decreased, max_increase, longest_streak)
    }
    let (north_increased, north_decreased, north_max_increase, north_longest_streak) = analyze(&north);
    let (south_increased, south_decreased, south_max_increase, south_longest_streak) = analyze(&south);
    println!("Sensor north");
    println!("{}", north_increased);
    println!("{}", north_decreased);
    println!("{}", north_max_increase);
    println!("{}", north_longest_streak);
    println!("Sensor south");
    println!("{}", south_increased);
    println!("{}", south_decreased);
    println!("{}", south_max_increase);
    println!("{}", south_longest_streak);
    if north_longest_streak > south_longest_streak {
        println!("north");
    } else {
        println!("south");
    }
}
