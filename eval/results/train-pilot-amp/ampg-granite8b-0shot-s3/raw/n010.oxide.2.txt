struct Int { value: Int }

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        false
    } else if n <= 3 {
        true
    } else if n % 2 == 0 || n % 3 == 0 {
        false
    } else {
        let mut i = 5
        while i * i <= n {
            if n % i == 0 || n % (i + 2) == 0 {
                return false
            }
            i = i + 6
        }
        true
    }
}

fn main() {
    let n = 91
    if is_prime(Int { value: n }) {
        print_str("yes")
    } else {
        print_str("no")
    }
}
