fn to_str(n: Int) -> Str {
    if n == 0 {
        return "0".to_str()
    }
    let mut digits: Vec<Str> = vec()
    let mut num = n
    while num > 0 {
        digits.push(int_to_str(num % 10))
        num = num / 10
    }
    digits.reverse()
    let mut s = "".to_str()
    for d in digits {
        s = concat(s, d)
    }
    s
}
