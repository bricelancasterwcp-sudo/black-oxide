struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 10 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let mut x = get(v, 1)?;
    x += 1;
    Some(x)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) };
    print(first_big(r.values));
    let r2 = Reading { label: "lab2", ..r };
    let largest = max(r2.values).unwrap();
    let smallest = min(r2.values).unwrap();
    println(smallest);
    println(largest - smallest);
}
