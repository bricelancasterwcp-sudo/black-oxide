fn main() {
    let phrase = "when in doubt go home".to_string()
    let word_count = split(phrase, " ").len()
    let character_count = str_len(phrase)
    print(word_count)
    print(character_count)
}

fn split(s: Str, delimiter: Str) -> Vec<Str> {
    let mut result = vec()
    let mut start = 0
    let mut end = 0
    while end < str_len(s) {
        if str_eq(substring(s, end, end + 1), delimiter) {
            push(result, substring(s, start, end))
            start = end + 1
        }
        end = end + 1
    }
    push(result, substring(s, start, end))
    result
}

fn substring(s: Str, start: Int, end: Int) -> Str {
    let mut result = ""
    for i in start < end {
        result = concat(result, substring(s, i, i + 1))
    }
    result
}

fn str_eq(a: Str, b: Str) -> Bool {
    let mut i = 0
    while i < str_len(a) && i < str_len(b) {
        if not(str_eq(substring(a, i, i + 1), substring(b, i, i + 1))) {
            return False
        }
        i = i + 1
    }
    str_len(a) == str_len(b)
}

fn str_len(s: Str) -> Int {
    let mut result = 0
    for _ in chars(s) {
        result = result + 1
    }
    result
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = vec()
    for i in 0 < str_len(s) {
        push(result, substring(s, i, i + 1))
    }
    result
}

fn concat(a: Str, b: Str) -> Str {
    let mut result = ""
    for s in vec(a, b) {
        result = result + s
    }
    result
}

fn vec_to_str(v: Vec<T>) -> Str {
    let mut result = "["
    for i in 0 < len(v) {
        result = concat(result, to_string(v.get(i).unwrap()))
        if i != len(v) - 1 {
            result = concat(result, ", ")
        }
    }
    concat(result, "]")
}

fn struct_to_str(s: Struct) -> Str {
    let mut result = "{"
    for f in struct_fields(s) {
        result = concat(result, concat(f.name, ": "))
        result = concat(result, to_string(f.value))
        result = concat(result, ", ")
    }
    concat(result, "}")
}

fn enum_to_str(e: Enum) -> Str {
    match e {
        Variant(args) => {
            let mut result = e.name
            for a in args {
                result = concat(result, concat(", ", to_string(a)))
            }
            result
        },
        Nullary => e.name.to_string(),
    }
}

fn tuple_to_str(t: Tuple) -> Str {
    let mut result = "("
    for i in 0 < len(t) {
        result = concat(result, to_string(t.get(i).unwrap()))
        if i != len(t) - 1 {
            result = concat(result, ", ")
        }
    }
    concat(result, ")")
}
