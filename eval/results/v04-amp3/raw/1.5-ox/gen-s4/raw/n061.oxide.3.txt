struct Rectangle {
    width: Int,
    height: Int,
}

fn main() {
    let rect = Rectangle { width: 9, height: 3 };
    print_area(rect);
    print_perimeter(rect);
}

fn area(r: Rectangle) -> Int {
    r.width * r.height
}

fn perimeter(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}
