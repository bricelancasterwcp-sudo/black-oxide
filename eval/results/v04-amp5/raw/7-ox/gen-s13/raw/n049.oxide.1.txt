fn min(a: Int, b: Int) -> Int {
    if a < b { a } else { b }
}

fn max(a: Int, b: Int) -> Int {
    if a > b { a } else { b }
}

fn main() {
    let figures = vec(8, 3, 11, 6);
    let smallest = min(min(figures[0], figures[1]), min(figures[2], figures[3]));
    let largest = max(max(figures[0], figures[1]), max(figures[2], figures[3]));
    print(smallest);
    print(max(largest, -smallest));
}
