```rust
struct Sensor {
    reading: Int,
}

struct Device {
    sensor: Sensor,
}

fn set_reading(device: Device, new_reading: Int) -> Device {
    let mut updated_device = device;
    updated_device.sensor.reading = new_reading;
    updated_device
}

fn main() {
    let initial_device = Device { sensor: Sensor { reading: 0 } };
    let updated_device = set_reading(initial_device, 7);
    print(updated_device.sensor.reading);
}
```
