struct Queue { values: Vec<Int> }

fn main() {
    let q = Queue { values: vec(1, 2, 3, 4, 5) }
    let len = len(q.values)
    if len > 1 {
        let first = get(q.values, 0).unwrap_or(-1)
        let last = get(q.values, len - 1).unwrap_or(-1)
        let mid = range(1, len - 1)
        let new_values = push(push(push(vec(), last), mid), first)
        for x in new_values {
            print(x)
            print_str("\n")
        }
    } else {
        print_str("Queue is too short")
    }
}
