struct Rectangle {
    width: Int,
    height: Int,
}

impl Rectangle {
    fn area(&self) -> Int {
        self.width * self.height
    }

    fn perimeter(&self) -> Int {
        2 * (self.width + self.height)
    }
}

fn main() {
    let rect = Rectangle { width: 9, height: 3 };
    print(int_to_str(rect.area()));
    print(str_len(concat("The area is ", int_to_str(rect.area()), " and the perimeter is ",
                         int_to_str(rect.perimeter()))));
}
