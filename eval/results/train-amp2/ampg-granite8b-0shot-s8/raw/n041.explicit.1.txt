fn main() {
    let mut measurements = vec(9, 2, 7, 4)
    let mut sorted = measurements.clone()
    sort(&mutsorted)
    for num in &sorted {
        print(num)
    }
}

fn sort(v: &Vec<Int>) {
    let mut i = 0
    while i < len(&v) {
        let mut j = i
        while j < len(&v) {
            if get(&v, j) < get(&v, j + 1) {
                let temp = get(&v, j)
                set(&mutv, j, get(&v, j + 1))
                set(&mutv, j + 1, temp)
            }
            j = j + 1
        }
        i = i + 1
    }
}
