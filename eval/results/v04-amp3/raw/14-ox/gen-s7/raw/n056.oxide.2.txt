struct NumberHolder { value: Option<Int> }

fn sum_valid_numbers(holders: Vec<NumberHolder>) -> Int {
    let mut sum = 0
    for holder in holders {
        match holder.value {
            Some(n) => sum = sum + n,
            None => continue,
        }
    }
    sum
}

fn count_invalid_numbers(holders: Vec<NumberHolder>) -> Int {
    let mut count = 0
    for holder in holders {
        match holder.value {
            Some(_) => continue,
            None => count = count + 1,
        }
    }
    count
}

fn main() {
    let numbers = vec(
        NumberHolder { value: Some(7) },
        NumberHolder { value: None },
        NumberHolder { value: Some(9) },
        NumberHolder { value: None },
    )
    print(sum_valid_numbers(numbers))
    print(count_invalid_numbers(numbers))
}
