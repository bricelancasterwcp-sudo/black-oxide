fn sum_of_first_six_multiples_of_four() -> Int {
    let mut total = 0
    let mut list = vec()
    for x in range(0, 24) {
        if x % 4 == 0 {
            list = push(list, x) // list is moved here
            total = total + x
        }
        if list.len() == 6 {
            break
        }
    }
    drop list
    total
}

fn main() {
    print(sum_of_first_six_multiples_of_four())
}
