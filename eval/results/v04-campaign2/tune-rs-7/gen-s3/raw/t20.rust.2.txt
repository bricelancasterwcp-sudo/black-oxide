fn main() {
    let items = vec![
        ("pen", 3, 4),
        ("pad", 9, 2),
        ("ink", 5, 3),
    ];
    let mut total_cost = 0;
    let mut max_cost_item = "";
    for (name, unit_price, quantity) in &items {
        let cost = unit_price * quantity;
        total_cost += cost;
        if cost > 0 && (max_cost_item.is_empty() || cost > items.iter().map(|(_, up, q)| up * q).max().unwrap_or(0)) {
            max_cost_item = name;
        }
    }
    println!("{}", total_cost);
    println!("{}", max_cost_item);
}
