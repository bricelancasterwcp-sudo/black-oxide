fn main() {
    let total = 0
    for x in range_inc_one_to_nine_hundred_and_forty_two() {
        if x % 3 == 0 || x % 5 == 0 {
            total = total + x
        }
    }
    print(total)
}

fn range_inc_one_to_nine_hundred_and_forty_two() -> Vec<Int> {
    let v = vec()
    for i in 1.to_int(900).iter() {
        push(v, i)
    }
    v
}

fn factor_sum(limit: Int) -> Int {
    let total = 0
    let v = range_inc_one_to_nine_hundred_and_forty_two()
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    drop v
    total
}

fn main() {
    print(factor_sum(1000))
}
