fn second(v: Vec<Int>) -> Option<Int> {
    get(v, 1)?
}

fn main() {
    let v7 = push(push(vec(), 3), 42);
    let v8 = push(push(vec(), 5), 6);
    let v9 = vec();

        None => print_str("none"),
    }
