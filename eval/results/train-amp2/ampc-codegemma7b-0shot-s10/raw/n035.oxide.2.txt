fn divide(a: Float, b: Float) -> Result<Float, Str> {
    if b == 0.0 {
        Err("division by zero".to_string())
    } else {
        Ok(a / b)
    }
}

fn main() {
    match divide(20.0, 4.0) {
        Ok(ans) => print(ans),
        Err(e) => print_str(e.as_str()),
    }
    match divide(20.0, 0.0) {
        Ok(ans) => print(ans),
        Err(e) => print_str(e.as_str()),
    }
}
