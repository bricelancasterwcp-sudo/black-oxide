fn main() {
    let arr = [8, "oak", 15];

    for (i, &item) in arr.iter().enumerate() {
        match item {
            8 => print!("{0} is a number"), i,
            "oak" => println!("{0} is a tree"), i,

            _ => print!("bad"),
        }
    }
}
