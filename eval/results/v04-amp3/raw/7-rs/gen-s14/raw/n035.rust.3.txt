fn divide(a: i32, b: i32) -> Result<f64, &'static str> {
    if b == 0 {
        Err("Refusing to divide by zero.")
    } else {
        Ok(a as f64 / b as f64)
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("Result: {}", result),
        Err(message) => println!("Bad: {}", message),
    }

    match divide(20, 0) {
        Ok(_) => panic!("This should not happen"),
        Err(message) => println!("Bad: {}", message),
    }
}
