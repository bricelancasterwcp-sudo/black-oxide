struct Keyboard { letters: Str }

fn count_vowels(s: Str) -> Int {
    let mut count = 0
    for c in s.chars() {
        if c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' ||
           c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U' {
            count += 1
        }
    }
    count
}

fn main() {
    let k = Keyboard { letters: concat("abcdefghijklmnopqrstuvwxyz", "ABCDEFGHIJKLMNOPQRSTUVWXYZ") }
    let vowels = count_vowels(k.letters)
    let consonants = len(k.letters) - vowels
    print(vowels)
    print(consonants)
}
