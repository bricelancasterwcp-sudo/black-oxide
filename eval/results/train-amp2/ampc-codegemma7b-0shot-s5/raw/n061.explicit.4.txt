fn area(r: &Rectangle) -> Float {
    r.width * r.height
}

fn perimeter(r: &Rectangle) -> Float {
    2.0 * (r.width + r.height)
}

fn main() {
    let rect = Rectangle { width: 9.0, height: 3.0 }
    print(area(&rect))
    print(perimeter(&rect))
    drop_any(&rect)
}

struct Rectangle {
    width: Float,
    height: Float,
}

fn print(x: Float) {
    print(format("{:.1}", x))
}

    print(s)

    let _ = x
}
