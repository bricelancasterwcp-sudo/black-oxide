fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    for i in range(2, trunc(sqrt(n)) + 1) {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let x = int_to_str(91)
    if is_prime(parse_int(x).unwrap()) {
        print_str("yes")
    } else {
        print_str("no")
    }
}
