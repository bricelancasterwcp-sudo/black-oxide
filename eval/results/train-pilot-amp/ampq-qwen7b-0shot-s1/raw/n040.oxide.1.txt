struct Shape { variant: Variant }

enum Variant {
    Square(Int),
    Circle(Int),
}

fn area(shape: Shape) -> Float {
    match shape.variant {
        Variant(square) => 3.0 * square * square,
        Variant(circle) => 3.0 * circle * circle,
    }
}

fn main() {
    let s = Shape { variant: Variant(Square(4)) }
    print(area(s))
}
