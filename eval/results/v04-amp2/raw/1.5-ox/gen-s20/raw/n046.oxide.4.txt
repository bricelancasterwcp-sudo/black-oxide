```rust
fn main() {
    let numbers = vec![5, 12, 3, 18, 9];
    let cutoff = 10;

    let count_below_cutoff = numbers.iter().filter(|&x| x < cutoff).count();
    let count_at_or_over_cutoff = numbers.iter().filter(|&x| x >= cutoff).count();

    println!("Numbers below the cutoff: {}", count_below_cutoff);
    println!("Numbers at or over the cutoff: {}", count_at_or_over_cutoff);
}
```
