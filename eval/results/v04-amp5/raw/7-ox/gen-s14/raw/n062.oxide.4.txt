```rust
fn main() {
    let signals = [("green", "go"), ("red", "stop"), ("yellow", "wait")];
    for signal in signals.iter() {
        match signal {
            ("green", _) => print!("go\n"),
            ("red", _) => print!("stop\n"),
            ("yellow", _) => print!("wait\n"),
            _ => unreachable!(),
        }
    }
}
```
