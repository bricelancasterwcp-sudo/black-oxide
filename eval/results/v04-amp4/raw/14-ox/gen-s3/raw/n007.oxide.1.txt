```rust
fn digit_sum(n: Int) -> Int {
    let s = int_to_str(n)
    let mut sum = 0
    for ch in chars(s) {
        let digit = parse_int(ch)?
        sum += digit
    }
    Some(sum)
}

fn main() {
    let number = 9875
    match digit_sum(number) {
        Some(sum) => print(sum),
        None => print_str("Error parsing digits"),
    }
}
```
