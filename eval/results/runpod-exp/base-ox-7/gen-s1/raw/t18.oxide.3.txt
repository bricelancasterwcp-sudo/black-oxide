```rust
struct Rectangle { width: Int, height: Int }
struct Triangle { sides: Vec<Int> }

fn perimeter_of_rectangle(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn perimeter_of_triangle(t: Triangle) -> Int {
    t.sides.iter().sum()
}

fn main() {
    let rect1 = Rectangle { width: 3, height: 4 };
    let tri = Triangle { sides: vec![3, 4, 5] };
    let rect2 = Rectangle { width: 2, height: 5 };

    print(perimeter_of_rectangle(rect1));
    print(perimeter_of_triangle(tri));
    print(perimeter_of_rectangle(rect2));
}
```
