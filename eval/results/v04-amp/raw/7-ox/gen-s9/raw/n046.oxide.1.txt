fn main() {
    let samples = vec(5, 12, 3, 18, 9);
    let below_cutoff = count_if(samples, |x| x < 10);
    let at_or_over_cutoff = count_if(samples, |x| x >= 10);
    print(below_cutoff);
    print_str(" ");
    print(at_or_over_cutoff);
}

fn count_if<T>(v: Vec<T>, f: fn(T) -> Bool) -> Int {
    let mut count = 0;
    for x in v {
        if f(x) {
            count += 1;
        }
    }
    count
}
