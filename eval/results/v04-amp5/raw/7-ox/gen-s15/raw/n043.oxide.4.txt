fn swap_first_last(v: Vec<Int>) -> Vec<Int> {
    let len = len(v);
    if len > 1 {
        let last = get(v.clone(), len - 1)?;
        set(swap(v.clone(), 0, last), len - 1, v[0])
    } else {
        v
    }
}

fn main() {
    let queue = push(push(push(push(vec(), 1), 2), 3), 4);
    let queue = push(queue, 5);
    let queue = swap_first_last(queue.clone());
    for x in queue {
        print(x);
    }
}
