fn fib(n: Int) -> Int {
    let a = 1
    let b = 1
    if n == 0 {
        return 1
    } else if n == 1 {
        return 1
    } else {
        let mut i = 2
        let mut result = 0
        while i <= n {
            result = a + b
            let temp = a
            a = b
            b = result
            i = i + 1
        }
        return result
    }
}
