fn main() {
    let mut balance = 0;

    deposit(balance, 3);
    println!("{}", balance);

    deposit(balance, 1);
    println!("{}", balance);

    deposit(balance, 4);
    println!("{}", balance);

    deposit(balance, 1);
    println!("{}", balance);

    deposit(balance, 5);
    println!("{}", balance);
}

fn deposit(mut balance: i32, amount: i32) {
    balance += amount;
}
