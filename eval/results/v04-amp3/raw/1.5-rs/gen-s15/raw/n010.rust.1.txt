```rust
fn is_prime(num: u32) -> bool {
    if num <= 1 {
        return false;
    }
    for i in 2..num {
        if num % i == 0 {
            return false;
        }
    }
    true
}

fn main() {
    let number = 91;
    if is_prime(number) {
        println!("yes");
    } else {
        println!("no");
    }
}
```
