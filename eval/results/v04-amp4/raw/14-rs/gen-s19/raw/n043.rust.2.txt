fn main() {
    let mut queue = vec![1, 2, 3, 4, 5];
    
    if let Some((first, last)) = queue.first().cloned().zip(queue.last().cloned()) {
        let len = queue.len();
        queue[0] = last;
        queue[len - 1] = first;
    }
    
    for entry in &queue {
        println!("{}", entry);
    }
}
