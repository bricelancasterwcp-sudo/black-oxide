fn main() {
    let state = get_signal()?
    if state == "green" {
        print_str("go")
    } else if state == "yellow" {
        print_str("wait")
    } else if state == "red" {
        print_str("stop")
    } else {
        print_str("unknown")
    }
}

fn get_signal() -> Option<Str> {
    print_str("Enter signal: ")
    let mut input = String.new()
    std_in().read_line_into(input)
    if input == "" {
        None
    } else {
        Some(input.trim().to_lowercase())
    }
}

fn std_in() -> StdIn {
    let mut stdin = std_in_raw()
    stdin.set_unget(false)
    stdin
}

fn std_in_raw() -> StdInRaw {
    unsafe
    std_in_raw_internal()
}

fn std_in_raw_internal() -> StdInRaw {
    let mut stdin = StdInRaw { buffer: 0 }
    stdin.init()
    stdin
}
