```rust
struct Readings { values: Vec<Int> }

fn count_exceeding_five(v: Vec<Int>) -> Int {
    let mut count = 0;
    for x in v {
        if x > 5 {
            count += 1;
        }
    }
    count
}

fn highest_reading(v: Vec<Int>) -> Option<Int> {
    max(v)
}

fn main() {
    let readings = Readings { values: push(push(push(push(vec(), 6), 11), 3), 14), 9) };
    print(count_exceeding_five(readings.values));
    match highest_reading(readings.values) {
        Some(n) => print(n),
        None => print_str("no readings"),
    }
}
```
