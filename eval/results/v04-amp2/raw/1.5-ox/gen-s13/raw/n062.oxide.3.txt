```rust
fn main() {
    let mut signal = "green";
    while true {
        match signal {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => panic!("invalid signal: {}", signal),
        }
        // Simulate arrival of a new signal
        if rand::random::<bool>() {
            signal = if rand::random::<bool>() { "green" } else { "red" };
        } else {
            signal = "yellow";
        }
    }
}
```
