```rust
fn main() {
    let n = 91;
    let mut divisors = Vec::new();
    for i in 2..n {
        if n % i == 0 {
            divisors.push(i);
        }
    }
    if divisors.len() > 2 || (divisors.len() == 2 && divisors[0] != 1) {
        print_str("yes");
    } else {
        print_str("no");
    }
}
```
