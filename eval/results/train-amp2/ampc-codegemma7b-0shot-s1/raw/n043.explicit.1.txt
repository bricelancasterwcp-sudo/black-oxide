fn queue_exchange(v: &Vec<Int>) -> Vec<Int> {
    let first = v.first().unwrap()
    let last = v.last().unwrap()
    let mut new_queue = v.clone()
    new_queue.remove(0)
    new_queue.push(first)
    new_queue.remove(new_queue.len() - 1)
    new_queue.insert(0, last)
    new_queue
}

fn main() {
    let mut queue = vec(1, 2, 3, 4, 5)
    queue = queue_exchange(&queue)
    for x in &queue {
        print(x)
    }
    drop queue
}
