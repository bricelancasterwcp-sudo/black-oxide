```rust
fn main() {
    let values = [5, 12, 3, 18, 9];
    let cutoff = 10;
    let below = values.iter().filter(|&&x| x < cutoff).count();
    let at_or_over = values.iter().filter(|&&x| x >= cutoff).count();
    println!("Values below the cutoff: {}", below);
    println!("Values at or over the cutoff: {}", at_or_over);
}
```
