use std::collections::HashSet;

fn main() {
    let samples = vec![5, 12, 3, 18, 9];
    let cutoff = 10;

    let under_cutoff: HashSet<_> = samples.iter().copied().filter(|x| x < &cutoff).collect();
    let at_or_over_cutoff: HashSet<_> = samples.iter().copied().filter(|x| x >= &cutoff).collect();

    println!("Under cutoff: {}", under_cutoff.len());
    println!("At or over cutoff: {}", at_or_over_cutoff.len());
}
