```rust
fn main() {
    let mut freq = [0; 26];
    for c in "coffee".chars() {
        if 'a' <= c && c <= 'z' {
            let idx = (c as u8 - b'a') as usize;
            freq[idx] += 1;
        }
    }

    for idx in 0..26 {
        if freq[idx] > 1 {
            print!("{}", (idx + b'a') as char);
        }
    }
}
```
