fn main() {
    let a = read_str("Enter first entry: ");
    match a {
        Some(n) => println!("{}", n),
        None => println!("bad"),
    }

    let b = read_str("Enter second entry: ");
    match b {
        Some(n) => println!("{}", n),
        None => println!("bad"),
    }

    let c = read_str("Enter third entry: ");
    match c {
        Some(n) => println!("{}", n),
        None => println!("bad"),
    }
}

fn read_str(prompt: &str) -> Option<&str> {
    print!("{}", prompt);
    let mut input = String::new();
    std::io::stdin().read_line(&mut input).ok()?;
    input.trim_end_matches('\n').parse()
}
