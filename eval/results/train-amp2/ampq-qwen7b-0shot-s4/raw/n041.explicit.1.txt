fn sort_four(a: Int, b: Int, c: Int, d: Int) -> Vec<Int> {
    let mut v = vec(a, b, c, d)
    for i in 0 < 3 {
        for j in 0 < 3 - i {
            if v(j) > v(j + 1) {
                let temp = v(j)
                v(j) == v(j + 1)
                v(j + 1) == temp
            }
        }
    }
    v
}

fn main() {
    let measurements = sort_four(9, 2, 7, 4)
    for m in &measurements {
        print(m)
        print_str("\n")
    }
    drop measurements
}
