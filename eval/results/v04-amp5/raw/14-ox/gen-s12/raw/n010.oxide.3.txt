fn is_prime(n: Int) -> Bool {
    if n < 2 {
        return false
    }
    for i in range(2, trunc(to_float(n) / 2.0) + 1) {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let num = 91
    if is_prime(num) {
        print_str("yes")
    } else {
        print_str("no")
    }
}
