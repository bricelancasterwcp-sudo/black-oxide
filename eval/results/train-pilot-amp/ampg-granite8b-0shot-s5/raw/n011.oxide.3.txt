fn fold(v: Vec<Int>, acc: Int, op: Int, x: Int) -> Int {
    match v {
        Vec(head, rest) => {
            let next = op(head, x)
            fold(rest, acc, op, next)
        },
        Vec(nullary) => acc,
    }
}

fn add(x: Int, y: Int) -> Int {
    x + y
}
