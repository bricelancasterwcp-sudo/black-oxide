fn main() {
    let square_side = 4;
    let pi = 3.0;
    let square_area = square_side * square_side;
    let circle_area = pi * (square_side as f32 * square_side as f32);
    println!("Area of square: {}", square_area);
    println!("Area of circle with radius {} as radius: {}", square_side, circle_area);
}
