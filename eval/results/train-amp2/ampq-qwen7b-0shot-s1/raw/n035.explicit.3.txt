fn div(x: Int, y: Int) -> Result<Int, Str> {
    if y == 0 {
        Err("bad".to_string())
    } else {
        Ok(x / y)
    }
}

fn main() {
    match div(20, 4)? {
        Ok(result) => {
            print(result)
        },
        Err(e) => {
            print_str(e)
        },
    }
    match div(20, 0)? {
        Ok(result) => {
            print(result)
        },
        Err(e) => {
            print_str(e)
        },
    }
}
