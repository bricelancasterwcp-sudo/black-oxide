fn second_entry(v: Vec<Int>) -> Option<Int> {
    match get(v, 1) {
        Some(x) => Some(x + 1),
        None => None,
    }
}

fn main() {
    print_str(if let Some(n) = second_entry(push(push(vec(), 7), 8)) { int_to_str(n) } else { "none".to_string() });
    print_str(if let Some(n) = second_entry(push(push(vec(), 8), 9)) { int_to_str(n) } else { "none".to_string() });
    print_str(if let Some(n) = second_entry(push(vec(), 9)) { int_to_str(n) } else { "none".to_string() });
}
