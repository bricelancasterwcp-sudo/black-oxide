fn walk(vec: Vec<Int>) -> Int {
    let total = 0
    for i in range(0, vec.len()) {
        let x = get(vec, i)?
        if x % 2 == 1 {
            let x2 = x * 2
            print_str(to_str(x2))
            print_str("\n")
        } else {
            total = total + x
        }
    }
    total
}
