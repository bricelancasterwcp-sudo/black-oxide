fn max(a: Int, b: Int) -> Int {
    if a > b {
        a
    } else {
        b
    }
}

fn second_largest(v: Vec<Int>) -> Int {
    let sorted = sort_vec(&v)
    max(sorted.get(0), sorted.get(1))
}

fn main() {
    let nums = push(push(vec(), 12), 5)
    print(max(40, 20))
    print(second_largest(&nums))
    drop nums
}

fn sort_vec(v: &Vec<Int>) -> Vec<Int> {
    let mut v = clone(&v)
    for i in 0.to_int(len(&v)) {
        for j in (i + 1).to_int(len(&v)) {
            if v.get(j) < v.get(i) {
                let t = v.get(i)
                v = functional_update(&v, i, v.get(j))
                v = functional_update(&v, j, t)
            }
        }
    }
    v
}

fn functional_update(v: &Vec<Int>, index: Int, new_value: Int) -> Vec<Int> {
    push(push(vec(), v.get(0)), new_value)
}

fn main() {
    let nums = push(push(vec(), 12), 5)
    print(max(40, 20))
    print(second_largest(&nums))
    drop nums
}
