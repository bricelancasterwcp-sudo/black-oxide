fn main() {
    let mut max_len = 0;
    let mut result = "";

    if "fig".len() > max_len {
        max_len = "fig".len();
        result = "fig";
    }
    if "banana".len() > max_len {
        max_len = "banana".len();
        result = "banana";
    }
    if "kiwi".len() > max_len {
        max_len = "kiwi".len();
        result = "kiwi";
    }

    println!("{} {}", result, max_len);
}
