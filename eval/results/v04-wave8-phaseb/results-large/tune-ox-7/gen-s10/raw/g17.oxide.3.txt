fn main() {
    let v1 = vec(2, 5, 9, 14, 21, 30)
    let v2 = vec(1, 5, 8, 15, 22)
    let mut merged = vec()
    let mut i = 0
    let mut j = 0
    while i < len(v1) && j < len(v2) {
        if get(v1, i).unwrap_or(-1) <= get(v2, j).unwrap_or(-1) {
            merged = push(merged, get(v1, i).unwrap_or(-1))
            i += 1
        } else {
            merged = push(merged, get(v2, j).unwrap_or(-1))
            j += 1
        }
    }
    while i < len(v1) {
        merged = push(merged, get(v1, i).unwrap_or(-1))
        i += 1
    }
    while j < len(v2) {
        merged = push(merged, get(v2, j).unwrap_or(-1))
        j += 1
    }
    print(len(merged))
    print(get(merged, 0).unwrap_or(-1))
    print(get(merged, len(merged) - 1).unwrap_or(-1))
    let mut sum = 0
    for x in merged {
        sum += x
    }
    print(sum)
    let mut non_decreasing = true
    let mut equal_followed = 0
    for i in range(1, len(merged)) {
        if get(merged, i).unwrap_or(-1) < get(merged, i - 1).unwrap_or(-1) {
            non_decreasing = false
        }
        if get(merged, i).unwrap_or(-1) == get(merged, i - 1).unwrap_or(-1) {
            equal_followed += 1
        }
    }
    print(non_decreasing)
    print(equal_followed)
}
