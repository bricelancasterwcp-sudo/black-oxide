struct Reading { label: Str, values: Vec<Int> }

fn count_above_five(v: Vec<Int>) -> Int {
    let count = 0
    for x in v {
        if x > 5 {
            count += 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max.is_none() || x > unwrap_or(max, -1) {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = vec(6, 11, 3, 14, 9)
    print(count_above_five(readings.clone()))
    let max_reading = find_max(readings)
    match max_reading {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
