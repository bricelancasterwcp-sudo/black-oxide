fn main() {
    let record = "name=widget;qty=17;price=245;tag=blue";
    let pairs: Vec<&str> = record.split(';').collect();
    println!("{}", pairs.len());
    for pair in pairs.iter() {
        let parts: Vec<&str> = pair.split('=').collect();
        if parts.len() == 2 {
            println!("{}", parts[0]);
            println!("{}", parts[1]);
        }
    }
    let qty: i32 = pairs.iter().find(|&p| p.starts_with("qty=")).unwrap_or("").split('=').nth(1).unwrap_or("0").parse().unwrap();
    let price: i32 = pairs.iter().find(|&p| p.starts_with("price=")).unwrap_or("").split('=').nth(1).unwrap_or("0").parse().unwrap();
    println!("{}", qty * price);
}
