struct Rect { width: Int, height: Int }

fn area(r: Rect) -> Int {
    r.width * r.height
}

fn perimeter(r: Rect) -> Int {
    2 * (r.width + r.height)
}

fn main() {
    let r = Rect { width: 9, height: 3 }
    print(area(r))          // prints 27
    print(perimeter(r))     // prints 24
}
