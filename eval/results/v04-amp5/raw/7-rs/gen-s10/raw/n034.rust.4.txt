fn get_second_entry<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}

fn main() {
    let list7 = vec![7];
    println!("{:?}", get_second_entry(&list7)); // None

    let list8: Vec<i32> = vec![];
    println!("{:?}", get_second_entry(&list8)); // None

    let list9 = vec![9, 10, 11];
    println!("{:?}", get_second_entry(&list9)); // Some(10)
}
