fn main() {
    assert_eq!(second_entry::<i32>(vec![1, 2, 3]), Some(2));
    assert_eq!(second_entry::<i32>(vec![1]), None);
    assert_eq!(second_entry::<i32>(vec![]), None);
}

fn second_entry<T: Clone>(xs: Vec<T>) -> Option<T> {
    match xs.get(1) {
        Some(x) => Some(x.clone()),
        None => None,
    }
}
