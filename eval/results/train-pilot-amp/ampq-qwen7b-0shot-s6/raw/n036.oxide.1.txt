struct Stock { units: Vec<Int>, values: Vec<Int> }

fn total_value(stock: Stock) -> Int {
    let mut total = 0
    for i in range(0, len(stock.units)) {
        total = total + stock.units.get(i)? * stock.values.get(i)?
    }
    total
}

fn main() {
    let stock = Stock { units: push(push(push(vec(), 4), 4), 4), values: push(push(push(vec(), 3), 3), 2) }
    print(total_value(stock))
}
