fn count_vowels(s: Str) -> Int {
    let mut vowel_count = 0;
    for c in chars(s) {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" ||
           c == "A" || c == "E" || c == "I" || c == "O" || c == "U" {
            vowel_count += 1;
        }
    }
    vowel_count
}

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm";
    let vowels = count_vowels(keyboard);
    let consonants = str_len(keyboard) - vowels;
    print(vowels);
    print_str(" ");
    print(consonants);
}
