fn count_vowels(word: &Str) -> Int {
    let vowels = "aeiouAEIOU"
    let mut count = 0
    for c in word.chars() {
        if vowels.contains(c) {
            count = count + 1
        }
    }
    count
}

fn main() {
    let word = "Hello, world!"
    let num_vowels = count_vowels(word)
    print_str("Number of vowels: ")
    print(num_vowels)
    drop word
    drop num_vowels
}
