fn main() {
    let mut numbers = vec![7, 4, 9, 10];
    let mut total_even = 0;

    // Iterate over only the first 4 elements (up to index 3) because there are 4 entries
    for &number in numbers.iter().take(4) {
        if number % 2 != 0 {
            println!("{}: {}", number, number * 2);
        } else {
            // Add to total of even entries only if it's odd (since we already handled odd numbers)
            total_even += number;
        }
    }

    // After walking through all numbers, print the total of even entries
    println!("Total of even entries: {}", total_even);
}
