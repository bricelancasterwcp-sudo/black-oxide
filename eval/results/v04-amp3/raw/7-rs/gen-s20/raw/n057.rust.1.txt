```rust
fn main() {
    let fruits = ["fig", "banana", "kiwi"];
    let mut max_length = 0;
    let mut fruit_name = "";

    for &fruit in fruits.iter() {
        if fruit.len() > max_length {
            max_length = fruit.len();
            fruit_name = fruit;
        }
    }

    println!("The fruit with the most letters is: {}, with {} letters", fruit_name, max_length);
}
```
