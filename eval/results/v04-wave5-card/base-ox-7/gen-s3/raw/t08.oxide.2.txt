fn count_positive(v: Vec<Int>) -> Int {
    count_if(v, |x| x > 0)
}

fn max_value(v: Vec<Int>) -> Option<Int> {
    max(v)
}

fn main() {
    let values = push(push(push(push(vec(), 3), 8), -2), 12);
    values.push(7);
    print(count_positive(values.clone()));
    match max_value(values) {
        Some(n) => print(n),
        None => print_str("empty"),
    }
}
