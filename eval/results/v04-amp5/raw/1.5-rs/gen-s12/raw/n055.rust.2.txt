```rust
fn main() {
    let input = ["8", "oak", "15"];

    for &item in input.iter() {
        match item.to_string().parse::<i32>() {
            Ok(num) => match num {
                8 => println!("good"),
                _ if num.to_string().contains("oak") => println!("good"),
                15 => println!("good"),
                _ => println!("bad")
            },
            Err(_) => println!("bad")
        }
    }
}
```
