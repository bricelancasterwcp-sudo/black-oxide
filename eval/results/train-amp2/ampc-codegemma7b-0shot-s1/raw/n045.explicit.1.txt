fn log_reader(log: &Vec<Int>) -> Int {
    let mut start = None
    let mut end = None
    for x in log {
        if start == None {
            start = Some(x)
        } else {
            end = Some(x)
        }
    }
    return start.unwrap()
}

fn main() {
    let log = vec(6, 7, 8, 9)
    print(log_reader(&log))
    drop log
}
