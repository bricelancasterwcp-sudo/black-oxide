```rust
fn main() {
    let text = "keyboard";

    let mut vowel_count = 0;
    let mut consonant_count = 0;

    for ch in text.chars() {
        if ["a", "e", "i", "o", "u"].contains(&ch.to_ascii_lowercase()) {
            vowel_count += 1;
        } else {
            consonant_count += 1;
        }
    }

    println!("Vowels: {}", vowel_count);
    println!("Consonants: {}", consonant_count);
}
```
