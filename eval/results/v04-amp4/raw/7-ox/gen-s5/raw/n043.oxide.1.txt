struct Queue { items: Vec<Int> }

fn enqueue(q: Queue, x: Int) -> Queue {
    Queue { items: push(q.items, x) }
}

fn dequeue(q: Queue) -> (Int, Queue) {
    let first = get(q.items, 0)?;
    (first, Queue { items: vec().concat(get(q.items, 1..q.items.len())?) })
}

fn swap_first_last(q: Queue) -> Queue {
    if q.items.len() < 2 {
        return q;
    }
    let last_index = q.items.len() - 1;
    let (first, rest1) = dequeue(q);
    let (_, rest2) = dequeue(rest1);
    enqueue(enqueue(Queue { items: push(vec(), get(rest2.items, last_index)?) }, first), get(rest2.items, 0)?)
}

fn main() {
    let mut q = Queue { items: vec(1, 2, 3, 4, 5) };
    q = swap_first_last(q);
    for item in q.items {
        print(item);
    }
}
