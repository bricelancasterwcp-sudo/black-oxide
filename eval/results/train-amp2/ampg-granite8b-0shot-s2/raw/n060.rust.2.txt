fn main() {
    let text = "a1b22";
    let mut digit_count = 0;
    let mut digit_total = 0;
    for c in text.chars() {
        if c.is_ascii_digit() {
            digit_count += 1;
            digit_total += c.to_digit(10).unwrap_or(0) as usize;
        }
    }
    println!("digit count: {}", digit_count);
    println!("digit total: {}", digit_total);
}
