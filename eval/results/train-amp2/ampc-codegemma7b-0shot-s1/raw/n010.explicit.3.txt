fn is_prime(num: Int) -> Bool {
    if num < 2 {
        return false
    }
    for i in range(2, num) {
        if num % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let num = 91
    if is_prime(num) {
        print("yes")
    } else {
        print("no")
    }
    drop num
}
