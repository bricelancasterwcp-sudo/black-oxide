fn reverse_digits(n: Int) -> Int {
    let mut reversed = 0
    let mut num = n
    while num > 0 {
        let digit = num % 10
        reversed = reversed * 10 + digit
        num = num / 10
    }
    reversed
}

fn main() {
    let number = 1234
    let reversed_number = reverse_digits(number)
    print(reversed_number)
}
