struct Queue { values: Vec<Int> }

fn swap_first_last(q: Queue) -> Queue {
    let len = len(q.values);
    if len > 0 {
        let first = get(q.values, 0)?;
        let last = get(q.values, len - 1)?;
        set(set(q, 0, last), len - 1, first)
    } else {
        q
    }
}

fn main() {
    let mut q = Queue { values: push(push(push(push(vec(), 1), 2), 3), 4) };
    print(concat(int_to_str(get(swap_first_last(q).values, 0)?), "\n"));
    for i in range(1, len(swap_first_last(q.values))) {
        print(concat(int_to_str(get(swap_first_last(q).values, i)?), "\n"));
    }
}
