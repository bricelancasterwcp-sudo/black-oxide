```rust
fn main() {
    let fruits = ["fig", "banana", "kiwi"];
    let mut max_fruit = "";
    let mut max_length = 0;

    for fruit in fruits {
        if fruit.len() > max_length {
            max_fruit = fruit;
            max_length = fruit.len();
        }
    }

    println!("The fruit with the most letters is: {}", max_fruit);
    println!("It has {} letters.", max_length);
}
```
