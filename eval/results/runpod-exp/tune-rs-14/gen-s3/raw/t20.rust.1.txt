fn main() {
    let items = vec![
        ("pen", 3, 4),
        ("pad", 9, 2),
        ("ink", 5, 3),
    ];
    let total_cost: u32 = items.iter().map(|(_, price, quantity)| price * quantity).sum();
    println!("{}", total_cost);
    let mut max_cost_item = "";
    let mut max_cost = 0;
    for (name, price, quantity) in items {
        let cost = price * quantity;
        if cost > max_cost {
            max_cost = cost;
            max_cost_item = name;
        }
    }
    println!("{}", max_cost_item);
}
