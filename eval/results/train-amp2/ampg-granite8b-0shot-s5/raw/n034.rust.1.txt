```rust
const fn sec(xs: [isize; 3]) -> Option<isize> {
    if xs.len() < 2 {
        None
    } else {
        Some(xs[1])
    }
}

fn main() {
    let a = sec([7]);
    let b = sec([8]);
    let c = sec([9, 10, 11]);

    println!("{:?}", a);
    println!("{:?}", b);
    println!("{:?}", c);
}
```
