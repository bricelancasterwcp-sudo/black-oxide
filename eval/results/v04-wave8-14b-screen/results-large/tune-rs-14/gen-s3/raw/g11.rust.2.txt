fn main() {
    let grid = [4, 9, 2, 3, 5, 7, 8, 1, 6];
    for i in 0..3 {
        println!("{}", grid[i * 3..(i + 1) * 3].iter().sum::<i32>());
    }
    for i in 0..3 {
        println!("{}", (0..3).map(|j| grid[j * 3 + i]).sum::<i32>());
    }
    println!("{}", (0..3).map(|i| grid[i * 4]).sum::<i32>());
    println!("{}", (0..3).map(|i| grid[2 - i]).sum::<i32>());
    let sums = [
        (0..3).map(|i| grid[i * 3..(i + 1) * 3].iter().sum::<i32>()).collect::<Vec<_>>(),
        (0..3).map(|i| (0..3).map(|j| grid[j * 3 + i]).sum::<i32>()).collect::<Vec<_>>(),
        vec![(0..3).map(|i| grid[i * 4]).sum::<i32>()],
        vec![(0..3).map(|i| grid[2 - i]).sum::<i32>()],
    ].concat();
    println!("{}", sums.iter().all(|&x| x == sums[0]));
}
