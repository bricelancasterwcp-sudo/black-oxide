```rust
fn main() {
    let mut list = vec![7, 4, 9, 10];
    let mut total = 0;

    for mut item in &mut list {
        if item % 2 != 0 {
            print!("{} {}\n", *item * 2, "\n");
        } else {
            total += *item;
        }
    }

    println!("Total of even entries: {}", total);
}
```
