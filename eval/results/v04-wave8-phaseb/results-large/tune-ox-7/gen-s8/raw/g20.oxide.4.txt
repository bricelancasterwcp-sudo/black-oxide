fn main() {
    let mut is_prime = vec(true, 101)
    for i in range(2, len(is_prime)) {
        if get(is_prime, i) {
            for j in range(i * i, 101, i) {
                set(is_prime, j, false)
            }
        }
    }
    let primes = filter(range(2, 101), |x| get(is_prime, x))
    print(len(primes))
    print(max(primes).unwrap_or(-1))
    print(sum(primes))
    let twin_count = count_if(zip(primes, drop(primes, 1)), |pair| *pair.1 - *pair.0 == 2)
    print(twin_count)
    print(get(primes, 9).unwrap_or(-1))
}
