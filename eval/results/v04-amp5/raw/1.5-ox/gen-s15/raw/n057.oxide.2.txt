```rust
fn main() {
    let fruit = vec!["fig", "banana", "kiwi"];
    let mut max_length = 0;
    let mut max_fruit = "";
    for &fruit_name in fruit.iter() {
        if fruit_name.len() > max_length {
            max_length = fruit_name.len();
            max_fruit = fruit_name.clone(); // Clone the string to avoid borrowing issues
        }
    }
    println!("The fruit with the most letters is {} and has {} letters.", max_fruit, max_length);
}
```
