fn swap_first_last(v: Vec<Int>) -> Vec<Int> {
    let len = len(&v) - 1
    let first = get(&v, 0).unwrap()
    let last = get(&v, len).unwrap()
    let mut result = vec()
    push(result, last)
    for i in range(1, len) {
        push(result, get(&v, i))
    }
    push(result, first)
    result
}

fn main() {
    let queue = vec(1, 2, 3, 4, 5)
    let swapped = swap_first_last(queue)
    for x in &swapped {
        print_str(int_to_str(x))
    }
}
