fn count_exceeds_five(readings: Vec<Int>) -> Int {
    let mut count = 0
    let mut max_reading = 0
    for reading in readings {
        if reading > max_reading {
            max_reading = reading
        }
        if reading > count {
            count = reading
        }
    }
    count
}

fn main() {
    let readings = vec(5, 11, 3, 14, 9)
    print(count_exceeds_five(readings))
}
