struct Queue { items: Vec<Int> }

fn exchange_first_last(q: Queue) -> Queue {
    let len = len(q.items)
    if len < 2 {
        return q
    }
    let first = get(q.items, 0)?
    let last = get(q.items, len - 1)?
    let mid = range(1, len - 1)
    let new_items = push(push(push(mid, last), first), get(q.items, len - 2)?)
    Queue { items: new_items }
}

fn print_queue(q: Queue) {
    for x in q.items {
        print(x)
    }
}

fn main() {
    let q = Queue { items: vec(1, 2, 3, 4, 5) }
    let exchanged_q = exchange_first_last(q)
    print_queue(exchanged_q)
}
