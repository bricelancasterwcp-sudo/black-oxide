fn arrange(s: Vec<Int>) -> Vec<Int> {
    let mut a = 0
    let mut b = 0
    let mut c = 0
    for x in &s {
        if x > a {
            c = b
            b = a
            a = x
        } else if x > b {
            c = b
            b = x
        } else if x > c {
            c = x
        }
    }
    push(push(push(vec(), a), b), c)
}

fn main() {
    let s = vec(2, 9, 5)
    let r = arrange(clone(&s))
    print(&r)
    drop r
    drop s
}
