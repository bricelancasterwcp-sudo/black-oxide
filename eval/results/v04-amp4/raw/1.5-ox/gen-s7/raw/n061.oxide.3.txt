```rust
struct Rectangle {
    width: Int,
    height: Int,
}

impl Rectangle {
    fn new(width: Int, height: Int) -> Rectangle {
        Rectangle { width, height }
    }

    fn area(&self) -> Int {
        self.width * self.height
    }

    fn perimeter(&self) -> Int {
        2 * (self.width + self.height)
    }
}

fn main() {
    let rect = Rectangle::new(9, 3);
    print(int_to_str(rect.area()));
    println(int_to_str(rect.perimeter()));
}
```
