struct Reading { label: Str, values: Vec<Int> }

fn reverse_string(s: Str) -> Str {
    let chars = chars(s)
    let mut reversed = vec()
    for c in chars {
        reversed = push(reversed, c)
    }
    concat(vec(), reversed)
}

fn main() {
    let s = int_to_str(1234)
    let reversed_chars = reverse_string(s.clone())
    let reversed_str = concat(vec(), reversed_chars)
    print_str(reversed_str)
}
