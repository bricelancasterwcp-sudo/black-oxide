fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn make_point(h: Int) -> Point {
    Point(h, 10)
}

fn print_point(p: &Point) {
    print_str("Point { ")
    print(p.x)
    print_str(", ")
    print(p.y)
    print_str(" }")
}

fn add_point(p1: &Point, p2: &Point) -> Point {
    Point(p1.x + p2.x, p1.y + p2.y)
}

fn main() {
    let p = make_point(5)
    print_point(&p)
    drop p
    let q = make_point(10)
    print_point(&q)
    drop q
    let p2 = add_point(&p, &q)
    print_str(" added to ")
    print_point(&q)
    print_str(" is ")
    print_point(&p2)
}
