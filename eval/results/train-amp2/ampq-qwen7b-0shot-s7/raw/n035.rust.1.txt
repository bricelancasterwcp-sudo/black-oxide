fn divide(numerator: i32, denominator: i32) -> Result<i32, &'static str> {
    if denominator == 0 {
        Err("bad")
    } else {
        Ok(numerator / denominator)
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("{}", result),
        Err(e) => println!("{}", e),
    }
}
