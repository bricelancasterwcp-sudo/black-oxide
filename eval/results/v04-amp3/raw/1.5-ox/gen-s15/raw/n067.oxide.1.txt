fn walk(v: Vec<Int>) -> Int {
    let mut total = 0;
    for i in v.iter().enumerate() {
        if i.0 % 2 != 0 && i.1 % 2 != 0 {
            println!("{}", i.1 * 2);
            total += i.1;
        }
    }
    return total;
}

fn main() {
    let v = vec![7, 4, 9, 10];
    let result = walk(v);
    println!("Total of even entries: {}", result);
}
