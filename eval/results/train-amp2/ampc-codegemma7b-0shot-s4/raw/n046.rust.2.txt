use std::collections::HashSet;

fn main() {
    let samples = vec![5, 12, 3, 18, 9];
    let cutoff = 10;

    let under_cutoff: HashSet<_> = samples.iter().copied().filter(|x| x < &cutoff).count();
    let at_or_over_cutoff: HashSet<_> = samples.iter().copied().filter(|x| x >= &cutoff).count();

    println!("Under cutoff: {}", under_cutoff);
    println!("At or over cutoff: {}", at_or_over_cutoff);
}
