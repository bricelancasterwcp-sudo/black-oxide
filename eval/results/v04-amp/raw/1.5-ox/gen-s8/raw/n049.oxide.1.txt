```rust
struct Figure {
    name: Str,
    size: Int,
}

fn main() {
    let figures = vec![
        Figure { name: "circle", size: 5 },
        Figure { name: "square", size: 10 },
        Figure { name: "triangle", size: 7 },
        Figure { name: "rectangle", size: 8 },
    ];

    let smallest = figures.iter().min_by(|a, b| a.size.cmp(&b.size)).unwrap();
    print(smallest.name);

    let largest = figures.iter().max_by(|a, b| a.size.cmp(&b.size)).unwrap();
    let gap = largest.size - smallest.size;
    print_str(format!("gap: {}", gap));
}
```
