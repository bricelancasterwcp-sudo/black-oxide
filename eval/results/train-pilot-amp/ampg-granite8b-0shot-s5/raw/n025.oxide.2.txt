struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(first_big(r.values))            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}

struct Point { x: Int, y: Int }

fn line_len(p1: Point, p2: Point) -> Float {
    let x = p2.x - p1.x
    let y = p2.y - p1.y
    to_float(x * x + y * y)
}

fn line_angle(p1: Point, p2: Point) -> Float {
    let x = to_float(p2.x - p1.x)
    let y = to_float(p2.y - p1.y)
    let d = trunc(to_float(x * x + y * y))
    trunc(to_float(y * d / (x * d)))
}

fn main() {
    let p1 = Point { x: 10, y: 20 }
    let p2 = Point { x: 30, y: 10 }
    print(line_len(p1, p2))
    print(line_angle(p1, p2))
}
