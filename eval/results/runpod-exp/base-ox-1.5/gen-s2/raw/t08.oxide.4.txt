```rust
fn main() {
    let v = vec![3, 8, -2, 12, 7];
    let mut count = 0;
    let mut max_value = *v.first().unwrap();

    for &x in v.iter() {
        if x > 0 {
            count += 1;
        }
        if x > max_value {
            max_value = x;
        }
    }

    println!("Positive values: {}", count);
    println!("Largest value: {}", max_value);
}
```
