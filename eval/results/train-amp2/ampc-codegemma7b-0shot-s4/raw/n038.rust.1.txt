use std::io::Write;

fn main() {
    let mut device = Device::new();
    device.sensor.reading = 7;
    println!("{}", device.sensor.reading);
}

struct Device {
    sensor: Sensor,
}

struct Sensor {
    reading: u8,
}

impl Device {
    fn new() -> Self {
        Self {
            sensor: Sensor { reading: 0 },
        }
    }
}
