fn process_numbers(texts: Vec<Str>) -> (Int, Int) {
    let mut sum = 0
    let mut count_non_numbers = 0

    for text in texts {
        match parse_int(text) {
            Some(num) => sum += num,
            None => count_non_numbers += 1,
        }
    }

    (sum, count_non_numbers)
}

fn main() {
    let texts = vec("7", "x", "9", "?")
    let (total, non_numbers_count) = process_numbers(texts)
    print(total)
    print(non_numbers_count)
}
