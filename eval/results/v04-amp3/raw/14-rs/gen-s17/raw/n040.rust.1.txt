```rust
enum Shape {
    Square { side: f64 },
    Circle { radius: f64 },
}

impl Shape {
    fn area(&self) -> f64 {
        match self {
            Shape::Square { side } => side * side,
            Shape::Circle { radius } => radius * radius * 3.0, // Using 3 as the value of pi
        }
    }
}

fn main() {
    let square = Shape::Square { side: 4.0 };
    println!("The area of the square is: {}", square.area());
}
```
