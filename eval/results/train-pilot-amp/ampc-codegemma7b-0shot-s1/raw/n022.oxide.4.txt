fn black(s: Str) -> Str {
    s.chars().map(c_black).collect()
}

fn c_black(c: Char) -> Char {
    if c == black {
        black
    } else {
        c
    }
}

struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(r.label)            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
    print(black(r.label))
}
