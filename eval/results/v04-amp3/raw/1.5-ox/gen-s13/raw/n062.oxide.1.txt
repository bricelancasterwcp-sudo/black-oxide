```rust
fn main() {
    let mut state = "green";
    loop {
        match state {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => panic!("unexpected state: {:?}", state),
        }
        // Simulate receiving signals
        receive_signal(&mut state);
    }
}

fn receive_signal(state: &mut Str) {
    if *state == "green" {
        *state = "yellow";
    } else if *state == "red" {
        *state = "stop";
    }
}
```
