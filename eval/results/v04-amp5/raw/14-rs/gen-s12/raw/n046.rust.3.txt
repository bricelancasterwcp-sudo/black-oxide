fn main() {
    let samples = vec![5, 12, 3, 18, 9];
    let cutoff = 10;
    let mut under_count = 0;
    let mut over_or_equal_count = 0;

    for &sample in &samples {
        if sample < cutoff {
            under_count += 1;
        } else {
            over_or_equal_count += 1;
        }
    }

    println!("Under {}: {}", cutoff, under_count);
    println!("At or over {}: {}", cutoff, over_or_equal_count);
}
