```rust
struct Figure { type_: String, size_: Int }

fn main() {
    let figures = vec![
        Figure { type_: "circle".to_string(), size_: 5 },
        Figure { type_: "square".to_string(), size_: 10 },
        Figure { type_: "triangle".to_string(), size_: 2 },
        Figure { type_: "rectangle".to_string(), size_: 7 },
    ];

    let mut smallest = figures[0];
    for f in &figures {
        if f.size_ < smallest.size_ {
            smallest = f;
        }
    }

    match smallest {
        Some(s) => print_str(&s.type_),
        None => print_str("No figures found."),
    }

    let largest = *figures.iter().max_by(|a, b| a.size_.cmp(&b.size_)).unwrap();
    println!("Gap: {}", largest.size_ - smallest.size_);
}
```
