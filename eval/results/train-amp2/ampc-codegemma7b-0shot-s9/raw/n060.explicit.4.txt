fn sum_digits(s: &Str) -> Int {
    let mut total = 0
    let mut count = 0
    for c in s.chars() {
        if c.is_digit(10) {
            total = total + c.to_digit(10).unwrap()
            count = count + 1
        }
    }
    print(total)
    print(count)
    Unit
}

fn main() {
    let str = "a1b22"
    sum_digits(&str)
    drop str
}
