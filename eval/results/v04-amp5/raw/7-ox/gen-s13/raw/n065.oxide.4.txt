struct Readings { values: Vec<Int> }

fn count_above(v: Vec<Int>) -> Int {
    let mut count = 0;
    for x in v {
        if x > 5 {
            count += 1;
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    max(v)
}

fn main() {
    let readings = Readings { values: push(push(push(push(vec(), 6), 11), 3), 14, 9) };
    print(count_above(readings.values.clone()));
    print_str("\n");
    match max_reading(readings.values.clone()) {
        Some(n) => print(n),
        None => print_str("no readings"),
    }
}
