```rust
fn second_entry<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}

fn main() {
    let list7 = [0, 7];
    let list8 = [0, 8, 9];
    let list9: [i32; 0] = [];

    println!("{:?}", second_entry(&list7)); // Some(&7)
    println!("{:?}", second_entry(&list8)); // Some(&8)
    println!("{:?}", second_entry(&list9)); // None
}
```
