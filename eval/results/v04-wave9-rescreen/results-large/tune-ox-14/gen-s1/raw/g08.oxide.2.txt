fn main() {
    let north = vec(12, 15, 19, 14, 14, 20, 31, 30)
    let south = vec(40, 38, 41, 45, 50, 44)
    let sensors = vec("north", "south")
    let readings = vec(north, south)
    for (i, sensor) in enumerate(sensors) {
        let reading = readings[i]
        let mut inc = 0
        let mut dec = 0
        let mut max_inc = 0
        let mut cur_streak = 0
        let mut max_streak = 0
        for j in range(1, len(reading)) {
            if reading[j] > reading[j - 1] {
                inc += 1
                cur_streak += 1
                max_inc = max(max_inc, reading[j] - reading[j - 1])
            } else if reading[j] < reading[j - 1] {
                dec += 1
                cur_streak = 0
            } else {
                cur_streak = 0
            }
            max_streak = max(max_streak, cur_streak)
        }
        print_str(sensor)
        print(inc)
        print(dec)
        print(max_inc)
        print(max_streak)
    }
    let north_streak = 3
    let south_streak = 5
    if north_streak > south_streak {
        print_str("north")
    } else {
        print_str("south")
    }
}
