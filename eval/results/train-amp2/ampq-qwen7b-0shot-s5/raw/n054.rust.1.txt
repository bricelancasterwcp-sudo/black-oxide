```rust
fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm";
    let vowels = "aeiou";
    let mut count_vowels = 0;
    let mut count_non_vowels = 0;

    for letter in keyboard.chars() {
        if vowels.contains(letter) {
            count_vowels += 1;
        } else {
            count_non_vowels += 1;
        }
    }

    println!("Vowels: {}, Non-vowels: {}", count_vowels, count_non_vowels);
}
```
