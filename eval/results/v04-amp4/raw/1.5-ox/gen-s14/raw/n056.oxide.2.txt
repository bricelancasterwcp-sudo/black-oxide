fn main() {
    let text = "The texts 7, x, 9 and ? each may or may not hold a number. Print the total of the ones that do, then how many do not.";

    let mut count_numbers = 0;
    let mut sum_numbers = 0;

    for c in text.chars() {
        if let Some(n) = c.to_digit(10) {
            count_numbers += 1;
            sum_numbers += n as i32;
        } else if c == '?' {
            // This is not a number, but it's allowed.
            // We can't add it to the sum or count.
            println!("Not a number");
        }
    }

    println!("Total numbers: {}", count_numbers);
    println!("Sum of numbers: {}", sum_numbers);
}
