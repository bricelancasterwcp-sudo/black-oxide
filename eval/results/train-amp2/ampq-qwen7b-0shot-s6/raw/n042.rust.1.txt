```rust
fn main() {
    let values = [12, 5, 20, 8, 17];
    let mut largest = i32::MIN;
    let mut second_largest = i32::MIN;

    for &value in values.iter() {
        if value > largest {
            second_largest = largest;
            largest = value;
        } else if value > second_largest {
            second_largest = value;
        }
    }

    println!("{}", largest);
    println!("{}", second_largest);
}
```
