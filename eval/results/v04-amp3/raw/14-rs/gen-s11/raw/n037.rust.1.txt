```rust
fn main() {
    let mut rect = Rectangle { width: 2, height: 3 };
    rect.width = 5;
    println!("The area of the rectangle is {}", rect.area());
}

#[derive(Debug)]
struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    fn area(&self) -> u32 {
        self.width * self.height
    }
}
```
