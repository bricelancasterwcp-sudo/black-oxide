use std::collections::LinkedList;

fn main() {
    let mut list = LinkedList::new();
    list.push_back(10);
    list.push_back(20);
    list.push_back(30);
    list.push_back(40);

    match list.get(2) {
        Some(entry) => println!("The entry at position two is: {}", entry),
        None => println!("Missing")
    }
}
