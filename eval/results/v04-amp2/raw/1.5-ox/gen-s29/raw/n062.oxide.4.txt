```rust
fn main() {
    let mut signal = "green";
    while true {
        match signal {
            "green" => println!("go for green"),
            "red" => println!("stop for red"),
            "yellow" => println!("wait for yellow"),
            _ => panic!("unknown signal: {}", signal),
        }
        // Simulate arrival of a new signal
        let new_signal = match rand::random::<u8>() % 3 {
            0 => "green",
            1 => "red",
            2 => "yellow",
            _ => unreachable!(),
        };
        signal = new_signal;
    }
}
```
