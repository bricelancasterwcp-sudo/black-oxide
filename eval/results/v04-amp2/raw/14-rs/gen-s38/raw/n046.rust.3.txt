```rust
fn main() {
    let samples = [5, 12, 3, 18, 9];
    let cutoff = 10;
    let mut below_cutoff = 0;
    let mut at_or_above_cutoff = 0;

    for &sample in &samples {
        if sample < cutoff {
            below_cutoff += 1;
        } else {
            at_or_above_cutoff += 1;
        }
    }

    println!("Below cutoff: {}", below_cutoff);
    println!("At or above cutoff: {}", at_or_above_cutoff);
}
```
