fn sum_of_numbers(v: Vec<Str>) -> Int {
    let mut sum = 0;
    let mut count_non_numbers = 0;
    for s in v {
        match parse_int(s) {
            Some(n) => sum += n,
            None => count_non_numbers += 1,
        }
    }
    print(sum);
    print(count_non_numbers);
    sum
}

fn main() {
    let values = vec("7", "x", "9", "?");
    sum_of_numbers(values);
}
