fn is_prime(x: i32) -> bool {
    if x <= 1 {
        return false;
    }
    for i in 2..=(x as f64).sqrt() as i32 {
        if x % i == 0 {
            return false;
        }
    }
    true
}

fn main() {
    let num = 91;
    if is_prime(num) {
        println!("yes");
    } else {
        println!("no");
    }
}
