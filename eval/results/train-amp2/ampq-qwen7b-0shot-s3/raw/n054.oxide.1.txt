struct LetterCounts { vowels: Int, consonants: Int }

fn count_letters(s: Str) -> LetterCounts {
    let mut counts = LetterCounts { vowels: 0, consonants: 0 }
    for c in chars(s) {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            counts.vowels = counts.vowels + 1
        } else {
            counts.consonants = counts.consonants + 1
        }
    }
    counts
}

fn main() {
    let s = "keyboard"
    let counts = count_letters(s)
    print(counts.vowels)
    print(counts.consonants)
}
