```rust
struct Queue {
    items: Vec<Int>,
}

impl Queue {
    fn new() -> Self {
        Queue { items: vec![] }
    }

    fn enqueue(&mut self, item: Int) {
        self.items.push(item);
    }

    fn dequeue(&mut self) -> Option<Int> {
        self.items.pop()
    }

    fn exchange_first_last(&mut self) {
        if let Some(first) = self.dequeue() {
            if let Some(last) = self.dequeue() {
                self.enqueue(first);
                self.enqueue(last);
            }
        }
    }
}

fn main() {
    let mut queue = Queue::new();
    for i in 1..=5 {
        queue.enqueue(i);
    }

    queue.exchange_first_last();

    while let Some(item) = queue.dequeue() {
        print!("{}", item);
    }
}
```
