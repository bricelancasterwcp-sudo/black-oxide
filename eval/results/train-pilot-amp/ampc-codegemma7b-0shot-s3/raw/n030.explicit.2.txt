fn count_vowels(s: &Str) -> Int {
    let mut count = 0
    for c in chars(s) {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            count = count + 1
        }
    }
    count
}

fn main() {
    let word = str_len("hello")
    let vowels = count_vowels(&word)
    print(vowels)
    drop word
    drop vowels
}
