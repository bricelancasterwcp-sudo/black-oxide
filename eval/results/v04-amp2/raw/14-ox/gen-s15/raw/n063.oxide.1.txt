struct Line { quantity: Int, price: Int }

fn calculate_value(line: Line) -> Int {
    line.quantity * line.price
}

fn main() {
    let line1 = Line { quantity: 4, price: 3 }
    let line2 = Line { quantity: 5, price: 2 }
    
    print(calculate_value(line1))
    print(calculate_value(line2))
    
    let total_value = calculate_value(line1) + calculate_value(line2)
    print(total_value)
}
