fn set_sensor(device: &Device) {
    let sensor = &device // a linear sensor is borrowed
    sensor.reading = 7
}

fn main() {
    let device = Device { sensor: Sensor { reading: 0 } }
    set_sensor(&device)
    print(device.sensor.reading)
}
