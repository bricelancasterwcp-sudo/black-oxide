struct Coffee {
    letters: Vec<Char>,
}

fn unique_letters(coffee: Coffee) -> Vec<Char> {
    let mut seen = vec()
    for letter in coffee.letters {
        if !contains(seen, letter) {
            seen = push(seen, letter)
        }
    }
    seen
}

fn main() {
    let letters = unique_letters(Coffee { letters: vec_from_str("coffee") })
    for letter in letters {
        print_str(str_from_char(letter))
    }
}

fn contains(v: Vec<Char>, x: Char) -> Bool {
    for y in v {
        if x == y {
            return true
        }
    }
    false
}

fn vec_from_str(s: Str) -> Vec<Char> {
    let mut v = vec()
    for c in s {
        v = push(v, c)
    }
    v
}

fn str_from_char(c: Char) -> Str {
    let mut s = str()
    s = push(s, c)
    s
}

fn str() -> Str {
    ""
}

fn push(s: Str, c: Char) -> Str {
    s + c
}

fn print_str(s: Str) {
    print(s)
}

fn main() {
    let letters = unique_letters(Coffee { letters: vec_from_str("coffee") })
    for letter in letters {
        print_str(str_from_char(letter))
    }
}
