fn helper(v: Vec<Int>) -> Option<Int> {
    let mut second = None;
    for x in v {
        if let Some(second_value) = &second {
            if x > *second_value {
                second = Some(x);
            }
        } else {
            second = Some(x);
        }
    }
    second
}

fn main() {
    println!("{:?}", helper(vec![7, 8, 9])); // prints: Some(8)
}
