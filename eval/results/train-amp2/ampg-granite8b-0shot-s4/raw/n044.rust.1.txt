fn main() {
    let mut balance = 0;
    deposit(3, &mut balance);
    deposit(1, &mut balance);
    deposit(4, &mut balance);
    deposit(1, &mut balance);
    deposit(5, &mut balance);
    println!("{}", balance);
}

fn deposit(amount: u32, balance: &mut u32) {
    *balance += amount;
    println!("{}", balance);
}
