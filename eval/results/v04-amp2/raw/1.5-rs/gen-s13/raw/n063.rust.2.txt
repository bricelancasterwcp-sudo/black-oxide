fn main() {
    let price1 = 3;
    let quantity1 = 4;
    let price2 = 2;
    let quantity2 = 5;

    println!("Line 1 value: {}", price1 * quantity1);
    println!("Line 2 value: {}", price2 * quantity2);

    println!("Total value of both lines: {}", (price1 as f64) * quantity1 + (price2 as f64) * quantity2);
}
