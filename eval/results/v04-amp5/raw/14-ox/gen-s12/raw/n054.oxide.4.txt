```rust
struct LetterCount { vowels: Int, consonants: Int }

fn is_vowel(c: Str) -> Bool {
    match c {
        "a" | "e" | "i" | "o" | "u" => true,
        _ => false,
    }
}

fn count_letters(s: Str) -> LetterCount {
    let mut vowels = 0
    let mut consonants = 0
    for ch in chars(s) {
        if is_vowel(ch) {
            vowels += 1
        } else if ch >= "a" && ch <= "z" || ch >= "A" && ch <= "Z" {
            consonants += 1
        }
    }
    LetterCount { vowels, consonants }
}

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    let count = count_letters(keyboard)
    print(count.vowels)
    print(count.consonants)
}
```
