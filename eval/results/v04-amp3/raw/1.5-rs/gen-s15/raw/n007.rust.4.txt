fn main() {
    let mut number = 9875;
    let mut sum = 0;

    while number > 0 {
        let digit = number % 10; // Get the last digit of number
        sum += digit; // Add it to the sum
        number /= 10; // Remove the last digit from number by dividing by 10
    }

    println!("The sum of the digits is: {}", sum);
}
