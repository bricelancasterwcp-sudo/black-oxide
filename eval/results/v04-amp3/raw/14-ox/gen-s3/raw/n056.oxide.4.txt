struct Reading { label: Str, values: Vec<Str> }

fn count_numbers(v: Vec<Str>) -> (Int, Int) {
    let mut sum = 0
    let mut count_non_numbers = 0
    for x in v {
        match parse_int(x) {
            Some(n) => sum += n,
            None => count_non_numbers += 1,
        }
    }
    (sum, count_non_numbers)
}

fn main() {
    let values = vec("7", "x", "9", "?")
    let (total, non_numbers_count) = count_numbers(values)
    print(total)
    print(non_numbers_count)
}
