```rust
fn main() {
    let mut signals = vec![0; 3];
    for i in 0..signals.len() {
        match signals[i] {
            0 => println!("go for green"),
            1 => println!("stop for red"),
            2 => println!("wait for yellow"),
            _ => panic!("invalid signal code: {}", signals[i]),
        }
    }
}
```
