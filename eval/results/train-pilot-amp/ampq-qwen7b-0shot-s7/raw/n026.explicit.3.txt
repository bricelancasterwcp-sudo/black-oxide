fn int_to_str(n: Int) -> Str {
    let mut result = ""
    let mut num = n
    if num == 0 {
        return "0"
    }
    while num > 0 {
        let digit = num % 10
        num = num / 10
        let digit_str = int_to_str(digit)
        result = concat(result, digit_str)
    }
    result
}

fn main() {
    let num = 2026
    let num_str = int_to_str(num)
    let len = str_len(num_str)
    print(len)
}
