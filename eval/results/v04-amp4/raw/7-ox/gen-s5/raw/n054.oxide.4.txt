struct Keyboard { letters: Str }

fn count_vowels(s: Str) -> Int {
    let mut count = 0
    for c in chars(s) {
        if c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' ||
           c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U' {
            count += 1
        }
    }
    count
}

fn main() {
    let k = Keyboard { letters: "qwertyuiopasdfghjklzxcvbnm" };
    let vowels = count_vowels(k.letters.clone());
    let not_vowels = str_len(k.letters) - vowels;
    print(vowels);
    print_str(" ");
    print(not_vowels);
}
