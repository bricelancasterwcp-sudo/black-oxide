enum Shape {
    Square(Int),
    Circle(Float),
}

fn area(shape: Shape) -> Float {
    match shape {
        Shape::Square(side) => to_float(side * side),
        Shape::Circle(radius) => radius * radius * 3.14159,
    }
}

fn main() {
    let square = Shape::Square(4);
    print(area(square));
}
