fn main() {
    let v = vec(3, 8, -2, 12, 7)
    let mut count_pos = 0
    let mut max_value = unwrap_or(max(v), -1)
    for x in v {
        if x > 0 {
            count_pos += 1
        }
        if x > max_value {
            max_value = x
        }
    }
    print(count_pos)
    print(max_value)
}
