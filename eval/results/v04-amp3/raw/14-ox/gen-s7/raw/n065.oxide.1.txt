struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding_five(v: Vec<Int>) -> Int {
    let count = 0
    for x in v {
        if x > 5 {
            count += 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max.is_none() || x > unwrap_or(max, 0) {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14)
    let readings = push(readings, 9)
    let r = Reading { label: "lab", values: readings }
    print(count_exceeding_five(r.values))            // prints the count of readings exceeding five
    match max_reading(r.values) {
        Some(n) => print(n),                        // prints the highest reading
        None => print_str("no readings"),
    }
}
