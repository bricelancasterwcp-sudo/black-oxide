struct Part { name: Str, count: Int, price: Int }
fn calculate_value(parts: Vec<Part>) -> Int {
    let mut total = 0
    for p in parts {
        total += p.count * p.price
    }
    total
}
fn find_low_quantity_parts(parts: Vec<Part>) -> Int {
    let mut count = 0
    for p in parts {
        if p.count < 10 {
            count += 1
        }
    }
    count
}
fn find_highest_value_part(parts: Vec<Part>) -> Str {
    let mut max_value = -1
    let mut max_name = ""
    for p in parts {
        let value = p.count * p.price
        if value > max_value {
            max_value = value
            max_name = p.name
        }
    }
    max_name
}
fn main() {
    let north_parts = vec(
        Part { name: "bolt", count: 40, price: 3 },
        Part { name: "hinge", count: 8, price: 25 },
        Part { name: "plate", count: 15, price: 12 },
    )
    let south_parts = vec(
        Part { name: "clamp", count: 6, price: 45 },
        Part { name: "rod", count: 22, price: 9 },
        Part { name: "washer", count: 100, price: 1 },
        Part { name: "gasket", count: 4, price: 30 },
    )
    let north_value = calculate_value(north_parts)
    let south_value = calculate_value(south_parts)
    let north_low_quantity = find_low_quantity_parts(north_parts)
    let south_low_quantity = find_low_quantity_parts(south_parts)
    let north_highest_value_part = find_highest_value_part(north_parts)
    let south_highest_value_part = find_highest_value_part(south_parts)
    print_str("North")
    print_int(north_value)
    print_int(north_low_quantity)
    print_str(north_highest_value_part)
    print_str("South")
    print_int(south_value)
    print_int(south_low_quantity)
    print_str(south_highest_value_part)
    if north_value > south_value {
        print_str("North")
    } else {
        print_str("South")
    }
}
