fn int_to_str(mut_x: Int) -> Str {
    let mut_chars = vec()
    while mut_x != 0 {
        let mut_digit = mut_x % 10
        mut_x = trunc(mut_x / 10)
        push(mut_chars, int_to_str(mut_digit))
    }
    let mut_result = str()
    for mut_ch in &mut_chars {
        mut_result = concat(mut_result, mut_ch)
    }
    if mut_result == str() {
        mut_result = str("0")
    }
    mut_result
}

fn concat(a: Str, b: Str) -> Str {
    let mut_result = a
    for mut_ch in chars(b) {
        mut_result = concat(mut_result, mut_ch)
    }
    mut_result
}

fn chars(s: Str) -> Vec<Str> {
    let mut_result = vec()
    let mut_cursor = 0
    let mut_top = 0
    let mut_bottom = str_len(s) - 1
    while mut_cursor <= mut_top && mut_cursor <= mut_bottom {
        if s.chars().nth(mut_cursor) == s.chars().nth(mut_bottom) {
            push(mut_result, str(&s).substring(mut_cursor, mut_cursor + 1))
            mut_cursor = mut_cursor + 1
            mut_bottom = mut_bottom - 1
        } else if s.chars().nth(mut_cursor) == s.chars().nth(mut_top) {
            push(mut_result, str(&s).substring(mut_cursor, mut_cursor + 1))
            mut_cursor = mut_cursor + 1
            mut_top = mut_top - 1
        } else {
            break
        }
    }
    mut_result
}

fn str(s: &Vec<Str>) -> Str {
    let mut_result = str()
    for mut_ch in s {
        mut_result = concat(mut_result, mut_ch)
    }
    mut_result
}

fn str_len(s: Str) -> Int {
    let mut_result = 0
    for mut_ch in chars(s) {
        mut_result = mut_result + 1
    }
    mut_result
}

fn trunc(x: Int) -> Int {
    if x < 0 {
        0
    } else {
        x
    }
}

fn push(v: Vec<T>, x: T) -> Vec<T> {
    let mut_result = v
    push_back(mut_result, x)
    mut_result
}

fn push_back(mut_v: Vec<T>, x: T) {
    let mut_size = len(mut_v)
    mut_v.set(mut_size, x)
    let mut_new_size = mut_size + 1
    while mut_new_size != len(mut_v) {
        mut_new_size = len(mut_v)
    }
}

fn len(v: Vec<T>) -> Int {
    let mut_result = 0
    for _ in v {
        mut_result = mut_result + 1
    }
    mut_result
}

    if index >= len(mut_v) {
        panic("index out of bounds")
    } else {
        mut_v.push(value)
    }
}
