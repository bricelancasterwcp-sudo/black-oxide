fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm";
    let mut vowels = 0;
    let mut consonants = 0;

    for c in keyboard.bytes() {
        match c.to_ascii_lowercase() as char {
            'a' | 'e' | 'i' | 'o' | 'u' => vowels += 1,
            _ if c.is_alphabetic() => consonants += 1,
            _ => (),
        }
    }

    print(vowels);
    print_str(" ");
    print(consonants);
}
