fn main() {
    let x = match read_int() {
        Some(n) => n,
        None => 0,
    };
    println!("{}", x);

    let y = match read_string() {
        Some(s) => s,
        None => "bad",
    };
    println!("{}", y);

    let z = match read_float() {
        Some(f) => f,
        None => 0.0,
    };
    println!("{}", z);
}

    // Read and parse integer
    // Assume valid input
    Some(8)
}

fn read_string() -> Option<Str> {
    // Read and parse string
    // Assume valid input
    Some("oak".to_string())
}

    // Read and parse float
