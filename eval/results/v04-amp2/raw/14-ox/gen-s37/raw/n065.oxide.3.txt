struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max == None || unwrap_or(max, -1) < x {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = push(push(push(push(push(vec(), 6), 11), 3), 14), 9)
    let count = count_exceeding(readings.clone(), 5)
    print(count)
    if let Some(max) = max_reading(readings) {
        print(max)
    } else {
        print_str("No readings")
    }
}
