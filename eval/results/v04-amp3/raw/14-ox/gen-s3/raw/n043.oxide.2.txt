struct Queue { items: Vec<Int> }

fn exchange_first_last(q: Queue) -> Queue {
    let len = len(q.items)
    if len < 2 {
        return q
    }
    let first = get(q.items, 0)?
    let last = get(q.items, len - 1)?
    let middle = range(1, len - 1)
    let new_items = push(push(push(vec(), last), middle), first)
    Queue { items: new_items }
}

fn main() {
    let q = Queue { items: vec(1, 2, 3, 4, 5) }
    let q = exchange_first_last(q)
    for x in q.items {
        print(x)
    }
}
