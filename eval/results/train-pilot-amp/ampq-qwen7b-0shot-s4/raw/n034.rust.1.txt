```rust
fn get_second<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}

fn main() {
    let list7 = vec![1, 2];
    let list8 = vec![3, 4, 5];
    let list9 = vec![6];

    println!("{:?}", get_second(&list7)); // Some(2)
    println!("{:?}", get_second(&list8)); // Some(4)
    println!("{:?}", get_second(&list9)); // None
}
```
