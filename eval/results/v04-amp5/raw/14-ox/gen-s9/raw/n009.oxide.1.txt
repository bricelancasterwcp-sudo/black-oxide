fn reverse_digits(n: Int) -> Str {
    let mut digits = vec()
    let mut num = n
    while num > 0 {
        digits = push(digits, num % 10)
        num = div(num, 10)
    }
    let result = ""
    for d in digits {
        result = concat(result, int_to_str(d))
    }
    result
}

fn main() {
    print(reverse_digits(1234))
}
