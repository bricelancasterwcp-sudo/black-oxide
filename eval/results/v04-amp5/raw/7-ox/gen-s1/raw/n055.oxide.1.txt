fn parse_int(s: Str) -> Option<Int> {
    str_len(s)? > 0 => Some(to_float(parse_int(s)) as Int),
    _ => None,
}

fn main() {
    let entries = vec!["8", "oak", "15"];
    for entry in entries {
        match parse_int(entry) {
            Some(n) => print(n),
            None => print_str("bad"),
        }
    }
}
