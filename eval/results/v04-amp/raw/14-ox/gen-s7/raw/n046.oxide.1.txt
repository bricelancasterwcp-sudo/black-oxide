```rust
fn count_values(v: Vec<Int>, cutoff: Int) -> (Int, Int) {
    let mut below = 0
    let mut above_or_equal = 0
    for x in v {
        if x < cutoff {
            below = below + 1
        } else {
            above_or_equal = above_or_equal + 1
        }
    }
    (below, above_or_equal)
}

fn main() {
    let values = vec![5, 12, 3, 18, 9]
    let cutoff = 10
    let (below, above_or_equal) = count_values(values, cutoff)
    print(below)
    print(above_or_equal)
}
```
