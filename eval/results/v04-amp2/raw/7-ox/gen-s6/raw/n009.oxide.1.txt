fn reverse_digits(n: Int) -> Str {
    let mut s = int_to_str(n)
    let chars = chars(s)
    let reversed_chars = push(push(vec(), chars[chars.len() - 1]), chars[chars.len() - 2])
    concat(concat(concat(concat(vec(), reversed_chars[0]), reversed_chars[1]), reversed_chars[2]), reversed_chars[3])
}

fn main() {
    print(reverse_digits(1234))
}
