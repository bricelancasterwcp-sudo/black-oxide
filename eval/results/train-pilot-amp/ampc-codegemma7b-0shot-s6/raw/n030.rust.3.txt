use std::io::{stdin, Result};

fn main() -> Result<()> {
    let mut vowels = 0;
    loop {
        let mut line = String::new();
        stdin().read_line(&mut line)?;
        if line.trim() == "" {
            break;
        }
        for c in line.chars() {
            match c.to_lowercase() {
                'a' | 'e' | 'i' | 'o' | 'u' => vowels += 1,
                _ => (),
            }
        }
    }
    println!("Total vowels: {}", vowels);
    Ok(())
}
