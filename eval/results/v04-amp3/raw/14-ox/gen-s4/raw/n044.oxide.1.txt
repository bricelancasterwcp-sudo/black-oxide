struct Account { balance: Int }

fn deposit(acc: Account, amount: Int) -> Account {
    let new_balance = acc.balance + amount
    Account { balance: new_balance }
}

fn main() {
    let mut acc = Account { balance: 0 }
    acc = deposit(acc, 3)
    print(acc.balance)
    acc = deposit(acc, 1)
    print(acc.balance)
    acc = deposit(acc, 4)
    print(acc.balance)
    acc = deposit(acc, 1)
    print(acc.balance)
    acc = deposit(acc, 5)
    print(acc.balance)
}
