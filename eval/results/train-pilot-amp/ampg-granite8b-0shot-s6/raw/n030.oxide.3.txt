struct Words { words: Vec<Str> }

fn count_vowels(w: Str) -> Int {
    let mut count = 0
    for c in w.chars() {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            count = count + 1
        }
    }
    count
}

fn main() {
    let words = Words { words: vec() }
    for w in words.words {
        print(count_vowels(w))
    }
}
