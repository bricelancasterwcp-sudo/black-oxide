```rust
fn main() {
    let square = Shape::Square(4.0);
    println!("Area of square: {}", area(square));
}

fn area(shape: Shape) -> f64 {
    match shape {
        Shape::Square(side) => side * side,
        Shape::Circle(radius) => 3.0 * radius * radius,
    }
}

enum Shape {
    Square(f64),
    Circle(f64),
}
```
