fn count_between(list1: &Vec<Int>, list2: &Vec<Int>) -> Int {
    let mut count = 0
    for num in list1 {
        if num > list2 {
            count = count + 1
        }
    }
    for num in list2 {
        if num > list1 {
            count = count + 1
        }
    }
    count
}

fn main() {
    let list1 = vec(1, 2, 3)
    let list2 = vec(4, 5)
    print(count_between(&list1, &list2))
    drop list1
    drop list2
}
