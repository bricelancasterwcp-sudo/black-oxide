fn count_under_and_over(vec: Vec<Int>, cutoff: Int) -> (Int, Int) {
    let mut under = 0;
    let mut over = 0;
    for x in vec {
        if x < cutoff {
            under += 1;
        } else {
            over += 1;
        }
    }
    (under, over)
}

fn main() {
    let values = push(push(push(push(vec(), 5), 12), 3), 18);
    let values = push(values, 9);
    let cutoff = 10;
    let (under, over) = count_under_and_over(values, cutoff);
    print_str("Under ");
    print(under);
    print_str(": ");
    print(over);
}
