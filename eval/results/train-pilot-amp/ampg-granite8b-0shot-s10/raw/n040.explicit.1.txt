fn main() {
    let square = Shape { kind: Square { side: 4.0 } }
    let circle = Shape { kind: Circle { radius: 4.0 } }
    print_area(&square)
    print_area(&circle)
}

fn print_area(s: &Shape) {
    print(area(s))
}

fn area(s: &Shape) -> Float {
    match s.kind {
        Square(refsq) => pi() * refsq.side * refsq.side,
        Circle(refcr) => pi() * refcr.radius * refcr.radius,
    }
}

fn pi() -> Float {
    3.0
}

enum Shape {
    Square(Square),
    Circle(Circle),
}

struct Square {
    side: Float,
}

struct Circle {
    radius: Float,
}

    let total = 0

        if x > limit {
            total = total + x
