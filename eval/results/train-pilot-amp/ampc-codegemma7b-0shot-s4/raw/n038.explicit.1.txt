fn main() {
    let mut device = Device { sensor: Sensor { reading: 0 } }
    device.sensor.reading = 7
    print(device.sensor.reading)
    drop device
}

struct Device {
    sensor: Sensor,
}

struct Sensor {
    reading: Int,
}

fn print(x: Int) {
    print_str(str_len(int_to_str(x)))
}

fn print_str(s: Int) {
    for c in chars(str_len(s)) {
        print(c)
    }
}

fn str_len(s: Int) -> Int {
    let mut len = 0
    for _ in range(0, s) {
        len = len + 1
    }
    len
}

fn chars(s: Int) -> Vec<Str> {
    let mut chars = vec()
    let mut idx = 0
    while idx < s {
        chars = push(chars, s)
        idx = idx + 1
    }
    chars
}

fn int_to_str(n: Int) -> Str {
    let mut s = 0
    let mut num = n
    while num > 0 {
        num = num / 10
        s = s + 1
    }
    s
}

fn drop_int(i: Int) {
    let _ = i
}

    x

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {

    v = push(v, x)
    v
}

fn len(v: &Vec<Int>) -> Int {
    let mut len = 0
    for _ in range(0, len(v)) {
        len = len + 1
    }
    len
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= len(v) {
        None
    } else {
        Some(v)
    }
}
