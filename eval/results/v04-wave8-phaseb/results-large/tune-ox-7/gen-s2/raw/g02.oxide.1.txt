fn main() {
    let words = vec("rotor", "banana", "level", "quiet", "deified", "system", "civic")
    let mut count_palindromes = 0
    for word in words {
        print_str(word)
        print(len(word))
        let reversed = reverse_str(word)
        print_str(reversed)
        if word == reversed {
            print_str("yes")
            count_palindromes += 1
        } else {
            print_str("no")
        }
        let vowel_count = count_if(chars(word), |c| c == "a" || c == "e" || c == "i" || c == "o" || c == "u")
        print(vowel_count)
        let unique_chars = len(set(chars(word)))
        print(unique_chars)
    }
    print(count_palindromes)
}
