fn main() {
    let mut is_prime = vec(true, 100)
    for i in range(2, 101) {
        if is_prime[i] {
            for j in range(i * i, 101, i) {
                is_prime[j] = false
            }
        }
    }
    let primes = filter(range(2, 101), |x| is_prime[x])
    print(len(primes))
    print(max(primes).unwrap_or(-1))
    print(sum(primes))
    let twin_count = count(zip(primes, primes[1..]), |(a, b)| b - a == 2)
    print(twin_count)
    if len(primes) >= 10 {
        print(primes[9])
    } else {
        print(-1)
    }
}
