struct Rectangle { width: Int, height: Int }

fn area(r: Rectangle) -> Int {
    r.width * r.height
}

fn perimeter(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn main() {
    let rect = Rectangle { width: 9, height: 3 };
    print(area(rect)); // missing semicolon after area call
    println(perimeter(rect)); // missing semicolon after perimeter call
}
