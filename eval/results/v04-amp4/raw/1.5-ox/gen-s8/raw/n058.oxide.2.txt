```rust
fn main() {
    let c = "coffee";
    let mut counts = [0; 26];
    
    for ch in c.chars() {
        if 'a' <= ch && ch <= 'z' {
            counts[ch as usize - 'a'] += 1;
        }
    }
    
    // Only need to print the first occurrence of each letter
    let mut printed = false;
    for (i, count) in counts.iter().enumerate() {
        if *count > 0 && !printed {
            print!("{}", (char)(i + 'a'));
            printed = true;
        }
    }
}
```
