fn concat(a: &Str, b: &Str) -> Str {
    let mut result = a
    let mut i = 0
    while i < str_len(b) {
        result = push(result, get(b, i))
        i = i + 1
    }
    result
}

fn to_str_helper(x: Int, base: Int) -> Str {
    if x < 0 {
        concat(to_str_helper(-x, base), "-")
    } else if x < base {
        concat(to_str_helper(x / base, base), get(to_str_digits(), x % base))
    } else if x == 0 {
        "0"
    } else {
        panic("Base conversion error: number too large")
    }
}
