struct Queue {
    values: Vec<Int>,
}

impl Queue {
    fn new(values: Vec<Int>) -> Self {
        Queue { values }
    }

    fn first(&self) -> Option<&Int> {
        self.values.get(0)
    }

    fn last(&self) -> Option<&Int> {
        self.values.last()
    }

    fn exchange_first_last(&mut self) {
        if let Some(first) = self.first() {
            if let Some(last) = self.last() {
                std::mem::swap(first, last);
            }
        }
    }

    fn print_values(&self) {
        for value in &self.values {
            println!("{}", value);
        }
    }
}

fn main() {
    let mut q = Queue::new(vec![1, 2, 3, 4, 5]);
    q.exchange_first_last();
    q.print_values();
}
