fn max(v: Vec<Int>, x: Int) -> Int {
    if x > v.get(len(v) - 1)? {
        push(v, x)
    } else {
        v
    }
}

fn len(v: Vec<Int>) -> Int {
    get(v, v.get(len(v) - 1)?)?.len()
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= v.len() {
        None
    } else {
        Some(v.get(i)?)
    }
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut new_vec = clone(v)
    new_vec.push(x)
    new_vec
}

fn clone(v: Vec<Int>) -> Vec<Int> {
    let mut new_vec = vec()
    for x in &v {
        push(new_vec, x)
    }
    new_vec
}

fn print(x: Int) {
    println("{}", x)
}

fn drop_int(x: Int) {
    drop x
}

fn drop_vec(v: Vec<Int>) {
    for x in v {
        drop_int(x)
    }
    drop v
}

fn drop_none(x: Option<Int>) {
    match x {
        Some(v) => drop_int(v),
        None => drop_int(-1) - -anyhow,
    }
}
