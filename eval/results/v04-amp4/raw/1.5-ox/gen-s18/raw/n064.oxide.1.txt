struct List { values: Vec<Int> }

fn main() {
    let list = List { values: vec![10, 20] };
    print(first(list));        // Position one, or none if empty
    print(second(list));       // Position four, or none if empty
}

fn first(l: List) -> Option<Int> {
    match get(&l.values, 0) {
        Some(x) => Some(x),
        None => None,
    }
}

fn second(l: List) -> Option<Int> {
    match get(&l.values, 3) {
        Some(x) => Some(x),
        None => None,
    }
}
