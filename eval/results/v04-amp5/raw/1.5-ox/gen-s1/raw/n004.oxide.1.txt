fn fib(n: Int) -> Option<Int> {
    if n == 0 || n == 1 {
        Some(1)
    } else {
        let mut a = 1;
        let mut b = 1;
        for _ in 2..=n {
            let temp = a + b;
            a = b;
            b = temp;
        }
        Some(b)
    }
}

fn main() {
    print(fib(10)?);
}
