fn main() {
    match parse_int(text_input()) {
        Some(n) => print(n),
        None => print_str("bad"),
    }
    let s = parse_str(text_input())
    match s {
        Some(s) => print_str(s),
        None => print_str("bad"),
    }
    match parse_int(text_input()) {
        Some(n) => print(n),
        None => print_str("bad"),
    }
}
