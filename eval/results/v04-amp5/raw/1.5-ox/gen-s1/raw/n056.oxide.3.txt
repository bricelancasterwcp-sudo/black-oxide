fn sum_numbers(numbers: Vec<&str>) -> (Int, Int) {
    let mut sum = 0;
    let mut count = 0;

    for num in numbers {
        match num.parse::<Int>() {
            Ok(n) => {
                sum += n;
                count += 1;
            }
            Err(_) => {}
        }
    }

    (sum, count)
}

fn main() {
    let numbers = vec!["7", "x", "9", "?"];
    let (total, count) = sum_numbers(numbers);
    print!("Total: {}, Count: {}", total, count);
}
