fn reverse_digits(mut n: Int) -> Int {
    let mut reversed = Int::zero();
    while n > Int::zero() {
        let digit = n % 10;
        reversed = push(reversed, digit);
        n /= 10; // Corrected this line from pop(n)
    }
    reversed
}

fn main() {
    print(reverse_digits(Int::from(1234)));
}
