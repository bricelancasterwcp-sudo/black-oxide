struct Line { units: Int, price_per_unit: Int }

fn line_value(line: Line) -> Int {
    line.units * line.price_per_unit
}

fn main() {
    let line1 = Line { units: 4, price_per_unit: 3 }
    let line2 = Line { units: 5, price_per_unit: 2 }
    print(line_value(line1))
    print(line_value(line2))
    let total_value = line_value(line1) + line_value(line2)
    print(total_value)
}
