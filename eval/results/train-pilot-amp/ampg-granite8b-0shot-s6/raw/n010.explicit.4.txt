fn is_prime(n: Int) -> Bool {
    2 <= n && (for_range(2, trunc(sqrt(to_float(n)))) || n % for_range(2, trunc(sqrt(to_float(n)))) != 0)
}

fn main() {
    let n = 91
    if is_prime(n) {
        print_str("yes\n")
    } else {
        print_str("no\n")
    }
}

fn for_range(a: Int, b: Int) -> Vec<Int> {
    let result = vec()
    for i in range(a, b) {
        result = push(result, i)
    }
    result
}

fn sqrt(n: Float) -> Float {
    let guess = 1.0
    let better_guess = guess - ((guess * guess) - n) / (2 * guess)
    if better_guess == guess {
        guess
    } else {
        sqrt(better_guess)
    }
}

fn main() {
    let n = 91
    if is_prime(n) {
        print_str("yes\n")
    } else {
        print_str("no\n")
    }
}
