fn main() {
    let v = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut max_value = i32::MIN;
    for &x in &v {
        if x > 0 {
            positive_count += 1;
        }
        if x > max_value {
            max_value = x;
        }
    }
    println!("{}", positive_count);
    println!("{}", max_value);
}
