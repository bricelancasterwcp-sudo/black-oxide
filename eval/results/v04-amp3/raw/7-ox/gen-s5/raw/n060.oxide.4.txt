fn main() {
    let s = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in chars(s) {
        if c >= '0' && c <= '9' {
            let digit = parse_int(concat("", str(c))).unwrap();
            sum += digit;
            count += 1;
        }
    }

    print(sum);
    print(count);
}
