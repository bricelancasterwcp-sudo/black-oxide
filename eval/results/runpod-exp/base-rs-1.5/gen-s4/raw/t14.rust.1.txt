```rust
fn main() {
    let text = "mississippi";
    let mut count_s = 0;
    let mut unique_chars: std::collections::HashSet<char> = HashSet::new();

    for c in text.chars() {
        if c == 's' {
            count_s += 1;
        }
        unique_chars.insert(c);
    }

    println!("The character 's' occurs {} times.", count_s);
    println!("There are {} distinct characters in the string.", unique_chars.len());
}
```
