fn filter_and_print_under_ten(numbers: Vec<Int>) -> Int {
    let mut count = 0
    for n in numbers {
        if n < 10 {
            print(n)
            count = count + 1
        }
    }
    count
}

fn main() {
    let numbers = vec(9, 2, 12, 5, 16)
    let kept_count = filter_and_print_under_ten(numbers)
    print(kept_count)
}
