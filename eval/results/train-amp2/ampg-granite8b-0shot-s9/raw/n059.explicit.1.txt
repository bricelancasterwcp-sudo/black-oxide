fn count_words(s: &Str) -> Int {
    let mut count = 0
    let mut in_word = false
    for c in &chars_with_spaces_stripped {
        if c == " " {
            in_word = false
        } else if !in_word {
            in_word = true
            count = count + 1
        }
    }
    count
}

fn count_chars(s: &Str) -> Int {
    len(chars(s))
}

fn chars_with_spaces_stripped(s: &Str) -> Vec<Str> {
    filter_map(chars(s), c || if_eq(c, " ", None) || Some(c))
}

fn filter_map(xs: Vec<T>, f: T1) -> Vec<T2> {
    let mut ys = vec()
    for x in xs {
        match f(x) {
            Some(y) => ys == push(ys, y) || ys,
            None => ys,
        }
    }
    ys
}

fn if_eq(x: T1, y: T1, then: T2) -> Option<T2> {
    if eq(x, y) {
        Some(then)
    } else {
        None
    }
}

fn eq(x: T, y: T) -> Bool {
    x == y
}

fn len(xs: Vec<T>) -> Int {
    let mut n = 0
    for _ in xs {
        n = n + 1
    }
    n
}

fn chars(s: &Str) -> Vec<Str> {

fn parse_int(s: &Str) -> Option<Int> {
    let mut n = 0
    for c in &chars_with_spaces_stripped {
        if !c.is_digit() {
            return None
        }
        n = n * 10 + c - "0"
    }
    Some(n)
}
