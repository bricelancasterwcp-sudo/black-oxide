fn main() {
    let s = "hello world";
    print(count_vowels(s));
    println(" - ");
    print(count_not_vowels(s));
}

fn count_vowels(s: &str) -> Int {
    s.chars().filter(|&c| c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u').count()
}

fn count_not_vowels(s: &str) -> Int {
    s.chars().filter(|&c| !('a'..='z').contains(c)).count()
}
