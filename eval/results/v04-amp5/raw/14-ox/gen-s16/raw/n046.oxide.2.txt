fn count_under(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x < cutoff {
            count += 1
        }
    }
    count
}

fn count_over_or_equal(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= cutoff {
            count += 1
        }
    }
    count
}

fn main() {
    let values = vec().push(5).push(12).push(3).push(18).push(9)
    let cutoff = 10
    print(count_under(values.clone(), cutoff))
    print(count_over_or_equal(values, cutoff))
}
