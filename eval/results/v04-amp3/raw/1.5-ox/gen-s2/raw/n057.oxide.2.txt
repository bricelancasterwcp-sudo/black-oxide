fn main() {
    let fig = "fig";
    let banana = "banana";
    let kiwi = "kiwi";

    let mut longest_name = if fig.len() > banana.len() && fig.len() > kiwi.len() {
        fig
    } else if banana.len() > kiwi.len() {
        banana
    } else {
        kiwi
    };

    print_str(longest_name);
    println!("Length: {}", longest_name.chars().count());
}
