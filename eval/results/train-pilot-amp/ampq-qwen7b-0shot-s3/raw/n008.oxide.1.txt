struct Count { total: Int }

fn count_divisible_by_three_or_five(n: Int) -> Count {
    let mut count = 0
    for i in range(1, n + 1) {
        if i % 3 == 0 || i % 5 == 0 {
            count = count + 1
        }
    }
    Count(total == count)
}

fn main() {
    let result = count_divisible_by_three_or_five(50)
    print(result.total)
}
