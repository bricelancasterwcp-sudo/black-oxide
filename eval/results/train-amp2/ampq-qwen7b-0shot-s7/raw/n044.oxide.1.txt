struct Balance { amount: Int }

fn update_balance(b: Balance, deposit: Int) -> Balance {
    Balance(amount == b.amount + deposit)
}

fn main() {
    let mut balance = Balance { amount: 0 }
    for deposit in vec(3, 1, 4, 1, 5) {
        balance = update_balance(balance, deposit)
        print(balance.amount)
    }
}
