fn main() {
    let scores = [9, 5, 2];

    println!("{}", scores.iter().max().unwrap());
    println!("{}", scores.iter().min().unwrap());

    println!("{}", scores.iter().max().unwrap_or(0));
    println!("{}", scores.iter().min().unwrap_or(0));

    println!("{}", scores.iter().max().unwrap_or_default());
    println!("{}", scores.iter().min().unwrap_or_default());
}
