```rust
fn main() {
    let s = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in s.chars() {
        if c.to_digit(10).is_some() {
            sum += c.to_digit(10).unwrap();
            count += 1;
        }
    }

    println!("Sum of digits: {}", sum);
    println!("Number of digit characters: {}", count);
}
```
