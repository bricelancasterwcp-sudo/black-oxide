fn square_area(side: Float) -> Float {
    side * side
}

fn area(shape: Shape) -> Float {
    match shape {
        Square(s) => square_area(s),
        Circle(r) => r * r * 3.14159,
    }
}

fn main() {
    print(area(Square(4.0)))
}
