fn min_and_gap(nums: &Vec<Int>) -> Result<Vec<Int>, String> {
    if nums.is_empty() {
        return Err("Vector is empty.".to_string())
    }
    let mut min = nums.first().unwrap().clone()
    let mut max = nums.first().unwrap().clone()
    for x in &nums {
        if x < &min {
            min = x.clone()
        }
        if x > &max {
            max = x.clone()
        }
    }
    Ok(vec_push(vec_push(vec(), min), max - min))
}
