struct Rectangle { w: Int, h: Int }
struct Triangle { a: Int, b: Int, c: Int }
fn rect_perimeter(r: Rectangle) -> Int {
    2 * (r.w + r.h)
}
fn tri_perimeter(t: Triangle) -> Int {
    t.a + t.b + t.c
}
fn main() {
    let r1 = Rectangle { w: 3, h: 4 }
    let t = Triangle { a: 3, b: 4, c: 5 }
    let r2 = Rectangle { w: 2, h: 5 }
    print(rect_perimeter(r1))
    print(tri_perimeter(t))
    print(rect_perimeter(r2))
}
