```rust
fn main() {
    let s = "mississippi";
    let count_s = s.chars().filter(|c| *c == 's').count();
    let unique_chars: Vec<char> = s.chars().collect();
    let count_unique = unique_chars.into_iter().collect::<std::collections::HashSet<_>>().len();
    print(count_s);
    print_str("\n");
    print(count_unique);
}
```
