```rust
fn main() {
    let fig = "fig";
    let banana = "banana";
    let kiwi = "kiwi";

    let mut longest_name = banana;
    let mut longest_count = banana.len();

    if fig.len() > longest_count {
        longest_name = fig;
        longest_count = fig.len();
    }

    if kiwi.len() > longest_count {
        longest_name = kiwi;
        longest_count = kiwi.len();
    }

    print_str(longest_name);
    print!(" ");
    println!("{}", longest_count);
}
```
