fn main() {
    let alpha = vec(-5, 12, 30, 22, 0, 35, 18)
    let beta = vec(8, -12, 5, 41, 27)
    for station in vec(alpha, beta) {
        print_str(station)
        let below_zero = count_if(station, |x| x <= 0)
        let above_thirty = count_if(station, |x| x >= 30)
        let max_fahrenheit = max(map(station, |c| (c * 9 / 5) + 32))
        let min_fahrenheit = min(map(station, |c| (c * 9 / 5) + 32))
        let mean_celsius = floor(mean(station))
        print(below_zero)
        print(above_thirty)
        print(max_fahrenheit)
        print(min_fahrenheit)
        print(mean_celsius)
    }
    let alpha_mean = floor(mean(alpha))
    let beta_mean = floor(mean(beta))
    if alpha_mean > beta_mean {
        print_str("alpha")
    } else {
        print_str("beta")
    }
}
