struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn min(rs: Vec<Reading>) -> Int {
    let mut smallest = first_big(get(rs, 0)?.values)
    for r in rs {
        smallest = min(smallest, first_big(r.values))
    }
    smallest
}

fn main() {
    let rs = vec(Reading { label: "lab", values: vec(5, 3, 9, 1, 7) })
    print(min(rs))
}
