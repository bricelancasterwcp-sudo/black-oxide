struct List { items: Vec<Int> }

fn count_entries(a: Vec<Int>, b: Vec<Int>) -> Int {
    a.len() + b.len()
}

fn main() {
    let list1 = List { items: push(push(vec(), 1), 2) }
    let list2 = List { items: push(push(vec(), 4), 5) }
    print(count_entries(list1.items.clone(), list2.items.clone()))
}
