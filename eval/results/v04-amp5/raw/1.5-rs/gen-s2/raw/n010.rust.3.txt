```rust
fn main() {
    let num = 91;
    let mut is_prime = true;

    for i in 2..num {
        if num % i == 0 {
            println!("No");
            is_prime = false;
            break;
        }
    }

    // Check if num is not equal to 2 (which is the only even prime number)
    if num != 2 && !is_prime {
        println!("No");
    } else {
        println!("Yes");
    }
}
```
