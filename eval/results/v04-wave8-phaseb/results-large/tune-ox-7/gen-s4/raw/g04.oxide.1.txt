fn main() {
    let numbers = vec(360, 97, 1024, 561, 143, 2003)
    let mut count_primes = 0
    for n in numbers {
        print(n)
        let (divisors_count, sum_divisors, max_divisor) = prime_factors(n)
        print(divisors_count)
        print(sum_divisors)
        print(max_divisor)
        if is_prime(n) {
            count_primes += 1
        }
    }
    print(count_primes)
}
fn prime_factors(n: Int) -> (Int, Int, Option<Int>) {
    let mut divisors = vec()
    let mut sum = 0
    let mut max_divisor = None
    for i in range(2, n + 1) {
        if n % i == 0 {
            divisors = push(divisors, i)
            sum += i
            max_divisor = Some(i)
        }
    }
    (len(divisors), sum, max_divisor)
}
fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    for i in range(2, n) {
        if n % i == 0 {
            return false
        }
    }
    true
}
