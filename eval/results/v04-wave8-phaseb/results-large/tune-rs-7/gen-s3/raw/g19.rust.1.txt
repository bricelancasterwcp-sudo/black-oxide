fn main() {
    let text = "order 47 shipped on 2026-03-15, invoice 1290 paid.";
    let mut letters = 0;
    let mut digits = 0;
    let mut spaces = 0;
    let mut others = 0;
    let mut longest_streak = 0;
    let mut current_streak = 0;
    let mut all_digits = String::new();
    for c in text.chars() {
        if c.is_alphabetic() {
            letters += 1;
        } else if c.is_digit(10) {
            digits += 1;
            current_streak += 1;
            longest_streak = longest_streak.max(current_streak);
            all_digits.push(c);
        } else if c == ' ' {
            spaces += 1;
            current_streak = 0;
        } else {
            others += 1;
            current_streak = 0;
        }
    }
    println!("{}", text.len());
    println!("{}", letters);
    println!("{}", digits);
    println!("{}", spaces);
    println!("{}", others);
    println!("{}", longest_streak);
    println!("{}", all_digits);
}
