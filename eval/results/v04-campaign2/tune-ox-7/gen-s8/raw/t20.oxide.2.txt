struct Item { name: Str, unit_price: Int, quantity: Int }
fn calculate_cost(item: Item) -> Int {
    item.unit_price * item.quantity
}
fn main() {
    let pen = Item { name: "pen", unit_price: 3, quantity: 4 }
    let pad = Item { name: "pad", unit_price: 9, quantity: 2 }
    let ink = Item { name: "ink", unit_price: 5, quantity: 3 }
    let total_cost = calculate_cost(pen) + calculate_cost(pad) + calculate_cost(ink)
    print(total_cost)
    let items = vec(pen, pad, ink)
    let mut max_cost = -1
    let mut item_with_max_cost = ""
    for i in range(0, len(items)) {
        let item = get(items, i).unwrap_or(Item { name: "", unit_price: 0, quantity: 0 })
        let cost = calculate_cost(item)
        if cost > max_cost {
            max_cost = cost
            item_with_max_cost = clone(get_field(item, "name").unwrap_or(""))
        }
    }
    print_str(item_with_max_cost)
}
