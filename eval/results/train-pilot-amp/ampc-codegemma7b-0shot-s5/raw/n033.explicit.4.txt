fn signal(c: Color) {
    match c {
        Green => print("go for green"),
        Yellow => print("wait for yellow"),
        Red => print("stop for red"),
    }
}

enum Color {
    Green,
    Yellow,
    Red,
}

fn main() {
    let green = Green
    let yellow = Yellow
    let red = Red
    signal(green)
    signal(yellow)
    signal(red)
}
