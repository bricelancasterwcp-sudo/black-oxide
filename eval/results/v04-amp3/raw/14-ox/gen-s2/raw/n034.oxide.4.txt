fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?;
    Some(x + 1)
}

fn main() {
    let numbers = vec(); // Corrected to use vec()
    numbers.push(7);     // Add elements to the vector
    numbers.push(8);
    numbers.push(9);
    match second(numbers) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
