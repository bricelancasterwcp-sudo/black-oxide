```rust
struct Shape {
    type_: Str,
    param: Float,
}

impl Shape {
    fn area(&self) -> Float {
        match self {
            Square(side) => side * side,
            Circle(radius) => radius * radius * 3.141592653589793,
        }
    }
}

fn main() {
    let s = Square(4);
    println!("Area: {}", s.area());
}
```
