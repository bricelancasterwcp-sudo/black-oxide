fn main() {
    let v = vec(10, 20, 30, 40)
    if len(&v) >= 2 {
        print(get(&v, 2).unwrap())
    } else {
        print("missing")
    }
    drop v
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < v.len() {
        Some(v.get(i).unwrap())
    } else {
        None
    }
}

fn len(v: &Vec<Int>) -> Int {
    v.len()
}

fn print(s: &Str) {
    print(s)
}

fn drop_vec(v: Vec<Int>) {
    drop v
}

fn main() {
    let v = vec(10, 20, 30, 40)
    if len(&v) >= 2 {
        print(get(&v, 2).unwrap())
    } else {
        print("missing")
    }
    drop_vec(v)
}
