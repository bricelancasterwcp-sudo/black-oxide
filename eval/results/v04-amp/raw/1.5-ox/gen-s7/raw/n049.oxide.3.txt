```rust
struct Figure {
    name: Str,
    value: Int,
}

fn main() {
    let figures = vec![
        Figure { name: "circle".to_string(), value: 3.14 },
        Figure { name: "square".to_string(), value: 16 },
        Figure { name: "triangle".to_string(), value: 9 },
        Figure { name: "rectangle".to_string(), value: 25 },
    ];

    let smallest = figures.iter().min_by(|a, b| a.value.cmp(&b.value)).unwrap();
    print_str(smallest.name);

    let largest = figures.iter().max_by(|a, b| a.value.cmp(&b.value)).unwrap();
    let gap = largest.value - smallest.value;
    print_int(gap);
}
```
