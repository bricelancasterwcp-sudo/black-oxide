fn main() {
    let rect = Rectangle { width: 7, height: 6 }
    print(area(&rect))
}

fn area(r: &Rectangle) -> Int {
    r.width * r.height
}

struct Rectangle {
    width: Int,
    height: Int,
}

fn drop_example() {
    let a = 3
    let b = 4
    drop a
    let c = 5
    print(a) // EX0002, reads a after it is dropped
    print(b) // EX0004, b is not dead
    print(c) // EX0004, c is not dead
}
