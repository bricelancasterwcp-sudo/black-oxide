fn count_vowels(s: Str) -> Int {
    let vowels = "aeiouAEIOU"
    let mut count = 0
    for c in chars(s) {
        if contains(vowels, c) {
            count += 1
        }
    }
    count
}

fn count_consonants(s: Str) -> Int {
    let vowels = "aeiouAEIOU"
    let mut count = 0
    for c in chars(s) {
        if !contains(vowels, c) && c != ' ' {
            count += 1
        }
    }
    count
}

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    print(count_vowels(keyboard))
    print(count_consonants(keyboard))
}
