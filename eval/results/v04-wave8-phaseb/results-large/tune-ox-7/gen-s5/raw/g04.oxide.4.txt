fn main() {
    let numbers = vec(360, 97, 1024, 561, 143, 2003)
    let mut count_prime = 0
    for n in numbers {
        print(n)
        let (divisors, sum_divisors, max_divisor) = prime_factors(n)
        print(len(divisors))
        print(sum_divisors)
        match max_divisor {
            Some(m) => print(m),
            None => print_str("None"),
        }
        if is_prime(n) {
            count_prime += 1
        }
    }
    print(count_prime)
}
fn prime_factors(n: Int) -> (Vec<Int>, Int, Option<Int>) {
    let mut divisors = vec()
    let mut sum_divisors = 0
    let mut max_divisor = None
    for i in range(2, n + 1) {
        if n % i == 0 {
            push(divisors, i)
            sum_divisors += i
            max_divisor = Some(i)
        }
    }
    (divisors, sum_divisors, max_divisor)
}
fn is_prime(n: Int) -> Bool {
    if n < 2 { return false }
    for i in range(2, n) {
        if n % i == 0 { return false }
    }
    true
}
