```rust
fn main() {
    let mut signal = "green";
    while true {
        match signal {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => panic!("unexpected signal: {}", signal),
        }
        // Simulate receiving a signal
        if signal == "red" {
            signal = "yellow";
        } else if signal == "green" {
            signal = "red";
        } else if signal == "yellow" {
            signal = "green";
        }
    }
}
```
