```rust
struct Queue {
    elements: Vec<Int>,
}

impl Queue {
    fn new() -> Queue {
        Queue { elements: vec![1, 2, 3, 4, 5] }
    }

    fn exchange_first_last(&mut self) {
        if self.elements.len() < 2 {
            panic!("Queue must have at least two elements to exchange first and last");
        }
        let temp = self.elements.remove(0);
        self.elements.push(temp);
    }

    fn print_elements(&self) {
        for &x in &self.elements {
            println!("{}", x);
        }
    }
}

fn main() {
    let mut q = Queue::new();
    q.exchange_first_last();
    q.print_elements();
}
```
