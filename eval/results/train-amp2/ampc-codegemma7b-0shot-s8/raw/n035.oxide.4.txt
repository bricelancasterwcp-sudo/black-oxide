fn divide(a: Float, b: Float) -> Result<Float, Str> {
    if b == 0.0 {
        Err("can't divide by zero".to_string())
    } else {
        Ok(a / b)
    }
}

fn main() {
    match divide(20.0, 4.0) {
        Ok(res) => print(res),
        Err(err) => print_str(err),
    }
    match divide(20.0, 0.0) {
        Ok(res) => print(res),
        Err(err) => print_str(err),
    }
}
