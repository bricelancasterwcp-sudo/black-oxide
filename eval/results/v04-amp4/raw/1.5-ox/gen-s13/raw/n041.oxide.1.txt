fn main() {
    let v = [4, 2, 9, 7];
    for i in 0..v.len() - 1 {
        for j in i + 1..v.len() {
            if v[i] > v[j] {
                let temp = v[i];
                v[i] = v[j];
                v[j] = temp;
            }
        }
    }
    for &x in &v {
        print!("{}", x);
        println!();
    }
}
