fn print_first_and_last(word: &Str) {
    print_char(word.chars().nth(0).unwrap())
    print_char(word.chars().nth(word.len() - 1).unwrap())
}

fn print(s: &Str, i: Int) {
    print_char(s.chars().nth(i).unwrap())
}

fn print_char(c: Char) {
    print(c)
}

    print_first_and_last("granite")
}
