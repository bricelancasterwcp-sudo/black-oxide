fn main() {
    let mut v = [4, 7, 9, 2];
    for i in 0..v.len() - 1 {
        for j in i + 1..v.len() {
            if v[i] > v[j] {
                v.swap(i, j);
            }
        }
    }
    println!("{}", v[0]);
    println!("{}", v[1]);
    println!("{}", v[2]);
    println!("{}", v[3]);
}
