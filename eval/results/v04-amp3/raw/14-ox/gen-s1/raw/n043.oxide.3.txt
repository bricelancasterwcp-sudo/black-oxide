struct Queue { data: Vec<Int> }

fn exchange_first_last(q: Queue) -> Queue {
    let len = len(q.data)
    if len < 2 {
        return q
    }
    let first = get(q.data, 0).unwrap()
    let last = get(q.data, len - 1).unwrap()
    let mut data = push(push(vec(), last), first)
    for i in range(1, len - 1) {
        data = push(data, get(q.data, i).unwrap())
    }
    Queue { data: data }
}

fn main() {
    let q = Queue { data: vec!(1, 2, 3, 4, 5) }
    let q = exchange_first_last(q)
    for x in q.data {
        print(x)
    }
}
