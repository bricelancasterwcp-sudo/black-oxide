fn main() {
    let items1 = vec(3, 3, 3, 3)
    let items2 = vec(2, 2, 2, 2, 2)
    let total = sum(&items1) + sum(&items2)
    print(total)
    drop items1
    drop items2
}

fn sum(items: &Vec<Int>) -> Int {
    let mut total = 0
    for item in items {
        total = total + item
    }
    total
}

fn vec(item: Int, count: Int) -> Vec<Int> {
    let mut v = Vec()
    for _ in 0 < count {
        push(&mut, v, item)
    }
    v
}

fn push(v: &Vec<Int>, item: Int) -> Vec<Int> {
    let mut v2 = v.clone()
    push(&mut, v2, item)
    v2
}
