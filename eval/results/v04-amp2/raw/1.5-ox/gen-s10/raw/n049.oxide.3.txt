struct Figure {
    name: Str,
    size: Int,
}

fn main() {
    let figures = vec![
        Figure { name: "circle".to_string(), size: 5 },
        Figure { name: "square".to_string(), size: 10 },
        Figure { name: "triangle".to_string(), size: 7 },
        Figure { name: "rectangle".to_string(), size: 8 },
    ];

    let smallest = figures.iter().min_by(|a, b| a.size.cmp(&b.size)).unwrap();
    print_str(smallest.name);

    let largest = figures.iter().max_by(|a, b| a.size.cmp(&b.size)).unwrap();
    let gap = largest.size - smallest.size;
    print_int(gap);
}
