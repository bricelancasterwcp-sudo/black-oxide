```rust
fn main() {
    let texts = vec![
        String::from("7"), 
        String::from("x"), 
        String::from("9"), 
        "?"
    ];

    let mut total = 0;
    let mut non_numbers_count = 0;

    for text in texts {
        if is_number(&text) {
            total += parse_number(&text).unwrap();
        } else {
            non_numbers_count += 1;
        }
    }

    println!("Total of numbers: {}", total);
    println!("Number of non-numbers: {}", non_numbers_count);
}

fn is_number(text: &str) -> bool {
    text.chars().all(|c| c.is_ascii_digit())
}

fn parse_number(text: &str) -> Result<i32, std::num::ParseIntError> {
    text.parse::<i32>()
}
```
