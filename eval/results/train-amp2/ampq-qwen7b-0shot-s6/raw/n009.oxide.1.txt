fn reverse_string(s: Str) -> Str {
    let mut chars = chars(s)
    let mut reversed = vec()
    while !chars.is_empty() {
        reversed = push(reversed, get(chars, 0)?)
        chars = concat(vec(), chars)?
    }
    concat(vec(), reversed)?
}
