struct Item { name: Str, unit_price: Int, quantity: Int }
fn main() {
    let items = vec(Item { name: "pen", unit_price: 3, quantity: 4 }, Item { name: "pad", unit_price: 9, quantity: 2 }, Item { name: "ink", unit_price: 5, quantity: 3 })
    let mut total_cost = 0
    let mut max_cost_item = ""
    for item in items {
        let cost = item.unit_price * item.quantity
        total_cost = total_cost + cost
        if cost > 0 {
            max_cost_item = item.name
        }
    }
    print(total_cost)
    print(max_cost_item)
}
