struct Line { items: Int, price: Int }

fn calculate(line: Line) -> Int {
    line.items * line.price
}

fn main() {
    let line1 = Line { items: 4, price: 3 };
    let line2 = Line { items: 5, price: 2 };
    print(calculate(line1));
    print(calculate(line2));
    print(calculate(Line { items: line1.items + line2.items, price: calculate(line1) / line1.items + calculate(line2) / line2.items }));
}
