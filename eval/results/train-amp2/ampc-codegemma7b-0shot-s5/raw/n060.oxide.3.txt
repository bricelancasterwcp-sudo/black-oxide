fn digits(s: Str) -> Int {
    let mut sum = 0
    let mut count = 0
    for c in s.chars() {
        if c.is_digit(10) {
            sum = sum + c.to_digit(10).unwrap()
            count = count + 1
        }
    }
    (sum * 10) + count
}

fn main() {
    let s = "a1b22"
    let result = digits(s)
    print(result)
}
