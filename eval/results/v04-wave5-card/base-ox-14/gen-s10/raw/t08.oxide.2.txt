fn count_positives(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count += 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max.is_none() || x > unwrap_or(max, -1) {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let values = vec(3, 8, -2, 12, 7)
    let positives = count_positives(clone(values))
    print(positives)
    match find_max(values) {
        Some(max) => print(max),
        None => print_str("No values"),
    }
}
