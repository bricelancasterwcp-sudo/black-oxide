fn main() {
    let figures = push(push(push(vec(), 8), 3), 11)
    let figures = push(figures, 6)

    let min_figure = match min(figures) {
        Some(n) => n,
        None => -1, // This should never happen for non-empty vectors
    }
    print(min_figure)

    let max_figure = match max(figures) {
        Some(n) => n,
        None => -1, // This should never happen for non-empty vectors
    }
    let gap = max_figure - min_figure
    print(gap)
}
