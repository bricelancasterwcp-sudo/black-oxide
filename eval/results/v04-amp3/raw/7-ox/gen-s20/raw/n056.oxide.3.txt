```rust
fn main() {
    let nums = vec![7, 'x', 9, '?'];
    let mut sum = 0;
    let mut count_non_nums = 0;

    for &num in &nums {
        match num {
            n if n.is_digit(10) => sum += n.to_digit(10).unwrap() as i32,
            _ => count_non_nums += 1,
        }
    }

    print(sum);
    print(count_non_nums);
}
```
