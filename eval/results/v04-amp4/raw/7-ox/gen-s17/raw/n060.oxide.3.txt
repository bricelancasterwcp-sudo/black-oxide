fn main() {
    let s = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in chars(s) {
        if is_digit(c) {
            sum += parse_int(concat(c, ""))?;
            count += 1;
        }
    }

    print(sum);
    print(count);
}

fn is_digit(c: Str) -> Bool {
    c >= "0" && c <= "9"
}
