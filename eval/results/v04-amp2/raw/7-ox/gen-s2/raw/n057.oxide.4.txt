fn main() {
    let fig = "fig";
    let banana = "banana";
    let kiwi = "kiwi";

    let mut max_len = 0;
    let mut fruit_name = "";

    if str_len(fig) > max_len {
        max_len = str_len(fig);
        fruit_name = fig;
    }

    if str_len(banana) > max_len {
        max_len = str_len(banana);
        fruit_name = banana;
    }

    if str_len(kiwi) > max_len {
        max_len = str_len(kiwi);
        fruit_name = kiwi;
    }

    print_str(fruit_name);
    print(max_len);
}
