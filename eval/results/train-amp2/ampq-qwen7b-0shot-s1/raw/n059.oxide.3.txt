fn count_words(s: Str) -> Int {
    let words = chars(s).filter_eq(" ").count()
    words + 1
}

fn count_chars(s: Str) -> Int {
    str_len(s)
}

fn main() {
    let phrase = "when in doubt go home"
    let word_count = count_words(phrase)
    let char_count = count_chars(phrase)
    print(word_count)
    print_str(" ")
    print(char_count)
}
