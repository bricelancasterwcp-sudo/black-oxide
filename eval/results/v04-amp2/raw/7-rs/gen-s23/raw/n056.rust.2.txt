fn main() {
    let texts = ["7", "x", "9", "?"];
    let mut sum = 0;
    let mut count_non_numbers = 0;

    for text in texts {
        match text.parse::<i32>() {
            Ok(num) => sum += num,
            Err(_) => count_non_numbers += 1,
        }
    }

    println!("Total of numbers: {}", sum);
    println!("Count of non-numbers: {}", count_non_numbers);
}
