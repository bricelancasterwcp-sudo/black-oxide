```rust
fn main() {
    let v: Vec<Int> = vec![7, 8, 9];
    if v.len() > 1 {
        match second(v) {
            Some(n) => println!("{}", n),
            None => println!("no second entry"),
        }
    } else {
        println!("no second entry");
    }
}

fn second(v: Vec<Int>) -> Option<Int> {
    if v.len() > 1 {
        Some(v[1])
    } else {
        None
    }
}
```
