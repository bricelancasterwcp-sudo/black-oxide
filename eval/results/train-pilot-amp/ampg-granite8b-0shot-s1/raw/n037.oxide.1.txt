fn main() {
    let rectangle = Rectangle { width: 2, height: 3 }
    rectangle = Rectangle { width: 5, ..rectangle }
    print(rectangle.area())
}

struct Rectangle { width: Int, height: Int }

fn area(r: Rectangle) -> Int {
    r.width * r.height
}

fn main() {
    let rectangle = Rectangle { width: 2, height: 3 }
    print(area(Rectangle { width: 5, ..rectangle }))
}
