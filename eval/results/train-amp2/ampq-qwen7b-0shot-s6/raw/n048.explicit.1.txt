fn find_positions(v: &Vec<Int>) -> Vec<Int> {
    let mut positions = vec()
    for i in range(0, len(v)) {
        if get(v, i) == Some(4) {
            push(positions, i)
        }
    }
    positions
}

fn main() {
    let nums = vec(4, 7, 4, 9, 4)
    let positions = find_positions(&nums)
    for pos in &positions {
        print(pos)
    }
    drop nums
    drop positions
}
