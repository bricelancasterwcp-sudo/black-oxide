fn len(s: &Str) -> Int {
    let mut total = 0
    for c in chars(s) {
        total = total + 1
    }
    total
}

fn chars(s: &Str) -> Vec<Str> {
    let mut chars = vec()
    for i in 0.to_int() < s.len() {
        chars = push(chars, str_index(s, i))
    }
    chars
}

fn str_index(s: &Str, i: Int) -> Str {
    Str(i)
}

fn str_len(s: &Str) -> Int {
    let mut total = 0
    for _ in chars(s) {
        total = total + 1
    }
    total
}

        chars = push(chars, str_index(&str_lit_digits, mod_int(x, 10)))

}
