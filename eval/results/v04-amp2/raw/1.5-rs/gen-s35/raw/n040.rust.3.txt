```rust
fn main() {
    let square_side = 4;
    let circle_radius = 3.0;

    println!("Area of square: {}", square_area(square_side));
    println!("Area of circle: {:.2}", circle_area(circle_radius));
}

fn square_area(side: u32) -> f64 {
    side as f64 * side as f64
}

fn circle_area(radius: f64) -> f64 {
    radius * radius * std::f64::consts::PI
}
```
