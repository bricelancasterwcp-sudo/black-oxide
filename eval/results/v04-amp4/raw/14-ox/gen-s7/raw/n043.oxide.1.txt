struct Queue { items: Vec<Int> }

fn main() {
    let q = Queue { items: push(push(push(push(push(vec(), 1), 2), 3), 4), 5) }
    let first = get(q.items, 0)?
    let last_index = len(q.items) - 1
    let last = get(q.items, last_index)?
    let q = set(set(q.items, 0, last), last_index, first)
    for item in q {
        print(item)
    }
}
