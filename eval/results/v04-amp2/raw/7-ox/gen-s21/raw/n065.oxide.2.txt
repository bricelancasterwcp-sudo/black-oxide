struct Reading { label: Str, values: Vec<Int> }

fn count_above_five(v: Vec<Int>) -> Int {
    let mut count = 0;
    for x in v {
        if x > 5 {
            count += 1;
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max_val = None;
    for &x in &v {
        match max_val {
            Some(m) => if x > m { max_val = Some(x); },
            None => max_val = Some(x),
        }
    }
    max_val
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 6), 11), 3), 14) };
    print(count_above_five(r.values)); // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r };
    match max_reading(r2.values) {
        Some(n) => print(n),
        None => print_str("no readings"),
    }
}
