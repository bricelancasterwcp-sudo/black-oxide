fn main() {
    let line1_units = 4;
    let line1_price_per_unit = 3;
    let line1_total = line1_units * line1_price_per_unit;

    let line2_units = 5;
    let line2_price_per_unit = 2;
    let line2_total = line2_units * line2_price_per_unit;

    println!("Line 1 value: {}", line1_total);
    println!("Line 2 value: {}", line2_total);
    println!("Total value: {}", line1_total + line2_total);
}
