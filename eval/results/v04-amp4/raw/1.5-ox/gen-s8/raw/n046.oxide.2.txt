fn main() {
    let v = vec![5, 12, 3, 18, 9];
    let mut count_under_cutoff = 0;
    let mut count_over_or_equal_cutoff = 0;

    for x in &v { // fix: use &v to avoid moving v
        if *x < 10 {
            count_under_cutoff += 1;
        } else {
            count_over_or_equal_cutoff += 1;
        }
    }

    println!("Count under cutoff: {}", count_under_cutoff);
    println!("Count over or equal cutoff: {}", count_over_or_equal_cutoff);
}
