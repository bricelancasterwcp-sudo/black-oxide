fn main() {
    let s = "mississippi";
    let count_s = s.chars().filter(|c| *c == 's').count();
    let unique_chars: Vec<Str> = s.chars().collect::<Vec<char>>().into_iter().collect::<std::collections::HashSet<_>>().iter().map(|&c| c.to_string()).collect();
    let distinct_count = unique_chars.len();
    print(count_s);
    print_str("\n");
    print(distinct_count);
}
