fn main() {
    let phrase = "when in doubt go home";
    let num_words = word_count(phrase);
    let num_chars = char_count(phrase);

    println!("The phrase contains {} words.", num_words);
    println!("The phrase spans {} characters including spaces.", num_chars);
}

fn word_count(s: Str) -> Int {
    s.split_whitespace().count()
}

fn char_count(s: Str) -> Int {
    s.len()
}
