```rust
struct Queue {
    items: Vec<Int>,
}

impl Queue {
    fn new() -> Self {
        Queue { items: vec![1, 2, 3, 4, 5] }
    }

    fn exchange_first_last(&mut self) {
        if self.items.len() > 1 {
            let first = self.items.remove(0);
            let last = self.items.pop().unwrap();
            self.items.insert(0, last);
            self.items.push(first);
        }
    }

    fn print_queue(&self) {
        for item in &self.items {
            println!("{}", item);
        }
    }
}

fn main() {
    let mut queue = Queue::new();
    queue.exchange_first_last();
    queue.print_queue();
}
```
