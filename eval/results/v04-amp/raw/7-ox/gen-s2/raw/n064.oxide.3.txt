struct List { value: Int, next: Option<Box<List>> }

fn get_at(list: &List, pos: Int) -> Option<Int> {
    let mut current = list;
    for _ in 0..pos {
        match current.next {
            Some(ref next) => current = next,
            None => return None
        }
    }
    Some(current.value)
}

fn main() {
    let list = List { value: 10, next: Some(Box::new(List { value: 20, next: None })) };
    match get_at(&list, 1) {
        Some(value) => print(value),
        None => print_str("none")
    }
    match get_at(&list, 4) {
        Some(value) => print(value),
        None => print_str("none")
    }
}
