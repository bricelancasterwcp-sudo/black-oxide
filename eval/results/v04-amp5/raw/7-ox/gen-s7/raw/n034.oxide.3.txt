fn second(v: Vec<Int>) -> Option<Int> {
    get(v, 1)?
}

fn main() {
    let v7 = push(push(vec(), 3), 42)
    match second(v7) {
        Some(n) => print(n),
        None => print_str("none"),
    }

    let v8 = push(push(vec(), 5), 6)
    match second(v8) {
        Some(n) => print(n),
        None => print_str("none"),
    }

    let v9 = vec()
    match second(v9) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
