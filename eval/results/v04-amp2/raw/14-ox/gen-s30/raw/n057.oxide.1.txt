fn main() {
    let fig = "fig"
    let banana = "banana"
    let kiwi = "kiwi"

    let max_len = max(str_len(fig), max(str_len(banana), str_len(kiwi)))?
    let name = if max_len == str_len(fig) {
        fig
    } else if max_len == str_len(banana) {
        banana
    } else {
        kiwi
    }

    print(name)
    print(max_len)
}
