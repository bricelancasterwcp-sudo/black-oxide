fn main() {
    let side = 4.0;
    let radius = 3.0;

    let area_square = side * side;
    let area_circle = 3.14159 * radius * radius;

    println!("The area of a square with side {} is {}", side, area_square);
    println!("The area of a circle with radius {} is {}", radius, area_circle);
}
