fn main() {
    let words = vec("rotor", "banana", "level", "quiet", "deified", "system", "civic")
    let mut palindrome_count = 0
    for word in words {
        print_str(word)
        let len = str_len(word)
        print(len)
        let reversed = reverse_str(word)
        print_str(reversed)
        if word == reversed {
            print_str("yes")
            palindrome_count += 1
        } else {
            print_str("no")
        }
        let vowel_count = count_vowels(word)
        print(vowel_count)
        let unique_char_count = count_unique_chars(word)
        print(unique_char_count)
    }
    print(palindrome_count)
}
fn reverse_str(s: Str) -> Str {
    let chars = chars(s)
    let len = len(chars)
    let mut reversed = ""
    for i in range(0, len) {
        let c = get(chars, len - 1 - i).unwrap_or("")
        reversed = concat(reversed, c)
    }
    reversed
}
fn count_vowels(s: Str) -> Int {
    let vowels = "aeiou"
    let mut count = 0
    for c in chars(s) {
        if contains(vowels, c) {
            count += 1
        }
    }
    count
}
fn count_unique_chars(s: Str) -> Int {
    let mut seen = vec()
    for c in chars(s) {
        if !contains(seen, c) {
            push(seen, c)
        }
    }
    len(seen)
}
