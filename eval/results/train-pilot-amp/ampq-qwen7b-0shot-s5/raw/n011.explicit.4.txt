fn sum_multiples(n: Int) -> Int {
    let mut total = 0
    for i in range(1, n + 1) {
        let multiple = i * 4
        total = total + multiple
    }
    total
}

fn main() {
    let mut multiples = vec()
    for i in range(1, 7) {
        let multiple = i * 4
        multiples = push(multiples, multiple)
    }
    let total = sum_multiples(6)
    print(total)
    drop multiples
}
