fn list_get(list: Vec<Int>, pos: Int) -> Option<Int> {
    match get(list, pos)? {
        Either(a, b) => Some(a + b),
        _ => None,
    }
}

fn main() {
    let l = push(push(push(push(vec(), 1), 1), 2), 2)
    let x = list_get(l, 5)
    match x {
        Some(v) => print(v),
        None => print_str("none"),
    }
}
