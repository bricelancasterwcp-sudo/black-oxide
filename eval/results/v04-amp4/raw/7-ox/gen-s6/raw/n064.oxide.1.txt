fn get_value(v: Vec<Int>, i: Int) -> Option<Int> {
    get(v, i)
}

fn main() {
    let list = push(push(vec(), 10), 20);
    match (get_value(list.clone(), 1), get_value(list, 4)) {
        (Some(a), Some(b)) => print(concat(int_to_str(a), " ", int_to_str(b))),
        _ => print_str("none"),
    }
}
