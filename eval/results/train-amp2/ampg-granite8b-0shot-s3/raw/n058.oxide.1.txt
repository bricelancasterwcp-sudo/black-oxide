fn main() {
    let mut coffee = "coffee".to_string().chars().collect().to_vec()
    loop_for_letter(coffee.clone(), "c")
    loop_for_letter(coffee.clone(), "o")
    loop_for_letter(coffee.clone(), "f")
    loop_for_letter(coffee.clone(), "e")
}

fn loop_for_letter(mutcoffee: Vec<Char>, letter: Str) {
    let mut i = 0
    loop_while_letter_not_found(coffee.clone(), letter, i)
}

fn loop_while_letter_not_found(mutcoffee: Vec<Char>, letter: Str, muti: Int) {
    if get(coffee, i)? == letter {
        return
    }
    i = i + 1
    loop_while_letter_not_found(coffee.clone(), letter, i)
}

fn str_concat(a: Str, b: Str) -> Str {
    let mut v = vec()
    for c in a.chars() {
        push(v, c)
    }
    for c in b.chars() {
        push(v, c)
    }
    chars_to_str(v)
}
