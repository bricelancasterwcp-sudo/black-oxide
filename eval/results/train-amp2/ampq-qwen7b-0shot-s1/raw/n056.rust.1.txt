```rust
fn main() {
    let texts = ["7", "x", "9", "?"];
    let mut total = 0;
    let mut missing = 0;

    for text in texts {
        if let Ok(num) = text.parse::<i32>() {
            total += num;
        } else {
            missing += 1;
        }
    }

    println!("Total: {}", total);
    println!("Missing: {}", missing);
}
```
