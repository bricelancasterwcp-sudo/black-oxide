fn int_to_str(n: Int) -> Str {
    let mut digits = vec()
    let mut x = n
    while x > 0 {
        digits = push(digits, trunc(x % 10))
        x = trunc(x / 10)
    }
    if len(digits) == 0 {
        digits = push(digits, 0)
    }
    let mut result = vec()
    for d in &digits {
        result = append(result, d)
    }
    result
}
