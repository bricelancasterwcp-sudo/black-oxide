fn main() {
    let texts: Vec<&str> = vec!("7", "x", "9", "?");
    let mut total_sum: Int = 0;
    let mut count = 0;

    for text in texts {
        match parse_int(text) {
            Ok(num) => {
                total_sum += num;
                count += 1;
            }
            Err(_) => {}
        }
    }

    println!("Total sum of numbers: {}", total_sum);
    println!("Count of numbers: {}", count);
}
