```rust
fn main() {
    let texts = ["7", "x", "9", "?"];
    let mut sum = 0;
    let mut count = 0;

    for text in texts {
        match text.parse::<i32>() {
            Ok(num) => sum += num,
            Err(_) => count += 1,
        }
    }

    println!("{}", sum);
    println!("{}", count);
}
```
