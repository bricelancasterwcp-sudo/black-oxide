struct LetterCount { letter: Str, count: Int }

fn main() {
    let word = "coffee"
    let mut counts = vec()
    for c in chars(word) {
        let found = false
        for i in range(0, len(counts)) {
            if get(counts, i)?.letter == c {
                let mut item = get(counts, i)?
                item.count += 1
                counts = set(counts, i, item)
                found = true
                break
            }
        }
        if !found {
            counts = push(counts, LetterCount { letter: c, count: 1 })
        }
    }
    for item in counts {
        if item.count > 1 {
            print(item.letter)
        }
    }
}
