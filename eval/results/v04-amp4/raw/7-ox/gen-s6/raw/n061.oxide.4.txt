struct Rect { width: Int, height: Int }

fn area(r: Rect) -> Int {
    r.width * r.height
}

fn perimeter(r: Rect) -> Int {
    2 * (r.width + r.height)
}

fn main() {
    let rect = Rect { width: 9, height: 3 };
    print(area(rect))           // field access copies: rect stays usable
    print(perimeter(rect))
}
