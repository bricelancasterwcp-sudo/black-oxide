```rust
fn main() {
    let data = [5, 12, 3, 18, 9];
    let cutoff = 10;
    
    let below_cutoff_count: usize = data.iter().filter(|&&x| x < cutoff).count();
    let over_cutoff_count: usize = data.iter().filter(|&&x| x >= cutoff).count();
    
    println!("Below cutoff: {}", below_cutoff_count);
    println!("Over cutoff: {}", over_cutoff_count);
}
```
