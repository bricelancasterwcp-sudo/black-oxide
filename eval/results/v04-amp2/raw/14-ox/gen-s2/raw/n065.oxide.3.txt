struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max_val = None
    for x in v {
        match max_val {
            Some(current_max) => {
                if x > current_max {
                    max_val = Some(x)
                }
            },
            None => {
                max_val = Some(x)
            }
        }
    }
    max_val
}

fn main() {
    let readings = vec()
    readings.push(6)
    readings.push(11)
    readings.push(3)
    readings.push(14)
    readings.push(9)

    let count = count_exceeding(clone(readings), 5)
    print(count)

    let readings_clone = clone(readings)
    let max_val = max_reading(readings_clone)
    match max_val {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
