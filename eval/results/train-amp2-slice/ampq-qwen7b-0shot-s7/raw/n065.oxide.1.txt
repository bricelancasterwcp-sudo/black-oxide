struct Reading { values: Vec<Int> }

fn count_above_five(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 5 {
            count = count + 1
        }
    }
    count
}

fn max_value(v: Vec<Int>) -> Int {
    let mut max = -1
    for x in v {
        if x > max {
            max = x
        }
    }
    max
}

fn main() {
    let readings = Reading { values: push(push(push(push(push(vec(), 6), 11), 3), 14), 9) }
    print(count_above_five(readings.values))
    print(max_value(readings.values))
}
