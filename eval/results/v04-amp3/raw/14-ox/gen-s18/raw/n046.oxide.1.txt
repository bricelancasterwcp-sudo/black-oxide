struct Reading { label: Str, values: Vec<Int> }

fn count_under_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x < cutoff {
            count += 1
        }
    }
    count
}

fn count_over_or_at_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= cutoff {
            count += 1
        }
    }
    count
}

fn main() {
    let values = push(push(push(push(vec(), 5), 12), 3), 18)
    let values = push(values, 9)
    let cutoff = 10
    let under_count = count_under_cutoff(values, cutoff)
    let over_count = count_over_or_at_cutoff(values, cutoff)
    print(under_count)
    print(over_count)
}
