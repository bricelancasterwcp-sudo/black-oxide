struct Reading { label: Str, values: Vec<Int> }

fn sum_of_numbers(v: Vec<Int>) -> Int {
    let mut sum = 0
    for x in v {
        if x != -1 {
            sum += x
        }
    }
    sum
}

fn count_non_numbers(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x == -1 {
            count += 1
        }
    }
    count
}

fn main() {
    let r = Reading { label: "lab", values: vec(-1, 7, -1, 9, -1) }
    let numbers = vec(7, -1, 9, -1)
    print(sum_of_numbers(numbers))
    print(count_non_numbers(numbers))
}
