fn main() {
    let values = vec();
    values.push(7);
    values.push(4);
    values.push(9);
    values.push(10);

    let mut sum_even = 0;
    for x in values {
        if x % 2 == 0 {
            sum_even += x;
        } else {
            print(x * 2);
        }
    }
    print(sum_even);
}
