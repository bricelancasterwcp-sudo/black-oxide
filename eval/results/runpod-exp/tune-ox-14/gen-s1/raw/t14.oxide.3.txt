fn count_s(text: Str) -> Int {
    let mut count = 0
    for c in chars(text) {
        if c == "s" {
            count = count + 1
        }
    }
    count
}
fn distinct_chars(text: Str) -> Int {
    let mut seen = set()
    for c in chars(text) {
        seen = insert(seen, c)
    }
    len(seen)
}
fn main() {
    let text = "mississippi"
    print(count_s(text))
    print(distinct_chars(text))
}
