fn sort(list: &Vec<Int>) -> Vec<Int> {
    if len(list) <= 1 {
        clone(&list)
    } else {
        let pivot = len(list) / 2
        let left = sort(slice(&list, 0, pivot))
        let right = sort(slice(&list, pivot, len(&list)))
        merge(&left, &right)
    }
}

fn merge(a: &Vec<Int>, b: &Vec<Int>) -> Vec<Int> {
    let mut result = vec()
    let mut i = 0
    let mut j = 0
    while i < len(a) && j < len(b) {
        if get(a, i) < get(b, j) {
            push(result, get(a, i))
            i = i + 1
        } else {
            push(result, get(b, j))
            j = j + 1
        }
    }
    while i < len(a) {
        push(result, get(a, i))
        i = i + 1
    }
    while j < len(b) {
        push(result, get(b, j))
        j = j + 1
    }
    result
}

fn main() {
    let numbers = push(push(push(vec(), 9), 7), 4)
    let sorted = sort(&numbers)
    print(get(&sorted, 0))
    print(get(&sorted, 1))
    print(get(&sorted, 2))
    drop sorted
    drop numbers
}
