struct Queue { items: Vec<Int> }

fn new_queue() -> Queue {
    Queue { items: vec(1, 2, 3, 4, 5) }
}

fn swap_first_last(q: Queue) -> Queue {
    let len = len(q.items)
    if len > 0 {
        q.items = swap(q.items, 0, len - 1)
    }
    q
}

fn main() {
    let mut q = new_queue()
    q = swap_first_last(q)
    for x in q.items {
        print(x)
    }
}
