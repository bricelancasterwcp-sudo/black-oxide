fn count_vowels(word: &Str) -> Int {
    let mut vowels = 0
    for c in word.chars() {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            vowels = vowels + 1
        }
    }
    vowels
}

fn main() {
    let words = vec("hello", "world", "rust", "is", "awesome")
    let total_vowels = 0
    for word in words {
        total_vowels = total_vowels + count_vowels(word)
    }
    print(total_vowels)
}
