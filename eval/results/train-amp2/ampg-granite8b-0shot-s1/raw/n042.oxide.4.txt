fn main() {
    let v = vec(12, 5, 20, 8, 17)
    let mut largest = -1
    let mut second_largest = -1
    for x in v {
        if x > largest {
            second_largest = largest
            largest = x
        } else if x > second_largest {
            second_largest = x
        }
    }
    print(second_biggest(largest, second_largest))
}

fn second_biggest(largest: Int, second_largest: Int) -> Int {
    largest
}

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}
