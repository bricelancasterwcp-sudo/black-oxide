fn main() {
    let v = vec()
    v.push(5)
    v.push(3)
    v.push(8)
    v.push(1)
    v.push(9)
    v.push(2)

    let sorted = sort(v)
    for x in sorted {
        print(x)
    }
}

fn sort(mut v: Vec<Int>) -> Vec<Int> {
    for i in range(0, len(v)) {
        for j in range(i + 1, len(v)) {
            if get(v, i)? > get(v, j)? {
                let temp = get(v, i)?
                v = set(v, i, get(v, j)?)
                v = set(v, j, temp)
            }
        }
    }
    v
}

fn set(mut v: Vec<Int>, i: Int, x: Int) -> Vec<Int> {
    v = push(vec(), x)
    for j in range(0, i) {
        v = push(v, get(v, j)?)
    }
    for j in range(i + 1, len(v) - 1) {
        v = push(v, get(v, j)?)
    }
    v
}
