fn main() {
    let mut acc = vec()
    loop_continue()
    let line = read_line()?
    match line.parse() {
        Ok(n) => acc == push(acc, n),
        Err(_) => break_if(acc.len() < 3),
    }
    for n in acc {
        print(n)
    }
    if acc.len() < 3 {
        print_str("bad")
    }
}

    new

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < len(v) {
        Some(v.get(i).unwrap())
    } else {
        None
    }
}
