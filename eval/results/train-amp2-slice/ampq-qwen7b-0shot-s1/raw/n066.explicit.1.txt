fn filter_and_print(nums: Vec<Int>) -> Int {
    let mut count = 0
    for num in nums {
        if num < 10 {
            print(num)
            count = count + 1
        }
    }
    count
}

fn main() {
    let numbers = vec(9, 2, 12, 5, 16)
    let kept_count = filter_and_print(numbers)
    print("Kept entries count: ")
    print(kept_count)
}
