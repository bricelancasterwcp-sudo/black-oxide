fn main() {
    let fruits = vec!["fig", "banana", "kiwi"];
    let mut max_len = 0;
    let mut longest_fruit = "";

    for fruit in fruits.iter() {
        if fruit.len() > max_len {
            max_len = fruit.len();
            longest_fruit = fruit;
        }
    }

    print_str(longest_fruit);
    print(max_len);
}
