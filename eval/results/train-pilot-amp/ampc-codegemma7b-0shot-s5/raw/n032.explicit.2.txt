    x: Int,

}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v = v
    v.push(x)
    v
}

fn print(x: Int) {
    if x != 0 {
        print(x / 10)
        print(x % 10)
    }
}

fn len(v: &Vec<Int>) -> Int {
    let mut total = 0
    for _ in &v {
        total = total + 1
    }
    total
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= len(&v) {
        None
    } else {
        Some(v.get(i))
    }
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a + 1 < b {
        v.push(i)
    }
    v
}

fn clone(x: &Point) -> Point {
    Point(x.x, x.y)
}

fn str_len(s: &Str) -> Int {
    let mut total = 0
    for _ in s {
        total = total + 1
    }
    total
}

fn concat(a: Str, b: Str) -> Str {
    let mut c = String.new()
    c.push_str(&a)
    c.push_str(&b)
    c
}

fn chars(s: &Str) -> Vec<Str> {
    let mut v = vec()
    for c in s.chars() {
        v.push(c.to_string())
    }
    v
}
