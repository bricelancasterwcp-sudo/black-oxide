fn main() {
    let readings = [6, 11, 3, 14, 9];

    let mut count = 0;
    for reading in readings {
        if reading > 5 {
            count += 1;
        }
    }

    let mut highest = readings[0];
    for reading in readings {
        if reading > highest {
            highest = reading;
        }
    }

    println!("Readings exceeding 5: {}", count);
    println!("Highest reading: {}", highest);
}
