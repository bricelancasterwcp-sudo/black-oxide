fn reverse(s: Str) -> Str {
    let mut result = String::with_capacity(s.len());
    for c in s.chars().rev() {
        result.push(c);
    }
    result
}

fn main() {
    let drum = "drum";
    let reversed_drum = reverse(drum);
    println!("Backwards: {}", reversed_drum);
    println!("Normal: {}", drum);
}
