fn main() {
    let p0 = Point { x: 1, y: 2 }
    let p1 = Point { x: 10, y: 5 }
    print(p0.x + p1.x)
}

struct Point {
    x: Int,
    y: Int,
}

fn print(x: Int) {
    println(x)
}

fn println(x: Int) {
    let mut out = "0"
    loop_while(x > 0, x == 0, x == 0, x == 0, x == 0, x == 0)
    drop out
}

fn loop_while(cond1: Bool, cond2: Bool, cond3: Bool, cond4: Bool, cond5: Bool, cond6: Bool) {
    if cond1 {
        if cond2 {
            if cond3 {
                if cond4 {
                    if cond5 {
                        if cond6 {
                            return
                        } else {
                            drop cond6
                        }
                    } else {
                        drop cond5
                    }
                } else {
                    drop cond4
                }
            } else {
                drop cond3
            }
        } else {
            drop cond2
        }
    } else {
        drop cond1
    }
}

fn drop_vec(v: &Vec<Int>) {
    for _ in v {
        drop _
    }
    drop v
}

fn clone(x: Int) -> Int {
    x
}

fn make_vec() -> Vec<Int> {
    vec()
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn to_float(x: Int) -> Float {
    to_float(x)
}

fn main() {
    let v = make_vec()
    v = push(v, 1)
    v = push(v, 2)
    v = push(v, 3)
    print(to_float(v.len()))
    drop_vec(&v)
}
