struct Line { units: Int, price: Int }

fn total_value(line: Line) -> Int {
    line.units * line.price
}

fn main() {
    let line1 = Line { units: 4, price: 3 }
    let line2 = Line { units: 5, price: 2 }
    
    print(total_value(line1))
    print(total_value(line2))
    
    let combined_value = total_value(line1) + total_value(line2)
    print(combined_value)
}
