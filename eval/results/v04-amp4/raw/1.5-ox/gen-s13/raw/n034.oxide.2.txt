fn second_entry(values: Vec<Int>) -> Option<Int> {
    match values.get(1) {
        Some(&x) => Some(x),
        None => None,
    }
}

fn main() {
    println!("{}", second_entry(vec![7]));          // prints "8"
    println!("{}", second_entry(vec![8, 9]));      // prints "9"
    println!("{:?}", second_entry(vec![]));         // prints "<std::option::None>"
}
