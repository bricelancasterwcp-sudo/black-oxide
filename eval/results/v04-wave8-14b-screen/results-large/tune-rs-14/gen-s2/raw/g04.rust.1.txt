fn main() {
    let numbers = [360, 97, 1024, 561, 143, 2003];
    let mut primes_count = 0;
    for &num in &numbers {
        let (divisors, sum, largest, is_prime) = analyze_number(num);
        println!("{}", num);
        println!("{}", divisors.len());
        println!("{}", sum);
        println!("{}", largest);
        println!("{}", is_prime);
        if is_prime {
            primes_count += 1;
        }
    }
    println!("{}", primes_count);
}
fn analyze_number(mut n: u32) -> (Vec<u32>, u32, u32, bool) {
    let mut divisors = Vec::new();
    let mut sum = 0;
    let mut largest = 0;
    let mut is_prime = true;
    for i in 2..=n {
        if n % i == 0 {
            if !divisors.contains(&i) {
                divisors.push(i);
                sum += i;
                if i > largest {
                    largest = i;
                }
            }
            n /= i;
            is_prime = false;
            if n == 1 {
                break;
            }
        }
    }
    (divisors, sum, largest, is_prime)
}
