struct Stats { vowels: Int, consonants: Int }

fn is_vowel(c: Str) -> Bool {
    match c {
        "a" | "e" | "i" | "o" | "u" => true,
        _ => false,
    }
}

fn count_letters(s: Str) -> Stats {
    let vowels = 0
    let consonants = 0
    for c in chars(s) {
        if is_vowel(c) {
            vowels = vowels + 1
        } else {
            consonants = consonants + 1
        }
    }
    Stats { vowels, consonants }
}

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    let stats = count_letters(keyboard)
    print(stats.vowels)
    print(stats.consonants)
}
