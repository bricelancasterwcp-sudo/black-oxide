    let mut v = vec()
    for i in range(1, 21) {
        v = push(v, i)
    }
    v
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let mut total = 0
    for x in &v {
        if x % 2 == 0 && x <= limit {
            total = total + x
        }
    }
    total
}
