fn walk(nums: &Vec<Int>) -> Int {
    let mut total = 0
    for x in &nums {
        if x % 2 != 0 {
            print_str(to_string(2 * x))
            println()
        } else {
            total = total + x
        }
    }
    total
}

fn main() {
    let nums = vec(7, 4, 9, 10)
    print(walk(&nums))
    drop nums
}
