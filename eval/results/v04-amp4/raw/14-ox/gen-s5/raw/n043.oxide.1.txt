fn main() {
    let queue = push(push(push(push(push(vec(), 1), 2), 3), 4), 5);
    let first = get(queue, 0).unwrap_or(-1);
    let last = get(queue, len(queue) - 1).unwrap_or(-1);
    let new_queue = set(set(queue, 0, last), len(queue) - 1, first);
    for i in range(0, len(new_queue)) {
        print(get(new_queue, i).unwrap_or(-1));
    }
}
