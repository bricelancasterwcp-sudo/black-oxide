let mut max_len = 0;
let mut longest_name = "";

fn check_name(name: Str) {
    let len = str_len(name);
    if len > max_len {
        max_len = len;
        longest_name = name;
    }
}

check_name("fig");
check_name("banana");
check_name("kiwi");

print_str(longest_name);
print(max_len);
