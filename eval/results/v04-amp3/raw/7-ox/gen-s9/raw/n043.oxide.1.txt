struct Queue { items: Vec<Int> }

fn exchange_first_last(q: &mut Queue) {
    if q.items.len() > 1 {
        let first = q.items[0];
        q.items[0] = q.items[q.items.len() - 1];
        q.items[q.items.len() - 1] = first;
    }
}

fn main() {
    let mut q = Queue { items: vec(1, 2, 3, 4, 5) };
    exchange_first_last(&mut q);
    for x in &q.items {
        print(x);
    }
}
