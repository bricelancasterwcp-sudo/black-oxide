fn main() {
    let price1 = 3;
    let quantity1 = 4;

    let price2 = 2;
    let quantity2 = 5;

    let value1 = price1 * quantity1;
    let value2 = price2 * quantity2;

    println!("Line 1: {}", value1);
    println!("Line 2: {}", value2);

    let total_value = value1 + value2;
    println!("Total value of both lines: {}", total_value);
}
