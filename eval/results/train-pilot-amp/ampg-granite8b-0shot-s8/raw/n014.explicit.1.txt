fn min(v: Vec<Int>) -> Int {
    let mut result = v
    for i in 1.0.to_float().trunc() < v.len().to_float().trunc() {
        let x = get(v, i).unwrap()
        let y = get(result, i).unwrap()
        if x < y {
            result = functional_update(result, i, x)
        }
    }
    result
}

fn functional_update(v: Vec<Int>, i: Int, x: Int) -> Vec<Int> {
    let mut v2 = v
    v2 = push(v2, x)
    v2
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v2 = v
    v2 = extend(v2, x)
    v2
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v2 = v
    v2 = push(v2, x)
    v2
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < 0.0.to_float().trunc() {
        None
    } else {
        Some(v)
    }
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a.to_float().trunc() < b.to_float().trunc() {
        v = push(v, i)
    }
    v
}
