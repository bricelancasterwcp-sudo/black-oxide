fn cutoff(v: &Vec<Int>, cutoff: Int) -> Unit {
    let mut under = 0
    let mut over = 0
    for x in &v {
        if x < &cutoff {
            under = under + 1
        } else {
            over = over + 1
        }
    }
    print(under)
    print(over)
}

fn main() {
    let nums = vec(5, 12, 3, 18, 9)
    cutoff(&nums, 10)
    drop nums
}
