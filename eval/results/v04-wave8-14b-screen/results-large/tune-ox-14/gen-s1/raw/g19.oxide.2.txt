fn main() {
    let text = "order 47 shipped on 2026-03-15, invoice 1290 paid."
    print(str_len(text))
    let letters = count_if(text, |c| is_alpha(c))
    print(letters)
    let digits = count_if(text, |c| is_digit(c))
    print(digits)
    let spaces = count_if(text, |c| c == " ")
    print(spaces)
    let others = str_len(text) - letters - digits - spaces
    print(others)
    let max_digit_streak = 0
    let mut current_streak = 0
    for c in chars(text) {
        if is_digit(c) {
            current_streak += 1
        } else {
            if current_streak > max_digit_streak {
                max_digit_streak = current_streak
            }
            current_streak = 0
        }
    }
    if current_streak > max_digit_streak {
        max_digit_streak = current_streak
    }
    print(max_digit_streak)
    let mut digit_str = ""
    for c in chars(text) {
        if is_digit(c) {
            digit_str = concat(digit_str, c)
        }
    }
    print(digit_str)
}
