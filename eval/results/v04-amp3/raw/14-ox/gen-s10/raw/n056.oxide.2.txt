struct Reading { label: Str, values: Vec<Option<Int>> }

fn sum_valid(v: Vec<Option<Int>>) -> Int {
    let mut sum = 0
    for x in v {
        if let Some(value) = x {
            sum += value
        }
    }
    sum
}

fn count_invalid(v: Vec<Option<Int>>) -> Int {
    let mut count = 0
    for x in v {
        if x == None {
            count += 1
        }
    }
    count
}

fn main() {
    let r = Reading { label: "data", values: vec![Some(7), None, Some(9), None] }
    print(sum_valid(r.values.clone()))
    print(count_invalid(r.values))
}
