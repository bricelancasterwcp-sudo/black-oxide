```rust
fn main() {
    let mut count = 0;
    let mut sum = 0;

    for &value in [7, 10, 9, 42] { // Replace ? with a valid number
        if value.is_digit(10) {
            sum += value;
            count += 1;
        }
    }

    println!("Sum of the ones that do: {}", sum);
    println!("Number of ones that do not: {}", 4 - count);
}
```
