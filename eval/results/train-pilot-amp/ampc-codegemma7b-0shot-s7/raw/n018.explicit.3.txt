    for x in &numbers {

fn vec(x: Int, y: Int, z: Int, w: Int) -> Vec<Int> {
    let mut v = Vec(x)
    push(v, y)
    push(v, z)
    push(v, w)
    v
}

fn push(mut: Vec<Int>, x: Int) -> Vec<Int> {
    mut.append(x)
    mut
}

fn print(x: Int) {
    print_str(to_str(x))
    print_str("\n")
}

fn to_str(x: Int) -> Str {
    str_from_chars(chars(int_to_str(x)))
}

fn str_from_chars(x: Vec<Str>) -> Str {
    let mut s = ""
    for c in x {
        s = concat(s, c)
    }
    s
}

fn print_str(x: Str) {
    let mut c = 0
    for b in chars(x) {
        print_char(b)
        c = c + 1
    }
}

fn print_char(x: Str) {
    if x == "0" {
        print_str("0")
    } else if x == "1" {
        print_str("1")
    } else if x == "2" {
        print_str("2")
    } else {
        print_str(".")
    }
}

fn chars(s: Str) -> Vec<Str> {
    let mut v = Vec("")
    let mut i = 0
    while i < len(s) {
        let c = get(s, i)
        if c != None {
            v.append(c)
            i = i + 1
        } else {
            break
        }
    }
    v
}

fn get(s: Str, i: Int) -> Option<Str> {
    if i >= 0 && i < len(s) {
        Some(str_from_chars(vec(to_str(i))))
    } else {
        None
    }
}

fn len(s: Str) -> Int {
    let mut c = 0
    for _ in chars(s) {
        c = c + 1
    }
    c
}

fn to_int(s: Str) -> Int {
    let mut i = 0
    let mut n = 0
    while i < len(s) {
        let c = get(s, i)
        if c != None {
            let d = to_digit(c)
            if d >= 0 && d <= 9 {
                n = n * 10 + d
            } else {
                return -1
            }
            i = i + 1
        } else {
            break
        }
    }
    n
}

fn to_digit(s: Option<Str>) -> Int {
    if s == None {
        return -1
    }
    let c = get(s, 0)
    if c == None {
        return -1
    }
    match c {
        Some(x) => {
            if x == "0" {
                return 0
            } else if x == "1" {
                return 1
            } else if x == "2" {
                return 2
            } else {
                return -1
            }
        },
        None => return0,
    }
}

fn int_to_str(n: Int) -> Str {
    if n < 0 {
        return str_from_chars(vec(to_str(-1)))
    } else {
        let mut digits = Vec("")
        let mut n = n
        while n > 0 {
            let r = n % 10
            digits.append(to_str(r))
            n = n / 10
        }
        if digits.len() == 0 {
            return str_from_chars(vec(to_str(0)))
        } else {
            let mut result = ""
            for i in 0.to_int(digits.len()) {
                result = concat(result, digits.get(i))
            }
            return result
        }
    }
}

fn concat(a: Str, b: Str) -> Str {
    let mut result = ""
    for c in chars(a) {
        result = concat(result, c)
    }
    for c in chars(b) {
        result = concat(result, c)
    }
    result
}
