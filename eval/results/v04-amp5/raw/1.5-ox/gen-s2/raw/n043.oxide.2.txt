struct Queue {
    items: Vec<Int>,
}

impl Queue {
    fn new(items: Vec<Int>) -> Self {
        Queue { items }
    }

    fn swap_first_last(&mut self) {
        let first = self.items.remove(0);
        let last = self.items.pop().unwrap();
        self.items.insert(0, last);
        self.items.push(first);
    }

    fn print_queue(&self) {
        for item in &self.items {
            println!("{}", item);
        }
    }
}

fn main() {
    let mut queue = Queue::new(vec![1, 2, 3, 4, 5]);
    queue.swap_first_last();
    queue.print_queue();
}
