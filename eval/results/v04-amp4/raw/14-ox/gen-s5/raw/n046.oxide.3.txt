struct Reading { label: Str, values: Vec<Int> }

fn count_under_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let count = 0
    for x in v {
        if x < cutoff {
            count += 1
        }
    }
    count
}

fn count_over_or_at_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let count = 0
    for x in v {
        if x >= cutoff {
            count += 1
        }
    }
    count
}

fn main() {
    let values = vec(5, 12, 3, 18, 9)
    let cutoff = 10
    print(count_under_cutoff(values.clone(), cutoff))
    print(count_over_or_at_cutoff(values, cutoff))
}
