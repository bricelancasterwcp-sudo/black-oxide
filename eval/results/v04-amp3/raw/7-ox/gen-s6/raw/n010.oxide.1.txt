fn is_prime(n: Int) -> Bool {
    let mut i = 2;
    while i * i <= n {
        if n % i == 0 {
            return false;
        }
        i += 1;
    }
    true
}

fn main() {
    let num = 91;
    if is_prime(num) {
        print_str("yes");
    } else {
        print_str("no");
    }
}
