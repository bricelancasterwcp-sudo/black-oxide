```rust
fn sum_digits(input: Str) -> Int {
    let mut total = 0;
    for ch in input.chars() {
        if ch.is_digit(10) {
            total += ch.to_digit(10).unwrap();
        }
    }
    total
}

fn count_digits(input: Str) -> Int {
    input.chars().filter(|&ch| ch.is_digit(10)).count()
}

fn main() {
    let input = "a1b22";
    println!("Sum of digits: {}", sum_digits(input));
    println!("Count of digits: {}", count_digits(input));
}
```
