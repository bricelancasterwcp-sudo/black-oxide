use std::io;

fn main() {
    let mut total_ones = 0;
    let mut total_not_ones = 0;

    for _ in 0..4 {
        let mut input = String::new();
        io::stdin().read_line(&mut input).unwrap();

        let input = input.trim();

        if input == "7" || input == "9" {
            total_ones += 1;
        } else if input == "?" {
            total_not_ones += 1;
        }
    }

    println!("{} ones", total_ones);
    println!("{} not ones", total_not_ones);
}
