fn main() {
    let mut list = vec_int()
    for x in range(1, 11) {
        if x % 2 == 0 {
            list = push(list, x)
        }
    }
    let mut index = len(&list) - 1
    while index >= 0 {
        print(get(&list, index))
        drop index
        index = index - 1
    }
    drop list
}
