fn main() {
    let mut queue = vec![1, 2, 3, 4, 5];
    if let Some((first, last)) = queue.split_first_mut().and_then(|(first, rest)| rest.split_last_mut().map(|(last, _)| (first, last))) {
        std::mem::swap(first, last);
    }
    for item in &queue {
        println!("{}", item);
    }
}
