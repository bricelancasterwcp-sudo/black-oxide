fn letter_count(string: &Str) -> Str {
    let vowels = 0
    let consonants = 0
    for c in chars(&string) {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            vowels = vowels + 1
        } else {
            consonants = consonants + 1
        }
    }
    format("{} vowels, {} consonants", vowels, consonants)
}

fn main() {
    print(letter_count("oxide"))
}

fn format(s: &Str) -> Str {
    let mut result = ""
    for c in chars(&s) {
        result = result + c
    }
    result
}

fn chars(s: &Str) -> Vec<Str> {
    let mut v = vec()
    for c in s {
        v = push(v, c)
    }
    v
}

fn push(mut_v: Vec<Str>, x: Str) -> Vec<Str> {
    mut_v.push(x)
    mut_v
}

fn vec() -> Vec<Str> {
    vec()
}

fn main() {
    let x = "oxide"
    print(letter_count(x))
}
