```rust
fn main() {
    let samples = [5, 12, 3, 18, 9];
    let cutoff = 10;
    let (under, over) = samples.iter().fold((0, 0), |(under, over), &x| {
        if x < cutoff {
            (under + 1, over)
        } else {
            (under, over + 1)
        }
    });
    println!("Under: {}", under);
    println!("Over or equal to: {}", over);
}
```
