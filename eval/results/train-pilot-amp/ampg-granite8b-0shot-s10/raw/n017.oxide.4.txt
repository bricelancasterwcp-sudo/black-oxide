struct List { value: Int, next: Option<Box<List>> }

fn get_second(l: Box<List>) -> Option<Int> {
    let second = match l.next {
        Some(mutRest) => match Boxeref(mutRest).next {
            Some(n) => Some(n.value),
            None => None,
        },
        None => None,
    }
    Some(l.value)
}

fn main() {
    let list = Box(List { value: 10, next: Some(Box(List { value: 20, next: Some(Box(List { value: 30, next: Some(Box()) })) })) })
    match get_second(list) {
        Some(n) => print(n),
        None => print_str("no second"),
    }
}
