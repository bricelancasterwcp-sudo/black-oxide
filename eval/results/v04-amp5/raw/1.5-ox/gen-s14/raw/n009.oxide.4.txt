fn reverse(v: Vec<Int>) -> Vec<Int> {
    let mut result = vec![];
    for i in (0..v.len()).rev() {
        result.push(v[i]);
    }
    return result;
}

fn main() {
    let n = 1234;
    let digits = format!("{n}");
    let reversed_digits: Vec<Int> = reverse(digits.chars().map(|c| c.to_digit(10).unwrap())?.collect());
    println!("{reversed_digits:?}");
}
