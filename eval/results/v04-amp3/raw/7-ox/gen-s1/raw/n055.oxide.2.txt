fn process_entry(entry: Str) -> Option<Int> {
    parse_int(entry)?
}

fn main() {
    let entries = vec!["8", "oak", "15"];
    for entry in entries.iter() {
        match process_entry(*entry) {
            Some(num) => print(num),
            None => print_str("bad"),
        }
        print_str(", ")
    }
}
