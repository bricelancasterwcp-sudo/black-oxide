fn main() {
    let numbers = [12, 5, 20, 8, 17];
    let mut largest = numbers[0];
    let mut second_largest = numbers[0];
    for &num in &numbers {
        if num > largest {
            second_largest = largest;
            largest = num;
        } else if num > second_largest {
            second_largest = num;
        }
    }
    println!("{}", largest);
    println!("{}", second_largest);
}
