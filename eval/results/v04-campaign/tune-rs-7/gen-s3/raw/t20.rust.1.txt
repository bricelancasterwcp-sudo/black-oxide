fn main() {
    let mut total_cost = 0;
    let items = vec![
        ("pen", 3, 4),
        ("pad", 9, 2),
        ("ink", 5, 3),
    ];
    for item in &items {
        let cost = item.1 * item.2;
        total_cost += cost;
        println!("{}", cost);
    }
    println!("{}", items.iter().max_by_key(|item| item.1 * item.2).unwrap().0);
}
