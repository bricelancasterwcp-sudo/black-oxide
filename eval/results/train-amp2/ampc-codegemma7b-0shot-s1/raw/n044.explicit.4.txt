fn deposits(mut_balance: &Int, deposits: &Vec<Int>) -> Int {
    for x in deposits {
        if x == mut_balance {
            mut_balance = mut_balance + x
        } else {
            mut_balance = mut_balance - x
        }
    }
    mut_balance.clone()
}

}
