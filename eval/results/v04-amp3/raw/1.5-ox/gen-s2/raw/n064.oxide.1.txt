fn main() {
    let list = vec![10, 20];

    if let Some(pos1) = get(list, 0) {
        println!("{}", pos1);
    } else {
        println!("Position 1 not found");
    }

    if let Some(pos4) = get(list, 3) {
        println!("{}", pos4);
    } else {
        println!("Position 4 not found");
    }
}

fn get<T>(vec: Vec<T>, index: Int) -> Option<T> {
    match vec.get(index as usize) {
        Some(val) => Some(*val),
        None => None,
    }
}
