fn main() {
    let mut count = 0;
    for num in 2..100 {
        if is_prime(num) {
            count += 1;
        }
    }
    println!("{}", count);
}
fn is_prime(num: u32) -> bool {
    if num <= 1 {
        return false;
    }
    for i in 2..num {
        if num % i == 0 {
            return false;
        }
    }
    true
}
