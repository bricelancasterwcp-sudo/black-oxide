struct Sitting { label: Str, scores: Vec<Int> }
fn count_range(v: Vec<Int>, min: Int, max: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= min && x <= max {
            count += 1
        }
    }
    count
}
fn main() {
    let morning = Sitting { label: "Morning", scores: vec(94, 71, 88, 65, 92, 79, 83) }
    let evening = Sitting { label: "Evening", scores: vec(55, 90, 77, 84, 68, 99) }
    for sitting in vec(morning.clone(), evening.clone()) {
        print_str(sitting.label)
        let above_90 = count_range(sitting.scores.clone(), 90, 100)
        let between_80_and_89 = count_range(sitting.scores.clone(), 80, 89)
        let between_70_and_79 = count_range(sitting.scores.clone(), 70, 79)
        let below_70 = count_range(sitting.scores.clone(), 0, 69)
        let max_score = match max(sitting.scores) {
            Some(n) => n,
            None => -1,
        }
        let mean_score = if len(sitting.scores) > 0 {
            sum(sitting.scores) / len(sitting.scores)
        } else {
            -1
        }
        print(above_90)
        print(between_80_and_89)
        print(between_70_and_79)
        print(below_70)
        print(max_score)
        print(mean_score)
    }
    let morning_mean = if len(morning.scores) > 0 {
        sum(morning.scores) / len(morning.scores)
    } else {
        -1
    }
    let evening_mean = if len(evening.scores) > 0 {
        sum(evening.scores) / len(evening.scores)
    } else {
        -1
    }
    if morning_mean >= evening_mean {
        print_str("Morning")
    } else {
        print_str("Evening")
    }
}
