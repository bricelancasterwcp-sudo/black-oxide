fn main() {
    let word = "drum"
    print_str(word)
    let letters = reverse_letters(word)
    print_str(letters)
}

fn reverse_letters(word: Str) -> Str {
    let mut result = String.new()
    for i in range(len(word), 0) {
        result = concat(result, get(word, i - 1).unwrap())
    }
    result
}

fn reverse_letters_mut(word: Str) -> Str {
    let mut result = String.new()
    for i in range(len(word), 0) {
        result = concat(result, get_mut(word, i - 1).unwrap())
    }
    result
}

fn reverse_letters_iter(word: Str) -> Str {
    let mut result = String.new()
    for ch in word.chars() {
        result = concat(result, ch)
    }
    result
}

fn reverse_letters_str_iter(word: Str) -> Str {
    let mut result = String.new()
    for ch in word {
        result = concat(result, ch)
    }
    result
}

fn reverse_letters_bytes(word: Str) -> Str {
    let mut result = String.new()
    for b in word.bytes() {
        result = concat(result, String.from_utf8_lossy(b.as_ref()))
    }
    result
}
