fn main() {
    let north = vec![12, 15, 19, 14, 14, 20, 31, 30];
    let south = vec![40, 38, 41, 45, 50, 44];
    fn analyze(sensor: &Vec<i32>, name: &str) {
        let mut increases = 0;
        let mut decreases = 0;
        let mut max_increase = 0;
        let mut current_streak = 0;
        let mut longest_streak = 0;
        for i in 1..sensor.len() {
            if sensor[i] > sensor[i - 1] {
                increases += 1;
                max_increase = max_increase.max(sensor[i] - sensor[i - 1]);
                current_streak += 1;
                longest_streak = longest_streak.max(current_streak);
            } else if sensor[i] < sensor[i - 1] {
                decreases += 1;
                current_streak = 0;
            }
        }
        println!("{}", name);
        println!("{}", increases);
        println!("{}", decreases);
        println!("{}", max_increase);
        println!("{}", longest_streak);
    }
    analyze(&north, "north");
    analyze(&south, "south");
    if north.len() > south.len() {
        println!("north");
    } else {
        println!("south");
    }
}
