use std::io::{stdin, Read};

fn main() {
    let mut buf = String::new();
    stdin().read_to_string(&mut buf).unwrap();

    let mut vowels = 0;
    let mut not_vowels = 0;

    for c in buf.chars() {
        match c {
            'a' | 'e' | 'i' | 'o' | 'u' => vowels += 1,
            _ => not_vowels += 1,
        }
    }

    println!("Vowels: {}", vowels);
    println!("Not vowels: {}", not_vowels);
}
