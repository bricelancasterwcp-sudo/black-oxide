fn digit_sum(s: &Str) -> Int {
    let mut sum = 0
    let mut count = 0
    for c in s.chars() {
        if c.is_digit(10) {
            sum = sum + c.to_digit(10).unwrap()
            count = count + 1
        }
    }
    sum
}

fn digit_count(s: &Str) -> Int {
    let mut count = 0
    for c in s.chars() {
        if c.is_digit(10) {
            count = count + 1
        }
    }
    count
}

fn main() {
    let text = "a1b22"
    print(digit_sum(&text))
    print(digit_count(&text))
}
