struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 8), 3), 11), 6) }
    let min_val = min(r.values)?.unwrap_or(-1)
    let max_val = max(r.values)?.unwrap_or(-1)
    let gap = max_val - min_val
    print(min_val)
    print(gap)
}
