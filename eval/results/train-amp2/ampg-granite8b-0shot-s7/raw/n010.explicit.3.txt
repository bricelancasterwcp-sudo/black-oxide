fn is_prime(number: Int) -> Bool {
    if number <= 1 {
        return false
    }
    for divisor in 2.to(number) {
        if number % divisor == 0 {
            return false
        }
    }
    true
}

fn main() {
    let number = 91
    if is_prime(number) {
        print_str("yes\n")
    } else {
        print_str("no\n")
    }
}
