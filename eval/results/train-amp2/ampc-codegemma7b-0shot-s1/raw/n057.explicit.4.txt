fn compare(x: &Str, y: &Str) -> Int {
    if x.len() > y.len() {
        1
    } else if x.len() < y.len() {
        -1
    } else {
        0
    }
}

fn main() {
    let fig = "fig".to_string()
    let banana = "banana".to_string()
    let kiwi = "kiwi".to_string()
    let largest = compare(&fig, &banana)
    if largest > 0 {
        print(fig)
        print(fig.len())
    } else if largest < 0 {
        print(banana)
        print(banana.len())
    } else {
        print(kiwi)
        print(kiwi.len())
    }
    drop fig
    drop banana
    drop kiwi
}

fn find_longest(x: &Str, y: &Str, z: &Str) -> String {
    if x.len() > y.len() && x.len() > z.len() {
        x.to_string()
    } else if y.len() > z.len() {
        y.to_string()
    } else {
        z.to_string()
    }
}

fn main() {
    let fig = "fig".to_string()
    let banana = "banana".to_string()
    let kiwi = "kiwi".to_string()
    let longest = find_longest(&fig, &banana, &kiwi)
    print(longest)
    print(longest.len())
    drop fig
    drop banana
    drop kiwi
}
