```rust
struct Queue<T> {
    items: Vec<T>,
}

impl<T> Queue<T> {
    fn new() -> Self {
        Queue { items: vec![] }
    }

    fn push(&mut self, item: T) {
        self.items.push(item);
    }

    fn pop(&mut self) -> Option<T> {
        self.items.pop()
    }

    fn exchange_first_last(&mut self) {
        if let Some(first) = self.pop() {
            if let Some(last) = self.pop() {
                self.push(first);
                self.push(last);
            } else {
                panic!("Queue underflow");
            }
        } else {
            panic!("Queue underflow");
        }
    }
}

fn main() {
    let mut queue = Queue::new();
    queue.push(1);
    queue.push(2);
    queue.push(3);
    queue.push(4);
    queue.push(5);

    queue.exchange_first_last();

    for item in queue.items {
        println!("{}", item);
    }
}
```
