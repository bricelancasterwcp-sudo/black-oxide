fn sort(scores: &Vec<Int>) -> Vec<Int> {
    let mut less = vec()
    let mut greater = vec()
    for x in &scores {
        if x < 5 {
            less = push(less, x)
        } else {
            greater = push(greater, x)
        }
    }
    let mut result = vec()
    let mut left = 0
    let mut right = 0
    while left < len(&less) && right < len(&greater) {
        if get(&less, left) < get(&greater, right) {
            result = push(result, get(&less, left))
            left = left + 1
        } else {
            result = push(result, get(&greater, right))
            right = right + 1
        }
    }
    while left < len(&less) {
        result = push(result, get(&less, left))
        left = left + 1
    }
    while right < len(&greater) {
        result = push(result, get(&greater, right))
        right = right + 1
    }
    result
}

fn str_split(s: Str, sep: Str) -> Vec<Str> {
    let mut result = vec()
    let i = chars(s)
    let j = chars(sep)
    let n = len(&i)
    let m = len(&j)
    let mut k = 0
    while k < n {
        let l = match_prefix(&i, &j, k)
        if l == -1 {
            result = push(result, get(i, k))
            k = k + 1
        } else {
            k = k + l
        }
    }
    result
}

fn match_prefix(i: &Vec<Str>, j: &Vec<Str>, k: Int) -> Int {
    let n = len(i)
    let m = len(j)
    let mut l = 0
    while l < m && k + l < n {
        if get(i, k + l) != get(j, l) {
            break
        }
        l = l + 1
    }
    if l == m {
        l
    } else {
        -1
    }
}

fn int_to_str(n: Int) -> Str {
    let mut s = vec()
    let mut x = n
    if x < 0 {
        s = push(s, "-")
        x = -x
    }
    if x < 10 {
        s = push(s, str_from_digit(x))
    } else {
        let q = x / 10
        let r = x % 10
        s = push(s, int_to_str(q))
        s = push(s, str_from_digit(r))
    }
    s
}
