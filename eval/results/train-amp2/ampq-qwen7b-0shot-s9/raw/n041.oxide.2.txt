struct Reading { label: Str, values: Vec<Int> }

fn sort(values: Vec<Int>) -> Vec<Int> {
    let mut sorted = values.clone()
    sorted.sort()
    sorted
}

fn main() {
    let r = Reading { label: "measurements", values: vec(9, 2, 7, 4) }
    let sorted = sort(r.values.clone())
    for x in sorted {
        print(x)
    }
}
