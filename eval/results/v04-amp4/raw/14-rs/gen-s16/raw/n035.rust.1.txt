fn main() {
    fn divide(a: i32, b: i32) -> Result<i32, &'static str> {
        if b == 0 {
            Err("bad")
        } else {
            Ok(a / b)
        }
    }

    match divide(20, 4) {
        Ok(result) => println!("{}", result),
        Err(e) => println!("{}", e),
    }
}
