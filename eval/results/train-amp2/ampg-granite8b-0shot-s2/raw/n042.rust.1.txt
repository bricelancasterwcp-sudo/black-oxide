fn main() {
    let v = vec![12, 5, 20, 8, 17];
    let mut largest = v[0];
    let mut second_largest = v[0];

    for i in v {
        if i > largest {
            second_largest = largest;
            largest = i;
        } else if i > second_largest {
            second_largest = i;
        }
    }

    println!("{}", largest);
    println!("{}", second_largest);
}
