fn fib(n: Int) -> Option<Int> {
    let mut a = 0;
    let mut b = 1;

    for _ in 2..=n { // Fixed: replaced '2' with 'n' to match the range up to n
        let temp = a + b;
        a = b;
        b = temp;
    }

    Some(b)
}

fn main() {
    match fib(10) {
        Some(n) => print(n),
        None => println_str("fib failed"),
    }
}
