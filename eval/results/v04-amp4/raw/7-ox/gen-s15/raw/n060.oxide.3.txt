fn main() {
    let s = "a1b22";
    let mut sum = 0;
    let mut count = 0;

    for c in chars(s) {
        if let Some(_) = parse_int(c) {
            if let Some(n) = parse_int(c) {
                sum += n;
                count += 1;
            }
        }
    }

    print(sum);
    print(count);
}
