    let pen = (3, 4)
    let pad = (9, 2)
    let ink = (5, 3)
    let total = pen.0 * pen.1 + pad.0 * pad.1 + ink.0 * ink.1
    print(total)
    if pen.0 * pen.1 > pad.0 * pad.1 && pen.0 * pen.1 > ink.0 * ink.1 {
        print_str("pen")
    } else if pad.0 * pad.1 > ink.0 * ink.1 {
        print_str("pad")
    } else {
        print_str("ink")
    }
}
