```rust
use std::collections::HashSet;

fn main() {
    let vowels: HashSet<char> = ['a', 'e', 'i', 'o', 'u'].iter().cloned().collect();
    let mut count = 0;
    for c in std::io::stdin().read_line().unwrap().chars() {
        if vowels.contains(&c.to_lowercase()) {
            count += 1;
        }
    }
    println!("The number of vowels is {}", count);
}
```
