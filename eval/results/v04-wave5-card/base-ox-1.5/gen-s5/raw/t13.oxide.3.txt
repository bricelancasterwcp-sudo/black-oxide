fn main() {
    for word in ["level", "stereo", "rotor"] {
        println!("{}", is_palindrome(word));
    }
}

fn is_palindrome(s: &str) -> bool {
    let mut left = 0;
    let mut right = s.len() - 1;

    while left < right {
        if s[left] != s[right] {
            return false;
        }
        left += 1;
        right -= 1;
    }

    true
}
