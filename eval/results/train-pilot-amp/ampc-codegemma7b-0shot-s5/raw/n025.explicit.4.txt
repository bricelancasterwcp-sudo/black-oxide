fn read_int() -> Option<Int> {
    let line = str_len(input())
    if line == 0 {
        return None
    }
    let mut chars = chars(line)
    let mut result = 0
    for c in chars {
        result = result * 10 + int_to_str(int_to_str(c).to_float().trunc())
    }
    Some(result)
}

        Some(x) => print(x + 1),

    }
}
