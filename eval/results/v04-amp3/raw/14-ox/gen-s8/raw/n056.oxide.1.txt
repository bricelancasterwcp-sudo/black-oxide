fn main() {
    let nums = vec("7", "x", "9", "?");
    let mut total = 0;
    let mut non_nums = 0;

    for s in nums {
        match parse_int(s) {
            Some(n) => total = total + n,
            None => non_nums = non_nums + 1,
        }
    }

    print(total);
    print(non_nums);
}
