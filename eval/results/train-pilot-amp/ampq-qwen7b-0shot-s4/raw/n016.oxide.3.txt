struct NumberList { numbers: Vec<Int> }

fn main() {
    let mut list = NumberList { numbers: vec() }
    for i in range(1, 11) {
        if i % 3 == 0 {
            list.numbers = push(list.numbers, i)
        }
    }
    let total: Int = sum(list.numbers)
    print(total)
}
