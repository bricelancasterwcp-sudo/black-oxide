struct Queue { entries: Vec<Int> }

fn main() {
    let mut q = Queue { entries: vec(1, 2, 3, 4, 5) }
    let first = q.entries.get(0)?.unwrap() + 1000
    let last = q.entries.get(q.entries.len() - 1)?.unwrap() + 1000
    q.entries = vec(last, q.entries.get(1)?.unwrap(), q.entries.get(2)?.unwrap(), q.entries.get(3)?.unwrap(), first)
    for e in q.entries {
        print(e)
    }
}
