fn main() {
    let mut longest_name = "";
    let mut max_length = 0;

    for name in ["fig", "banana", "kiwi"] {
        if name.len() > max_length {
            longest_name = name;
            max_length = name.len();
        }
    }

    println!("Longest name: {}", longest_name);
    println!("Letter count: {}", max_length);
}
