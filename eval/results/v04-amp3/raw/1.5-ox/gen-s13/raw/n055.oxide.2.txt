fn main() {
    let mut v = Vec::new();
    read(&mut v);
    if let Some(n) = pop(&v) {
        println!("{}", n);
    } else {
        println!("bad");
    }
    read(&mut v);
    if let Some(n) = pop(&v) {
        println!("{}", n);
    } else {
        println!("bad");
    }
}

fn read(mut v: &mut Vec<String>) {
    for line in std::io::stdin().lines() {
        let s = line.unwrap();
        if s == "8" { push(&mut v, s.to_string()) }
        else if s == "oak" { push(&mut v, s.to_string()) }
        else if s == "15" { push(&mut v, s.to_string()) }
    }
}

fn pop(v: &Vec<String>) -> Option<String> {
    v.pop()
}
