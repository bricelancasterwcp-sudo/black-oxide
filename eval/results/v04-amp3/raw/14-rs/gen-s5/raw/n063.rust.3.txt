use std::io;

fn main() {
    let line1_units = 4;
    let line1_price = 3;
    let line1_value = line1_units * line1_price;

    let line2_units = 5;
    let line2_price = 2;
    let line2_value = line2_units * line2_price;

    let total_value = line1_value + line2_value;

    println!("Line 1 value: {}", line1_value);
    println!("Line 2 value: {}", line2_value);
    println!("Total value: {}", total_value);
}
