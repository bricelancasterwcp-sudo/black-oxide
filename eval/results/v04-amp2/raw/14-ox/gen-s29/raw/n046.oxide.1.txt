fn count_under(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x < cutoff {
            count += 1
        }
    }
    count
}

fn count_over_or_equal(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= cutoff {
            count += 1
        }
    }
    count
}

fn main() {
    let values = vec()
    let values = push(push(push(push(push(values, 5), 12), 3), 18), 9)
    let cutoff = 10
    let under_count = count_under(values, cutoff)
    let over_count = count_over_or_equal(values, cutoff)
    print(under_count)
    print(over_count)
}
