fn main() {
    let input = vec_str("8", "oak", "15")
    for entry in input {
        match entry {
            Some(num) => print_int(num),
            None => print_str("bad"),
        }
    }
}

    vec(Some(to_int(s1)), None, Some(to_int(s3)))
}
