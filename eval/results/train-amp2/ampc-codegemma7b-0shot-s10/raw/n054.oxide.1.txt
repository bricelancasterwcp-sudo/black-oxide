fn is_vowel(c: Str) -> Bool {
    match c {
        Str(s) => match s {
            _ => false,
            _ => true,
        },
        _ => false,
    }
}

fn main() {
    let word = "keyboard"
    let mut vowels = 0
    let mut consonants = 0
    for c in word.chars() {
        if is_vowel(Str(c)) {
            vowels = vowels + 1
        } else {
            consonants = consonants + 1
        }
    }
    print(vowels)
    print(consonants)
}
