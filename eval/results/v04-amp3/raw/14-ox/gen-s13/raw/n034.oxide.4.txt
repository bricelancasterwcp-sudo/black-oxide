fn second_entry(v: Vec<Int>) -> Option<Int> { get(v, 1)? }

fn main() {
    let list1 = push(push(vec(), 7), 8);
    let list2 = push(push(vec(), 7), 9);

    match second_entry(list1) {
        Some(n) => print(n),
        None => print_str("none"),
    }

    match second_entry(list2) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
