fn count_and_sum_digits(s: Str) -> (Int, Int) {
    let sum = 0;
    let count = 0;
    for c in chars(s) {
        if parse_int(c).is_some() {
            sum = sum + parse_int(c).unwrap();
            count = count + 1;
        }
    }
    (sum, count)
}

fn main() {
    let (sum, count) = count_and_sum_digits("a1b22");
    print(sum);
    print(count);
}
