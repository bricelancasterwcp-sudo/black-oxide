fn main() {
    let numbers = vec("7", "x", "9", "?");
    let mut sum = 0;
    let mut non_numbers = 0;

    for n in numbers {
        match parse_int(n) {
            Some(num) => sum += num,
            None => non_numbers += 1,
        }
    }

    print(sum);
    print(non_numbers);
}
