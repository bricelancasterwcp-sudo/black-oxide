fn main() {
    let stock = push(push(push(push(vec(), 4), 4), 4), 5)
    let units = 4
    let total = sum_big(&stock, units)
    print(total)
    drop total
    drop stock
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
