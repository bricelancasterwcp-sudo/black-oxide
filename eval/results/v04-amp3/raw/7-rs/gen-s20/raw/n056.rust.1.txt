```rust
fn main() {
    let mut sum = 0;
    let mut count_non_numbers = 0;

    for &item in &[7, 'x', 9, '?'] {
        match item {
            num @ 0..=9 => sum += num,
            _ => count_non_numbers += 1,
        }
    }

    println!("Total of numbers: {}", sum);
    println!("Count of non-numbers: {}", count_non_numbers);
}
```
