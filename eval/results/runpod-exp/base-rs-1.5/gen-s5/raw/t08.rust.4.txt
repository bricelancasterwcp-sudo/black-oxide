```rust
fn main() {
    let numbers = [3, 8, -2, 12, 7];
    let mut count_positive = 0;
    let mut max_value = numbers[0];

    for &num in &numbers {
        if num > 0 {
            count_positive += 1;
        }
        if num > max_value {
            max_value = num;
        }
    }

    println!("Number of positive values: {}", count_positive);
    println!("Largest value: {}", max_value);
}
```
