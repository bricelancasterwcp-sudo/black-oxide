fn main() {
    let text = "order 47 shipped on 2026-03-15, invoice 1290 paid."
    let char_count = str_len(text)
    let letter_count = count_if(text, |c| is_alpha(c))
    let digit_count = count_if(text, |c| is_digit(c))
    let space_count = count_if(text, |c| c == ' ')
    let other_count = char_count - (letter_count + digit_count + space_count)
    let max_digit_streak = max_digit_streak(text)
    let digits = collect(filter_chars(text, is_digit), "")
    print(char_count)
    print(letter_count)
    print(digit_count)
    print(space_count)
    print(other_count)
    print(max_digit_streak)
    print(digits)
}
fn count_if(s: Str, pred: Fn(Char) -> Bool) -> Int {
    let mut count = 0
    for c in chars(s) {
        if pred(c) {
            count += 1
        }
    }
    count
}
fn max_digit_streak(s: Str) -> Int {
    let mut max_streak = 0
    let mut current_streak = 0
    for c in chars(s) {
        if is_digit(c) {
            current_streak += 1
        } else {
            if current_streak > max_streak {
                max_streak = current_streak
            }
            current_streak = 0
        }
    }
    if current_streak > max_streak {
        max_streak = current_streak
    }
    max_streak
}
fn filter_chars(s: Str, pred: Fn(Char) -> Bool) -> Vec<Char> {
    let mut result = vec()
    for c in chars(s) {
        if pred(c) {
            result = push(result, c)
        }
    }
    result
}
