fn main() {
    let p1 = Point { x: 1, y: 5 }
    let p2 = Point { x: 10, y: p1.y }
    print(p1.x + p2.x)
    drop p1
    drop p2
}

struct Point {
    x: Int,
    y: Int,
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v = v
    v.push(x)
    v
}

fn print(x: Int) {
    let s = int_to_str(x)
    print_str(&s)
}

    let mut chars = chars(s)
    for c in &chars {
        print_char(c)
    }
}

    let mut bytes = chars(c)
    for b in &bytes {
        write_byte(b)
    }
}
