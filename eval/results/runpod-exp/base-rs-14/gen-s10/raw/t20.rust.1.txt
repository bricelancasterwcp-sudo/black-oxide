```rust
fn main() {
    let pen_price = 3;
    let pen_quantity = 4;
    let pad_price = 9;
    let pad_quantity = 2;
    let ink_price = 5;
    let ink_quantity = 3;

    let pen_cost = pen_price * pen_quantity;
    let pad_cost = pad_price * pad_quantity;
    let ink_cost = ink_price * ink_quantity;

    let total_cost = pen_cost + pad_cost + ink_cost;
    println!("Total cost: {}", total_cost);

    let mut highest_cost = pen_cost;
    let mut highest_item = "pen";

    if pad_cost > highest_cost {
        highest_cost = pad_cost;
        highest_item = "pad";
    }

    if ink_cost > highest_cost {
        highest_cost = ink_cost;
        highest_item = "ink";
    }

    println!("Item with the highest cost: {}", highest_item);
}
```
