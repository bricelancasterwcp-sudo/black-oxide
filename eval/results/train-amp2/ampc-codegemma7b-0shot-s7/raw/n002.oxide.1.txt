    n - m * (trunc(n / m))

fn trunc(x: Float) -> Int {
    let x2 = x.to_int_lossy()
    if x - x2 >= 0.5 {
        x2 + 1.to_int_lossy()
    } else {
        x2
    }
}

fn len(v: Vec<Int>) -> Int {
    let mut result = 0.to_int_lossy()
    for _ in v {
        result = result + 1.to_int_lossy()
    }
    result
}

    if i.to_int_lossy() >= len(v).to_int_lossy() {
        return None
    }
    Some(v.chars().nth(i.to_int_lossy()).unwrap())

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut result = vec()
    for item in v {
        result = push(result, item)
    }
    result = push(result, x)
    result
}

fn fold(v: Vec<Int>, init: Int, f: FnMut) -> Int {
    let mut result = init
    for item in v {
        result = f(result, item)
    }
    result
}
