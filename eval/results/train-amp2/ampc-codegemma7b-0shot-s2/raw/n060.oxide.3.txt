fn is_digit(c: Str) -> Bool {
    c >= "0" && c <= "9"
}

fn main() {
    let input = "a1b22"
    let mut sum = 0
    let mut count = 0
    for c in input.chars() {
        if is_digit(c) {
            sum = sum + c.to_digit(10).unwrap()
            count = count + 1
        }
    }
    print(sum)
    print(count)
}
