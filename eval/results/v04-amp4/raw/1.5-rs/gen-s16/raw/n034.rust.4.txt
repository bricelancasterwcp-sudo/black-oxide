```rust
fn main() {
    println!("{:?}", get_second_entry(&[7, 8, 9]));
    println!("{:?}", get_second_entry(&[123]));
    println!("{:?}", get_second_entry(&[]));
}

fn get_second_entry(list: &[isize]) -> Option<isize> {
    if list.len() > 1 {
        Some(list.get(1).copied())
    } else {
        None
    }
}
```
