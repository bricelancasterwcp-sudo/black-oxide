fn main() {
    match read_entry(8) {
        Some(n) => println!("{}", n),
        None => println!("bad"),
    }
    match read_entry("oak") {
        Some(n) => println!("{}", n),
        None => println!("bad"),
    }
    match read_entry(15) {
        Some(n) => println!("{}", n),
        None => println!("bad"),
    }
}

fn read_entry(x: Int) -> Option<Int> {
    if x == 8 {
        Some(x)
    } else {
        None
    }
}
