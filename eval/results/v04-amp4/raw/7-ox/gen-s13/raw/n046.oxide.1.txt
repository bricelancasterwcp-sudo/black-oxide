fn count_under_cutoff(v: Vec<Int>, cutoff: Int) -> (Int, Int) {
    let mut under = 0;
    let mut over = 0;
    for x in v {
        if x < cutoff {
            under += 1;
        } else {
            over += 1;
        }
    }
    (under, over)
}

fn main() {
    let samples = push(push(push(push(vec(), 5), 12), 3), 18);
    let samples2 = push(samples, 9);
    let (under, over) = count_under_cutoff(samples2, 10);
    print(under);
    print_str(", ");
    print(over);
}
