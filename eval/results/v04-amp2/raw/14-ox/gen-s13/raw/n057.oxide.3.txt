fn main() {
    let fig = "fig"
    let banana = "banana"
    let kiwi = "kiwi"

    let max_name = if str_len(fig) > str_len(banana) && str_len(fig) > str_len(kiwi) {
        fig
    } else if str_len(banana) > str_len(kiwi) {
        banana
    } else {
        kiwi
    }

    print_str(max_name)
    print(str_len(max_name))
}
