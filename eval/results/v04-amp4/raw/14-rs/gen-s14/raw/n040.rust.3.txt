use std::f64::consts::PI;

enum Shape {
    Square { side: f64 },
    Circle { radius: f64 },
}

impl Shape {
    fn area(&self) -> f64 {
        match self {
            Shape::Square { side } => side * side,
            Shape::Circle { radius } => 3.0 * radius * radius,
        }
    }
}

fn main() {
    let square = Shape::Square { side: 4.0 };
    println!("Area of the square: {}", square.area());
}
