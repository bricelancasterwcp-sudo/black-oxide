```rust
struct Square { side: Int }
struct Circle { radius: Float }

fn area(shape: Shape) -> Float {
    match shape {
        Square(side) => to_float(side * side),
        Circle(radius) => 3.0 * radius * radius,
    }
}

enum Shape {
    Square(Int),
    Circle(Float),
}

fn main() {
    let square = Square { side: 4 };
    print(area(Shape::Square(square.side)));
}
```
