fn main() {
    let mut total = 0
    let mut count = 0
    for x in &vec_from_file {
        if x > 7 {
            total = total + x
            count = count + 1
        }
    }
    print(total)
    print(count)
    drop total
    drop count
}

fn vec_from_file() -> Vec<Int> {
    let mut v = vec()
    for line in file("7.txt").iter_lines() {
        let x = parse_int(line).unwrap()
        push(v, x)
    }
    v
}

fn file(name: &Str) -> Vec<Str> {
    let mut v = vec()
    for line in open(name).iter_lines() {
        push(v, line)
    }
    v
}

fn open(name: &Str) -> File {
    File(name)
}
