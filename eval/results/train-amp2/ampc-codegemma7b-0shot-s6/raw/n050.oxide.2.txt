fn arrange(mutnums: Vec<Int>) -> Vec<Int> {
    for i in 0.upto(nums.len() - 1) {
        for j in (i + 1).upto(nums.len()) {
            if nums.get(i).unwrap_or_default() < nums.get(j).unwrap_or_default() {
                let tmp = nums.get(i).unwrap_or_default()
                nums.set(i, nums.get(j).unwrap_or_default())
                nums.set(j, tmp)
            }
        }
    }
    nums
}

fn main() {
    let mut scores = vec(2, 9, 5)
    for score in arrange(scores) {
        print_str(to_string(score))
        print_str("\n")
    }
}

fn to_string(n: Int) -> Str {
    let mut digits = vec()
    let mut number = n
    while number > 0 {
        digits.push(number % 10)
        number = number / 10
    }
    let mut result = vec()
    for digit in digits.iter().rev() {
        result.push(to_char(digit))
    }
    Str(result)
}

fn to_char(n: Int) -> Str {
    if n == 0 {
        return Str("0")
    }
    let mut number = n
    let mut result = vec()
    while number > 0 {
        let digit = number % 10
        result.push(if digit < 10 { to_char(digit) } else { to_char(digit - 10 + 65) })
        number = number / 10
    }
    Str(result)
}

fn up_to(n: Int, upper: Int) -> Vec<Int> {
    let mut result = vec()
    for i in 0.upto(upper) {
        if i <= n {
            result.push(i)
        }
    }
    result
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < 0 || i >= v.len() {
        return None
    }
    Some(v.get(i).unwrap_or_default())
}

fn set(v: Vec<Int>, i: Int, x: Int) -> Vec<Int> {
    if i < 0 || i >= v.len() {
        return v
    }
    let mut result = v
    result.set(i, x)
    result
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut result = v
    result.push(x)
    result
}

fn len(v: Vec<Int>) -> Int {
    v.len()
}

fn print_str(s: Str) {
    for c in s.chars {
        print(c)
    }
}

fn print(i: Int) {
    print_str(to_string(i))
}

fn str_len(s: Str) -> Int {
    s.chars.len()
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = vec()
    for c in s.chars {
        result.push(Str(c))
    }
    result
}

fn parse_int(s: Str) -> Option<Int> {
    let mut number = 0
    let mut multiplier = 1
    for c in s.chars.rev() {
        let digit = if c >= 48 && c <= 57 { c - 48 } else { returnNone }
        number = number + multiplier * digit
        multiplier = multiplier * 10
    }
    Some(number)
}

fn to_float(n: Int) -> Float {
    Float(n)
}

fn trunc(x: Float) -> Int {
    Int(x.value.floor() + 0.5)
}

fn returnNone() -> Option<Int> {
    None
}

fn int_to_str(n: Int) -> Str {
    let mut digits = vec()
    let mut number = n
    while number > 0 {
        digits.push(number % 10)
        number = number / 10
    }
    let mut result = vec()
    for digit in digits.iter().rev() {
        result.push(to_char(digit))
    }
    Str(result)
}

    }
}
