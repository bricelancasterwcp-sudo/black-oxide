fn reverse_num(n: Int) -> Str {
    let mut s = int_to_str(n)
    let mut chars = chars(s)
    let mut reversed_chars = vec()
    for c in chars {
        reversed_chars = push(reversed_chars, c)
    }
    let reversed_str = concat(reversed_chars)
    reversed_str
}

fn main() {
    let num = 1234
    let reversed_num = reverse_num(num)
    print_str(reversed_num)
}
