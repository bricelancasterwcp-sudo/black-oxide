fn main() {
    let signal = read_signal(); // Assume this function reads a signal
    match signal {
        "green" => println!("go"),
        "red" => println!("stop"),
        "yellow" => println!("wait"),
        _ => println!("unknown signal"),
    }
}

fn read_signal() -> String {
    // Simulate reading a signal from some source
    "green".to_string()
}
