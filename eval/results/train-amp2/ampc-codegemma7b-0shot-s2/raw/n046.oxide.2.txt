fn count(v: Vec<Int>, cutoff: Int) -> Vec<Int> {
    let mut under = 0
    let mut over = 0
    for x in v {
        if x < cutoff {
            under = under + 1
        } else {
            over = over + 1
        }
    }
    push(push(vec(), over), under)
}

fn main() {
    let samples = vec(5, 12, 3, 18, 9)
    let cutoff = 10
    let result = count(samples, cutoff)
    print(result)
}

fn print(v: Vec<Int>) {
    for x in v {
        print(x)
    }
}

fn vec(items: Int, rest: Int) -> Vec<Int> {
    let mut v = vec()
    for _ in 0.range(items) {
        v.push(rest)
    }
    v
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut v2 = v
    v2.push(x)
    v2
}

fn len(v: Vec<Int>) -> Int {
    let mut count = 0
    for _ in v {
        count = count + 1
    }
    count
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if len(v) <= i {
        None
    } else {
        Some(v.get(i).unwrap())
    }
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a.range(b) {
        v.push(i)
    }
    v
}

fn clone(x: Int) -> Int {
    x
}

fn str_len(s: Str) -> Int {
    let mut count = 0
    for _ in s.chars() {
        count = count + 1
    }
    count
}

fn concat(a: Str, b: Str) -> Str {
    a + b
}

fn chars(s: Str) -> Vec<Str> {
    let mut v = vec()
    for c in s.chars() {
        v.push(c.to_string())
    }
    v
}

fn int_to_str(n: Int) -> Str {
    n.to_string()
}

fn parse_int(s: Str) -> Option<Int> {
    match s.parse() {
        Ok(n) => Some(n),
        Err(_) => None,
    }
}

fn to_float(n: Int) -> Float {
    n.float()
}

fn trunc(x: Float) -> Int {
    x.trunc()
}

fn print_str(s: Str) {
    print(s)
}

    let samples = vec(5, 12, 3, 18, 9)
    let cutoff = 10
    let result = count(samples, cutoff)
    print(result)
}
