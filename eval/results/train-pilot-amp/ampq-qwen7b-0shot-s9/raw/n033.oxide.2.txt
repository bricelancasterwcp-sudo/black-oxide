struct Signal { color: Str }

fn main() {
    let signal = Signal { color: "green" }
    match signal.color {
        Str(s) => match s {
            Str(_) => print_str("go for green"),
            Str(_) => print_str("wait for yellow"),
            Str(_) => print_str("stop for red"),
        },
        _ => print_str("unknown signal"),
    }
}
