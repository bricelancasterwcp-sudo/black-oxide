fn count_digits(s: Str) -> Int {
    let mut count = 0
    let mut sum = 0
    for ch in chars(s) {
        if parse_int(ch).is_some() {
            count += 1
            sum += parse_int(ch).unwrap()
        }
    }
    (sum, count)
}

fn main() {
    let input = "a1b22"
    let (total, num_digits) = count_digits(input)
    print(total)
    print(num_digits)
}
