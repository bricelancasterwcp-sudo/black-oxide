fn reverse_digits(n: Int) -> Str {
    let digits = Vec<Int>()
    let mut num = n
    while num > 0 {
        let digit = num % 10
        digits = push(digits, digit)
        num = num / 10
    }
    let result = Vec<Str>()
    for digit in digits {
        result = push(result, int_to_str(digit))
    }
    concat(result)
}

fn main() {
    print_str(reverse_digits(1234))
}
