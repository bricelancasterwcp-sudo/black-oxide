fn print_unique(s: &Str) {
    let mut seen: Vec<Char> = vec()
    for c in s.chars() {
        if !seen.contains(&c) {
            print(c)
            seen.push(c)
        }
    }
}

    print_unique(&str_len)

    print_unique(&trunc)
}
