fn count_vowels(s: Str) -> Unit {
    let mut vowels = 0
    let mut consonants = 0
    for c in chars(s) {
        match c {
            "a" => vowels += 1,
            "e" => vowels += 1,
            "i" => vowels += 1,
            "o" => vowels += 1,
            "u" => vowels += 1,
            _ => consonants += 1,
        }
    }
    print(vowels)
    print(consonants)
}

fn main() {
    count_vowels("keyboard")
}
