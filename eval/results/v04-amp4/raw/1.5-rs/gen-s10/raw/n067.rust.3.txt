fn main() {
    let numbers = [7, 4, 9, 10];
    let mut total_even = 0;

    for &num in &numbers {
        if num % 2 == 1 { // Correct condition to check for odd numbers
            println!("{}", num * 2);
        } else {
            total_even += num;
        }
    }

    println!("Total of even entries: {}", total_even);
}
