fn vec() -> Vec<T> {
    let mut v = vec_with_capacity(0)
    v
}

fn push(v: Vec<T>, x: T) -> Vec<T> {
    let len = len(v)
    let mut w = vec_with_capacity(len + 1)
    for i in range(0, len) {
        w = push(w, get(v, i))
    }
    w = push(w, x)
    w
}

fn len(v: Vec<T>) -> Int {
    let mut i = 0
    while get(v, i).is_some() {
        i = i + 1
    }
    i
}

fn get(v: Vec<T>, i: Int) -> Option<T> {
    if i < len(v) {
        Some(v.nth(i))
    } else {
        None
    }
}
