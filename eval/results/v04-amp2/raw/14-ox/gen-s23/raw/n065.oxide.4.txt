struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max_val = None
    for x in v {
        if max_val.is_none() || x > unwrap_or(max_val, -1) {
            max_val = Some(x)
        }
    }
    max_val
}

fn main() {
    let readings = push(push(push(push(push(vec(), 6), 11), 3), 14), 9)
    let count = count_exceeding(clone(readings), 5)
    print(count)
    let max_val = max_reading(clone(readings))
    match max_val {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
