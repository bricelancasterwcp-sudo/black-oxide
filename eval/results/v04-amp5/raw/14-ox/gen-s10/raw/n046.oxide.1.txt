fn count_under(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x < cutoff {
            count += 1
        }
    }
    count
}

fn count_over_or_equal(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= cutoff {
            count += 1
        }
    }
    count
}

fn main() {
    let values = vec!([5, 12, 3, 18, 9])
    let cutoff = 10
    print(count_under(values, cutoff))
    let values = vec!([5, 12, 3, 18, 9])
    print(count_over_or_equal(values, cutoff))
}
