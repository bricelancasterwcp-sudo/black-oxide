fn main() {
    let number = match std::env::args().nth(1) {
        Some(s) => match s.parse::<i32>() {
            Ok(n) => n,
            Err(_) => {
                println!("bad");
                return;
            }
        },
        None => {
            println!("bad");
            return;
        }
    };

    println!("{}", number + 1);
}
