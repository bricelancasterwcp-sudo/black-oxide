fn main() {
    let grid = [4, 9, 2, 3, 5, 7, 8, 1, 6];
    for i in 0..3 {
        println!("{}", grid[i*3] + grid[i*3+1] + grid[i*3+2]);
    }
    for j in 0..3 {
        println!("{}", grid[j] + grid[j+3] + grid[j+6]);
    }
    println!("{}", grid[0] + grid[4] + grid[8]);
    println!("{}", grid[2] + grid[4] + grid[6]);
    let mut all_equal = true;
    for i in 1..8 {
        if grid[i] != grid[0] {
            all_equal = false;
            break;
        }
    }
    println!("{}", all_equal);
}
