```rust
use std::io;

fn main() {
    let numerator = 20;
    let denominator = 4;

    match divide(numerator, denominator) {
        Ok(result) => println!("Result: {}", result),
        Err(_) => println!("bad"),
    }
}

fn divide(a: i32, b: i32) -> Result<i32, io::Error> {
    if b == 0 {
        return Err(io::Error::new(io::ErrorKind::Other, "Division by zero"));
    }
    Ok(a / b)
}
```
