fn main() {
    let mut v = Vec::new();
    read(&mut v);
    if let Some(n) = pop(&v) {
        println!("{}", n);
    } else {
        println!("bad");
    }
    read(&mut v);
    if let Some(n) = pop(&v) {
        println!("{}", n);
    } else {
        println!("bad");
    }
}

fn read(v: &mut Vec<String>) {
    for line in std::io::stdin().lines() {
        match line.unwrap() {
            "8" => v.push(String::from(line.unwrap())),
            "oak" => v.push(String::from(line.unwrap())),
            "15" => v.push(String::from(line.unwrap())),
            _ => (),
        }
    }
}

fn pop(v: &Vec<String>) -> Option<String> {
    v.pop()
}
