    let s = int_to_str(x)
    push(vec(), s)
}

fn int_to_str(n: Int) -> Str {
    if n < 0 {
        return concat(str_len("-"), int_to_str(-n))
    }
    if n < 10 {
        return str_len("0") + str_len(n)
    }
    let mut digits = vec()
    while n != 0 {
        digits = push(digits, n % 10)
        n = trunc(n / 10)
    }
    let digits = reverse(digits)
    let mut s = str_len("")
    for d in &digits {
        s = concat(s, int_to_str(d))
    }
    return s
}

fn reverse(mutv: Vec<Int>) -> Vec<Int> {
    let mut result = vec()
    while len(mutv) != 0 {
        result = push(result, pop(&mutv))
    }
    return result
}

fn pop(v: &Vec<Int>) -> Int {
    let len = len(v)
    if len == 0 {
        return 0
    }
    let x = v.get(len - 1)
    drop v
    return x
}

fn str_len(s: Str) -> Str {
    let mut n = 0
    for _ in &s {
        n = n + 1
    }
    return n
}
