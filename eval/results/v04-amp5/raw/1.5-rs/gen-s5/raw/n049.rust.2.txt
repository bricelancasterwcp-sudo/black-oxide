```rust
fn main() {
    let mut figures = vec![8, 3, 11, 6];

    // Sort the vector to find the smallest and largest values easily
    figures.sort_unstable();

    // Print the smallest figure
    println!("Smallest Figure: {}", figures[0]);

    // Calculate and print the gap between the largest and the smallest figures
    let max = figures.last().unwrap();
    let min = figures.first().unwrap();
    println!("Gap Between Largest and Smallest: {}", max - min);
}
```
