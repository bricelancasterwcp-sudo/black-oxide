struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 5 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?;
    x += 1;
    Some(x)
}

fn main() {
    let readings = vec![6, 11, 3, 14, 9];
    let highest = first_big(readings.clone());
    println_str(concat("Highest reading is ", str_to_str(highest)));
}
