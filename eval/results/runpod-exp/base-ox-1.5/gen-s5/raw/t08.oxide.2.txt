```rust
fn main() {
    let mut v = vec![3, 8, -2, 12, 7];
    
    // Count positive numbers
    let count_positive = v.iter().filter(|&&x| x > 0).count();
    print!("Number of positive values: {}", count_positive);
    
    // Find the largest value
    let max_value = *v.iter().max().unwrap_or(&-1);
    print!("Largest value: {}", max_value);
}
```
