fn main() {
    let mut rect = Rectangle {
        width: 2,
        height: 3,
    };
    rect.change_width(5);
    println!(
        "The area of the rectangle is {} square pixels.",
        rect.area()
    );
}

struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    fn area(&self) -> u32 {
        self.width * self.height
    }

    fn change_width(&mut self, width: u32) {
        self.width = width;
    }
}
