fn print_sorted(v: Vec<Int>) {
    let sorted = sort(v)
    for x in sorted {
        print(x)
    }
}

fn main() {
    let measurements = vec()
    measurements = push(measurements, 9)

    measurements = push(measurements, 7)
    measurements = push(measurements, 4)
    print_sorted(measurements)
