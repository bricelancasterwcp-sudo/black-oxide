fn main() {
    let mut c = 9;
    let mut s = 0;
    let mut h = 0;
    let mut a = 0;
    let mut t = 0;
    let mut m = 0;
    let mut n = 0;
    let mut l = 1000;
    for i in 1..=30 {
        if c % 2 == 0 {
            h += 1;
            c /= 2;
        } else if c % 3 == 0 {
            a += 1;
            c += 5;
        } else {
            t += 1;
            c = (c * 3 + 1) % 1000;
        }
        s += c;
        if c > m {
            m = c;
            n = i;
        }
        if c < l {
            l = c;
        }
    }
    println!("{}", c);
    println!("{}", h);
    println!("{}", a);
    println!("{}", t);
    println!("{}", m);
    println!("{}", n);
    println!("{}", l);
    println!("{}", s);
    println!("{}", if c >= 100 { "high" } else { "low" });
}
