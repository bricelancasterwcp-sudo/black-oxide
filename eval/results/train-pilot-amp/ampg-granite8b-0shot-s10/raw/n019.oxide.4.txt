struct List { head: Int, tail: Option<Box<List>> }

fn len(l: Box<List>) -> Int {
    let x = match l.tail {
        None => 1,
        Some(refRest) => 1 + len(refRest),
    }
    x
}

fn main() {
    let l1 = Box { head: 1, tail: Some(Box { head: 2, tail: Some(Box { head: 3, tail: None }) }) }
    let l2 = Box { head: 4, tail: Some(Box { head: 5, tail: None }) }
    print(len(l1) + len(l2))
}
