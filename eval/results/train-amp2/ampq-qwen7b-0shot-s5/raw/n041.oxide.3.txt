fn sort(mutvec: Vec<Int>) -> Vec<Int> {
    let mut sorted = vec()
    while letSome(x) == get(mutvec, 0)? {
        sorted = push(sorted, x)
        mutvec = get(mutvec, 1)?
    }
    sorted
}

fn get(vec: Vec<Int>, index: Int) -> Option<Int> {
    if index >= 0 && index < vec.len() {
        Some(vec(index))
    } else {
        None
    }
}
