```rust
fn main() {
    let word = "keyboard";
    let mut vowels_count = 0;
    let mut consonants_count = 0;

    for letter in word.chars() {
        match letter {
            'a' | 'e' | 'i' | 'o' | 'u' | 'A' | 'E' | 'I' | 'O' | 'U' => vowels_count += 1,
            _ => consonants_count += 1,
        }
    }

    println!("Vowels: {}", vowels_count);
    println!("Consonants: {}", consonants_count);
}
```
