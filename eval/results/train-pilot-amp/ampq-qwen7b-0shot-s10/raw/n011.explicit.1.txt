fn sum_multiples_four(limit: Int) -> Int {
    let mut total = 0
    let mut count = 0
    let mut multiple = 4
    while count < limit {
        total = total + multiple
        count = count + 1
        multiple = multiple + 4
    }
    total
}

fn main() {
    let total = sum_multiples_four(6)
    print(total)
}
