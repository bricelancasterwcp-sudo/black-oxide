```rust
struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 5 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = match get(v, 1)? {
        n @ 9...14 => n + 1,
        _ => None,
    };
    x
}

fn main() {
    let r = Reading { label: "lab", values: vec![6, 11, 3, 14, 9] };
    print(first_big(r.values));            // field access copies: r stays usable
    println(second(r.values).unwrap_or(0)); // on the next line the highest reading of all
}
```
