fn main() {
    let north = [12, 15, 19, 14, 14, 20, 31, 30];
    let south = [40, 38, 41, 45, 50, 44];
    fn analyze(sensor: &[i32]) -> (String, i32, i32, i32, i32) {
        let mut increases = 0;
        let mut decreases = 0;
        let mut max_increase = 0;
        let mut longest_streak = 0;
        let mut current_streak = 0;
        for i in 1..sensor.len() {
            if sensor[i] > sensor[i - 1] {
                increases += 1;
                current_streak += 1;
                max_increase = max_increase.max(sensor[i] - sensor[i - 1]);
            } else if sensor[i] < sensor[i - 1] {
                decreases += 1;
                current_streak = 0;
            }
            longest_streak = longest_streak.max(current_streak);
        }
        (format!("{:?}", sensor), increases, decreases, max_increase, longest_streak)
    }
    let (north_info, north_increases, north_decreases, north_max_increase, north_longest_streak) =
        analyze(&north);
    let (south_info, south_increases, south_decreases, south_max_increase, south_longest_streak) =
        analyze(&south);
    println!("{}", north_info);
    println!("{}", south_info);
    if north_longest_streak > south_longest_streak {
        println!("North");
    } else {
        println!("South");
    }
}
