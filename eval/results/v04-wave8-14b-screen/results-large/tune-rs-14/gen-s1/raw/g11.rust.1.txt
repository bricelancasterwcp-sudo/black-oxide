fn main() {
    let grid = [4, 9, 2, 3, 5, 7, 8, 1, 6];
    for i in 0..3 {
        println!("{}", grid[i * 3..(i + 1) * 3].iter().sum::<i32>());
    }
    for i in 0..3 {
        println!("{}", (0..3).map(|j| grid[j * 3 + i]).sum::<i32>());
    }
    println!("{}", (0..3).map(|i| grid[i * 4]).sum::<i32>());
    println!("{}", (0..3).map(|i| grid[2 - i * 2]).sum::<i32>());
    let sums = [
        grid[0..3].iter().sum::<i32>(),
        grid[3..6].iter().sum::<i32>(),
        grid[6..9].iter().sum::<i32>(),
        (0..3).map(|i| grid[i * 3]).sum::<i32>(),
        (0..3).map(|i| grid[i * 3 + 1]).sum::<i32>(),
        (0..3).map(|i| grid[i * 3 + 2]).sum::<i32>(),
        (0..3).map(|i| grid[i * 4]).sum::<i32>(),
        (0..3).map(|i| grid[2 - i * 2]).sum::<i32>(),
    ];
    println!("{}", sums.iter().all(|&x| x == sums[0]));
}
