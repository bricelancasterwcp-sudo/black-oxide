fn pop(mutv: Vec<Int>) -> Option<Int> {
    if len(mutv) == 0 {
        return None
    }
    let x = get(mutv, len(mutv) - 1)?
    mutv = truncate(mutv, len(mutv) - 1)
    Some(x)
}

fn truncate(mutv: Vec<Int>, n: Int) -> Vec<Int> {
    if n < 0 {
        return vec()
    }
    for i in n + 1 {
        pop(mutv)
    }
    mutv
}

fn len(v: Vec<Int>) -> Int {
    let mut acc = 0
    for _ in v {
        acc = acc + 1
    }
    acc
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < 0 || i >= len(v) {
        return None
    }
    Some(v.clone(i))
}

fn int_to_str(n: Int) -> Str {
    if n == 0 {
        return "0"
    }
    let mut acc = ""
    let mut x = n
    while x > 0 {
        acc = char(trunc(x % 10) + 48) + acc
        x = trunc(x / 10)
    }
    acc
}
