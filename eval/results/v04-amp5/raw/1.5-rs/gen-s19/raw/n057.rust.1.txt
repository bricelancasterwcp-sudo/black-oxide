```rust
fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut max_count = 0;
    let mut max_name = "";

    for name in names {
        let count = name.len();
        if count > max_count {
            max_count = count;
            max_name = name;
        }
    }

    println!("The name with the most letters is {} and it has {} letters.", max_name, max_count);
}
```
