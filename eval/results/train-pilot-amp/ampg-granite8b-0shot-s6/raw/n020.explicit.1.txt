fn mean(nums: &Vec<Int>) -> Float {
    let sum = 0
    let len = len(nums)
    for x in &nums {
        sum = sum + x
    }
    to_float(sum) / to_float(len)
}

fn main() {
    let nums = push(push(vec(), 4), 8)
    let mean_num = mean(&nums)
    drop nums
    print(mean_num)
    let nums = push(push(vec(), 15), 16)
    let mean_num = mean(&nums)
    drop nums
    print(mean_num)
    let nums = push(push(vec(), 23), 42)
    let mean_num = mean(&nums)
    drop nums
    print(mean_num)
}
