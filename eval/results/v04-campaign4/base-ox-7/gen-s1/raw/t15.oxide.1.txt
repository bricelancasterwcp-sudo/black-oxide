fn main() {
    let s1 = "12";
    let s2 = "x";
    let s3 = "30";

    let mut sum = 0;
    let mut count_failed = 0;

    if let Some(n) = parse_int(s1) {
        sum += n;
    } else {
        count_failed += 1;
    }

    if let Some(n) = parse_int(s2) {
        sum += n;
    } else {
        count_failed += 1;
    }

    if let Some(n) = parse_int(s3) {
        sum += n;
    } else {
        count_failed += 1;
    }

    print(sum);
    print_str(" ");
    print(count_failed);
}
