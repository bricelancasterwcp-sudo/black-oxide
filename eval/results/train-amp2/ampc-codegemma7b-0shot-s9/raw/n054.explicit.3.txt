fn is_vowel(c: &Char) -> Bool {
    match c {
        Char(c) => {
            match c {
                _ => false,
                _ => true,
            }
        },
        _ => false,
    }
}

fn count_vowels(s: &Str) -> Int {
    let mut count = 0
    for c in chars(s) {
        if is_vowel(&c) {
            count = count + 1
        }
    }
    count
}

fn count_consonants(s: &Str) -> Int {
    let mut count = 0
    for c in chars(s) {
        if !is_vowel(&c) {
            count = count + 1
        }
    }
    count
}

fn main() {
    let word = "keyboard"
    let vowels = count_vowels(&word)
    let consonants = count_consonants(&word)
    print(vowels)
    print(consonants)
    drop vowels
    drop consonants
    drop word
}
