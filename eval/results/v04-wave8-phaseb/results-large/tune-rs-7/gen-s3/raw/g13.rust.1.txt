fn main() {
    let mut c = 9;
    let mut h = 0;
    let mut a = 0;
    let mut t = 0;
    let mut l = 0;
    let mut n = 0;
    let mut s = 0;
    let mut b = 0;
    for i in 1..=30 {
        if c % 2 == 0 {
            h += 1;
            c /= 2;
        } else if c % 3 == 0 {
            a += 1;
            c += 5;
        } else {
            t += 1;
            c = (c * 3 + 1) % 1000;
        }
        if c > l {
            l = c;
            n = i;
        }
        if c < s || s == 0 {
            s = c;
        }
        if c < 10 {
            b += 1;
        }
    }
    println!("{}", c);
    println!("{}", h);
    println!("{}", a);
    println!("{}", t);
    println!("{}", l);
    println!("{}", n);
    println!("{}", s);
    println!("{}", s);
    println!("{}", b);
    if c > 100 {
        println!("high");
    } else {
        println!("low");
    }
}
