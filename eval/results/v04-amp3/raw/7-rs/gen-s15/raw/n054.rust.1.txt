```rust
fn main() {
    let word = "keyboard";
    let mut vowels = 0;
    let mut consonants = 0;

    for letter in word.chars() {
        match letter.to_lowercase().next() {
            Some('a' | 'e' | 'i' | 'o' | 'u') => vowels += 1,
            _ if letter.is_alphabetic() => consonants += 1,
            _ => {}
        }
    }

    println!("Vowels: {}", vowels);
    println!("Consonants: {}", consonants);
}
```
