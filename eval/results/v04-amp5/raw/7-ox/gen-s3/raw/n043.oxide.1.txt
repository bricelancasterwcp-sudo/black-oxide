```rust
struct Queue { items: Vec<Int> }

fn new_queue() -> Queue {
    Queue { items: vec![] }
}

fn enqueue(q: Queue, x: Int) -> Queue {
    Queue { items: push(q.items, x) }
}

fn dequeue(q: Queue) -> (Option<Int>, Queue) {
    let len = len(q.items);
    if len == 0 {
        (None, q)
    } else {
        (Some(get(q.items, 0)?), Queue { items: set(q.items, 0, get(q.items, len - 1)?) })
    }
}

fn main() {
    let mut q = new_queue();
    q = enqueue(q, 1);
    q = enqueue(q, 2);
    q = enqueue(q, 3);
    q = enqueue(q, 4);
    q = enqueue(q, 5);

    let (first, q) = dequeue(q);
    let (_, q) = dequeue(q);
    let (_, q) = dequeue(q);
    let (_, q) = dequeue(q);
    let (last, _) = dequeue(q);

    print(first.unwrap());
    print(last.unwrap());
}
```
