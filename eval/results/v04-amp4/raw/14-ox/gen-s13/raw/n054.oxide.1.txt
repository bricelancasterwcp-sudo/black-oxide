fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    let mut vowels = 0
    let mut consonants = 0

    for c in chars(keyboard) {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            vowels += 1
        } else {
            consonants += 1
        }
    }

    print(vowels)
    print(consonants)
}
