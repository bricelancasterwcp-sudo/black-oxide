fn main() {
    let samples = vec![5, 12, 3, 18, 9];
    let cutoff = 10;
    
    let under_count = count(samples.iter().filter(|&&x| x < cutoff).collect::<Vec<_>>(), cutoff);
    let over_or_equal_count = count(samples.iter().filter(|&&x| x >= cutoff).collect::<Vec<_>>(), cutoff);

    print(under_count);
    print_str(" ");
    print(over_or_equal_count);
}

fn count(v: Vec<i32>, _cutoff: i32) -> i32 {
    v.len() as i32
}
