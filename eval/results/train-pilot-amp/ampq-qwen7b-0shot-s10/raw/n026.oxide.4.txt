struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn int_to_str(n: Int) -> Str {
    let mut s = ""
    let mut m = n
    while m > 0 {
        s = concat(char_to_str(m % 10 + 48), s)
        m = trunc(m / 10)
    }
    s
}

fn char_to_str(c: Int) -> Str {
    let mut s = ""
    if c >= 48 && c <= 57 {
        s = concat(s, str(c))
    }
    s
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(first_big(r.values))            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
    let num = 2026
    let num_str = int_to_str(num)
    print_str(num_str)
    print(str_len(num_str))
}
