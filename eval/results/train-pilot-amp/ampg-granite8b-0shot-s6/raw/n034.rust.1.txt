fn main() {
    let list: Vec<i32> = vec![7, 8, 9];
    let second = get_second_element(list);
    match second {
        Some(x) => println!("{}", x),
        None => println!("None"),
    }
}

fn get_second_element(list: Vec<i32>) -> Option<i32> {
    if list.len() < 2 {
        return None;
    }
    Some(list[1])
}
