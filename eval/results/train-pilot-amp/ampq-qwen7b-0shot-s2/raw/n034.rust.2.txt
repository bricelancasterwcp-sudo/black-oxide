fn second_item<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}

fn main() {
    let list7 = vec![1, 2, 3, 4, 5, 6, 7];
    let list8: Vec<_> = vec![1, 2];
    let list9: Vec<_> = vec![];
    
    println!("{:?}", second_item(&list7)); // Some(2)
    println!("{:?}", second_item(&list8)); // Some(2)
    println!("{:?}", second_item(&list9)); // None
}
