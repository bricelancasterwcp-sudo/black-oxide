fn main() {
    let morning = [94, 71, 88, 65, 92, 79, 83];
    let evening = [55, 90, 77, 84, 68, 99];
    fn analyze(sitting: &[i32]) {
        let mut above_90 = 0;
        let mut between_80_and_89 = 0;
        let mut between_70_and_79 = 0;
        let mut below_70 = 0;
        let mut max_mark = sitting[0];
        let sum: i32 = sitting.iter().sum();
        for &mark in sitting {
            if mark >= 90 { above_90 += 1; }
            else if mark >= 80 { between_80_and_89 += 1; }
            else if mark >= 70 { between_70_and_79 += 1; }
            else { below_70 += 1; }
            if mark > max_mark { max_mark = mark; }
        }
        println!("Morning");
        println!("{}", above_90);
        println!("{}", between_80_and_89);
        println!("{}", between_70_and_79);
        println!("{}", below_70);
        println!("{}", max_mark);
        println!("{}", sum / sitting.len() as i32);
    }
    analyze(&morning);
    analyze(&evening);
}
