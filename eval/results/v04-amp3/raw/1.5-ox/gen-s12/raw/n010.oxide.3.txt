fn main() {
    let n = 91;
    if is_prime(n) {
        print_str("yes");
    } else {
        print_str("no");
    }
}

fn is_prime(num: Int) -> Bool {
    if num <= 1 {
        return false;
    }

    for i in 2..num {
        if num % i == 0 {
            return false;
        }
    }

    true
}
