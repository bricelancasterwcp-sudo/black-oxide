```rust
use std::collections::VecDeque;

fn main() {
    let mut queue = VecDeque::new();
    queue.push_back(1);
    queue.push_back(2);
    queue.push_back(3);
    queue.push_back(4);
    queue.push_back(5);

    queue.push_back(queue.pop_front().unwrap());
    queue.push_front(queue.pop_back().unwrap());

    for entry in queue {
        println!("{}", entry);
    }
}
```
