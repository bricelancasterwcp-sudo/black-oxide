```rust
fn main() {
    let mut rect = Rectangle {width: 2, height: 3};
    
    // Change width to 5
    rect.width = 5;
    
    // Print the new area (width * height)
    println!("Area of the rectangle now is {}", rect.area());
}

struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    fn area(&self) -> u32 {
        self.width * self.height
    }
}
```
