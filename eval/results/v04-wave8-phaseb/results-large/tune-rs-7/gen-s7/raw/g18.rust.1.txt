fn main() {
    let mut settings = (4, 2, 30, "base");
    for _ in 0..5 {
        match settings {
            (_, _, _, "tuned") => settings.0 = 8,
            (_, _, _, "patient") => settings.1 = 90,
            (_, _, _, "settled") => settings.2 = 5,
            (_, _, _, "wide") => settings.0 = 16,
            (_, _, _, "last") => settings.1 = 120,
            _ => {}
        }
        println!("{}", settings.3);
        println!("{}", settings.0);
        println!("{}", settings.1);
        println!("{}", settings.2);
        println!("{}", settings.0 * settings.2);
    }
    println!("{}", settings.0 + settings.1 + settings.2);
}
