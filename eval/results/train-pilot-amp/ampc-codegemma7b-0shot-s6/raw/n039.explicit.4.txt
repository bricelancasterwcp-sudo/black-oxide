fn main() {
    let list = vec(1, 2)
    let entry = get(&list, 5)
    match entry {
        Some(x) => print(x),
        None => print("None"),
    }
    drop list
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i >= v.len() {
        None
    } else {
        Some(v.get(i).unwrap())
    }
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut new_vec = v
    new_vec.push(x)
    new_vec
}

fn print(x: Int) {
    println(x)
}

fn print_str(s: Str) {
    println(s)
}

fn vec(x: Int, y: Int) -> Vec<Int> {
    let mut new_vec = Vec.new()
    new_vec.push(x)
    new_vec.push(y)
    new_vec
}

fn clone(x: &Vec<Int>) -> Vec<Int> {
    let mut new_vec = Vec.new()
    for i in 0.to(x.len()) {
        new_vec.push(x.get(i).unwrap())
    }
    new_vec
}

fn len(v: &Vec<Int>) -> Int {
    v.len()
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut new_vec = Vec.new()
    for i in a.to(b) {
        new_vec.push(i)
    }
    new_vec
}

fn chars(s: &Str) -> Vec<Str> {
    let mut new_vec = Vec.new()
    for i in 0.to(s.len()) {
        new_vec.push(s.chars().nth(i).unwrap())
    }
    new_vec
}

fn int_to_str(n: Int) -> Str {
    n.to_string()
}

fn parse_int(s: &Str) -> Option<Int> {
    match s.parse() {
        Ok(i) => Some(i),
        Err(_) => None,
    }
}

fn to_float(n: Int) -> Float {
    n.to_float()
}

fn trunc(x: Float) -> Int {
    x.trunc()
}
