fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut sum = 0;
    let mut count_non_numbers = 0;

    for text in texts.iter().filter_map(|text| text.parse::<i32>().ok()) {
        sum += text;
    }

    count_non_numbers = texts.len() - count_non_numbers;

    println!("Sum: {}", sum);
    println!("Count of non-numbers: {}", count_non_numbers);
}
