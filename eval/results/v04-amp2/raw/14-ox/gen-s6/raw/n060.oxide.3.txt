fn count_digits(s: Str) -> Int {
    let mut count = 0;
    for c in chars(s) {
        if parse_int(c).is_some() {
            count += 1;
        }
    }
    count
}

fn sum_digits(s: Str) -> Int {
    let mut sum = 0;
    for c in chars(s) {
        if let Some(digit) = parse_int(c) {
            sum += digit;
        }
    }
    sum
}

fn main() {
    let input = "a1b22";
    print(sum_digits(input));
    print(count_digits(input));
}
