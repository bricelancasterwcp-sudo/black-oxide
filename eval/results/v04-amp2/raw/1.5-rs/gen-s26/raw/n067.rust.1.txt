```rust
fn main() {
    let numbers = vec![7, 4, 9, 10];
    let mut sum_of_evens = 0;

    for &number in &numbers {
        if number % 2 != 0 {
            println!("{}", number * 2);
        } else {
            sum_of_evens += number;
        }
    }

    println!("Sum of even entries: {}", sum_of_evens);
}
```
