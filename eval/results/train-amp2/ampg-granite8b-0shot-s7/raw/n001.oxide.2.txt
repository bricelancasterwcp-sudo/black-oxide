struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn sum_cubes(n: Int) -> Int {
    let mut total = 0
    for x in range(1, n + 1) {
        total = total + (x * x * x)
    }
    total
}

fn main() {
    let n = 5
    print(sum_cubes(n))
}
