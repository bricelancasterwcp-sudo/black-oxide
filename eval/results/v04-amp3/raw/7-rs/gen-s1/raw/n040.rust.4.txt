```rust
fn main() {
    let square = Shape::Square(4.0);
    println!("Area of the square: {}", square.area());
}

enum Shape {
    Square(f64),
    Circle(f64),
}

impl Shape {
    fn area(&self) -> f64 {
        match self {
            Shape::Square(side) => side * side,
            Shape::Circle(radius) => 3.0 * radius * radius,
        }
    }
}
```
