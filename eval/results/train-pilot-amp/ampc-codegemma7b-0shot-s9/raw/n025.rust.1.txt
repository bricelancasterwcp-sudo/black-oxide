```rust
use std::io::{stdin, stdout, Write};

fn main() {
    print!("Enter a number: ");
    stdout().flush().unwrap();

    let mut input = String::new();
    stdin().read_line(&mut input).unwrap();

    let number: u32 = input.trim().parse().unwrap();

    let result = number + 1;

    println!("The result is: {}", result);
}
```
