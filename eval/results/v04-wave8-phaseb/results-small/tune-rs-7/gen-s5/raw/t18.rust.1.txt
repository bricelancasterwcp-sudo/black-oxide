fn main() {
    let shapes = [
        (3, 4), // Rectangle 3x4
        (3, 4, 5), // Triangle 3, 4, 5
        (2, 5), // Rectangle 2x5
    ];
    for shape in &shapes {
        match *shape {
            (a, b) => println!("{}", 2 * (a + b)), // Perimeter of rectangle
            (a, b, c) if a*a + b*b == c*c => println!("{}", a + b + c), // Perimeter of triangle
            _ => unreachable!(), // Should not happen for valid shapes
        }
    }
}
