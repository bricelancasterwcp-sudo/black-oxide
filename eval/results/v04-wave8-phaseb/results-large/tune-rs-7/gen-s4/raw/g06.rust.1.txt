fn main() {
    let morning = vec![94, 71, 88, 65, 92, 79, 83];
    let evening = vec![55, 90, 77, 84, 68, 99];
    fn analyze(sitting: &str, marks: &[i32]) {
        let (mut above_90, mut between_80_89, mut between_70_79, mut below_70) = (0, 0, 0, 0);
        let mut max_mark = i32::MIN;
        let sum: i32 = marks.iter().sum();
        for &mark in marks {
            if mark >= 90 { above_90 += 1; }
            else if mark >= 80 { between_80_89 += 1; }
            else if mark >= 70 { between_70_79 += 1; }
            else { below_70 += 1; }
            max_mark = max_mark.max(mark);
        }
        let mean = sum / marks.len() as i32;
        println!("{}", sitting);
        println!("{}", above_90);
        println!("{}", between_80_89);
        println!("{}", between_70_79);
        println!("{}", below_70);
        println!("{}", max_mark);
        println!("{}", mean);
    }
    analyze("Morning", &morning);
    analyze("Evening", &evening);
    if morning.iter().sum::<i32>() / morning.len() as i32 > evening.iter().sum::<i32>() / evening.len() as i32 {
        println!("Morning");
    } else {
        println!("Evening");
    }
}
