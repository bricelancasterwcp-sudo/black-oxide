fn main() {
    let input = "a1b22";

    // Initialize variables to keep track of the sum and count
    let mut sum = 0;
    let mut count = 0;

    // Iterate through each character in the input string
    for c in input.chars() {
        if c.is_digit(10) {
            // Convert the digit character to an integer and add it to the total
            sum += c.to_digit(10).unwrap();
            // Increment the count of digit characters
            count += 1;
        }
    }

    // Print the total sum and count of digit characters
    println!("Sum of digit characters: {}", sum);
    println!("Count of digit characters: {}", count);
}
