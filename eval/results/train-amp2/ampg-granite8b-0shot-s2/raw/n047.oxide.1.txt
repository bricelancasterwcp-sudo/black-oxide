fn collect_evens() -> Vec<Int> {
    let mut evens = vec()
    for i in range(1, 11) {
        if i % 2 == 0 {
            evens = push(evens, i)
        }
    }
    evens
}

fn print_in_reverse(v: Vec<Int>) {
    for i in range(len(v) - 1, -1, -1) {
        print(get(v, i)?)
    }
}

fn main() {
    let evens = collect_evens()
    print_in_reverse(evens)
}
