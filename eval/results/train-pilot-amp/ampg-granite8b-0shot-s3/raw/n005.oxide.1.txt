    if n % 2 == 0 {
        Some(n)
    } else {
        None
    }
}

    let mut v = vec()
    for i in a.to(b - 1) {
        v = push(v, i)
    }
    v

        }
    }

fn print(n: Int) {
    print_str(n.to_str())
}

fn print_str(s: Str) {
    let s = clone(s)
    let v = chars(s)
    for c in v {
        let c = clone(c)
        print_str_char(c)
    }
}

fn print_str_char(c: Str) {
    match c {
        StringChar(ch) => print_char(ch),
        _ => print_str("?"),
    }
}

fn print_char(ch: Char) {
    match ch {
        CharPrintable(ch) => {
            let c = ch + 0
            print_str(c.to_str())
        },
        _ => print_str("?"),
    }
}

fn to_str(n: Int) -> Str {
    int_to_str(n)
}

fn char_to_str(ch: Char) -> Str {
    ch.to_str()
}

fn clone(x: T) -> T {
    match x {
        Int(n) => Int(n),
        Float(f) => Float(f),
        Bool(b) => Bool(b),
        Str(s) => Str(s),
        Unit => Unit,
        Vec(v) => {
            let mut v2 = vec()
            for x in v {
                let x = clone(x)
                v2 = push(v2, x)
            }
            Vec(v2)
        },
        Option(o) => {
            match o {
                Some(x) => Some(clone(x)),
                None => None,
            }
        },
        Result(r) => {
            match r {
                Ok(x) => Ok(clone(x)),
                Err(e) => Err(clone(e)),
            }
        },
        Struct(fields) => {
            let fields2 = functional_update_clone(fields)
            Struct(fields2)
        },
        Enum(name, fields) => {
            let fields2 = functional_update_clone(fields)
            Enum(name, fields2)
        },
    }
}

fn functional_update_clone(fields: Vec<Field>) -> Vec<Field> {
    let mut fields2 = vec()
    for f in fields {
        let f = match f {
            Field(name, x) => {
                let x = clone(x)
                Field(name, x)
            },
        }
        fields2 = push(fields2, f)
    }
    fields2
}

enum T {
    Int(Int),
    Float(Float),
    Bool(Bool),
    Str(Str),
    Unit,
    Vec(Vec<T>),
    Option(Option<T>),
    Result(Result<T, T>),
    Struct(Vec<Field>),
    Enum(Str, Vec<Field>),
}

enum Field {
    Field(Str, T),
}
