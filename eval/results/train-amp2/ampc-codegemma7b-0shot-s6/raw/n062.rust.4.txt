use std::io::{stdin, Result};

fn main() -> Result<()> {
    loop {
        let mut input = String::new();
        stdin().read_line(&mut input)?;
        let input = input.trim();

        match input {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => continue,
        }
    }

    Ok(())
}
