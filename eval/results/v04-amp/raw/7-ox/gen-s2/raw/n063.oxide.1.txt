struct Line { items: Vec<Int>, price_per_item: Int }

fn main() {
    let line1 = Line { items: push(push(vec(), 4), 4), price_per_item: 3 };
    let line2 = Line { items: push(push(vec(), 5), 5), price_per_item: 2 };

    print(line1.items.len() * line1.price_per_item);
    print(line2.items.len() * line2.price_per_item);

    let total_value = (line1.items.len() * line1.price_per_item) + (line2.items.len() * line2.price_per_item);
    print(total_value);
}
