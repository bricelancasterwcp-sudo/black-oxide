fn main() {
    let mut counter = 9
    let mut halved = 0
    let mut added_five = 0
    let mut tripled = 0
    let mut max_value = 0
    let mut max_round = 0
    let mut min_value = 1000
    let mut running_total = 0
    let mut below_ten = 0
    for i in range(1, 31) {
        if counter % 2 == 0 {
            counter /= 2
            halved += 1
        } else if counter % 3 == 0 {
            counter += 5
            added_five += 1
        } else {
            counter = (counter * 3 + 1) % 1000
            tripled += 1
        }
        running_total += counter
        if counter > max_value {
            max_value = counter
            max_round = i
        }
        if counter < min_value {
            min_value = counter
        }
        if counter < 10 {
            below_ten += 1
        }
    }
    print(max_value)
    print(halved)
    print(added_five)
    print(tripled)
    print(max_round)
    print(min_value)
    print(running_total)
    print(below_ten)
    if max_value > 100 {
        print_str("high")
    } else {
        print_str("low")
    }
}
