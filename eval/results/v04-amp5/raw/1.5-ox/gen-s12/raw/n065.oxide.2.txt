struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 10 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    match get(v, 1) {
        Some(x) => Some(x + 1),
        None => None,
    }
}

fn main() {
    let readings = vec![6, 11, 3, 14, 9];
    let count_exceeding_five = readings.iter().filter(|&&x| x > 5).count();
    print(count_exceeding_five);
    println(max(readings));
}
