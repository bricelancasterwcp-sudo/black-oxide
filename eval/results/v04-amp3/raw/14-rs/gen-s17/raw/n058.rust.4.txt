fn main() {
    let word = "coffee";
    let mut seen = [false; 26];
    for c in word.chars() {
        if let Some(idx) = (c.to_ascii_lowercase() as u8).wrapping_sub(b'a').checked_sub(0) as usize {
            if !seen[idx] {
                seen[idx] = true;
                println!("{}", c);
            }
        }
    }
}
