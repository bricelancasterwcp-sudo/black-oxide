```rust
struct Square {
    side: Int,
}

struct Circle {
    radius: Float,
}

trait Shape {
    fn area(&self) -> Float;
}

impl Shape for Square {
    fn area(&self) -> Float {
        self.side * self.side
    }
}

impl Shape for Circle {
    fn area(&self) -> Float {
        std::f64::consts::PI * self.radius * self.radius
    }
}

fn main() {
    let square = Square { side: 4 };
    println!("{}", square.area());
}
```
