fn main() {
    let line1 = push(push(vec(), 3), 3)
    let line2 = push(push(vec(), 2), 2)
    let total = push(push(line1.clone(), 3), 2)
    print(sum_big(&line1, 0))
    drop line1
    print(sum_big(&line2, 0))
    drop line2
    print(sum_big(&total, 0))
    drop total
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut new_v = v
    new_v.push(x)
    new_v
}

fn clone(x: &Int) -> Int {
    x.clone()
}

fn clone_vec(v: Vec<Int>) -> Vec<Int> {
    let mut new_v = Vec(Int)
    for x in v {
        new_v.push(clone(&x))
    }
    new_v
}

fn print_str(s: &Str) {
    for c in chars(s) {
        print_char(c)
    }
}

    print_str(&c)

fn int_to_str_clone(n: Int) -> Str {
    int_to_str(n)
}

fn int_to_str(n: Int) -> Str {
    let mut s = Str
    let mut n = n
    while n > 0 {
        let d = n % 10
        n = n / 10
        s = concat(int_to_char_clone(d), s)
    }
    s
}

fn int_to_char_clone(n: Int) -> Str {
    int_to_char(n)
}

fn int_to_char(n: Int) -> Str {
    let mut c = Char
    if n < 10 {
        c = Char(n + 48)
    } else {
        c = Char(n + 55)
    }
    Str(c)
}

fn chars(s: &Str) -> Vec<Str> {
    let mut chars = Vec(Str)
    let mut i = 0
    while i < str_len(s) {
        let c = s.get(i)
        chars.push(c)
        i = i + 1
    }
    chars
}

fn str_len(s: &Str) -> Int {
    s.len()
}

fn concat(a: Str, b: Str) -> Str {
    let mut new_str = Str
    let mut i = 0
    while i < str_len(&a) {
        let c = a.get(i)
        new_str = concat_char(new_str, c)
        i = i + 1
    }
    i = 0
    while i < str_len(&b) {
        let c = b.get(i)
        new_str = concat_char(new_str, c)
        i = i + 1
    }
    new_str
}

fn concat_char(s: Str, c: Str) -> Str {
    let new_str = concat(s, c)
    new_str
}
