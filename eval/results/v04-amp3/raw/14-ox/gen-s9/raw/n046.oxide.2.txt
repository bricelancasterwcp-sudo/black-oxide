fn count_below_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut below_count = 0;
    for x in v {
        if x < cutoff {
            below_count += 1;
        }
    }
    below_count
}

fn count_at_or_above_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut above_count = 0;
    for x in v {
        if x >= cutoff {
            above_count += 1;
        }
    }
    above_count
}

fn main() {
    let values = vec(5, 12, 3, 18, 9);
    let cutoff = 10;
    let below_count = count_below_cutoff(values, cutoff);
    let values = vec(5, 12, 3, 18, 9);
    let above_count = count_at_or_above_cutoff(values, cutoff);
    print(below_count);
    print(above_count);
}
