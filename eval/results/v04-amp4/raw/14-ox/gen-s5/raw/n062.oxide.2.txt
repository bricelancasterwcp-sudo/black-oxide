struct Signal { color: Str }

fn print_signal_meaning(s: Signal) {
    match s.color {
        "green" => print_str("go"),
        "red" => print_str("stop"),
        "yellow" => print_str("wait"),
        _ => print_str("unknown signal"),
    }
}

fn main() {
    let signals = vec!("green", "red", "yellow");
    for color in signals {
        print_signal_meaning(Signal { color: color.clone() });
    }
}
