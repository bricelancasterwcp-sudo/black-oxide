```rust
fn main() {
    let fruits = ["fig", "banana", "kiwi"];
    let mut longest_fruit = "";
    let mut max_length = 0;

    for fruit in &fruits {
        if fruit.len() > max_length {
            longest_fruit = fruit;
            max_length = fruit.len();
        }
    }

    println!("The longest fruit is {} with {} letters.", longest_fruit, max_length);
}
```
