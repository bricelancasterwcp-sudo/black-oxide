struct Entry { value: Option<Int> }

fn parse_entry(s: Str) -> Option<Entry> {
    let v = parse_int(s)?
    Some(Entry { value: Some(v) })
}

fn main() {
    for entry in vec("8", "oak", "15") {
        match parse_entry(entry)? {
            Entry { value: Some(n) } => print(n),
            _ => print_str("bad"),
        }
    }
}
