```rust
fn main() {
    let s = "mississippi";
    let mut count_s = 0;
    let mut seen_chars = HashSet::new();

    for c in s.chars() {
        if c == 's' {
            count_s += 1;
        }
        seen_chars.insert(c);
    }

    println!("The character 's' occurs {} times.", count_s);
    println!("There are {} distinct characters.", seen_chars.len());
}
```
