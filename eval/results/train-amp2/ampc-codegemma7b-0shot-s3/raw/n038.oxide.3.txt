fn main() {
    let mut device = Device { sensor: Sensor { reading: 0 } }
    device.sensor.reading = 7
    print(device.sensor.reading)
}

struct Device { sensor: Sensor }

struct Sensor { reading: Int }

fn print(x: Int) {
    if x < 0 {
        print_str("-")
    }
    print_str(int_to_str(x))
}

fn print_str(s: Str) {
    for c in chars(s) {
        print(c)
    }
}

fn int_to_str(n: Int) -> Str {
    if n < 0 {
        return concat("-", int_to_str(-n))
    }
    if n < 10 {
        return Str(char(n + 48))
    }
    return concat(int_to_str(n / 10), Str(char((n % 10) + 48)))
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = vec()
    for i in 0.to_int() < len(s) {
        result.push(Str(get(s, i)?)).expect("get failed")
    }
    result
}

fn concat(a: Str, b: Str) -> Str {
    let mut result = vec()
    for c in chars(a) {
        result.push(c)
    }
    for c in chars(b) {
        result.push(c)
    }
    Str(result)
}

fn len(v: Vec<Int>) -> Int {
    let mut result = 0
    for _ in v {
        result = result + 1
    }
    result
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= len(v) {
        return None
    }
    return Some(v.get(i))
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut result = v
    result.push(x)
    result
}

fn main() {
    let mut device = Device { sensor: Sensor { reading: 0 } }
    device.sensor.reading = 7
    print(device.sensor.reading)
}
