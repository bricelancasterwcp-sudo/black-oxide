```rust
fn main() {
    let data = [5, 12, 3, 18, 9];
    let cutoff = 10;

    let mut count_under_cutoff = 0;
    let mut count_over_or_equal_cutoff = 0;

    for &number in &data {
        if number < cutoff {
            count_under_cutoff += 1;
        } else {
            count_over_or_equal_cutoff += 1;
        }
    }

    println!("Numbers under the cutoff: {}", count_under_cutoff);
    println!("Numbers over or equal to the cutoff: {}", count_over_or_equal_cutoff);
}
```
