fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a.trunc() + 1.trunc() - 1.trunc() + 1.trunc() {
        if i >= b {
            break
        }
        v = push(v, i)
    }
    v
}

struct VecIterator { vec: Vec<Int>, idx: Int }

fn next(mut: VecIterator) -> Option<Int> {
    if mut.idx >= len(mut.vec) {
        None
    } else {
        let elem = get(mut.vec, mut.idx)?
        mut.idx = mut.idx + 1
        Some(elem)
    }
}

fn fold(v: Vec<Int>, init: Int, f: FnMutIntIntInt) -> Int {
    let mut acc = init
    for x in v {
        acc = f(acc, x)
    }
    acc
}

fn get(v: Vec<Int>, idx: Int) -> Option<Int> {
    if idx >= len(v) {
        None
    } else {
        Some(v.iter().nth(idx).unwrap())
    }
}
