    }
    let mut total = 0

        if x % 4 == 0 {

    drop list
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    let mut i = a
    while i < b {
        v = push(v, i)
        i = i + 1
    }
    v
}
