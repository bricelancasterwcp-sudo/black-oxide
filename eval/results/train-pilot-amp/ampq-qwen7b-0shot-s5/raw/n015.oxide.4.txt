struct Reading { values: Vec<Int> }

fn count_larger_than_four(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 4 {
            count = count + 1
        }
    }
    count
}

fn main() {
    let readings = push(push(vec(), 5), push(vec(), 3), vec(), push(vec(), 9), vec(), push(vec(), 7))
    print(count_larger_than_four(readings))
}
