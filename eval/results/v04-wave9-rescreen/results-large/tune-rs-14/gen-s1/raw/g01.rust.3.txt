fn main() {
    let group_a = [120, 95, 143, 87, 210, 156, 99];
    let group_b = [175, 132, 88, 240, 119, 167];
    for (label, group) in [("A", &group_a), ("B", &group_b)] {
        let sum: i32 = group.iter().sum();
        let count = group.len() as i32;
        let average = sum / count;
        let above_average_count = group.iter().filter(|&&x| x > average).count() as i32;
        let max_min_diff = *group.iter().max().unwrap() - *group.iter().min().unwrap();
        println!("{}", label);
        println!("{}", count);
        println!("{}", average);
        println!("{}", above_average_count);
        println!("{}", max_min_diff);
    }
    if group_a.iter().sum::<i32>() > group_b.iter().sum::<i32>() {
        println!("A");
    } else {
        println!("B");
    }
}
