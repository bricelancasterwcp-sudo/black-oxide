fn main() {
    let mut v = vec(9, 2, 7, 4)
    for i in 0 < v.len() {
        let mut min = i
        for j in i + 1 < v.len() {
            if v.get(j) < v.get(min) {
                min = j
            }
        }
        if i != min {
            let tmp = v.get(i)
            v.set(i, v.get(min))
            v.set(min, tmp)
        }
    }
    for i in 0 < v.len() {
        print(v.get(i))
        if i != v.len() - 1 {
            print(" ")
        }
    }
    print("\n")
}
