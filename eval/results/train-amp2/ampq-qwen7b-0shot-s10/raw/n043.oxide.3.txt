struct Queue { entries: Vec<Int> }

fn main() {
    let mut q = Queue { entries: vec(1, 2, 3, 4, 5) }
    let first = get(q.entries, 0)?.unwrap() + 1000
    let last = get(q.entries, q.entries.len() - 1)?.unwrap() + 1000
    q.entries = vec(last, get(q.entries, 1)?.unwrap(), get(q.entries, 2)?.unwrap(), get(q.entries, 3)?.unwrap(), first)
    for e in q.entries {
        print(e)
    }
}
