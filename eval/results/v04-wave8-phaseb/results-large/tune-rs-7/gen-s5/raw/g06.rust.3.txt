fn main() {
    let morning = vec![94, 71, 88, 65, 92, 79, 83];
    let evening = vec![55, 90, 77, 84, 68, 99];
    fn analyze(sitting: &str, marks: &[i32]) {
        let (mut above_90, mut between_80_and_89, mut between_70_and_79, mut below_70) = (0, 0, 0, 0);
        let mut highest = i32::MIN;
        for &mark in marks {
            if mark >= 90 { above_90 += 1; }
            else if mark >= 80 { between_80_and_89 += 1; }
            else if mark >= 70 { between_70_and_79 += 1; }
            else { below_70 += 1; }
            highest = highest.max(mark);
        }
        let mean = marks.iter().sum::<i32>() / marks.len() as i32;
        println!("{}", sitting);
        println!("{}", above_90);
        println!("{}", between_80_and_89);
        println!("{}", between_70_and_79);
        println!("{}", below_70);
        println!("{}", highest);
        println!("{}", mean);
    }
    analyze("Morning", &morning);
    analyze("Evening", &evening);
    let morning_mean = morning.iter().sum::<i32>() / morning.len() as i32;
    let evening_mean = evening.iter().sum::<i32>() / evening.len() as i32;
    if morning_mean > evening_mean {
        println!("Morning");
    } else {
        println!("Evening");
    }
}
