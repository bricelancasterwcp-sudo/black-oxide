struct Reading { label: Str, values: Vec<Int> }
fn count_changes(v: Vec<Int>) -> (Int, Int) {
    let mut inc = 0
    let mut dec = 0
    for i in range(1, len(v)) {
        if v[i] > v[i - 1] {
            inc += 1
        } else if v[i] < v[i - 1] {
            dec += 1
        }
    }
    (inc, dec)
}
fn largest_increase(v: Vec<Int>) -> Int {
    let mut max_inc = 0
    for i in range(1, len(v)) {
        let inc = v[i] - v[i - 1]
        if inc > max_inc {
            max_inc = inc
        }
    }
    max_inc
}
fn longest_stretch(v: Vec<Int>) -> Int {
    let mut max_stretch = 0
    let mut current_stretch = 0
    for i in range(1, len(v)) {
        if v[i] > v[i - 1] {
            current_stretch += 1
        } else {
            if current_stretch > max_stretch {
                max_stretch = current_stretch
            }
            current_stretch = 0
        }
    }
    if current_stretch > max_stretch {
        max_stretch = current_stretch
    }
    max_stretch
}
fn main() {
    let north = Reading { label: "north", values: vec(12, 15, 19, 14, 14, 20, 31, 30) }
    let south = Reading { label: "south", values: vec(40, 38, 41, 45, 50, 44) }
    let readings = vec(north, south)
    for r in readings {
        let (inc, dec) = count_changes(r.values)
        let max_inc = largest_increase(r.values)
        let max_stretch = longest_stretch(r.values)
        print_str(r.label)
        print(inc)
        print(dec)
        print(max_inc)
        print(max_stretch)
    }
    if longest_stretch(north.values) > longest_stretch(south.values) {
        print_str("north")
    } else {
        print_str("south")
    }
}
