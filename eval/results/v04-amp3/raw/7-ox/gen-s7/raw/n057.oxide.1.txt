fn main() {
    let fruits = vec("fig", "banana", "kiwi");
    let mut max_len = 0;
    let mut longest_name = "";
    for fruit in fruits {
        if str_len(fruit) > max_len {
            max_len = str_len(fruit);
            longest_name = fruit;
        }
    }
    print(longest_name);
    print(max_len);
}
