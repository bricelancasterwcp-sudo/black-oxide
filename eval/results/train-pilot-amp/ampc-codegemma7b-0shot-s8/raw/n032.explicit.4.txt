fn print(n: Int) {
    print_str(int_to_str(n))
}

fn int_to_str(n: Int) -> Str {
    if n < 0 {
        concat("-", int_to_str(-n))
    } else if n < 10 {
        Str(n + 48)
    } else {
        concat(int_to_str(n / 10), int_to_str(n % 10))
    }
}

fn concat(a: Str, b: Str) -> Str {
    Str(a.chars().into_iter().chain(b.chars().into_iter()).collect())
}

fn print_str(s: Str) {
    for c in s.chars() {
        print(c)
    }
}

fn print(c: Char) {
    if c < 10 {
        print_str(Str(c + 48))
    } else {
        let mut digits = vec()
        let mut n = c - 10
        while n >= 10 {
            digits.push(n % 10)
            n = n / 10
        }
        digits.push(n)
        for d in digits.iter().rev() {
            print_str(Str(d + 48))
        }
    }
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec()
    for i in a.upto(b) {
        result.push(i)
    }
    result
}

fn trunc(x: Float) -> Int {
    if x < 0.0 {
        (x - 0.5).floor()
    } else {
        (x + 0.5).floor()
    }
}
