```rust
fn main() {
    let s = Shape::Square(4.0);
    print(s.area());
}

enum Shape {
    Square(f64),
    Circle(f64),
}

impl Shape {
    fn area(&self) -> f64 {
        match self {
            Shape::Square(side) => side * side,
            Shape::Circle(radius) => std::f64::consts::PI * radius * radius,
        }
    }
}
```
