struct List { values: Vec<Int> }

fn main() {
    let list = List { values: push(push(vec(), 10), 20) };
    
    if let Some(first) = get(list.values, 1) {
        print(first);
    } else {
        print_str("position one is empty");
    }
    
    if let Some(fourth) = get(list.values, 3) {
        print(fourth);
    } else {
        print_str("position four is empty");
    }
}
