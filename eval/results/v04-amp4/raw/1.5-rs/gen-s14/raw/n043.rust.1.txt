fn main() {
    let mut q = [1, 2, 3, 4, 5];
    let temp = q[0]; // Save the first element
    q[0] = q[q.len() - 1]; // Exchange the first and last elements
    q[q.len() - 1] = temp; // Restore the saved first element
    for &x in q.iter() {
        println!("{}", x); // Print each entry on a new line
    }
}
