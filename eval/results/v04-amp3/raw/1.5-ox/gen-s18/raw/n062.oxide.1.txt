```rust
use std::io;

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();

    let codes: Vec<char> = input.chars().collect();
    for &code in codes.iter() {
        match code {
            'g' => print_str("go for green\n"),
            'r' => print_str("stop for red\n"),
            'y' => print_str("wait for yellow\n"),
            _ => print_str("invalid signal code: {}\n", code),
        }
    }
}
```
