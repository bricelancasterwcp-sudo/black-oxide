struct Letter { character: Str, is_vowel: Bool }

fn count_vowels_and_consonants(letters: Vec<Letter>) -> (Int, Int) {
    let vowels = 0
    let consonants = 0
    for letter in letters {
        if letter.is_vowel {
            vowels += 1
        } else {
            consonants += 1
        }
    }
    (vowels, consonants)
}
