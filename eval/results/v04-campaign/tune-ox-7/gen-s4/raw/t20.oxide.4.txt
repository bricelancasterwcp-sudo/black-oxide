struct Item { name: Str, unit_price: Int, quantity: Int }
fn main() {
    let pen = Item { name: "pen", unit_price: 3, quantity: 4 }
    let pad = Item { name: "pad", unit_price: 9, quantity: 2 }
    let ink = Item { name: "ink", unit_price: 5, quantity: 3 }
    let items = vec(pen, pad, ink)
    let mut total_cost = 0
    for item in items {
        let cost = item.unit_price * item.quantity
        total_cost += cost
    }
    print(total_cost)
    let mut max_cost_item = None
    for item in items {
        let cost = item.unit_price * item.quantity
        if max_cost_item == None || cost > max_cost_item.unwrap().unit_price * max_cost_item.unwrap().quantity {
            max_cost_item = Some(item)
        }
    }
    match max_cost_item {
        Some(item) => print(item.name),
        None => print("No items"),
    }
}
