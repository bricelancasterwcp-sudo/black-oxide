```rust
fn main() {
    let mut vowels = 0;
    let mut consonants = 0;

    for c in "keyboard".chars() {
        match c.to_ascii_lowercase() {
            'a' | 'e' | 'i' | 'o' | 'u' => vowels += 1,
            _ => consonants += 1,
        }
    }

    println!("vowels: {}", vowels);
    println!("consonants: {}", consonants);
}
```
