struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max == None || unwrap_or(clone(max), -1) < x {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14)
    let readings = push(readings, 9)
    let r = Reading { label: "lab", values: readings }
    print(count_exceeding(clone(r.values), 5))
    match max_reading(clone(r.values)) {
        Some(n) => print(n),
        None => print_str("no readings"),
    }
}
