fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    print(second(push(push(vec(), 7), 8))) // Should print 9
    print(second(push(push(vec(), 8), 9))) // Should print 10
    print(second(push(push(vec(), 9), 7))) // Should print none (too short)
}
