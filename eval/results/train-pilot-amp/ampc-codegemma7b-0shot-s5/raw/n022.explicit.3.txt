fn int_to_str(n: Int) -> Str {
    let mut s = vec()
    let mut n = n
    while n > 0 {
        let r = n % 10
        n = trunc(n / 10)
        s.push(int_to_char(r))
    }
    let mut t = ""
    for c in s {
        t = t + &c
    }
    return t
}

fn int_to_char(n: Int) -> Str {
    if n < 10 {
        return int_to_str(n)
    } else {
        let mut c = int_to_str(n - 10)
        c.push(char(n + 48))
        return c
    }
}

fn parse_int(s: Str) -> Option<Int> {
    let mut n = 0
    let mut s = s
    while len(s) > 0 {
        let c = s.get(0).unwrap()
        let mut i = 0
        for d in "0123456789" {
            if c == d {
                i = i + 1
            }
        }
        if i == 0 {
            return None
        }
        s = drop_first(s)
        n = n * 10 + (i - 1)
    }
    return Some(n)
}

fn drop_first(s: Str) -> Str {
    if len(s) > 0 {
        return get(s, 1).unwrap()
    } else {
        return ""
    }
}

fn trunc(x: Int) -> Int {
    if x >= 0 {
        return x
    } else {
        return x - 1
    }
}

fn main() {
    let nums = push(push(vec(), 5), 40)
    let bigger = extend(clone(&nums), 70)
    print(sum_big(&nums, 10))
    drop nums
    print(sum_big(&bigger, 10))
    drop bigger
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}
