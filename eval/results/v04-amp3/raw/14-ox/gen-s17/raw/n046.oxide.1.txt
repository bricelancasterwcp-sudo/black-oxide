struct Reading { label: Str, values: Vec<Int> }

fn count_under_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x < cutoff {
            count += 1
        }
    }
    count
}

fn count_over_or_equal_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= cutoff {
            count += 1
        }
    }
    count
}

fn main() {
    let readings = push(push(push(push(vec(), 5), 12), 3), 18)
    let readings = push(readings, 9)
    let cutoff = 10
    print(count_under_cutoff(readings, cutoff))
    print(count_over_or_equal_cutoff(readings, cutoff))
}
