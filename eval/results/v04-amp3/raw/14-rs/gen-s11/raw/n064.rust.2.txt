fn main() {
    let numbers = vec![10, 20];

    if let Some(&value) = numbers.get(0) {
        println!("Position one: {}", value);
    } else {
        println!("Position one: None");
    }

    if let Some(&value) = numbers.get(3) {
        println!("Position four: {}", value);
    } else {
        println!("Position four: None");
    }
}
