fn main() {
    let mut vec = vec(9, 2, 7, 4)
    for _ in range(0, len(vec)) {
        let mut min = 0
        for y in range(0, len(vec)) {
            if get(vec, y) < get(vec, min) {
                min = y
            }
        }
        print(get(vec, min))
        vec = push(remove(vec, min))
    }
}
