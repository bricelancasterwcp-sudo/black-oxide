    let s = "a1b22"
    let mut total = 0
    let mut count = 0
    for c in s.chars() {
        if c >= "0" && c <= "9" {
            let num = parse_int(c).unwrap_or(0)
            total = total + num
            count = count + 1
        }
    }
    print(total)
    print(count)
}
