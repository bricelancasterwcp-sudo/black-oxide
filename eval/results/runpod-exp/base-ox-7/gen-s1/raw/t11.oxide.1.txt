fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut sorted = v;
    for i in range(0, len(sorted)) {
        for j in range(i + 1, len(sorted)) {
            if get(sorted, i)? > get(sorted, j)? {
                let temp = get(sorted, i)?;
                set(sorted, i, get(sorted, j)?);
                set(sorted, j, temp);
            }
        }
    }
    sorted
}

fn main() {
    let v = vec(5, 3, 8, 1, 9, 2);
    let sorted = sort(v);
    for x in sorted {
        print(x);
    }
}
