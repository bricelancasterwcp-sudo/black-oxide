struct Queue {
    items: Vec<Int>,
}

impl Queue {
    fn new() -> Self {
        Queue { items: vec![] }
    }

    fn push(&mut self, item: Int) {
        self.items.push(item);
    }

    fn pop(&mut self) -> Option<Int> {
        if let Some(last) = self.items.pop() {
            last
        } else {
            None
        }
    }

    fn first(&self) -> Option<&Int> {
        self.items.get(0)
    }

    fn last(&self) -> Option<&Int> {
        self.items.last()
    }
}

fn main() {
    let mut q = Queue::new();
    for i in 1..=5 {
        q.push(i);
    }

    if let Some(first) = q.first() {
        if let Some(last) = q.last() {
            q.pop(); // remove first
            q.push(*last); // move last to front
            q.pop(); // remove last again
            q.push(*first); // move first back to end
        }
    }

    for item in q.items.iter() {
        print!("{}", item);
        println!();
    }
}
