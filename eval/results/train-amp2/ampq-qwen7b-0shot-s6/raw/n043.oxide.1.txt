struct Queue { items: Vec<Int> }

fn main() {
    let mut queue = Queue { items: vec(1, 2, 3, 4, 5) }
    let first = queue.items.get(0)?.clone()
    let last = queue.items.pop().unwrap()
    queue.items.insert(0, last)
    queue.items.push(first)
    for item in queue.items {
        print(item)
    }
}
