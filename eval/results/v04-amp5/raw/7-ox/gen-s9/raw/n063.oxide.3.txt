struct Line { items: Vec<Int>, price: Int }

fn value(line: Line) -> Int {
    line.items.len() * line.price
}

fn main() {
    let line1 = Line { items: push(push(vec(), 4), 4), price: 3 };
    let line2 = Line { items: push(push(push(vec(), 5), 5), 5), price: 2 };
    print(value(line1)); 
    print_str("\n"); 
    print(value(line2)); 
    print_str("\n"); 
    let combined = Line { items: concat(line1.items.clone(), line2.items.clone()), price: value(Line { items: vec(), price: 0 }) + value(line2) };
    print(value(combined));
}
