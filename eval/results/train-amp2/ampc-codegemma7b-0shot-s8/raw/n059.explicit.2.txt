fn count_words(phrase: &Str) -> Int {
    let mut words = 0
    for word in phrase.split_whitespace() {
        words = words + 1
    }
    words
}

fn count_chars(phrase: &Str) -> Int {
    let mut chars = 0
    for c in phrase.chars() {
        chars = chars + 1
    }
    chars
}

fn main() {
    let phrase = "when in doubt go home"
    print(count_words(&phrase))
    print(count_chars(&phrase))
}
