```rust
fn main() {
    let mut v = vec![10, 20, 30];
    println!("{}", get(&mut v, 1)); // prints 20
    println!("{}", get(&mut v, 5)); // prints -1 (position out of bounds)
}

fn get(v: &mut Vec<Int>, i: Int) -> Option<Int> {
    if i < 0 || i >= v.len() {
        None
    } else {
        Some(v[i])
    }
}
```
