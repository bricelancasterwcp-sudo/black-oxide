```rust
fn main() {
    let numbers = vec![12, 5, 20, 8, 17];
    let mut largest = 0;
    let mut second_largest = 0;
    
    for num in numbers {
        if num > largest {
            second_largest = largest;
            largest = num;
        } else if num > second_largest {
            second_largest = num;
        }
    }
    
    println!("{}", largest);
    println!("{}", second_largest);
}
```
