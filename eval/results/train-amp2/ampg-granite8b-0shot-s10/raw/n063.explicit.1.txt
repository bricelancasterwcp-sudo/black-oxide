    let mut units = 0
    for _ in 0.range(4) {
        units = units + 1
    }
    units * 3
}

    }

}

}

    let mut s = ""
    let mut n = n
    while n > 0 {
        n = n - 1
        s = "0" + s
    }
    s
}
