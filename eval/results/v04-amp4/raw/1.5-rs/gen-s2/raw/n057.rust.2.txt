```rust
fn main() {
    let names = vec!["fig", "banana", "kiwi"];
    let mut longest_name = "";
    let mut longest_length = 0;

    for name in &names {
        if name.len() > longest_length {
            longest_length = name.len();
            longest_name = name;
        }
    }

    println!("The longest name is: {} with length {}", longest_name, longest_length);
}
```
