fn main() {
    let numbers = vec![1, 2, 3, 4, 5];
    match get_second(&numbers) {
        Some(num) => println!("Second number: {}", num),
        None => println!("No second number"),
    }

    let numbers = vec![7];
    match get_second(&numbers) {
        Some(num) => println!("Second number: {}", num),
        None => println!("No second number"),
    }

    let numbers = vec![8, 9];
    match get_second(&numbers) {
        Some(num) => println!("Second number: {}", num),
        None => println!("No second number"),
    }
}

fn get_second<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}
