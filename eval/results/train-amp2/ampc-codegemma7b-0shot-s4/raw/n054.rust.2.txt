fn main() {
    let word = "keyboard";
    let mut vowels = 0;
    let mut consonants = 0;

    for c in word.chars() {
        match c {
            'a' | 'e' | 'i' | 'o' | 'u' => vowels += 1,
            'A' | 'E' | 'I' | 'O' | 'U' => vowels += 1,
            _ => consonants += 1,
        }
    }

    println!("Vowels: {}", vowels);
    println!("Consonants: {}", consonants);
}
