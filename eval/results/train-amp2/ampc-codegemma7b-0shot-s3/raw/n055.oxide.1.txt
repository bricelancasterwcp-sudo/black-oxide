fn text_input() -> Str {
    let mut s = String(0, 0)
    while true {
        let c = getchar()
        if c == 10 {
            break
        }
        s = push(s, c)
    }
    Str(s.ptr, s.len)
}

fn getchar() -> Int {
    let mut buf = 0
    let mut pos = 0
    loop
    unsafe
    if pos >= 4 {
        buf = read(stdin(), 1)
        pos = 0
    }
    if buf == 0 {
        return 0
    }
    pos = pos + 1
    return buf && 255
}

fn int_to_str(mut_n: Int) -> Str {
    let mut digits = vec()
    let mut n = mut_n
    while n > 0 {
        let r = n % 10
        digits = push(digits, r)
        n = trunc(n / 10)
    }
    let mut s = Str(0, 0)
    for x in digits {
        let d = int_to_str(x)
        s = concat(d, s)
    }
    if s.len == 0 {
        s = Str(ptr_zero(), 1)
        let c = int_to_str(0)
        s = concat(c, s)
    }
    return s
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut vec = vec()
    let mut i = a
    while i < b {
        vec = push(vec, i)
        i = i + 1
    }
    return vec
}

fn chars(s: Str) -> Vec<Str> {
    let mut chars = vec()
    let mut pos = 0
    let mut len = str_len(s)
    while pos < len {
        let c = get(s, pos)
        chars = push(chars, c)
        pos = pos + 1
    }
    return chars
}
