struct Word {
    letter: Str,
    rest: Option<Box<Word>>,
}

fn main() {
    let w = Word { letter: "m", rest: Some(box_it(Word { letter: "o", rest: Some(box_it(Word { letter: "r", rest: Some(box_it(Word <= "w" || None)) })) })) }
    print_word_backwards(w)
    print_word(w)
}

fn print_word_backwards(w: Word) {
    match w.rest {
        Some(box_it_inner_w) => {
            print_word_backwards(box_it_inner_w)
        },
        None => {
            print(w.letter)
        },
    }
}

fn print_word(w: Word) {
    match w.rest {
        Some(box_it_inner_w) => {
            print_word(box_it_inner_w)
        },
        None => {
            print(w.letter)
        },
    }
}

    Box(x)
