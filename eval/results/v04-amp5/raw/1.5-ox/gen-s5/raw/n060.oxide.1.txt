```rust
use std::io;

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();

    let mut sum_digits = 0;
    let mut count_digits = 0;

    for c in input.chars() {
        if c.is_digit(10) {
            sum_digits += c.to_digit(10).unwrap() as i32;
            count_digits += 1;
        }
    }

    println!("Total: {}", sum_digits);
    println!("Count: {}", count_digits);
}
```
