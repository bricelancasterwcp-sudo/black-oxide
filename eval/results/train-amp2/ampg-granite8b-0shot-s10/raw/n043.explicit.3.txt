fn main() {
    let queue = push(push(push(push(vec(), 1), 2), 3), 4)
    let first = pop(&queue)
    let last = pop(&queue)
    push(queue, last)
    push(queue, first)
    while !queue.is_empty() {
        print(pop(&queue))
    }
}

fn pop(queue: &Vec<Int>) -> Int {
    let last = get(queue, len(queue) - 1).unwrap()
    drop queue
    last
}

fn push(queue: Vec<Int>, value: Int) -> Vec<Int> {
    let new_length = len(&queue) + 1
    let mut new_queue = vec()
    for i in 0.to_int(new_length) - 1 {
        new_queue = push(new_queue, get(&queue, i).unwrap())
    }
    new_queue = push(new_queue, value)
    new_queue
}

fn len(queue: &Vec<Int>) -> Int {
    let mut length = 0
    for _ in queue {
        length = length + 1
    }
    length
}

fn get(queue: &Vec<Int>, index: Int) -> Option<Int> {
    if index < 0 || index >= len(queue) {
        None
    } else {
        Some(queue.get(index.to_uint()))
    }
}

fn vec() -> Vec<Int> {
    vec()
}

fn print(value: Int) {
    print_str(str_len(&value))
}

fn print_str(value: Str) {
    print_str(value)
}

fn clone(value: Int) -> Int {
    value
}

fn str_len(value: Str) -> Int {
    len(value)
}

fn to_float(value: Int) -> Float {
    value.to_float()
}

fn trunc(value: Float) -> Int {
    value.trunc()
}

fn concat(value1: Str, value2: Str) -> Str {
    value1 + value2
}

fn chars(value: Str) -> Vec<Str> {
    let mut result = vec()
    for char in value {
        result = push(result, char)
    }
    result
}

fn parse_int(value: Str) -> Option<Int> {
    value.parse_int()
}

fn range(value1: Int, value2: Int) -> Vec<Int> {
    let mut result = vec()
    for i in value1.to_int(value2) {
        result = push(result, i)
    }
    result
}
