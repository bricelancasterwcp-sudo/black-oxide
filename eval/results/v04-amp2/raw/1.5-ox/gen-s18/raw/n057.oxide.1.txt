```rust
fn main() {
    let mut max_name = String::new();
    let mut max_length = 0;

    for &name in ["fig", "banana", "kiwi"].iter() {
        if name.len() > max_length {
            max_name = name.to_string();
            max_length = name.len();
        }
    }

    println!("The name with the most letters is '{}', which has {} letters.", max_name, max_length);
}
```
