fn readings(readings: Vec<Int>) -> Int {
    let mut count = 0
    let mut highest = 0
    for reading in readings {
        if reading > 5 {
            count = count + 1
        }
        if reading > highest {
            highest = reading
        }
    }
    count + highest
}

fn main() {
    let readings = vec(6, 11, 3, 14, 9)
    print(readings(readings))
}
