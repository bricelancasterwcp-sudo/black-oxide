    let fig11 = 11
    let fig6 = 6
    let min = min_of(min_of(fig8, fig3), min_of(fig11, fig6))
    let max = max_of(max_of(fig8, fig3), max_of(fig11, fig6))

    print(min)
    print(gap)

    drop gap
}
