struct Square { side: Int }
struct Circle { radius: Float }

const PI: Float = 3.0

fn area(shape: Variant<Square, Circle>) -> Float {
    match shape {
        Variant::Square(Square { side }) => (side * side).to_float(),
        Variant::Circle(Circle { radius }) => PI * radius * radius,
    }
}

fn main() {
    let square = Square { side: 4 }
    print(area(Variant::Square(square)))
}
