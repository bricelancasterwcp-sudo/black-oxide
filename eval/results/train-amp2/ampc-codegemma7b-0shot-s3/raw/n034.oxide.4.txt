fn second(v: Vec<Int>) -> Option<Int> {
    match get(v, 1) {
        Some(x) => Some(x + 1),
        None => None,
    }
}

fn main() {
    print(second(vec(7)).unwrap_or_else(print_str("none")))
    print(second(vec(7, 8)).unwrap_or_else(print_str("none")))
    print(second(vec(7, 8, 9)).unwrap_or_else(print_str("none")))
}
