fn main() {
    let numbers = vec![7, 8, 9];
    match get_second(&numbers) {
        Some(value) => println!("The second value is: {}", value),
        None => println!("None"),
    }
}

fn get_second<T>(list: &[T]) -> Option<&T> {
    if list.len() > 1 {
        Some(&list[1])
    } else {
        None
    }
}
