```rust
fn main() {
    let text = "7x9?";

    let mut sum = 0;
    let mut count = 0;

    for c in text.chars() {
        match c {
            '0'..='9' => {
                if let Ok(n) = c.to_digit(10) {
                    sum += n;
                    count += 1;
                }
            },
            _ => {}
        }
    }

    println!("Sum: {}", sum);
    println!("Count: {}", count);
}
```
