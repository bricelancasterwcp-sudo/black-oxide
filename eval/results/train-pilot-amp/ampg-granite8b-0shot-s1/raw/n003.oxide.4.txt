struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(first_big(r.values))            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}

fn is_divisible_by(n: Int, m: Int) -> Bool {
    n % m == 0
}

fn count_divisors_of(n: Int) -> Int {
    let mut count = 0
    for m in range(1, n + 1) {
        if is_divisible_by(n, m) {
            count = count + 1
        }
    }
    count
}

fn main() {
    let count = count_divisors_of(36)
    if count == 2 {
        print(" exactly")
    } else if count == 1 {
        print(" not exactly")
    } else {
        print_str("divisors: ")
        print(count)
    }
}
