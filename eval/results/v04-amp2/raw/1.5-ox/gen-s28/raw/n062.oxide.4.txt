```rust
fn main() {
    let mut state = "green";
    loop {
        match state {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => panic!("invalid state: {}", state),
        }
        // Simulate arrival of a signal
        if rand::random::<bool>() {
            state = "green";
        } else if rand::random::<bool>() {
            state = "red";
        } else {
            state = "yellow";
        }
    }
}
```
