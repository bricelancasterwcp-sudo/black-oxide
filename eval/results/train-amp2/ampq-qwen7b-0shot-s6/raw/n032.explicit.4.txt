    println(s)
}

fn int_to_str(n: Int) -> Str {
    let mut result = vec()
    let mut n = n
    while n > 0 {
        let digit = n % 10
        let digit = int_to_str(digit)
        push(result, &digit)
        n = n / 10
    }
    let result = concat(result, vec())
    result
}

fn push(v: Vec<Str>, x: Str) -> Vec<Str> {
    let mut result = v
    push(result, x)
    result
}
