struct Pair { label: Str, value: Str }
fn parse_record(record: Str) -> Vec<Pair> {
    let pairs = vec()
    for pair_str in str_split(record, ";") {
        let parts = str_split(pair_str, "=")
        if len(parts) == 2 {
            let label = get(parts, 0).unwrap_or("")
            let value = get(parts, 1).unwrap_or("")
            pairs = push(pairs, Pair { label: label, value: value })
        }
    }
    pairs
}
fn main() {
    let record = "name=widget;qty=17;price=245;tag=blue"
    let pairs = parse_record(record)
    print(len(pairs))
    for pair in pairs {
        print(pair.label)
        print(pair.value)
    }
    let qty_str = get(find(pairs, |p| p.label == "qty").unwrap_or(Pair { label: "", value: "" }), "value").unwrap_or("")
    let price_str = get(find(pairs, |p| p.label == "price").unwrap_or(Pair { label: "", value: "" }), "value").unwrap_or("")
    let qty = parse_int(qty_str).unwrap_or(0)
    let price = parse_int(price_str).unwrap_or(0)
    print(qty * price)
}
