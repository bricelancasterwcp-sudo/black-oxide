fn main() {
    let vowels = "aeiouAEIOU";
    let mut count_vowels = 0;
    let mut count_non_vowels = 0;

    for ch in "keyboard".chars() {
        if vowels.contains(ch) {
            count_vowels += 1;
        } else {
            count_non_vowels += 1;
        }
    }

    print(count_vowels);
    print_str(", ");
    print(count_non_vowels);
}
