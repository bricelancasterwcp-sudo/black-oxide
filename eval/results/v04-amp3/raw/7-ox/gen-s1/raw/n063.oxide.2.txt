struct Stock { items: Vec<Item> }

struct Item { price: Int, quantity: Int }

fn total_price(stock: Stock) -> Int {
    let mut sum = 0;
    for item in stock.items {
        sum += item.price * item.quantity;
    }
    sum
}

fn main() {
    let line1 = Stock { items: vec![Item { price: 3, quantity: 4 }, Item { price: 2, quantity: 5 }] };
    print(total_price(line1));
    print(total_price(line1)); // Fix: reassign the variable inside the loop
}
