struct Item { name: Str, unit_price: Int, quantity: Int }

fn total_cost(items: Vec<Item>) -> Int {
    let mut total = 0
    for item in items {
        total = total + (item.unit_price * item.quantity)
    }
    total
}

fn most_expensive_item(items: Vec<Item>) -> Str {
    let mut max_cost = -1
    let mut max_name = "unknown"
    for item in items {
        let cost = item.unit_price * item.quantity
        if cost > max_cost {
            max_cost = cost
            max_name = item.name
        }
    }
    max_name
}

fn main() {
    let items = vec![
        Item { name: "pen", unit_price: 3, quantity: 4 },
        Item { name: "pad", unit_price: 9, quantity: 2 },
        Item { name: "ink", unit_price: 5, quantity: 3 }
    ]
    print(total_cost(items))
    let items = vec![
        Item { name: "pen", unit_price: 3, quantity: 4 },
        Item { name: "pad", unit_price: 9, quantity: 2 },
        Item { name: "ink", unit_price: 5, quantity: 3 }
    ]
    print_str(most_expensive_item(items))
}
