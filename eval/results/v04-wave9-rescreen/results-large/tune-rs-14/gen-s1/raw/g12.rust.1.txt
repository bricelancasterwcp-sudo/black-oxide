fn main() {
    let figures = [
        ("rect", 7, 4),
        ("square", 5, 0),
        ("circle", 3, 0),
        ("rect", 2, 9),
    ];
    let mut total_area = 0;
    let mut max_area = 0;
    let mut max_figure = "";
    for &(kind, width, height) in &figures {
        let area = match kind {
            "rect" => width * height,
            "square" => width * width,
            "circle" => 3 * width * width,
            _ => 0,
        };
        println!("{}", kind);
        println!("{}", area);
        total_area += area;
        if area > max_area {
            max_area = area;
            max_figure = kind;
        }
    }
    println!("{}", total_area);
    println!("{}", max_figure);
}
