fn main() {
    let samples = vec![5, 12, 3, 18, 9];
    let cutoff = 10;

    let below_count = samples.iter().filter(|&&x| x < cutoff).count();
    let above_or_equal_count = samples.iter().filter(|&&x| x >= cutoff).count();

    println!("{} fall under the cutoff", below_count);
    println!("{} sit at or over it", above_or_equal_count);
}
