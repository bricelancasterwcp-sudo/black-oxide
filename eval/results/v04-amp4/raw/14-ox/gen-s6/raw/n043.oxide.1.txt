struct Queue { values: Vec<Int> }

fn swap_first_last(q: Queue) -> Queue {
    let len = len(q.values)
    if len < 2 {
        return q
    }
    let first = get(q.values, 0).unwrap()
    let last = get(q.values, len - 1).unwrap()
    let middle = slice(q.values, 1, len - 1)
    Queue { values: push(push(push(middle, first), last), get(q.values, len - 2).unwrap()) }
}

fn main() {
    let q = Queue { values: vec!(1, 2, 3, 4, 5) }
    let q = swap_first_last(q)
    for x in q.values {
        print(x)
        print_str("\n")
    }
}
