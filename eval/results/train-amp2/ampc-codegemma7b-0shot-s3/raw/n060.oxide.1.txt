fn digits(s: Str) -> Int {
    let mut sum = 0
    let mut count = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            sum = sum + c.to_int()?
            count = count + 1
        }
    }
    return (sum + count)
}

fn main() {
    let a1b22 = "a1b22"
    print(digits(a1b22))
}
