fn main() {
    let s = "coffee";
    let mut seen = vec![];
    for c in chars(s) {
        if !seen.contains(&c) && count_chars(s, c.to_string()) > 1 {
            print_str(c);
            seen.push(c);
        }
    }
}

fn count_chars(s: Str, c: Str) -> Int {
    count(chars(s), |x| x == &c)
}
