struct Reading { values: Vec<Int> }

fn min(v: Vec<Int>) -> Option<Int> {
    let mut min = v.get(0)?.clone()
    for x in v {
        if x < min {
            min = x
        }
    }
    Some(min)
}

fn main() {
    let r = Reading { values: vec(5, 3, 9, 1, 7) }
    match min(r.values) {
        Some(n) => print(n),
        None => print_str("empty vector"),
    }
}
