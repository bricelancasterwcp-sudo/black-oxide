fn main() {
    let batch = vec(9, 2, 12, 5, 16);
    let mut kept: Vec<Int> = vec();

    for x in batch {
        if x < 10 {
            push(kept, x);
        }
    }

    for y in kept {
        print(y); 
        print_str("\n"); // Print each entry on its own line
    }

    print(len(kept));
}
