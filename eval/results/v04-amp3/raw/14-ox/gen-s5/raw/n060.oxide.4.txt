fn count_digits(s: Str) -> (Int, Int) {
    let mut total = 0
    let mut count = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            total += parse_int(c.clone())?
            count += 1
        }
    }
    (total, count)
}

fn main() {
    let s = "a1b22"
    let (digits, count) = count_digits(s)
    print(digits)
    print(count)
}
