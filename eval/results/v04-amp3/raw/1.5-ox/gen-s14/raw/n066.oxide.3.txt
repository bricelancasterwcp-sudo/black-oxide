fn main() {
    let mut batch = vec![9, 2, 12, 5, 16];
    for i in 0..batch.len() {
        if batch[i] < 10 {
            println!("{}", batch[i]);
            batch.remove(i);

        }
    }
    println!("Kept {} entries", batch.len());
}
