struct Count { vowels: Int, consonants: Int }

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    let mut count = Count { vowels: 0, consonants: 0 }
    
    for ch in chars(keyboard) {
        match ch {
            "a" | "e" | "i" | "o" | "u" => count.vowels += 1,
            _ => count.consonants += 1,
        }
    }
    
    print(count.vowels)
    print(count.consonants)
}
