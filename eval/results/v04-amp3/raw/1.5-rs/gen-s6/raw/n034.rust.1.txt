```rust
fn get_second_entry(numbers: Vec<i32>) -> Option<&i32> {
    numbers.get(1)
}

fn main() {
    let test_numbers = vec![7, 8, 9];
    println!("{:?}", get_second_entry(test_numbers)); // Output: Some(&8)
    
    let empty_list: Vec<i32> = vec![];
    println!("{:?}", get_second_entry(empty_list)); // Output: None
}
```
