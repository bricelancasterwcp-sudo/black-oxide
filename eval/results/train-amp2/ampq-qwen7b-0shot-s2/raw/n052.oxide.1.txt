fn if_char_eq_n(i: Int, c: Str) -> Option<Int> {
    if c == "n" {
        Some(i)
    } else {
        None
    }
}

fn print_positions(self: Vec<Int>) {
    for pos in self {
        println(pos)
    }
}

    self.into_iter().enumerate().map(move || Tuple(_, _)).collect()
}

fn filter_map(self: Vec<Tuple<Int, Str>>, f: Fn<Int, Str>) -> Vec<Int> {
    self.into_iter().filter_map(move || f).collect()
}

struct VecIterator {
    iter: VecIteratorImpl,
}

struct VecIteratorImpl {
    index: Int,
    values: Vec<Str>,
}

fn next(self: VecIterator) -> Option<Str> {
    let next = self.iter.next()
    if next.is_some() {
        Some(next.unwrap())
    } else {
        None
    }
}

fn len(self: VecIterator) -> Int {
    self.iter.len()
}

    VecIterator(self.iter.enumerate().map(move || Tuple(_, _)))

    VecIterator(self.into_iter().filter_map(move || f))
}

    VecIteratorImpl(self.iter.into_iter())
}

fn map(self: VecIteratorImpl, f: Fn<Str>) -> VecIteratorImpl {
    VecIteratorImpl(self.index, self.values.map(move || f))

    if self.index < self.values.len() {
