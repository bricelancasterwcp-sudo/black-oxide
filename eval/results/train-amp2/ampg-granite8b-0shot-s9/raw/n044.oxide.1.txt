struct Deposit { amount: Int, balance: Int }

fn main() {
    let mut balance = 0
    for d in deposits() {
        balance = push(balance, d)
    }
    print(balance)
}

fn deposits() -> Vec<Deposit> {
    vec(Deposit { amount: 3, balance: 0 }, Deposit { amount: 1, balance: 0 }, Deposit { amount: 4, balance: 0 }, Deposit { amount: 1, balance: 0 }, Deposit { amount: 5, balance: 0 })
}

fn push(vec: Vec<Deposit>, x: Deposit) -> Vec<Deposit> {
    vec(x, vec)
}

enum List {
    Cell(Int, Box<List>),
    Nil,
}

fn main() {
    let list = push(push(Nil, 1), 2)
    print(list_sum(list))
}

fn list_sum(list: List) -> Int {
    match list {
        Nil => 0,
        Cell(head, tail) => head + list_sum(tail),
    }
}

fn push(list: List, x: Int) -> List {
    Cell(x, Box(list))
}

struct List {
    head: Int,
    tail: Option<Box<List>>,
}

    let list = push(push(Nil, 1), 2)
    print(list_sum(list))
}

fn list_sum(list: List) -> Int {
    match list {
        Nil => 0,
        Cell(head, tail) => head + list_sum(tail),
    }
}

fn push(list: List, x: Int) -> List {
    List(x, Some(Box(list)))
