struct TextBuilder { content: Str }

fn build_text() -> TextBuilder {
    let mut builder = TextBuilder { content: "" }
    builder.content = to_string(builder.content) + "one"
    builder.content = to_string(builder.content) + "two"
    builder.content = to_string(builder.content) + "three"
    builder
}

fn main() {
    let text = build_text()
    print_str(text.content)
}
