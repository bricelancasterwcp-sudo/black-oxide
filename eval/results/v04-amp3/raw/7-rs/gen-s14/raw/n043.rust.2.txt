fn main() {
    let mut queue = vec![1, 2, 3, 4, 5];
    if let Some(first) = queue.first_mut().and_then(|&mut f| Some(f)) {
        if let Some(last) = queue.last_mut().and_then(|&mut l| Some(l)) {
            std::mem::swap(&mut first, &mut last);
        }
    }
    for item in &queue {
        println!("{}", item);
    }
}
