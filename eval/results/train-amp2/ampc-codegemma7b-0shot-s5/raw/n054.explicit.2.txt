fn count_vowels(s: &Str) -> Int {
    let mut vowels = 0
    for c in s.chars() {
        match c {
            _ => {
                if "aeiouAEIOU".contains(c) {
                    vowels = vowels + 1
                }
            },
        }
    }
    vowels
}

fn count_non_vowels(s: &Str) -> Int {
    let mut non_vowels = 0
    for c in s.chars() {
        match c {
            _ => {
                if !"aeiouAEIOU".contains(c) {
                    non_vowels = non_vowels + 1
                }
            },
        }
    }
    non_vowels
}

fn main() {
    let word = "keyboard"
    let vowels = count_vowels(word)
    let non_vowels = count_non_vowels(word)
    print("Number of vowels: {}\n", vowels)
    print("Number of non-vowels: {}\n", non_vowels)
}
