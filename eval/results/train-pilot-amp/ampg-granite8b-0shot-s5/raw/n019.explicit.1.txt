fn list_difference(list1: &Vec<Int>, list2: &Vec<Int>) -> Int {
    let mut count = 0
    for i in list1 {
        let present = 0
        for j in list2 {
            if i == j {
                present = 1
            }
        }
        if present == 0 {
            count = count + 1
        }
    }
    count
}

fn main() {
    let list1 = push(push(vec(), 1), 2)
    let list2 = push(vec(), 4)
    let list3 = push(vec(), 5)
    let count = list_difference(&list1, &list2)
    drop list1
    drop list2
    print(count)
    drop count
}
