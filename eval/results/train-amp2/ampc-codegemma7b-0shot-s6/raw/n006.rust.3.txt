use std::cmp::max;

fn find_biggest_power_of_two(n: u32) -> u32 {
    let mut result = 1;
    while result < n {
        result *= 2;
    }
    if result > n {
        result /= 2;
    }
    result
}

fn main() {
    let biggest_power_of_two = find_biggest_power_of_two(100);
    println!("The biggest power of two that does not exceed 100 is {}", biggest_power_of_two);
}
