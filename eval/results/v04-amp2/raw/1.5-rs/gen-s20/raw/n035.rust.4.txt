```rust
fn divide_by_zero(a: i32, b: i32) -> Result<f64, &'static str> {
    if b == 0 {
        Err("Cannot divide by zero")
    } else {
        Ok(a as f64 / b as f64)
    }
}

fn main() {
    match divide_by_zero(20, 4) {
        Ok(result) => println!("The result is {}", result),
        Err(error) => println!("{}", error),
    }
}
```
