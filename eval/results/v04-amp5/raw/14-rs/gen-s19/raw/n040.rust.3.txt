```rust
enum Shape {
    Square(f64),
    Circle(f64),
}

impl Shape {
    fn area(&self) -> f64 {
        match self {
            Shape::Square(side) => side * side,
            Shape::Circle(radius) => 3.0 * radius * radius,
        }
    }
}

fn main() {
    let square = Shape::Square(4.0);
    let area = square.area();
    println!("The area of the square is: {}", area);
}
```
