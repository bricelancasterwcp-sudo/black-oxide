fn reverse_number(n: Int) -> Str {
    let s = int_to_str(n)
    let chars = chars(s)
    let reversed_chars = chars(vec())
    for i in range(0, len(chars)) {
        reversed_chars.push(get(chars, len(chars) - i - 1))
    }
    concat(reversed_chars, vec())
}

fn main() {
    print(reverse_number(1234))
}
