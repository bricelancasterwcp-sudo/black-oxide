fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return False
    }
    for i in range(2, trunc(n ** 0.5) + 1) {
        if n % i == 0 {
            return False
        }
    }
    True
}

fn main() {
    let number = 91
    if is_prime(number) {
        print_str("yes")
    } else {
        print_str("no")
    }
}
