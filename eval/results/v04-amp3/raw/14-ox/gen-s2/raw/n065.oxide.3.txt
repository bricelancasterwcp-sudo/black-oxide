struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max_val = None
    for x in v {
        max_val = match max_val {
            Some(current_max) => {
                if x > current_max {
                    Some(x)
                } else {
                    Some(current_max)
                }
            },
            None => {
                Some(x)
            }
        }
    }
    max_val
}

fn main() {
    let readings = vec(6, 11, 3, 14, 9)
    let count = count_exceeding(readings, 5)
    print(count)
    let max_reading = find_max(vec(6, 11, 3, 14, 9))
    match max_reading {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
