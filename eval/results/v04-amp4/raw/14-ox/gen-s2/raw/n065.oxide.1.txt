struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let count = 0
    for x in v {
        if x > threshold {
            count += 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    if len(v) == 0 {
        None
    } else {
        let mut max_val = get(v, 0)?
        for x in v {
            if x > max_val {
                max_val = x
            }
        }
        Some(max_val)
    }
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14)
    let readings = push(readings, 9)
    print(count_exceeding(readings, 5))
    match find_max(readings) {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
