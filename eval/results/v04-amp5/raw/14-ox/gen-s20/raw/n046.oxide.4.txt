fn count_below_and_above(vec: Vec<Int>, cutoff: Int) -> (Int, Int) {
    let below = 0
    let above = 0
    for x in vec {
        if x < cutoff {
            below += 1
        } else {
            above += 1
        }
    }
    (below, above)
}

fn main() {
    let values = vec().push(5).push(12).push(3).push(18).push(9)
    let cutoff = 10
    let (below, above) = count_below_and_above(values, cutoff)
    print(below)
    print(above)
}
