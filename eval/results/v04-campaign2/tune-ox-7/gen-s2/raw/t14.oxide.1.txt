fn main() {
    let s = "mississippi"
    let count_s = 0
    for c in chars(s) {
        if c == "s" {
            count_s += 1
        }
    }
    print(count_s)
    let distinct_chars = vec()
    for c in chars(s) {
        if !contains(distinct_chars, c) {
            distinct_chars = push(distinct_chars, c)
        }
    }
    print(len(distinct_chars))
}
