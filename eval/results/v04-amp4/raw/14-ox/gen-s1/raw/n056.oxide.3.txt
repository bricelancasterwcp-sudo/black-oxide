fn parse_and_sum(texts: Vec<Str>) -> (Int, Int) {
    let mut sum = 0;
    let mut count_non_numbers = 0;
    for text in texts {
        match parse_int(text) {
            Some(n) => sum = sum + n,
            None => count_non_numbers = count_non_numbers + 1,
        }
    }
    (sum, count_non_numbers)
}

fn main() {
    let texts = vec("7", "x", "9", "?");
    let (total, non_numbers) = parse_and_sum(texts);
    print(total);
    print(non_numbers);
}
