struct Square { side: Int }

struct Circle { radius: Int }

enum Shape {
    Square(Square),
    Circle(Circle),
}

fn area(s: Shape) -> Int {
    match s {
        Shape(s) => match s {
            Square(s) => s.side * s.side,
            Circle(c) => c.radius * c.radius * 3,
        },
    }
}

fn main() {
    let square = Shape(Square { side: 4 })
    print(area(square))
}
