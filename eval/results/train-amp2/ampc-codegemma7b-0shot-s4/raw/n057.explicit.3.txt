fn longest_fruit(fruits: &Vec<String>) -> String {
    let mut longest = fruits.first().unwrap().clone()
    for fruit in fruits {
        if fruit.len() > longest.len() {
            longest = fruit.clone()
        }
    }
    longest
}

    let fruits = vec_str("fig", "banana", "kiwi")
    print(longest_fruit(&fruits))
    print(longest_fruit(&fruits).len())
}
