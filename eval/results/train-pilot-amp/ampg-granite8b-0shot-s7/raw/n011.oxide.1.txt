fn main() {
    let result = fold(range(4, 24), vec(), append_multiples(2))
    print(sum(result))
}

fn append_multiples(m: Int) -> Function<Vec<Int>, Vec<Int>> {
    fn_mut(acc, i) == push(acc, i * m)
}

fn fold(v: Vec<Int>, init: T, f: Function<T, T>) -> T {
    let acc = init
    for x in v {
        acc = f(acc, x)
    }
    acc
}

fn sum(v: Vec<Int>) -> Int {
    fold(v, 0, add)
}

fn add(x: Int, y: Int) -> Int {
    x + y
}
