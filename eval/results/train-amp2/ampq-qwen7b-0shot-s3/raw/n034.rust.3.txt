fn second_element<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}

fn main() {
    let list7 = vec![1, 2, 3, 4, 5, 6, 7];
    let list8 = vec![1, 2];
    let list9 = Vec::<i32>::new();

    match second_element(&list7) {
        Some(value) => println!("Second element: {}", value),
        None => println!("No second element"),
    }

    match second_element(&list8) {
        Some(value) => println!("Second element: {}", value),
        None => println!("No second element"),
    }

    match second_element(&list9) {
        Some(value) => println!("Second element: {}", value),
        None => println!("No second element"),
    }
}
