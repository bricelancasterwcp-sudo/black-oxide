struct Keyboard {
    letters: Vec<Char>,
}

fn count_vowels(keyboard: Keyboard) -> Int {
    let mut num_vowels = 0
    for c in keyboard.letters {
        let is_vowel = match c {
            Char(v) => match v {
                Vowel => true,
                _ => false,
            },
        }
        if is_vowel {
            num_vowels = num_vowels + 1
        }
    }
    num_vowels
}

fn count_consonants(keyboard: Keyboard) -> Int {
    len(keyboard.letters) - count_vowels(keyboard)
}

fn main() {
    let keyboard = Keyboard { letters: push(push(vec(), Char(Vowel)), Char(Consonant)) }
    let num_vowels = count_vowels(keyboard)
    print(num_vowels)
}
