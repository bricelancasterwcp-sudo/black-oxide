struct Square { side: Float }
struct Circle { radius: Float }

fn area(shape: Shape) -> Float {
    match shape {
        Square { side } => side * side,
        Circle { radius } => 3.0 * radius * radius,
    }
}

enum Shape {
    Square(Square),
    Circle(Circle),
}

fn main() {
    let square = Square { side: 4.0 };
    let area_value = area(Shape::Square(square));
    print(area_value);
}
