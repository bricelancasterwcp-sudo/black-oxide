```rust
fn main() {
    let mut signals = vec![0; 3];

    for i in 0..signals.len() {
        match signals[i] {
            1 => println!("Stop"),
            2 => println!("Go"),
            3 => println!("Wait"),
            _ => panic!("Invalid signal: {}", signals[i]),
        }

        if i == 0 {
            signals[i] = 1; // Start with red
        } else if i > 0 && signals[i - 1] == 1 { // Red followed by green
            signals[i] = 2;
        } else if i > 0 && signals[i - 1] == 3 { // Yellow followed by red
            signals[i] = 1; // Stop as a signal change
        }
    }
}
```
