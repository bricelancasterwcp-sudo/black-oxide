fn reverse_digits(n: Int) -> Str {
    let s = int_to_str(n)
    let mut chars = chars(s)
    let mut reversed = ""
    while !chars.is_empty() {
        let last_char = chars.pop().unwrap()
        reversed = concat(reversed, last_char)
    }
    reversed
}

fn main() {
    print(reverse_digits(1234))
}
