fn divide(a: Float, b: Float) -> Result<Float, String> {
    if b == 0.0 {
        Err("Division by zero!".to_string())
    } else {
        Ok(a / b)
    }
}

fn main() {
    let result = divide(20.0, 4.0).unwrap_or_else(println)
    print(result)
    let result = divide(20.0, 0.0).unwrap_or_else(println)
    print(result)
}

fn println(msg: &Str) {
    print(msg)
    print("\n")
}

fn print(msg: &Str) {
    for c in msg.chars() {
        print(c)
    }
}

fn print(n: Int) {
    if n >= 10 {
        print(n / 10)
        print(n % 10)
    } else {
        print(n)
    }
}

fn print(f: Float) {
    let s = int_to_str(trunc(f * 100.0))
    print(".")
    print(s)
}
