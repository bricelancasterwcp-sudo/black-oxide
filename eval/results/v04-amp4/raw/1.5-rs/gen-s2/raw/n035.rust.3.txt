```rust
fn divide(a: i32, b: i32) -> Result<f64, &'static str> {
    if b == 0 {
        Err("Divide by zero")
    } else {
        Ok((a as f64 / b as f64).round())
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("{} / {} = {}", 20, 4, result),
        Err(msg) => println!("{}", msg),
    }
}
```
