fn main() {
    let s = "keyboard";
    let mut vowels = 0;
    let mut consonants = 0;

    for c in s.chars() {
        match c.to_lowercase().next() {
            Some('a' | 'e' | 'i' | 'o' | 'u') => vowels += 1,
            _ => consonants += 1
        }
    }

    print(vowels);
    print_str(" ");
    print(consonants);
}
