```rust
fn main() {
    let mut rect = Rectangle { width: 2, height: 3 };
    println!("Before change: {}", rect.area());
    
    // Change width to 5
    rect.width = 5;
    
    println!("After change: {}", rect.area());
}

struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    fn area(&self) -> u32 {
        self.width * self.height
    }
}
```
