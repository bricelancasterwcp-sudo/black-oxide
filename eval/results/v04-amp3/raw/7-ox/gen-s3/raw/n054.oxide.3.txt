fn main() {
    let s = "keyboard";
    let mut count_vowels = 0;
    let mut count_non_vowels = 0;

    for c in s.chars() {
        match c.to_lowercase().next() {
            Some('a' | 'e' | 'i' | 'o' | 'u') => count_vowels += 1,
            _ => count_non_vowels += 1,
        }
    }

    print(count_vowels);
    print_str(", ");
    print(count_non_vowels);
}
