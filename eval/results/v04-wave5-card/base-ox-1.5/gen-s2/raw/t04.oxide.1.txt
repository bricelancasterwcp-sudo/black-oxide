fn main() {
    let mut count = 0;
    for i in 2..100 {
        if is_prime(i) {
            count += 1;
        }
    }
    print(count);
}

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return False;
    }
    for i in 2..(n as f64).sqrt() as Int {
        if n % i == 0 {
            return False;
        }
    }
    True
}
