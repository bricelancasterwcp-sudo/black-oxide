fn main() {
    let item1_price = 3;
    let item2_price = 2;

    // Calculate the total price for four units of each type
    let total_price_line_1 = 4 * item1_price + 5 * item2_price;

    // Print the value of the first line
    println!("The value of the first line: {}", total_price_line_1);

    // Calculate the total value for both lines combined
    let total_price_combined = total_price_line_1 * 2;

    // Print the value of both lines together
    println!("The total value of both lines combined: {}", total_price_combined);
}
