fn sum_and_count_digits(s: Str) -> (Int, Int) {
    let mut sum = 0;
    let mut count = 0;
    for c in chars(s) {
        match parse_int(c) {
            Some(n) => { sum += n; count += 1 },
            None => ()
        }
    }
    (sum, count)
}

fn main() {
    let s = "a1b22";
    let (total, count) = sum_and_count_digits(s);
    print(total); print_str("\n"); print(count);
}
