struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 5 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    match get(v, 1) {
        Some(n) if n + 1 <= 9 => None,
        Some(n) => Some(n + 1),
    }
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) };
    println(first_big(r.values));
    println(second(r.values).unwrap_or(0));
}
