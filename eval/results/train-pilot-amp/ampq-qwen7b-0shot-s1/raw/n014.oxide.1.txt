fn min_in_set(v: Vec<Int>) -> Int {
    let mut min = v.pop().unwrap()
    for x in v {
        if x < min {
            min = x
        }
    }
    min
}

fn main() {
    let readings = push(push(push(push(vec(), 5), 3), 9), 1)
    let readings = push(readings, 7)
    print(min_in_set(readings))
}
