fn main() {
    let mut counter = 9
    let mut total = 0
    let mut halved = 0
    let mut added_five = 0
    let mut tripled = 0
    let mut max_value = 9
    let mut max_round = 1
    let mut min_value = 9
    let mut below_ten_count = 0
    for round in range(1, 31) {
        if counter % 2 == 0 {
            counter /= 2
            halved += 1
        } else if counter % 3 == 0 {
            counter += 5
            added_five += 1
        } else {
            counter = (counter * 3 + 1) % 1000
            tripled += 1
        }
        total += counter
        if counter > max_value {
            max_value = counter
            max_round = round
        }
        if counter < min_value {
            min_value = counter
        }
        if counter < 10 {
            below_ten_count += 1
        }
    }
    print(counter)
    print(halved)
    print(added_five)
    print(tripled)
    print(max_value)
    print(max_round)
    print(min_value)
    print(total)
    print(below_ten_count)
    if counter > 100 {
        print_str("high")
    } else {
        print_str("low")
    }
}
