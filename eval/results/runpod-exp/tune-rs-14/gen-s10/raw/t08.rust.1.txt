fn main() {
    let v = vec![3, 8, -2, 12, 7];
    let mut pos_count = 0;
    let mut max_val = i32::MIN;
    for &x in &v {
        if x > 0 {
            pos_count += 1;
        }
        if x > max_val {
            max_val = x;
        }
    }
    println!("{}", pos_count);
    println!("{}", max_val);
}
