fn main() {
    let mut texts = vec("7", "x", "9", "?")
    let mut sum = 0
    let mut count = 0
    for text in &muttexts {
        if letOk(num) == parse_int(text) {
            sum = sum + num
            count = count + 1
        } else {
            texts.remove(text)
        }
    }
    print_str("Sum: ")
    print(sum)
    print_str("\nCount: ")
    print(count)
}
