fn log_start_end(log: Vec<Int>) -> Int {
    let start = get(log, 0)?
    let end = get(log, len(log) - 1)?
    print(start)
    print(end)
    start
}

fn main() {
    let log = vec(6, 7, 8, 9)
    let first = log_start_end(log)
    match second(log) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    print(first_big(r.values))            // field access copies: r stays usable
    let r2 = Reading { label: "lab2", ..r }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }
}
