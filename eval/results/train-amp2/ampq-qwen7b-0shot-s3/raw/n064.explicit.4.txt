fn get_value(v: &Vec<Int>, i: Int) -> Option<Int> {
    get(v, i)
}

fn main() {
    let nums = vec(10, 20)
    let first = get_value(&nums, 1)
    let second = get_value(&nums, 4)
    print_str("First: ")
    print(first.map_or("None", to_float))
    print_str("\nSecond: ")
    print(second.map_or("None", to_float))
    drop nums
}
