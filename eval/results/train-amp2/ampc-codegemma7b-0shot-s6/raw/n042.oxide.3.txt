    let mut v = vec(12, 5, 20, 8, 17)
    v.sort_by_key(std_cmp_int)
    print(get(v, len(v) - 1)?)
    print(get(v, len(v) - 2)?)
}

fn std_cmp_int(i: Int, j: Int) -> Ordering {
    match i.cmp(j) {
        Ordering(lt) => lt,
        Ordering(gt) => gt,
        Ordering(eq) => eq,
    }
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < len(v) {
        Some(v.get(i))
    } else {
        None
    }
}

fn len(v: Vec<Int>) -> Int {
    v.len()
}

enum Ordering {
    Lt,
    Gt,
    Eq,
}

fn cmp(i: Int, j: Int) -> Ordering {
    if i < j {
        Ordering(Lt)
    } else if i > j {
        Ordering(Gt)
    } else {
        Ordering(Eq)
    }
}

fn std_cmp_int(i: Int, j: Int) -> Ordering {
    cmp(i, j)
}
