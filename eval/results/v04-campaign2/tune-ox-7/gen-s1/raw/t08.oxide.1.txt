struct Reading { label: Str, values: Vec<Int> }
fn count_positive(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count += 1
        }
    }
    count
}
fn max_value(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        match max {
            Some(current_max) => {
                if x > current_max {
                    max = Some(x)
                }
            },
            None => {
                max = Some(x)
            },
        }
    }
    max
}
fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 3), 8), -2), 12) }
    let r2 = Reading { label: "lab2", values: push(push(push(push(vec(), 7), 4), -5), 9) }
    print(count_positive(r.values))
    match max_value(r.values) {
        Some(n) => print(n),
        None => print_str("no values"),
    }
    print(count_positive(r2.values))
    match max_value(r2.values) {
        Some(n) => print(n),
        None => print_str("no values"),
    }
}
