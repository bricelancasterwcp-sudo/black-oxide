struct Letter {
    char: Str,
    count: Int,
}

fn find_duplicates(s: Str) -> Vec<Letter> {
    let mut letters = vec()
    let mut duplicates = vec()
    let mut new_duplicates = vec()

    for c in chars(s) {
        let found = false
        for l in letters {
            if l.char == c {
                found = true
                break
            }
        }
        if found {
            let mut found_in_dups = false
            for l in duplicates {
                if l.char == c {
                    let new_l = Letter { char: c, count: l.count + 1 }
                    new_duplicates = push(new_duplicates, new_l)
                    found_in_dups = true
                    break
                }
            }
            if !found_in_dups {
                let l = Letter { char: c, count: 2 }
                new_duplicates = push(new_duplicates, l)
            }
        } else {
            let l = Letter { char: c, count: 1 }
            letters = push(letters, l)
        }
    }

    duplicates = append(duplicates, new_duplicates)
    duplicates
}

fn main() {
    let word = "coffee"
    let duplicates = find_duplicates(word)
    for d in duplicates {
        if d.count > 1 {
            print(d.char)
        }
    }
}
