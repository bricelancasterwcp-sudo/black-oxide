fn main() {
    let rect = Point { x: 2, y: 3 }
    let rect = Point { x: 5, ..rect }
    print_str(&rect)
}

struct Point {
    x: Int,
    y: Int,
}

fn print_str(s: &Str) {
    let mut output = String.new()
    for ch in s.chars() {
        output.push_str(&ch)
    }
    drop output
}

fn clone(x: &Point) -> Point {
    Point(x.x, x.y)
}

fn main() {
    let rect = Point { x: 2, y: 3 }
    let rect = Point { x: 5, ..rect }
    print_str(&rect)
}
