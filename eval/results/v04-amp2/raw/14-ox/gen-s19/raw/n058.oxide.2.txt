struct LetterCount { letter: Str, count: Int }

fn find_repeated_letters(s: Str) -> Vec<Str> {
    let mut counts = vec()
    let mut result = vec()

    for c in chars(s) {
        let found = false
        for i in range(0, len(counts)) {
            if get(counts, i)?.letter == c {
                let count = get(counts, i)?.count + 1
                counts = push(counts, LetterCount { letter: c, count: count })
                found = true
                break
            }
        }
        if !found {
            counts = push(counts, LetterCount { letter: c, count: 1 })
        }
    }

    for item in counts {
        if item.count > 1 {
            result = push(result, item.letter)
        }
    }

    result
}

fn main() {
    let word = "coffee"
    let repeated_letters = find_repeated_letters(word)
    for letter in repeated_letters {
        print_str(letter)
    }
}
