```rust
fn main() {
    let a1b22 = "a1b22";
    
    let mut num_count = 0;
    let mut total_digit_sum = 0;

    for c in a1b22.chars() {
        if c.is_digit(10) {
            total_digit_sum += c.to_digit(10).unwrap();
            num_count += 1;
        }
    }

    println!("Total digit sum: {}", total_digit_sum);
    println!("Number of digits: {}", num_count);
}
```
