```rust
use rand::Rng; // Import the Rng trait from the rand crate

fn main() {
    loop {
        let mut rng = rand::thread_rng(); // Create a thread-local random number generator
        let signal = rng.gen_range(0..=2); // Generate a random number between 0 and 2 (inclusive)
        match signal {
            0 => println!("Go"),         // Green light: go
            1 => println!("Stop"),       // Red light: stop
            2 => println!("Wait"),      // Yellow light: wait
            _ => unreachable!(),      // Unreachable case, should never happen
        }
    }
}
```

To compile the program with Rust, run `rustc program.rs`.
