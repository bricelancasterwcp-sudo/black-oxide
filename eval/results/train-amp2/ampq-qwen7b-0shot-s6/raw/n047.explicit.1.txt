fn main() {
    let mut even_nums: Vec<Int> = vec()
    for i in range(1, 11) {
        if i % 2 == 0 {
            push(even_nums, i)
        }
    }
    let mut i = len(even_nums) - 1
    while i >= 0 {
        print(even_nums.get(i).unwrap())
        print_str("\n")
        i = i - 1
    }
    drop even_nums
}
