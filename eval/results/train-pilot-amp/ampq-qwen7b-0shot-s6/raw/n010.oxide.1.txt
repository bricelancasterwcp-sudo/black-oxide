struct PrimeCheck { n: Int }

fn is_prime(pc: PrimeCheck) -> Bool {
    let PrimeCheck { n } = pc
    if n <= 1 {
        return False
    }
    for i in range(2, n) {
        if n % i == 0 {
            return False
        }
    }
    True
}

fn main() {
    let pc = PrimeCheck { n: 91 }
    if is_prime(pc) {
        print_str("yes")
    } else {
        print_str("no")
    }
}
