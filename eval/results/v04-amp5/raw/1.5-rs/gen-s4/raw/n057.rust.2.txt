fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut max_name = &names[0];
    let mut max_length = names[0].len();

    for name in names.iter().skip(1) {
        if name.len() > max_length {
            max_name = name;
            max_length = name.len();
        }
    }

    println!("The name with the most letters is '{}' with {} characters.", max_name, max_length);
}
