```rust
fn main() {
    let side = 4;
    let radius = 3.0;

    println!("Area of square: {}", area_square(side));
    println!("Area of circle: {:.2}", area_circle(radius));
}

fn area_square(side: u32) -> f64 {
    side as f64 * side as f64
}

fn area_circle(radius: f64) -> f64 {
    radius * radius * std::f64::consts::PI
}
```
