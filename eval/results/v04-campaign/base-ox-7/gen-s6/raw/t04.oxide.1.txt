fn main() {
    let mut count = 0;
    for num in range(2, 100) {
        if is_prime(num) {
            count += 1;
        }
    }
    print(count);
}

fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false;
    }
    for i in range(2, trunc(to_float(n).sqrt()) + 1) {
        if n % i == 0 {
            return false;
        }
    }
    true
}
