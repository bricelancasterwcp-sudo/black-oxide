```rust
fn main() {
    let mut count = 0;
    let mut sum = 0;

    for &value in [7, 'x', 9, '?'].iter() {
        match value {
            7 => sum += 7,
            x if x.is_digit(10) => sum += x.to_digit(10).unwrap(),
            _ => count += 1,
        }
    }

    println!("Sum: {}", sum);
    println!("Count of non-numeric values: {}", count);
}
```
