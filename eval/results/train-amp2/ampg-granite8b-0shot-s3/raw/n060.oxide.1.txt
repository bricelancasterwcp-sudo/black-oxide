fn main() {
    let s = "a1b22"
    let mut total = 0
    let mut digit_count = 0
    for c in s.chars() {
        if c.is_ascii_digit() {
            digit_count = digit_count + 1
            total = total + c.to_digit(10).unwrap()
        }
    }
    print_str(total.to_string())
    print_str(digit_count.to_string())
}
