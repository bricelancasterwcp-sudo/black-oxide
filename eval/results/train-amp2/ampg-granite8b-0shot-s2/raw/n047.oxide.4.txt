fn collect_evens() -> Vec<Int> {
    let mut evens = vec()
    for i in 1.upto(10) {
        if i % 2 == 0 {
            evens = push(evens, i)
        }
    }
    evens
}

fn print_in_reverse(v: Vec<Int>) {
    for i in (0.until(len(v))).rev() {
        print(get(v, i)?)
    }
}

fn main() {
    let evens = collect_evens()
    print_in_reverse(evens)
}
