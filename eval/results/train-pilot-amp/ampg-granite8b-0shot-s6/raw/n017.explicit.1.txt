fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < 0 || i >= len(&v) {
        None
    } else {
        Some(get_helper(&v, i))
    }
}

fn get_helper(v: &Vec<Int>, i: Int) -> Int {
    let x = get(v, i - 1)
    if x == None {
        v.get(i)
    } else {
        x.unwrap()
    }
}

    let list = push(push(push(vec(), 10), 20), 30)
    let result = get(&list, 1)
    match result {
        None => print("missing"),
        Some(x) => print(x) < println("missing"),
    }
}
