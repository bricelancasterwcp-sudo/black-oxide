struct Digit { value: Int }

fn digit_sum(n: Int) -> Int {
    let v = int_to_str(n)
    let mut sum = 0
    for c in chars(v) {
        let d = parse_int(c)?
        sum += d
    }
    Some(sum)
}

fn main() {
    let num = 9875
    match digit_sum(num) {
        Some(sum) => print(sum),
        None => print_str("Invalid number"),
    }
}
