fn divide(a: f64, b: f64) -> Result<f64, &'static str> {
    if b == 0.0 {
        Err("Cannot divide by zero")
    } else {
        Ok(a / b)
    }
}

fn main() {
    let result = divide(20.0, 4.0);
    match result {
        Ok(answer) => println!("Twenty divided by four is {}", answer),
        Err(error) => println!("Error: {}", error),
    }
}
