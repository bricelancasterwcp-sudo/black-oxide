struct Reading { label: Str, values: Vec<Int> }

fn count_numbers(v: Vec<Str>) -> (Int, Int) {
    let mut sum = 0
    let mut count_non_numeric = 0
    for s in v {
        match parse_int(s) {
            Some(n) => sum += n,
            None => count_non_numeric += 1,
        }
    }
    (sum, count_non_numeric)
}

fn main() {
    let readings = Vec<Str> { "7", "x", "9", "?" }
    let (total, non_numeric_count) = count_numbers(readings)
    print(total)
    print(non_numeric_count)
}
