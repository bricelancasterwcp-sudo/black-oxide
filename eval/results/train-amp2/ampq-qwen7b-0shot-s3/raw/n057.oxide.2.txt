fn main() {
    let fig_len = "fig".len()
    let banana_len = "banana".len()
    let kiwi_len = "kiwi".len()
    let fig = fig_len + 3 * banana_len + 3 * kiwi_len
    let max_len = fig
    let max_word = if max_len == fig { "fig" } else { if max_len == 3 * banana_len + 3 * kiwi_len { "banana" } else { "kiwi" } }
    print(max_word)
    print(max_len)
}
