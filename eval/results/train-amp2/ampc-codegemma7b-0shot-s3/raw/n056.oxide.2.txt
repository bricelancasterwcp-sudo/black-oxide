fn main() {
    let seven = 7.to_string().parse().ok().transpose()?.unwrap_or(0)
    let nine = 9.to_string().parse().ok().transpose()?.unwrap_or(0)
    let x = "".parse().ok().transpose()?.unwrap_or(0)
    let question = "".parse().ok().transpose()?.unwrap_or(0)
    print(seven + nine + x + question)
    print(4 - (seven + nine + x + question))
}
