```rust
fn main() {
    let mut v = vec![3, 8, -2, 12, 7];
    let count_positive = v.iter().filter(|&&x| x > 0).count();
    let largest_positive = *v.iter().filter(|&&x| x > 0).max().unwrap_or(&-1);
    println!("Number of positive values: {}", count_positive);
    println!("Largest positive value: {}", largest_positive);
}
```
