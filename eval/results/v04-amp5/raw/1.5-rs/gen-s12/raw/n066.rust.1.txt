```rust
fn main() {
    let numbers = vec![9, 2, 12, 5, 16];
    let count = numbers.into_iter().filter(|&num| num < 10).count();
    println!("Kept entries:");
    for &num in &numbers {
        if num < 10 {
            println!("{}", num);
        }
    }
    println!("Total kept: {}", count);
}
```
