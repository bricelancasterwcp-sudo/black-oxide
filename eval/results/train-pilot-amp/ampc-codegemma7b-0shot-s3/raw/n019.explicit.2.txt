fn count_entries(list1: &Vec<Int>, list2: &Vec<Int>) -> Int {
    let mut count = 0
    for x in list1 {
        if x >= &list2 {
            count = count + 1
        }
    }
    for y in list2 {
        if y >= &list1 {
            count = count + 1
        }
    }
    count
}

fn main() {
    let list1 = vec(1, 2, 3)
    let list2 = vec(4, 5)
    print(count_entries(&list1, &list2))
    drop list1
    drop list2
}
