struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding_five(v: Vec<Int>) -> Int {
    count_if(v, |x| x > 5)
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    max(v)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 6), 11), 3), 14) }
    print(count_exceeding_five(clone(r.values)))
    let max_value = match max_reading(clone(r.values)) {
        Some(n) => n,
        None => -1
    }
    print(max_value)
}
