fn sum_and_count_digits(s: Str) -> Int {
    let mut sum = 0
    let mut count = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            sum = sum + parse_int(c).unwrap()
            count = count + 1
        }
    }
    (sum * 100) + count
}

fn main() {
    let s = "a1b22"
    let result = sum_and_count_digits(s)
    print(result / 100)
    print_str("\n")
    print(result % 100)
}
