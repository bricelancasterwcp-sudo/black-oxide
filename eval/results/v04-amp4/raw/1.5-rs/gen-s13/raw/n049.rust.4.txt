```rust
fn main() {
    let numbers = vec![8, 3, 11, 6];
    let min_value = *numbers.iter().min().unwrap();
    let max_value = *numbers.iter().max().unwrap();
    println!("The smallest figure: {}", min_value);
    println!("The gap between the largest and the smallest: {}", max_value - min_value);
}
```
