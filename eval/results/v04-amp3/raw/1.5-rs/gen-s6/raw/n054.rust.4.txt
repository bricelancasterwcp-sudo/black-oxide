fn count_vowels_and_consonants(word: &str) {
    let mut vowel_count = 0;
    let mut consonant_count = 0;

    for c in word.chars() {
        if c.is_ascii_lowercase() && matches!(c, 'a' | 'e' | 'i' | 'o' | 'u') || 
           c.is_ascii_uppercase() && matches!(c, 'A' | 'E' | 'I' | 'O' | 'U') {
            vowel_count += 1;
        } else if !matches!(c, ' ') && !c.is_whitespace() {
            consonant_count += 1;
        }
    }

    println!("Vowels: {}", vowel_count);
    println!("Consonants: {}", consonant_count);
}

fn main() {
    let word = "hello world";
    count_vowels_and_consonants(word);
}
