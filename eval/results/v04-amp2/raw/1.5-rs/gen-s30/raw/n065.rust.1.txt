```rust
fn main() {
    let readings = [6, 11, 3, 14, 9];
    let count_exceed_five = readings.iter().filter(|&&x| x > 5).count();
    println!("{} of the readings exceed five", count_exceed_five);

    let max_reading = *readings.iter().max().unwrap();
    println!("The highest reading is {}", max_reading);
}
```
