fn main() {
    let text = "order 47 shipped on 2026-03-15, invoice 1290 paid."
    let chars = str_len(text)
    let letters = count_if(str_chars(text), |c| is_ascii_alpha(c))
    let digits = count_if(str_chars(text), |c| is_ascii_digit(c))
    let spaces = count_if(str_chars(text), |c| c == ' ')
    let others = chars - (letters + digits + spaces)
    let max_digits_streak = max_streak(str_chars(text), |c| is_ascii_digit(c))
    let all_digits = concat(filter(str_chars(text), |c| is_ascii_digit(c)))
    print(chars)
    print(letters)
    print(digits)
    print(spaces)
    print(others)
    print(max_digits_streak)
    print_str(all_digits)
}
