fn enqueue(v: &Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn first(v: &Vec<Int>) -> Int {
    get(v, 0).unwrap()
}

fn last(v: &Vec<Int>) -> Int {
    get(v, len(v) - 1).unwrap()
}

fn swap(v: &Vec<Int>, i: Int, j: Int) -> Vec<Int> {
    let a = get(v, i).unwrap()
    let b = get(v, j).unwrap()
    set(v, i, b)
    set(v, j, a)
    v.clone()
}

fn main() {
    let mut queue = vec(1, 2, 3, 4, 5)
    queue = swap(&queue, 0, len(&queue) - 1)
    for x in &queue {
        print(x)
        print("\n")
    }
    drop queue
}

fn set(v: &Vec<Int>, i: Int, x: Int) -> Vec<Int> {
    let mut result = v.clone()
    result.insert(i, x)
    result.remove(i + 1)
    result
}
