fn div(n: Int, d: Int) -> Option<Int> {
    if d == 0 {
        None
    } else {
        Some(n / d)
    }
}
fn main() {
    match div(100, 5) {
        Some(q) => print(q),
        None => print_str("undefined"),
    }
    match div(100, 0) {
        Some(q) => print(q),
        None => print_str("undefined"),
    }
    match div(100, 4) {
        Some(q) => print(q),
        None => print_str("undefined"),
    }
}
