fn divide(a: i32, b: i32) -> Result<f64, &'static str> {
    if b == 0 {
        Err("Cannot divide by zero")
    } else {
        Ok((a as f64) / (b as f64))
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("{}", result),
        Err(e) => println!("bad: {}", e),
    }

    match divide(20, 0) {
        Ok(_) => panic!("This should not happen"),
        Err(e) => println!("bad: {}", e),
    }
}
