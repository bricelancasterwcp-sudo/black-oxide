```rust
struct Rectangle { width: Int, height: Int }
struct Triangle { side1: Int, side2: Int, side3: Int }

fn perimeter_rectangle(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn perimeter_triangle(t: Triangle) -> Int {
    t.side1 + t.side2 + t.side3
}

fn main() {
    let shapes = vec(
        Rectangle { width: 3, height: 4 },
        Triangle { side1: 3, side2: 4, side3: 5 },
        Rectangle { width: 2, height: 5 }
    );

    for shape in shapes {
        match shape {
            Rectangle { width, height } => print(perimeter_rectangle(Rectangle { width, height })),
            Triangle { side1, side2, side3 } => print(perimeter_triangle(Triangle { side1, side2, side3 }))
        }
    }
}
```
