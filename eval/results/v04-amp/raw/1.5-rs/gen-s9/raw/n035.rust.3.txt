```rust
fn divide(a: i32, b: i32) -> Result<f64, &'static str> {
    if b == 0 {
        Err("Cannot divide by zero")
    } else {
        Ok(f64::from(a) / f64::from(b))
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("The result is {}", result),
        Err(error) => println!("{}", error),
    }
}
```
