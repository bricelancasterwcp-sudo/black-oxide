fn a1b22(s: Str) -> Int {
    let mut total = 0
    let mut digits = 0
    let mut i = 0
    while i < str_len(&s) {
        let c = chars(&s).get(i)
        if c >= "0" && c <= "9" {
            total = total + c.to_int()
            digits = digits + 1
        }
        i = i + 1
    }
    print(total)
    print(digits)
}

fn main() {
    a1b22("a1b22")
}
