fn main() {
    let v = vec![7, 'x'.to_string(), 9, "?".to_string()];
    let mut sum = 0;
    let mut count_non_numbers = 0;

    for item in v {
        match parse_int(item) {
            Some(num) => sum += num,
            None => count_non_numbers += 1,
        }
    }

    print(sum);
    print(count_non_numbers);
}
