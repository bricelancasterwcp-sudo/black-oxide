fn main() {
    let fig = match figure() {
        Figure8 => "8",
        Figure3 => "3",
        Figure11 => "11",
        Figure6 => "6",
    }
    let mut prev = fig
    for fig in figure_sequence() {
        print_str(prev)
        print_str(" ")
        prev = match fig {
            Figure8 => "8",
            Figure3 => "3",
            Figure11 => "11",
            Figure6 => "6",
        }
    }
}

enum Figure {
    Figure3,
    Figure6,
    Figure8,
    Figure11,
}

fn figure() -> Figure {
    let n = rand_int(12)
    match n {
        Rng(i) => match i % 4 {
            Rng(i) => match i {
                Rng(i) => match i {
                    Rng(i) => match i {
                        Rng(i) => match i {
                            Rng(i) => match i {
                                Rng(i) => match i {
                                    Rng(i) => match i {
                                        Rng(i) => match_figure()._1,
                                        Rng(i) => match_figure()._1,
                                        Rng(i) => match_figure()._1,
                                    },
                                    Rng(i) => match_figure()._1,
                                },
                                Rng(i) => match_figure()._1,
                            },
                            Rng(i) => match_figure()._1,
                        },
                    },
                },
            },
        },
    }
}

fn figure_sequence() -> Vec<Figure> {
    let mut figs = vec()
    loop_infinitely(figs, figure(), figs.push)
}

enum Range {
    Rng(Int),
}

fn rand_int(n: Int) -> Range {
    Range(0)
}
