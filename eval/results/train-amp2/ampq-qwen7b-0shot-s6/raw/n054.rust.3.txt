fn main() {
    let word = "keyboard";
    let vowels = "aeiouAEIOU";
    let mut count_vowels = 0;
    let mut count_non_vowels = 0;

    for letter in word.chars() {
        if vowels.contains(letter) {
            count_vowels += 1;
        } else {
            count_non_vowels += 1;
        }
    }

    println!("Vowels: {}", count_vowels);
    println!("Non-vowels: {}", count_non_vowels);
}
