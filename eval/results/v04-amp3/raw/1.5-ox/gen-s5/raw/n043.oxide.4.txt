```rust
struct Queue<T> {
    values: VecDeque<T>,
}

impl<T> Queue<T> {
    fn new() -> Self {
        Queue { values: VecDeque::new() }
    }

    fn push(&mut self, item: T) {
        self.values.push_back(item);
    }

    fn pop(&mut self) -> Option<T> {
        self.values.pop_front()
    }

    fn first(&self) -> Option<&T> {
        self.values.front()
    }

    fn last(&self) -> Option<&T> {
        self.values.back()
    }
}

fn main() {
    let mut q = Queue::new();
    for i in 1..=5 {
        q.push(i);
    }

    if let Some(first) = q.first() {
        println!("{}", first);

        if let Some(last) = q.last() {
            let temp = last.clone(); // Move the last element to a temporary variable
            q.pop();
            q.push(temp); // Replace the popped last element with the temporary

            while !q.is_empty() {
                print!("{}", q.pop().unwrap());
            }
        } else {
            println!("The queue is empty");
        }
    }
}
```
