```rust
fn main() {
    let s = "keyboard".to_string();
    let mut vowel_count = 0;
    let mut consonant_count = 0;

    for c in s.chars() {
        match c.to_ascii_lowercase() {
            'a' | 'e' | 'i' | 'o' | 'u' => vowel_count += 1,
            _ => consonant_count += 1,
        }
    }

    println!("Number of vowels: {}", vowel_count);
    println!("Number of consonants: {}", consonant_count);
}
```
