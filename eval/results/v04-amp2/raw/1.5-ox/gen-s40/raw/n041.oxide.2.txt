fn main() {
    let mut v = vec![9, 2, 7, 4];
    for i in 0..v.len() - 1 {
        for j in (i + 1)..v.len() {
            if v[i] > v[j] {
                let temp = v[i];
                v[i] = v[j];
                v[j] = temp;
            }
        }
    }
    for x in &v {
        print!("{}", x);
        println!();
    }
}
