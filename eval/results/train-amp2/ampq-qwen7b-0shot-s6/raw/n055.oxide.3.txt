fn parse_int(s: Str) -> Option<Int> {
    let mut chars = chars(s)
    if chars.len() == 0 {
        return None
    }
    let mut negative = false
    if chars.get(0) == Some("-") {
        negative = true
        chars = chars.slice(1, chars.len())
    }
    let mut result = 0
    for char in chars {
        result = result * 10 + char.to_int()
    }
    if negative {
        Some(-result)
    } else {
        Some(result)
    }
}
