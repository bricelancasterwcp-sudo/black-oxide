struct Reading { label: Str, values: Vec<Int> }

fn count_above_five(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 5 {
            count += 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    match max(v) {
        Some(n) => Some(n),
        None => None,
    }
}

fn main() {
    let r = Reading { label: "lab", values: vec![6, 11, 3, 14, 9] };
    print(count_above_five(r.values))
    print_str("\n")
    match max_reading(r.values.clone()) {
        Some(n) => print(n),
        None => print_str("empty"),
    }
}
