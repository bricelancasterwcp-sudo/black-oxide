struct Square { side: Int }

struct Circle { radius: Float }

fn area(s: Square) -> Float {
    return s.side * s.side
}

fn area(c: Circle) -> Float {
    return 3.14159 * c.radius * c.radius
}

fn main() {
    let s = Square { side: 4 }
    print(area(s))
    let c = Circle { radius: 5.0 }
    print(area(c))
}
