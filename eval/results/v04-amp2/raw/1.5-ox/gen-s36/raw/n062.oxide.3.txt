```rust
fn main() {
    let mut state = "green";
    loop {
        match state {
            "green" => println!("go"),
            "red" => println!("stop"),
            "yellow" => println!("wait"),
            _ => panic!("unexpected state: {}", state),
        }
        // Simulate arrival of signals
        if state == "green" && rand::random::<bool>() {
            state = "red";
        } else if state == "red" && rand::random::<bool>() {
            state = "yellow";
        }
    }
}
```
