fn merge(v: Vec<Int>, w: Vec<Int>) -> Vec<Int> {
    let mut result = vec()
    let mut i = 0
    let mut j = 0
    while i < len(v) && j < len(w) {
        if v[i] <= w[j] {
            result = push(result, v[i])
            i += 1
        } else {
            result = push(result, w[j])
            j += 1
        }
    }
    while i < len(v) {
        result = push(result, v[i])
        i += 1
    }
    while j < len(w) {
        result = push(result, w[j])
        j += 1
    }
    result
}
fn main() {
    let a = vec(2, 5, 9, 14, 21, 30)
    let b = vec(1, 5, 8, 15, 22)
    let merged = merge(a, b)
    print(len(merged))
    print(get(merged, 0).unwrap_or(-1))
    print(get(merged, len(merged) - 1).unwrap_or(-1))
    let sum = fold(merged, 0, |acc, x| acc + x)
    print(sum)
    let is_sorted = all(range(1, len(merged)), |i| merged[i] >= merged[i - 1])
    print(is_sorted)
    let count_equal = count(range(0, len(merged) - 1), |i| get(merged, i).unwrap_or(-1) == get(merged, i + 1).unwrap_or(-1))
    print(count_equal)
}
