fn count_exceeds_five(readings: Vec<Int>) -> Int {
    let mut count = 0
    let mut max_reading = 0
    for reading in &readings {
        if reading > &max_reading {
            max_reading = reading.clone()
        }
        if reading > &count {
            count = reading.clone()
        }
    }
    count
}

fn main() {
    let readings = push(push(vec(), 6), push(push(push(vec(), 11), 3), push(push(push(vec(), 14), 9), vec())))
    print(count_exceeds_five(&readings))
    drop readings
}
