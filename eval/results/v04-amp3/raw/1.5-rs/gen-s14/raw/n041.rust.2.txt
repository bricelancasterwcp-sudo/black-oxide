fn main() {
    let mut numbers = [9, 2, 7, 4];

    // Sort the array in ascending order
    for i in 0..numbers.len() {
        for j in (i + 1)..numbers.len() {
            if numbers[i] > numbers[j] {
                // Swap the elements if they are out of order
                let temp = numbers[i];
                numbers[i] = numbers[j];
                numbers[j] = temp;
            }
        }
    }

    // Print each number in sorted order
    for &num in &numbers {
        println!("{}", num);
    }
}
