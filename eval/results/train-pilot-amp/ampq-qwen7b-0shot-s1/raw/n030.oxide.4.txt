fn count_vowels(word: Str) -> Int {
    let vowels = "aeiouAEIOU"
    let mut count = 0
    for c in chars(word) {
        if vowels.find(c).is_some() {
            count = count + 1
        }
    }
    count
}

fn main() {
    let word = "hello"
    let count = count_vowels(word)
    print(count)
}
