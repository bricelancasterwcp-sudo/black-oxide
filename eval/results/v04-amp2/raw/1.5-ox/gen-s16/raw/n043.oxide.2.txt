```rust
struct Queue {
    data: Vec<Int>,
}

impl Queue {
    fn new() -> Self {
        Queue { data: vec![1, 2, 3, 4, 5] }
    }

    fn exchange_first_last(&mut self) {
        if self.data.len() < 2 {
            panic!("Queue must have at least two elements to exchange");
        }
        let first = self.data.remove(0);
        let last = self.data.pop().unwrap();
        self.data.insert(0, last);
        self.data.push(first);
    }

    fn print(&self) {
        for &value in &self.data {
            print!("{}", value);
        }
        println!();
    }
}

fn main() {
    let mut q = Queue::new();
    q.exchange_first_last();
    q.print();
}
```
