fn area(s: &Square) -> Float {
    s.side * s.side
}

fn area(c: &Circle) -> Float {
    PI * c.radius * c.radius
}

    let circle = Circle { radius: 1.0 }
    print(area(&square))
    print(area(&circle))
}

}
