fn main() {
    let price_four_units = 3;
    let quantity_four_units = 4;
    let price_five_units = 2;
    let quantity_five_units = 5;

    let value_four_units = price_four_units * quantity_four_units;
    let value_five_units = price_five_units * quantity_five_units;
    let total_value = value_four_units + value_five_units;

    println!("Value of the first line: {}", value_four_units);
    println!("Value of the second line: {}", value_five_units);
    println!("Total value of both lines: {}", total_value);
}
