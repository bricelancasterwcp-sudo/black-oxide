    let fruits = vec("fig", "banana", "kiwi")
    let longest = find_longest(fruits)
    if longest.len() > 3 {
        print(longest)

fn find_longest(v: Vec<Str>) -> Str {
    let mut longest = ""
    for s in v {
        if s.len() > longest.len() {
            longest = s
        }
    }
    longest
}
