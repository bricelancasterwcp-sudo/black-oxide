struct List { elements: [Int; 4] }

fn main() {
    let mut list = List { elements: [10, 20, 30, 40] };

    match get(list.elements, 1) {
        Some(n) => print(n),
        None => print_str("position 1 is empty"),
    }

    match get(list.elements, 3) {
        Some(n) => print(n),
        None => print_str("position 4 is empty"),
    }
}

fn get(arr: [Int; 4], index: Int) -> Option<Int> {
    if index < 0 || index >= arr.len() {
        None
    } else {
        Some(arr[index])
    }
}
