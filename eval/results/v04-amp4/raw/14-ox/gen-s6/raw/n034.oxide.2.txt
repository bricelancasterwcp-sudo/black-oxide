fn second_entry(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x)
}

fn main() {
    print_str("Second entry of [7, 8, 9]: ")
    match second_entry(push(push(push(vec(), 7), 8), 9)) {
        Some(n) => print(n),
        None => print_str("None"),
    }
    print_str("\nSecond entry of [7, 8]: ")
    match second_entry(push(push(vec(), 7), 8)) {
        Some(n) => print(n),
        None => print_str("None"),
    }
    print_str("\nSecond entry of [7]: ")
    match second_entry(push(vec(), 7)) {
        Some(n) => print(n),
        None => print_str("None"),
    }
}
