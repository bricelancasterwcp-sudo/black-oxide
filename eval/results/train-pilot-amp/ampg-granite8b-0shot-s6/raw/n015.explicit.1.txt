    let numbers = push(push(vec(), 5), 40)

    print(sum_big(&numbers, 10))
    drop numbers
    print(sum_big(&bigger, 10))
    drop bigger
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut v = vec()
    for i in a.to(b - 1) {
        v = push(v, i)
    }
    v
}

fn parse_int(s: Str) -> Option<Int> {
    match s.parse() {
        Some(n) => Some(n),
        None => None,
    }
}
