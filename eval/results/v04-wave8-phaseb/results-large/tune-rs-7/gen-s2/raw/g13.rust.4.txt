fn main() {
    let mut c = 9;
    let mut t = 0;
    let mut h = 0;
    let mut a = 0;
    let mut m = 0;
    let mut l = 1_000;
    let mut s = 0;
    for i in 1..=30 {
        if c % 2 == 0 {
            c /= 2;
            h += 1;
        } else if c % 3 == 0 {
            c += 5;
            a += 1;
        } else {
            c = (c * 3 + 1) % 1_000;
        }
        t += c;
        if c > m {
            m = c;
            l = i;
        }
        if c < s || i == 1 {
            s = c;
        }
    }
    println!("{}", c);
    println!("{}", h);
    println!("{}", a);
    println!("{}", m);
    println!("{}", l);
    println!("{}", s);
    println!("{}", t);
    println!("{}", if c < 10 { "low" } else { "high" });
}
