struct Line { items: Vec<(Int, Int)> }

fn total(line: &Line) -> Int {
    line.items.iter().map(|&(price, count)| price * count).sum()
}

fn main() {
    let line1 = Line { items: vec![(3, 4), (2, 5)] };
    print(total(&line1));
    print_str("\n");
    let line2 = Line { items: vec![(3, 4), (2, 5)] };
    print(total(&line2));
    print_str("\n");
    print(total(&Line { items: vec![(3, 4), (2, 5)] }));
}
