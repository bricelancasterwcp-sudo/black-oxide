```rust
fn divide(dividend: i32, divisor: i32) -> Result<f64, &'static str> {
    if divisor == 0 {
        Err("Cannot divide by zero")
    } else {
        Ok(dividend as f64 / divisor as f64)
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("Result: {}", result),
        Err(error) => println!("Error: {}", error),
    }
}
```
