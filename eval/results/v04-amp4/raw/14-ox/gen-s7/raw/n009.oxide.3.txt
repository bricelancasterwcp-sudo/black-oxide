fn reverse_digits(n: Int) -> Option<Str> {
    let digits = str_len(int_to_str(n))
    let mut result = ""
    for i in range(0, digits) {
        let d = get(chars(int_to_str(n)), digits - i - 1)?
        result = concat(result, int_to_str(d))
    }
    Some(result)
}

fn main() {
    print_str(reverse_digits(1234)?.unwrap_or(""))
}
