fn main() {
    let numbers = [3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut max_value = i32::MIN;
    for &number in &numbers {
        if number > 0 {
            positive_count += 1;
        }
        if number > max_value {
            max_value = number;
        }
    }
    println!("{}", positive_count);
    println!("{}", max_value);
}
