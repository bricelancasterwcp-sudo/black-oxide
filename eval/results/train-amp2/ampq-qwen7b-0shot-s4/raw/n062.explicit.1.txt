fn signal(s: &Str) -> Str {
    match s {
        Str(b) => match b.as_str() {
            String(c) => match c.as_str() {
                String(g) => match g.as_str() {
                    String(green) => green.to_string(),
                    String(red) => red.to_string(),
                    String(yellow) => yellow.to_string(),
                    _ => String("Unknown"),
                },
                _ => String("Unknown"),
            },
            _ => String("Unknown"),
        },
        _ => String("Unknown"),
    }
}
