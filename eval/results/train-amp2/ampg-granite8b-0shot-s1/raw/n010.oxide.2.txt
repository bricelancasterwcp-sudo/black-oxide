struct Int { value: Int }

fn is_prime(n: Int) -> Bool {
    if n <= Int {
        value = 1
    }
    let mut i = Int { value: 2 }
    while i * i <= n {
        if n % i == Int {
            value = 0
        }
        i = i + 1
    }
    n > i
}

struct Reading { label: Str, values: Vec }

fn first_big(v: Vec) -> Int {
    let found = -1
    let mut i = Int
    while i < len(v) {
        if get(v, i).unwrap() > 10 {
            found = get(v, i).unwrap()
            break
        }
        i = i + 1
    }
    found
}

fn second(v: Vec) -> Option {
    let x = get(v, 1).unwrap()
    Some(x + 1)
}
