fn main() {
    let s = "mississippi";
    let count_s = s.chars().filter(|c| *c == 's').count();
    let unique_chars: Vec<Str> = s.chars().collect::<std::collections::HashSet<_>>().into_iter().map(|c| c.to_string()).collect();
    let count_unique_chars = unique_chars.len();
    print(count_s);
    print_str("\n");
    print(count_unique_chars);
}
