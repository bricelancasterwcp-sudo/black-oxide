fn main() {
    let color = read_color()
    if color == "green" {
        print("go")
    } else if color == "red" {
        print("stop")
    } else if color == "yellow" {
        print("wait")
    }
}

fn read_color() -> String {
    let mut input = String.new()
    std.io.stdin().read_line(&mut).expect("error reading input")
    input.trim().to_lowercase()
}
