    println(highest_reading(vec(6, 11, 3, 14, 9)))
}

fn how_many_over_five(readings: Vec<Int>) -> Int {
    let mut count = 0
    for reading in readings {
        if reading > 5 {
            count = count + 1
        }
    }
    count
}

fn highest_reading(readings: Vec<Int>) -> Int {
    let mut max = 0
    for reading in readings {
        if reading > max {
            max = reading
        }
    }
    max
}
