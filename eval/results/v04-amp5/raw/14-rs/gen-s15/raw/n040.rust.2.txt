enum Shape {
    Square(f32),
    Circle(f32),
}

impl Shape {
    fn area(&self) -> f32 {
        match self {
            Shape::Square(side) => side * side,
            Shape::Circle(radius) => 3.0 * radius * radius, // Using 3 as the value of pi
        }
    }
}

fn main() {
    let square = Shape::Square(4.0);
    println!("Area of the square: {}", square.area());
}
