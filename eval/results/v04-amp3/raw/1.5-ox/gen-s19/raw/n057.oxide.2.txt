```rust
fn main() {
    let names = vec!["fig", "banana", "kiwi"];
    let mut max_name = names[0];
    let mut max_len = max_name.len();
    for &name in &names {
        if name.len() > max_len {
            max_name = name;
            max_len = name.len();
        }
    }
    println!("{} ({})", max_name, max_len);
}
```
