fn list() -> List {
    push(push(vec(), 10), 20)
}

fn at_pos(l: List, pos: Int) -> Option<Int> {
    let mut p = l
    let mut i = 0
    while p != nil {
        if i == pos {
            return get(p, 0)
        }
        p = get(p, 1)
        i = i + 1
    }
    none
}

fn print_opt(o: Option<Int>) {
    match o {
        Some(x) => {
            print(x)
            print("\n")
        },
        None => {
            print("None\n")
        },
    }
}

fn main() {
    let l = list()
    print_opt(at_pos(l, 0))
    drop l
    print_opt(at_pos(l, 1))
}

fn nil() -> List {
    nil
}

fn none() -> Option<Int> {
    none
}

fn get(l: List, i: Int) -> List {
    get(l, i)
}

fn push(l: List, e: Int) -> List {
    push(l, e)
}

fn vec() -> List {
    vec()
}

fn print(n: Int) {
    print(n)
}

fn match_some(o: Option<Int>) -> Int {
    match o {
        Some(x) => x,
        None => 1,
    }
}

fn match_none(o: Option<Int>) -> Int {
    match o {
        Some(x) => 1,
        None => 1,
    }
}

fn main() {
    let nil = nil()
    let none = none()
    let one = match_some(none)
    print(one)
    drop nil
    drop none
    let one = match_some(none)
    print(one)
    let one = match_none(none)
    print(one)
}
