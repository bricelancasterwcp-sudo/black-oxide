```rust
fn main() {
    let input = "a1b22";
    let mut total_digits = 0;
    let mut count_digits = 0;

    for c in input.chars() {
        if c.is_digit(10) {
            total_digits += c.to_digit(10).unwrap();
            count_digits += 1;
        }
    }

    println!("Total digits: {}", total_digits);
    println!("Number of digits: {}", count_digits);
}
```
