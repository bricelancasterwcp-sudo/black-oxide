    let nums = vec(7, 4, 9, 10)
    let mut total = 0
    for x in &nums {
        if x % 2 != 0 {
            print_str(&int_to_str_x2)
        } else {
            total = total + x
        }
    }
    print(&total)
    drop nums
}

    let s = int_to_str(n)
    concat(s, s)

fn print_str(s: Str) {
    let mut i = 0
    while i < str_len(s) {
        print(&s)
        i = i + 1
    }
}

    let mut i = 0
    while i < str_len(s) {
        print_char(&s)
        i = i + 1
    }
}

    let mut i = 0
    while i < str_len(c) {
        print_char(&c)
        i = i + 1
    }
}
