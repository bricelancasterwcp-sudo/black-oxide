fn main() {
    let nums = push(push(vec(), 5), 40)
    let mut result = vec()
    for x in &nums {
        if x % 2 == 0 {
            result = push(result, x)
        }
    }
    for i in 1 + range(1, 11) {
        print(get(&result, i - 1).unwrap())
    }
    drop result
    drop nums
}

fn sum_big(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn main() {
    let nums = push(push(vec(), 5), 40)
    let bigger = extend(clone(&nums), 70)
    print(sum_big(&nums, 10))
    drop nums
    print(sum_big(&bigger, 10))
    drop bigger
}
