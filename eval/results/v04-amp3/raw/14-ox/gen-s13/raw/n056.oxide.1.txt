fn sum_of_numbers(v: Vec<Str>) -> Int {
    let mut sum = 0
    for s in v {
        match parse_int(s) {
            Some(n) => sum += n,
            None => (),
        }
    }
    sum
}

fn count_non_numbers(v: Vec<Str>) -> Int {
    let mut count = 0
    for s in v {
        if parse_int(s).is_none() {
            count += 1
        }
    }
    count
}

fn main() {
    let values = vec("7", "x", "9", "?")
    let total = sum_of_numbers(values)
    print(total)
    let non_numbers = count_non_numbers(vec("7", "x", "9", "?"))
    print(non_numbers)
}
