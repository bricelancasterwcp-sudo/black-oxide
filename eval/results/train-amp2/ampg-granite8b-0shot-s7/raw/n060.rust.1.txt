fn main() {
  let text = "a1b22";
  let mut total = 0;
  let mut count = 0;
  for c in text.chars() {
    match c {
      '0'..='9' => {
        total += c as usize - '0' as usize;
        count += 1;
      }
      _ => ()
    }
  }
  println!("Total: {}", total);
  println!("Count: {}", count);
}
