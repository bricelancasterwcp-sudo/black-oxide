```rust
fn main() {
    let signals = [String::from("Green"), String::from("Red"), String::from("Yellow")];
    
    for &signal in signals.iter() {
        match signal.as_str() {
            "Green" => println!("Go for green"),
            "Red" => println!("Stop for red"),
            "Yellow" => println!("Wait for yellow"),
            _ => println!("Unknown signal"),
        }
    }
}
```
