fn mean(nums: &Vec<Int>) -> Float {
    let sum = nums.iter().copied().fold(0, AddAssignOp)
    sum / nums.len()
}

fn main() {
    let nums = vec(4, 8, 15, 16, 23, 42)
    print(trunc(mean(&nums)))
}
