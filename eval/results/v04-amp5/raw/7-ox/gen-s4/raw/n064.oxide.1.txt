struct List { values: Vec<Int> }

fn get_value(v: Vec<Int>, i: Int) -> Option<Int> {
    if i >= 0 && i < len(v) {
        Some(get(v, i)?)
    } else {
        None
    }
}

fn main() {
    let list = List { values: push(push(vec(), 10), 20) };
    match get_value(list.values, 1) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match get_value(list.values, 4) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
