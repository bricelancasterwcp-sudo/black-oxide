struct Reading { label: Str, values: Vec<Int> }

fn count_non_numbers(values: Vec<Str>) -> Int {
    let mut count = 0
    for v in values {
        if parse_int(v).is_none() {
            count += 1
        }
    }
    count
}

fn sum_numbers(values: Vec<Str>) -> Int {
    let mut sum = 0
    for v in values {
        if let Some(n) = parse_int(v) {
            sum += n
        }
    }
    sum
}

fn main() {
    let values = vec!("7", "x", "9", "?")
    print(sum_numbers(values.clone()))
    print(count_non_numbers(values))
}
