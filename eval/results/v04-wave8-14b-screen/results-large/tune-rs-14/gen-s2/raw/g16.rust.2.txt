fn main() {
    let mut value = 44;
    let mut rounds = 0;
    let mut max_value = 44;
    let mut min_value = 44;
    let mut sum_values = 0;
    let mut even_rounds = 0;
    let mut sum_digits = 0;
    let mut reached_one = false;
    let mut previous_value = 0;
    while rounds < 50 && !reached_one {
        let mut new_value = 0;
        let mut temp = value;
        while temp > 0 {
            let digit = temp % 10;
            new_value += digit * digit;
            sum_digits += digit;
            temp /= 10;
        }
        previous_value = value;
        value = new_value;
        if value == 1 {
            reached_one = true;
        }
        if value > max_value {
            max_value = value;
        }
        if value < min_value {
            min_value = value;
        }
        sum_values += value;
        if value % 2 == 0 {
            even_rounds += 1;
        }
        rounds += 1;
    }
    println!("{}", rounds);
    println!("{}", value);
    println!("{}", previous_value);
    println!("{}", max_value);
    println!("{}", min_value);
    println!("{}", sum_values);
    println!("{}", even_rounds);
    println!("{}", sum_digits);
    println!("{}", reached_one);
}
