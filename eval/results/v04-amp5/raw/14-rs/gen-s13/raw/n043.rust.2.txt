fn main() {
    let mut queue = vec![1, 2, 3, 4, 5];
    
    if let Some(first) = queue.first().cloned() {
        if let Some(last) = queue.pop() {
            queue.insert(0, last);
            queue.push(first);
        }
    }
    
    for entry in &queue {
        println!("{}", entry);
    }
}
