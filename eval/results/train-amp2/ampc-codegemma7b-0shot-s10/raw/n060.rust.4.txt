fn main() {
    let text = "a1b22";
    let mut total = 0;
    let mut count = 0;
    for c in text.chars() {
        if c.is_digit(10) {
            total += c.to_digit(10).unwrap() as u32;
            count += 1;
        }
    }
    println!("Total: {}", total);
    println!("Count: {}", count);
}
