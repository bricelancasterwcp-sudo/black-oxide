struct VecInt { values: Vec<Int> }

fn collect_evens() -> VecInt {
    let mut evens = VecInt { values: vec() }
    for i in range(1, 11) {
        if i % 2 == 0 {
            evens.values = push(evens.values, i)
        }
    }
    evens
}

fn main() {
    let evens = collect_evens()
    for i in evens.values.iter().rev() {
        print(i)
    }
}
