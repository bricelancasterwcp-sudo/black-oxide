struct Reading { label: Str, values: Vec<Int> }

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    let r1 = Reading { label: "lab1", values: push(push(vec(), 7), 8) }
    let r2 = Reading { label: "lab2", values: push(push(vec(), 8), 9) }
    let r3 = Reading { label: "lab3", values: vec() }
    match second(r1.values) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match second(r2.values) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match second(r3.values) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
