fn main() {
    let numbers: [Option<i32>; 4] = [
        Some(7),
        None,
        Some(9),
        Some(?), // This is a placeholder for an actual number that you want to add if it exists.
    ];

    let total: i32 = numbers.iter().filter_map(|&num| num).sum();

    println!("The total of the ones that do: {}", total);
    println!("Number of elements that do not have a value: {:?}", numbers.iter().filter(|&&n| n.is_none()).count());
}
