fn is_prime(n: u32) -> bool {
    if n <= 1 {
        return false;
    }
    for i in 2..=((n as f64).sqrt() as u32) + 1 { // Corrected the range to include the square root plus one
        if n % i == 0 {
            return false;
        }
    }
    true
}

fn main() {
    let mut count = 0;
    for num in 2..100 {
        if is_prime(num) {
            count += 1;
        }
    }
    println!("There are {} prime numbers strictly less than 100.", count);
}
