struct List { value: Int, next: Option<Box<List>> }

fn get_at(list: &List, pos: Int) -> Option<Int> {
    if pos == 0 {
        Some(list.value)
    } else if let Some(next) = list.next {
        get_at(&next, pos - 1)
    } else {
        None
    }
}

fn main() {
    let list = Box::new(List { value: 20, next: Some(Box::new(List { value: 10, next: None })) });
    match get_at(&list, 1) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match get_at(&list, 4) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
