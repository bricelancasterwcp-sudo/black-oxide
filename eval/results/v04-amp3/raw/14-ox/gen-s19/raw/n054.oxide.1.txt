fn is_vowel(c: Str) -> Bool {
    match c {
        "a" | "e" | "i" | "o" | "u" => true,
        _ => false,
    }
}

fn count_vowels_and_consonants(s: Str) -> (Int, Int) {
    let mut vowels = 0
    let mut consonants = 0
    for c in chars(s) {
        if is_vowel(c) {
            vowels += 1
        } else if c >= "a" && c <= "z" || c >= "A" && c <= "Z" {
            consonants += 1
        }
    }
    (vowels, consonants)
}

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    let (vowels, consonants) = count_vowels_and_consonants(keyboard)
    print(vowels)
    print(consonants)
}
