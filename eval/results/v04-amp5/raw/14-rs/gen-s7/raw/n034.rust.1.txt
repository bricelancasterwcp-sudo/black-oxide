```rust
fn main() {
    let numbers = vec![7, 8, 9];
    match get_second_entry(&numbers) {
        Some(value) => println!("{}", value),
        None => println!("none"),
    }
}

fn get_second_entry<T>(list: &[T]) -> Option<&T> {
    if list.len() > 1 {
        Some(&list[1])
    } else {
        None
    }
}
```
