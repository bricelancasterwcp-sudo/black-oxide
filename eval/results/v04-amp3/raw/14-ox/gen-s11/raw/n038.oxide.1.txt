struct Device { sensor: Sensor }
struct Sensor { reading: Int }

fn set_reading(device: Device, value: Int) -> Device {
    let sensor = device.sensor;
    let new_sensor = Sensor { reading: value };
    Device { sensor: new_sensor }
}

fn print_reading(device: Device) {
    print(device.sensor.reading)
}

fn main() {
    let initial_device = Device { sensor: Sensor { reading: 0 } };
    let updated_device = set_reading(initial_device, 7);
    print_reading(updated_device);
}
