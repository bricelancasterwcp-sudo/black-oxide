fn main() {
    let phrase = "when in doubt go home"
    let word_count = len(split_str(phrase, " "))
    let char_count = str_len(phrase)
    print(word_count)
    print(char_count)
}

fn split_str(s: Str, delimiter: Str) -> Vec<Str> {
    let mut result = vec()
    let mut current = ""
    for c in chars(s) {
        if c == delimiter {
            result = push(result, current)
            current = ""
        } else {
            current = concat(current, c)
        }
    }
    push(result, current)
}
