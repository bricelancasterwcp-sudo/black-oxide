fn count_above_five(v: Vec<Int>) -> Int {
    let mut count = 0;
    for x in v {
        if x > 5 {
            count += 1;
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    sort(v).pop()
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14);
    print(count_above_five(readings)) // field access copies: readings stays usable
    let readings2 = push(push(push(push(vec(), 9), 6), 11), 3);
    match max_reading(readings2) {
        Some(n) => print(n),
        None => print_str("empty"),
    }
}
