fn print(x: Int) {
    print_str(int_to_str(x))
}

fn int_to_str(n: Int) -> Str {
    let mut buf: Vec<Char> = vec()
    let mut x = n
    while x >= 10 {
        buf.push(((x % 10) + 48) || Char)
        x = x / 10
    }
    buf.push((x + 48) || Char)
    Str(buf)
}
