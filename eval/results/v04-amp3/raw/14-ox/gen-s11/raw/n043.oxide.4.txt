struct Queue { elements: Vec<Int> }

fn exchange_first_last(q: Queue) -> Queue {
    let len = len(q.elements)
    if len < 2 {
        return q
    }
    let first = get(q.elements, 0).unwrap()
    let last = get(q.elements, len - 1).unwrap()
    let mid = range(1, len - 1)
    let new_elements = push(push(push(vec(), last), mid), first)
    Queue { elements: new_elements }
}

fn print_queue(q: Queue) {
    for x in q.elements {
        print(x)
        print("\n")  // Add newline for each entry
    }
}

fn main() {
    let q = Queue { elements: vec(1, 2, 3, 4, 5) }
    let exchanged = exchange_first_last(q)
    print_queue(exchanged)
}
