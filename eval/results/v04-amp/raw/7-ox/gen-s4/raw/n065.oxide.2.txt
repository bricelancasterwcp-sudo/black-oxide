fn count_above_five(v: Vec<Int>) -> Int {
    let mut count = 0;
    for x in v {
        if x > 5 {
            count += 1;
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max_val = None;
    for &x in &v {
        match max_val {
            Some(m) => max_val = Some(if x > m { x } else { m }),
            None => max_val = Some(x),
        }
    }
    max_val
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14);
    readings.push(9);
    print(count_above_five(readings.clone())); // field access copies: readings stays usable
    match max_reading(readings) {
        Some(n) => print(n),
        None => print_str("empty"),
    }
}
