```rust

fn main() {
    let mut total = 0;
    let count = [7, 8, 9, ?].iter().map(|&x| if x.is_digit(10) { &x } else { &"" }).count();

    for x in [7, 8, 9, ?].iter() {
        if x.is_digit(10) {
            total += x.to_string().parse::<i32>().unwrap();
        }
    }

    println!("Total: {}", total);
    println!("Count: {}", count);
}
```
