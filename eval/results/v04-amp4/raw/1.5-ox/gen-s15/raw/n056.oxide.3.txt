fn sum_of_numbers(texts: Vec<&str>) -> (Int, Int) {
    let mut total = 0;
    let mut count = 0;

    for text in texts {
        if let Ok(num) = parse_int(text) {
            total += num;
            count += 1;
        }
    }

    (total, count)
}

fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let (sum, count) = sum_of_numbers(texts);
    println!("Total: {}", sum);
    println!("Count of numbers: {}", count);
}
