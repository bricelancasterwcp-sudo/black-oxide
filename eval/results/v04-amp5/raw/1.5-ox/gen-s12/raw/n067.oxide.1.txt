fn main() {
    let mut numbers: Vec<Int> = vec![7, 4, 9, 10];
    let mut total_evens = 0;

    // Walk the list
    for i in 0..numbers.len() {
        if numbers[i] % 2 != 0 {
            println!("{}", numbers[i] * 2);
        } else {
            total_evens += numbers[i];
        }
    }

    // Print the total of even entries
    print!("Total of even entries: {}", total_evens);
}
