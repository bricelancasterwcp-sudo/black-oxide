```rust
fn main() {
    let numbers = vec![5, 12, 3, 18, 9];
    let below_cutoff = numbers.iter().filter(|&&x| x < 10).count();
    let above_or_equal_cutoff = numbers.iter().filter(|&&x| x >= 10).count();
    println!("Numbers below cutoff (10): {}", below_cutoff);
    println!("Numbers at or over cutoff (10): {}", above_or_equal_cutoff);
}
```
