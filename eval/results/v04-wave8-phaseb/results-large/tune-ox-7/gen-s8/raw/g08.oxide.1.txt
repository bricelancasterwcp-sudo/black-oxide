struct Reading { label: Str, values: Vec<Int> }
fn analyze(r: Reading) -> Option<Str> {
    let mut last = -1
    let mut increases = 0
    let mut decreases = 0
    let mut max_increase = 0
    let mut longest_streak = 0
    let mut current_streak = 0
    for v in r.values {
        if last != -1 {
            let diff = v - last
            if diff > 0 {
                increases += 1
                current_streak += 1
                max_increase = max(max_increase, diff)
            } else if diff < 0 {
                decreases += 1
                current_streak = 0
            }
        }
        last = v
    }
    longest_streak = max(longest_streak, current_streak)
    print_str(r.label)
    print(increases)
    print(decreases)
    print(max_increase)
    print(longest_streak)
    None
}
fn main() {
    let north = Reading { label: "north", values: vec(12, 15, 19, 14, 14, 20, 31, 30) }
    let south = Reading { label: "south", values: vec(40, 38, 41, 45, 50, 44) }
    analyze(north)
    analyze(south)
    let north_streak = analyze(north).unwrap_or(-1)
    let south_streak = analyze(south).unwrap_or(-1)
    if north_streak > south_streak {
        print_str("north")
    } else {
        print_str("south")
    }
}
