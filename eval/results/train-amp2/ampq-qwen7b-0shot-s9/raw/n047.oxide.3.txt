struct List { items: Vec<Int> }

fn add_to_list(list: List, item: Int) -> List {
    push(list, item)
}

fn main() {
    let mut even_list = List { items: vec() }
    for i in range(1, 11) {
        if i % 2 == 0 {
            even_list = add_to_list(even_list, i)
        }
    }
    while !even_list.items.is_empty() {
        let last_item = pop(even_list.items)?
        print(last_item)
    }
}
