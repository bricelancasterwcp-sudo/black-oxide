fn main() {
    let sensor = Sensor { reading: 0 }
    let device = Device { sensor: &sensor }
    set_reading(&mut * device.sensor, 7)
    print(device.sensor.reading)
}

struct Device {
    sensor: Sensor,
}

struct Sensor {
    reading: Int,
}

fn set_reading(sensor: &Sensor, reading: Int) {
    sensor.reading = reading
}
