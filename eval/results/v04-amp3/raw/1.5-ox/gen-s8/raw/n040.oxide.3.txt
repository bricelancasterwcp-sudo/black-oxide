```rust
struct Shape {
    kind: String,
    data: Vec<f64>,
}

impl Shape {
    fn area(&self) -> f64 {
        match self {
            Square(side) => side * side,
            Circle(radius) => 3.14 * radius * radius,
        }
    }
}

fn main() {
    let square = Shape { kind: String::from("Square"), data: vec![4.0] };
    println(square.area());   // Print the area of a square with side 4, using pi
}
```
