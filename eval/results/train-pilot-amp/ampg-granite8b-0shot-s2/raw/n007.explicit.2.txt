fn sum_digits(v: Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in v {
        if x > limit {
            total = total + x
        }
    }
    total
}

fn extend(v: Vec<Int>, x: Int) -> Vec<Int> {
    push(v, x)
}

fn main() {
    let nums = push(push(vec(), 5), 40)
    let bigger = extend(clone(&nums), 70)
    print(sum_digits(nums, 10))
    drop nums
    print(sum_digits(bigger, 10))
    drop bigger
}
