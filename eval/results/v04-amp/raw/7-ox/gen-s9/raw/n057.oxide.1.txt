fn main() {
    let fig = "fig";
    let banana = "banana";
    let kiwi = "kiwi";

    let mut max_len = 0;
    let mut longest_name = "";

    if str_len(fig) > max_len {
        max_len = str_len(fig);
        longest_name = fig;
    }

    if str_len(banana) > max_len {
        max_len = str_len(banana);
        longest_name = banana;
    }

    if str_len(kiwi) > max_len {
        max_len = str_len(kiwi);
        longest_name = kiwi;
    }

    print_str(longest_name);
    print(max_len);
}
