fn read_str(prompt: &str) -> Option<String> {
    use std::io::{self, BufRead};
    let stdin = io::stdin();
    let mut reader = stdin.lock();
    println!("{}", prompt);
    let mut line = String::new();
    if reader.read_line(&mut line).is_err() {
        None
    } else {
        Some(line.trim().to_string())
    }
}
