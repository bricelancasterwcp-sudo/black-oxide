fn main() {
    let figures = vec!(8, 3, 11, 6);
    
    let min_figure = match figures.iter().min_by_key(|&&x| x) {
        Some(&x) => x,
        None => 0,
    };
    
    print(min_figure);
    
    let max_figure = match figures.iter().max_by_key(|&&x| x) {
        Some(&x) => x,
        None => 0,
    };
    
    let gap = max_figure - min_figure;
    print(gap);
}
