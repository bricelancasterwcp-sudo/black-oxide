struct Scores { values: Vec<Int> }

fn main() {
    let scores = Scores { values: vec(5, 2, 9) }
    let mut sorted = sort(scores.values.clone())
    for score in sorted {
        print_str(score.to_string())
    }
}

fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut result = vec()
    for x in v {
        if result.is_empty() {
            result.push(x)
        } else {
            let mut i = 0
            while i < result.len() && result.get(i) < x {
                i = i + 1
            }
            result.insert(i, x)
        }
    }
    result
}

fn print_str(s: Str) {
    println(s)
}

    v + vec(x)

    v.len() == 0
}

fn len(v: Vec<Int>) -> Int {
    v.len()
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    if i < 0 || i >= len(v) {
        None
    } else {
        Some(v.get(i))
    }
}

    v + vec(x, v.get(i))
