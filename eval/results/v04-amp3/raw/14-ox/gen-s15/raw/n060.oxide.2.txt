fn count_digits_and_sum(s: Str) -> (Int, Int) { let mut sum = 0; let mut count = 0; for c in chars(s) { if parse_int(c).is_some() { let digit = parse_int(c).unwrap(); sum += digit; count += 1; } } (sum, count) } fn main() { let input = "a1b22"; let (sum, count) = count_digits_and_sum(input); print(sum); print(count); }
