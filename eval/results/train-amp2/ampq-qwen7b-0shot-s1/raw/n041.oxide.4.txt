fn main() {
    let mut vec = vec(9, 2, 7, 4)
    for _ in range(0, len(vec)) {
        let mut min = 0
        for y in range(0, len(vec)) {
            if get(vec, y) < get(vec, min) {
                min = y
            }
        }
        print(get(vec, min))
        vec = push(remove(vec, min))
    }
}

fn remove(v: Vec<Int>, i: Int) -> Vec<Int> {
    let mut result = vec()
    let mut j = 0
    for x in v {
        if j != i {
            result = push(result, x)
        }
        j = j + 1
    }
    result
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut result = vec()
    for y in v {
        result = push(result, y)
    }
    result = push(result, x)
    result
}
