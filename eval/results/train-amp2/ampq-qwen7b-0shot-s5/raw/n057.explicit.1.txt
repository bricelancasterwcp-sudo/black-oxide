fn max_len(s1: &Str, s2: &Str) -> Int {
    if str_len(s1) > str_len(s2) {
        str_len(s1)
    } else {
        str_len(s2)
    }
}

fn main() {
    let fig = "fig"
    let banana = "banana"
    let kiwi = "kiwi"
    let max_length = max_len(max_len(fig, banana), kiwi)
    print_str(fig)
    print(max_length)
    drop fig
    print_str(banana)
    print(max_length)
    drop banana
    print_str(kiwi)
    print(max_length)
    drop kiwi
}
