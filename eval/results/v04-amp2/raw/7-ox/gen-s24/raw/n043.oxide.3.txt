struct Queue { items: Vec<Int> }

fn new_queue() -> Queue {
    Queue { items: vec(1, 2, 3, 4, 5) }
}

fn swap_first_last(q: &mut Queue) {
    let len = len(q.items);
    if len > 0 {
        let first = get(q.items, 0)?;
        let last = get(q.items, len - 1)?;
        q.items[0] = last;
        q.items[len - 1] = first;
    }
}

fn main() {
    let mut q = new_queue();
    swap_first_last(&mut q);
    for x in q.items {
        print(x);
        print_str("\n");
    }
}
