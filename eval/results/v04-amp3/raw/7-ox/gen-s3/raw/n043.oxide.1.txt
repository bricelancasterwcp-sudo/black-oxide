struct Queue { items: Vec<Int> }

fn main() {
    let mut q = Queue { items: vec(1, 2, 3, 4, 5) };
    let first = get(q.items, 0)?;
    let last = get(q.items, len(q.items) - 1)?;
    q.items = push(pop(q.items)?, last);
    q.items = push(vec(), first);
    for item in q.items {
        print(item)
    }
}
