```rust
fn main() {
    let text = "a1b22";
    let mut count_digit = 0;
    let mut sum_digit = 0;

    for c in text.chars() {
        if c.is_digit(10) {
            let digit = c.to_digit(10).unwrap();
            sum_digit += digit;
            count_digit += 1;
        }
    }

    println!("The total of all digit characters is: {}", sum_digit);
    println!("The number of digit characters is: {}", count_digit);
}
```
