struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let found = -1
    for x in v {
        if x > 10 {
            found = x
            break
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn int_to_str(n: Int) -> Str {
    match n {
        Digit(d) => d.to_string(),
        _ => "0123456789".chars().skip(n).next().unwrap().to_string(),
    }
}
