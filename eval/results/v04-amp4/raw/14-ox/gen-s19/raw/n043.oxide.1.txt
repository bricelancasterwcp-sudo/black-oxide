struct Queue { items: Vec<Int> }

fn main() {
    let q = Queue { items: vec() }
    let q = push(push(push(push(push(q.items, 1), 2), 3), 4), 5)
    let len = len(q)
    if len > 0 {
        let first = get(q, 0).unwrap_or(-1)
        let last = get(q, len - 1).unwrap_or(-1)
        let q = set(set(q, 0, last), len - 1, first)
        for x in q {
            print(x)
        }
    }
}
