struct Rectangle { width: Int, height: Int }

fn area(r: Rectangle) -> Int {
    r.width * r.height
}

fn perimeter(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}

fn main() {
    let rect = Rectangle { width: 9, height: 3 };
    print(area(rect)); // fixed the missing semicolons at lines 13 and 14
    let distance = perimeter(rect); // fixed the missing semicolon at line 15
    println!("Area: {}, Perimeter: {}", area(rect), distance);
}
