```rust
fn main() {
    let mut sum = 0;
    let mut count = 0;

    for s in ["12", "x", "30"] {
        match s.parse::<i32>() {
            Ok(num) => {
                sum += num;
                count += 1;
            }
            Err(_) => {
                println!("Failed to parse: {}", s);
            }
        }
    }

    println!("Sum: {}, Count: {}", sum, count);
}
```
