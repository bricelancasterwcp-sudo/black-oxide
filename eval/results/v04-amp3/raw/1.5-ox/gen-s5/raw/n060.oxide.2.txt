```rust
fn main() {
    let mut sum = 0;
    let mut count = 0;
    if let Some(input) = io::stdin().read_line() {
        for c in input.chars() {
            if c.is_digit(10) {
                sum += c.to_digit(10).unwrap();
                count += 1;
            }
        }
        println!("Sum: {}", sum);
        println!("Count: {}", count);
    }
}
```
