fn count_s(s: Str) -> Int {
    let mut count = 0;
    for c in chars(s) {
        if c == "s" {
            count += 1;
        }
    }
    count
}

fn distinct_chars(s: Str) -> Int {
    let mut seen = Vec::new();
    for c in chars(s) {
        if !contains(seen, c) {
            push(seen, c);
        }
    }
    len(seen)
}

fn main() {
    let s = "mississippi";
    print(count_s(s));
    print_str(" ");
    print(distinct_chars(s));
}
