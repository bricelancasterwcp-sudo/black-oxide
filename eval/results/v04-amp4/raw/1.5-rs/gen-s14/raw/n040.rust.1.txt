```rust
fn main() {
    let square_side = 4;
    let shape_area_square = square_side * square_side;

    let circle_radius = 3.0;
    const PI: f64 = 3.141592653589793;
    let shape_area_circle = PI * circle_radius * circle_radius;

    println!("The area of the square is {:.2}", shape_area_square);
    println!("The area of the circle is {:.2}", shape_area_circle);
}
```
