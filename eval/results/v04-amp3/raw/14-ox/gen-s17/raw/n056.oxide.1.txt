struct Reading { label: Str, values: Vec<Int> }

fn sum_of_numbers(v: Vec<Str>) -> Int {
    let mut sum = 0
    for s in v {
        if parse_int(s)? {
            sum += to_float(parse_int(s).unwrap())
        }
    }
    sum
}

fn count_non_numbers(v: Vec<Str>) -> Int {
    let mut count = 0
    for s in v {
        if !parse_int(s)? {
            count += 1
        }
    }
    count
}

fn main() {
    let r = Reading { label: "lab", values: vec("7", "x", "9", "?") }
    print(sum_of_numbers(r.values))
    print(count_non_numbers(r.values))
}
