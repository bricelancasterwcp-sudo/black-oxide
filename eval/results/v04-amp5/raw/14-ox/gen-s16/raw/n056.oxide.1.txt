fn sum_numbers(values: Vec<Str>) -> Int {
    let mut sum = 0
    let mut count_non_numbers = 0
    for v in values {
        match parse_int(v) {
            Some(n) => sum += n,
            None => count_non_numbers += 1,
        }
    }
    print(sum)
    print(count_non_numbers)
}

fn main() {
    let values = vec("7", "x", "9", "?")
    sum_numbers(values)
}
