fn main() {
    let mut sum = 0;
    let mut count = 0;
    let numbers = [Some(7), Some(42), Some(9), Some(37)];
    
    for number in numbers.iter() {
        if let Some(n) = number {
            sum += n;
            count += 1;
        }
    }
    
    println!("The sum of the numbers is: {}", sum);
    println!("The count of numbers is: {}", count);
}
