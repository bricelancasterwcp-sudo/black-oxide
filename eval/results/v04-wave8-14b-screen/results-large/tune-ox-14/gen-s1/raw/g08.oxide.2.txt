fn main() {
    let north = vec(12, 15, 19, 14, 14, 20, 31, 30)
    let south = vec(40, 38, 41, 45, 50, 44)
    let (inc_n, dec_n, max_inc_n, longest_inc_n) = analyze(north)
    let (inc_s, dec_s, max_inc_s, longest_inc_s) = analyze(south)
    print_str("north")
    print(inc_n)
    print(dec_n)
    print(max_inc_n)
    print(longest_inc_n)
    print_str("south")
    print(inc_s)
    print(dec_s)
    print(max_inc_s)
    print(longest_inc_s)
    if longest_inc_n > longest_inc_s {
        print_str("north")
    } else {
        print_str("south")
    }
}
fn analyze(readings: Vec<Int>) -> (Int, Int, Int, Int) {
    let mut inc = 0
    let mut dec = 0
    let mut max_inc = 0
    let mut longest_inc = 0
    let mut current_inc = 0
    for i in range(1, len(readings)) {
        if readings[i] > readings[i - 1] {
            inc += 1
            current_inc += 1
            if readings[i] - readings[i - 1] > max_inc {
                max_inc = readings[i] - readings[i - 1]
            }
        } else if readings[i] < readings[i - 1] {
            dec += 1
            current_inc = 0
        } else {
            current_inc = 0
        }
        if current_inc > longest_inc {
            longest_inc = current_inc
        }
    }
    (inc, dec, max_inc, longest_inc)
}
