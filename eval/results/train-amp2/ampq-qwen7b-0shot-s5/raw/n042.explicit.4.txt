fn max(a: Int, b: Int) -> Int {
    if a > b {
        a
    } else {
        b
    }
}

fn main() {
    let nums = vec(12, 5, 20, 8, 17)
    let largest = max(nums.clone(), nums.clone())
    print(largest)
    let second_largest = max(max(nums.clone(), nums.clone()), nums.clone())
    print(second_largest)
    drop nums
}
