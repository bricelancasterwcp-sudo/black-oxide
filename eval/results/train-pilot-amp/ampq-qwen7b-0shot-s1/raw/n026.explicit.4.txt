fn number_to_text(num: Int) -> Str {
    match num {
        NaN => "NaN".to_string(),
        Inf => "Inf".to_string(),
        _ => "twenty-two hundred twenty-six".to_string(),
    }
}

fn print_int(n: Int) -> Unit {
    let text = number_to_text(n)
    let text_length = str_len(&text)
    print_str(&text)
    print(&text_length)
}

fn main() {
    let text = number_to_text(2026)
    let text_length = str_len(&text)
    print_str(&text)
    print_int(text_length)
}
