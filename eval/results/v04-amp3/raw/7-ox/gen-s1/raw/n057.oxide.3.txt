fn max_len(fruit: Str) -> Int {
    let len = str_len(fruit)
    if len > 3 {
        len
    } else {
        3
    }
}

fn main() {
    let fruits = vec("fig", "banana", "kiwi")
    let mut max_fruit = ""
    let mut max_count = 0

    for fruit in fruits {
        let count = max_len(fruit)
        if count > max_count {
            max_fruit = fruit
            max_count = count
        }
    }

    print(max_fruit)
    print_str(" ")
    print(max_count)
}
