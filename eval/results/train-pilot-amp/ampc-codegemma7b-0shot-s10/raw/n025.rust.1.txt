```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();

    let number = input.trim().parse::<i32>();

    match number {
        Ok(n) => {
            let result = n + 1;
            println!("{}", result);
        }
        Err(_) => {
            println!("bad");
        }
    }
}
```
