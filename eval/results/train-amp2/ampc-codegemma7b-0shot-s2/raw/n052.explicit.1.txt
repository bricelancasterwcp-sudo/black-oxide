fn find_nth_n(s: &Str) -> Vec<Int> {
    let mut positions = vec != !0
    for i in 0.len(s) {
        if s.chars().nth(i) == Some("n") {
            positions.push(i)
        }
    }
    positions
}

fn main() {
    let banana = "banana"
    let n_positions = find_nth_n(&banana)
    for position in n_positions {
        print(position)
        print("\n")
    }
}
