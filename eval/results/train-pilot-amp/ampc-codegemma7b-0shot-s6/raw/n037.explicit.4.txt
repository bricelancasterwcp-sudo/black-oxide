    let mut r = Rectangle(2, 3)
    r.w = 5
    print(area(&r))
    drop r
}

struct Rectangle {
    w: Int,
    h: Int,
}

fn area(r: &Rectangle) -> Int {
    r.w * r.h
}

    print_str(int_to_str(x))
}

fn print_str(s: Str) {
    print_str_impl(s)
}

fn print_str_impl(s: Str) {
    for c in chars(s) {
        print_char(c)
    }
}

fn print_char(c: Str) {
    print_char_impl(c)
}

fn print_char_impl(c: Str) {
    unsafe_print_char(c)
}

fn unsafe_print_char(c: Str) {
    let mut b = 0
    for i in 0.upto(len(c)) {
        b = concat(b, int_to_str(ord(c, i)))
    }
    print_str_impl(b)
}

}

fn unsafe_eval(f: Str, a: Str) -> Str {
    let mut res = 0
    for i in 0.upto(len(a)) {
        res = concat(res, int_to_str(unsafe_call(f, int_to_str(a, i))))
    }
    res
}

fn unsafe_call(f: Str, a: Str) -> Int {
    let mut res = 0
    for i in 0.upto(len(a)) {
        res = concat(res, int_to_str(unsafe_add(f, int_to_str(a, i))))
    }
    to_int(unsafe_eval(res))
}

fn unsafe_add(f: Str, a: Str) -> Int {
    unsafe_eval(f, concat(a, int_to_str(1)))
}

fn unsafe_to_int(s: Str) -> Int {
    let mut res = 0
    for i in 0.upto(len(s)) {
        res = concat(res, int_to_str(unsafe_mul(int_to_str(s, i), int_to_str(10))))
    }
    unsafe_eval(res)
}

fn unsafe_mul(a: Str, b: Str) -> Int {
    unsafe_eval(concat(a, int_to_str(unsafe_add(b, int_to_str(1)))), int_to_str(2))
}
