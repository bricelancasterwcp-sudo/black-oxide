fn get(s: &Str, i: Int) -> Option<Char> {
    if i < 0 || i >= s.len() {
        None
    } else {
        Some(s.get(i))
    }
}

fn is_n(c: &Char) -> Bool {
    match c {
        Some(n) => n == "n",
        None => false,
    }
}
