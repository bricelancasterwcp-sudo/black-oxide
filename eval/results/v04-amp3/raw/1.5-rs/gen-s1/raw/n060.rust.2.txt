fn main() {
    let s = "a1b22";
    let mut digit_count = 0;
    let mut sum_of_digits = 0;

    for c in s.chars() {
        if c.is_digit(10) {
            digit_count += 1;
            sum_of_digits += c.to_digit(10).unwrap();
        }
    }

    println!("Total of all digit characters: {}", sum_of_digits);
    println!("Number of digit characters: {}", digit_count);
}
