fn main() {
    let items = [("pen", 3, 4), ("pad", 9, 2), ("ink", 5, 3)];
    let mut total_cost = 0;
    let mut max_cost_item = "";
    let mut max_cost = 0;
    for item in &items {
        let cost = item.1 * item.2;
        total_cost += cost;
        if cost > max_cost {
            max_cost = cost;
            max_cost_item = item.0;
        }
    }
    println!("{}", total_cost);
    println!("{}", max_cost_item);
}
