fn is_prime(x: Int) -> Bool {
    if x == 1 {
        return false
    }
    let mut i = 2
    while i * i <= x {
        if x % i == 0 {
            return false
        }
        i = i + 1
    }
    true
}

fn main() {
    let n = 91
    if is_prime(n) {
        print("Yes")
    } else {
        print("No")
    }
}

fn to_string(i: Int) -> Str {
    if i == 0 {
        return "0"
    }
    let mut s = ""
    let mut n = i
    while n > 0 {
        let d = n % 10
        n = n / 10
        s = char_to_str(d) + s
    }
    s
}

fn to_string(f: Float) -> Str {
    let i = int(f)
    let frac = f - i
    let frac_str = to_string(frac)
    if frac_str.len() > 5 {
        return string(i) + "." + frac_str.substring(0, 6)
    }
    string(i) + "." + frac_str
}

fn substring(s: Str, start: Int, end: Int) -> Str {
    let mut result = ""
    for i in start + 1.to_int() {
        if i < end {
            result = result + string(s.charAt(i))
        } else {
            break
        }
    }
    return result
}

fn charAt(s: Str, i: Int) -> Char {
    if i >= 0 && i < s.len() {
        return s.chars().nth(i).unwrap()
    }
    panic("Index out of bounds")
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec()
    for i in a + 1.to_int() {
        if i <= b {
            result.push(i)

            break
        }
    }
    return result
}

fn int_to_str(n: Int) -> Str {
    if n == 0 {
        return "0"
    }
    let mut s = ""
    let mut m = n
    while m > 0 {
        let d = m % 10
        m = m / 10
        s = char_to_str(d) + s
    }
    s
}

fn parse_int(s: Str) -> Option<Int> {
    let mut result = 0
    let mut multiplier = 1
    for c in s.chars().rev() {
        if c >= 48 && c <= 57 {
            result = result + multiplier * (c - 48)
            multiplier = multiplier * 10
        } else {
            return None
        }
    }
    Some(result)
}
