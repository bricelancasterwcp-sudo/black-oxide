struct Rectangle { width: Int, height: Int }
fn perimeter_rectangle(r: Rectangle) -> Int {
    2 * (r.width + r.height)
}
struct Triangle { a: Int, b: Int, c: Int }
fn perimeter_triangle(t: Triangle) -> Int {
    t.a + t.b + t.c
}
fn main() {
    let r1 = Rectangle { width: 3, height: 4 }
    print(perimeter_rectangle(r1))
    let t1 = Triangle { a: 3, b: 4, c: 5 }
    print(perimeter_triangle(t1))
    let r2 = Rectangle { width: 2, height: 5 }
    print(perimeter_rectangle(r2))
}
