fn main() {
    let samples = vec![5, 12, 3, 18, 9];
    let cutoff = 10;
    let (under, over_or_eq) = samples.iter().fold((0, 0), |(acc_under, acc_over_or_eq), &x| {
        if x < cutoff {
            (acc_under + 1, acc_over_or_eq)
        } else {
            (acc_under, acc_over_or_eq + 1)
        }
    });
    println!("Under: {}, Over or Equal: {}", under, over_or_eq);
}
