fn count_digits(s: Str) -> (Int, Int) {
    let mut count = 0
    let mut sum = 0
    for ch in chars(s) {
        if parse_int(ch).is_some() {
            let digit = unwrap_or(parse_int(ch), 0)
            count += 1
            sum += digit
        }
    }
    (sum, count)
}

fn main() {
    let input = "a1b22"
    let (total, num_digits) = count_digits(input)
    print(total)
    print(num_digits)
}
