struct Log { entries: Vec<Int> }

fn first(log: Log) -> Int {
    get(log.entries, 0)?
}

fn last(log: Log) -> Int {
    get(log.entries, len(log.entries) - 1)?
}

fn main() {
    let log = Log { entries: vec(6, 7, 8, 9) }
    print(first(log))
    print(last(log))
}
