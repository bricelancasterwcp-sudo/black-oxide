struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding(v: Vec<Int>, threshold: Int) -> Int {
    let mut count = 0
    for x in v {
        if x > threshold {
            count = count + 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let mut max_val = None
    for x in v {
        match max_val {
            Some(current_max) => {
                if x > current_max {
                    max_val = Some(x)
                }
            },
            None => {
                max_val = Some(x)
            },
        }
    }
    max_val
}

fn main() {
    let readings = Vec<Int> { 6, 11, 3, 14, 9 }
    let count = count_exceeding(readings, 5)
    print(count)
    let readings = Vec<Int> { 6, 11, 3, 14, 9 } // Reassign to use again
    match max_reading(readings) {
        Some(max_val) => print(max_val),
        None => print_str("No readings"),
    }
}
