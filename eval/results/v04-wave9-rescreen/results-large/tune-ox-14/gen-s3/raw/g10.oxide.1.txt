fn main() {
    let first = "the harbour lights flickered"
    let second = "a small boat drifted past silently"
    for p in vec(first, second) {
        let words = split(p, " ")
        print_str(if p == first { "first" } else { "second" })
        print(len(words))
        let mut max_len = 0
        let mut min_len = 1000
        let mut long_count = 0
        for w in words {
            let len = str_len(w)
            if len > max_len {
                max_len = len
            }
            if len < min_len {
                min_len = len
            }
            if len > 4 {
                long_count += 1
            }
        }
        print(max_len)
        print(min_len)
        print(long_count)
    }
    if len(split(first, " ")) > len(split(second, " ")) {
        print_str("first")
    } else {
        print_str("second")
    }
}
