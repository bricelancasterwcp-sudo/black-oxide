fn main() {
    let a = [120, 95, 143, 87, 210, 156, 99];
    let b = [175, 132, 88, 240, 119, 167];
    fn summarize(g: &[i32]) {
        println!("{}", if g == &a { "A" } else { "B" });
        println!("{}", g.len());
        let sum = g.iter().sum::<i32>();
        let avg = sum / g.len() as i32;
        println!("{}", avg);
        let above_avg = g.iter().filter(|&&x| x > avg).count();
        println!("{}", above_avg);
        let diff = g.iter().max().unwrap() - g.iter().min().unwrap();
        println!("{}", diff);
    }
    summarize(&a);
    summarize(&b);
    let sum_a: i32 = a.iter().sum();
    let sum_b: i32 = b.iter().sum();
    if sum_a > sum_b {
        println!("A");
    } else {
        println!("B");
    }
}
