fn main() {
    let mut n = 44
    let mut seen = vec()
    let mut max_val = -1
    let mut min_val = -1
    let mut sum_vals = 0
    let mut even_count = 0
    let mut digit_sum_total = 0
    let mut reached_one = false
    for i in range(0, 50) {
        seen = push(seen, n)
        if max_val == -1 || n > max_val { max_val = n }
        if min_val == -1 || n < min_val { min_val = n }
        sum_vals += n
        let mut next_n = 0
        for digit in digits(n) {
            next_n += digit * digit
            digit_sum_total += digit
        }
        n = next_n
        if n == 1 {
            reached_one = true
            break
        }
    }
    print(len(seen))
    print(n)
    print(get(seen, len(seen) - 2).unwrap_or(-1))
    print(max_val)
    print(min_val)
    print(sum_vals)
    for x in seen {
        if x % 2 == 0 { even_count += 1 }
    }
    print(even_count)
    print(digit_sum_total)
    print(reached_one)
}
