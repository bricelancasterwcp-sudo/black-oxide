struct Line { items: Vec<Str>, price_per_item: Int }

fn total_value(line: Line) -> Int {
    len(line.items) * line.price_per_item
}

fn main() {
    let line1 = Line { items: push(push(vec(), "unit"), "unit"), price_per_item: 3 };
    let line2 = Line { items: push(push(push(vec(), "unit"), "unit"), "unit"), price_per_item: 2 };
    print(total_value(line1))
    print(total_value(line2))
    print(total_value(Line { items: concat(line1.items, line2.items), price_per_item: 0 }))
}
