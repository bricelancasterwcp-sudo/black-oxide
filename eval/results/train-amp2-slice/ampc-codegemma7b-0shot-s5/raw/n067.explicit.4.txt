fn to_str(x: Int) -> Str {
    if x < 10 {
        if x == 0 {
            return str_of("0")
        }
        return str_of(char(x + 48))
    } else {
        let mut digits = vec()
        let mut n = x
        while n >= 10 {
            digits.push(char((n % 10) + 48))
            n = n / 10
        }
        digits.push(char(n + 48))
        let mut s = str_of("")
        for d in digits.iter().rev() {
            s = concat(s, str_of(d))
        }
        return s
    }
}

fn parse_int(s: &Str) -> Option<Int> {
    let mut n = 0
    let mut s = s
    if s.chars().nth(0) == Some(char(45)) {
        s = &s && s.chars().skip(1)
        n = n - 1
    }
    for c in s.chars() {
        if c >= char(48) && c <= char(57) {
            n = n * 10 + (c - char(48))
        } else {
            return None
        }
    }
    return Some(n)
}
