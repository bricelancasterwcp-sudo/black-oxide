fn is_prime(n: Int) -> Bool {
    if n <= 1 { return False }
    for i in range(2, to_int(to_float(n).sqrt() + 1)) {
        if n % i == 0 { return False }
    }
    True
}

fn count_primes(limit: Int) -> Int {
    let mut count = 0
    for num in range(2, limit) {
        if is_prime(num) {
            count += 1
        }
    }
    count
}

fn main() {
    print(count_primes(100))
}
