struct Queue { items: Vec<Int> }

fn main() {
    let q = Queue { items: push(push(push(push(push(vec(), 1), 2), 3), 4), 5) }
    let first = get(q.items, 0).unwrap_or(-1)
    let last = get(q.items, len(q.items) - 1).unwrap_or(-1)
    let new_items = set(set(q.items, 0, last), len(q.items) - 1, first)
    let q2 = Queue { items: new_items }
    for x in q2.items {
        print(x)
    }
}
