fn letter_count(string: &Str) -> Str {
    let vowels = 0
    let consonants = 0
    for c in chars(&string) {
        if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
            vowels = vowels + 1
        } else {
            consonants = consonants + 1
        }
    }
    format("{} vowels, {} consonants", vowels, consonants)
}

fn main() {
    print(letter_count("oxide"))
}
