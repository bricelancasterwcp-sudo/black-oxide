```rust
fn main() {
    let input = "a1b22";

    let mut digits_count = 0;
    let mut sum_of_digits = 0;

    for c in input.chars() {
        if c.is_digit(10) {
            digits_count += 1;
            sum_of_digits += c.to_digit(10).unwrap();
        }
    }

    println!("Total of digit characters: {}", sum_of_digits);
    println!("Number of digit characters: {}", digits_count);
}
```
