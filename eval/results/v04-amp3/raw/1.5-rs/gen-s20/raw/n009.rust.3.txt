fn main() {
    let num: u32 = 1234;
    
    // Reverse the number by converting it to a string, then reversing the characters
    let reversed_num: u32 = i32::from_str_radix(&(num.to_string().chars().rev()).collect::<String>(), 10).unwrap().try_into().expect("Invalid reversed number");
    
    println!("Reversed number: {}", reversed_num);
}
