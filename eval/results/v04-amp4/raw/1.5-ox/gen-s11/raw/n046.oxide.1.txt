```rust
fn main() {
    let vec = vec![5, 12, 3, 18, 9];
    let mut count_below_cutoff = 0;
    let mut count_at_or_above_cutoff = 0;

    for value in &vec {
        if *value < 10 {
            count_below_cutoff += 1;
        } else {
            count_at_or_above_cutoff += 1;
        }
    }

    println!("Count below cutoff: {}", count_below_cutoff);
    println!("Count at or above cutoff: {}", count_at_or_above_cutoff);
}
```
