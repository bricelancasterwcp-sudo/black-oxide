fn count_larger_than_four(v: &Vec<Int>) -> Int {
    let count = 0
    for x in v {
        if x > 4 {
            count = count + 1
        }
    }
    count
}

fn main() {
    let nums = vec(5, 3, 9, 1, 7)
    let larger_than_four_count = count_larger_than_four(&nums)
    print(larger_than_four_count)
    drop nums
}
