```rust
fn main() {
    let mut sum = 0;
    let mut count = 0;

    for c in "7x9?".chars() {
        match c {
            '7' | 'x' | '9' => {
                if let Some(n) = c.to_digit(10) {
                    sum += n;
                    count += 1;
                }
            },
            '?' => {
                if let Ok(n) = c.parse::<i32>() {
                    sum += n;
                    count += 1;
                }
            },
            _ => {}
        }
    }

    println!("Sum: {}", sum);
    println!("Count: {}", count);
}
```
