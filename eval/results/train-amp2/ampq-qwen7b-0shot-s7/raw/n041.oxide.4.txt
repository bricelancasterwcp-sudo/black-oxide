struct Sorting { values: Vec<Int> }

fn sort(v: Vec<Int>) -> Vec<Int> {
    let mut sorted = v
    let n = len(sorted)
    for i in range(0, n) {
        for j in range(i + 1, n) {
            if get(sorted, i)? > get(sorted, j)? {
                let temp = get(sorted, i)?
                set(sorted, i, get(sorted, j)?)
                set(sorted, j, temp)
            }
        }
    }
    sorted
}

fn main() {
    let measurements = push(push(push(vec(), 9), 2), push(vec(), 7))
    let sorted_measurements = sort(measurements)
    for x in sorted_measurements {
        print(x)
    }
}
