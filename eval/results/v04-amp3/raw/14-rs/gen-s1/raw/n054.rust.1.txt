```rust
use std::collections::HashMap;

fn main() {
    let word = "keyboard";
    let mut counts = HashMap::new();

    for letter in word.chars() {
        if is_vowel(letter) {
            *counts.entry('vowels').or_insert(0) += 1;
        } else {
            *counts.entry('consonants').or_insert(0) += 1;
        }
    }

    println!("Vowels: {}", counts.get(&'vowels').unwrap_or(&0));
    println!("Consonants: {}", counts.get(&'consonants').unwrap_or(&0));
}

fn is_vowel(c: char) -> bool {
    matches!(c, 'a' | 'e' | 'i' | 'o' | 'u')
}
```
