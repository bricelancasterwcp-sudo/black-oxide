fn digits(n: Int) -> Vec<Int> {
    let mut digits = vec()
    let mut nn = n
    while nn != 0 {
        digits = push(digits, nn % 10)
        nn = nn / 10
    }
    digits
}

fn main() {
    let num = 1234
    let digits = digits(num)
    for digit in &digits {
        print(digit)
    }
    drop digits
    print(digits(num))
}
