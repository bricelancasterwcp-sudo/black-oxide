struct Queue { items: Vec<Int> }

fn main() {
    let mut q = Queue { items: vec(1, 2, 3, 4, 5) }
    if len(q.items) > 1 {
        let first = get(q.items, 0).unwrap_or(-1)
        let last = get(q.items, len(q.items) - 1).unwrap_or(-1)
        q.items = push(push(vec(), last), first)
        for i in range(1, len(q.items) - 1) {
            q.items = push(q.items, get(q.items, i).unwrap_or(-1))
        }
    }
    for item in q.items {
        print(item)
    }
}
