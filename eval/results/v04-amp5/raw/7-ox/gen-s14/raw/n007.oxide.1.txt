```rust
fn main() {
    let num = 9875;
    let str_num = int_to_str(num);
    let chars_vec: Vec<Str> = chars(str_num);
    let sum_of_digits = filter(chars_vec, |c| c >= "0" && c <= "9")
                                .map(|c| parse_int(c).unwrap())
                                .sum();
    print(sum_of_digits);
}
```
