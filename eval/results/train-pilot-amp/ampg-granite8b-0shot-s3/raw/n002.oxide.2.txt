fn factorial(n: Int) -> Int {
    if n <= 1 {
        1
    } else {
        n * factorial(n - 1)
    }
}

    if n <= 1 {
        acc
    } else {
        factorial_2(n * acc, n - 1)
    }
}

fn reduce(v: Vec<Int>, f: Int) -> Int {
    let mutacc = v.get(0)?
    for x in v.iter().rest() {
        acc = f(acc, x)
    }
    acc
}
