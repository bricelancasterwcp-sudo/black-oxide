```rust
fn main() {
    let arr = [7, x, 9, ?];

    let mut sum = 0;
    let mut count = 0;

    for &num in arr.iter() {
        if num.is_some() {
            sum += num.unwrap();
            count += 1;
        }
    }

    println!("Sum of numbers: {}", sum);
    println!("Number of valid numbers: {}", count);
}
```
