```rust
struct Sensor {
    reading: Int,
}

struct Device {
    sensor: Sensor,
}

fn set_reading(device: Device, new_value: Int) -> Device {
    let updated_sensor = Sensor { reading: new_value };
    Device { sensor: updated_sensor }
}

fn main() {
    let initial_device = Device { sensor: Sensor { reading: 0 } };
    let updated_device = set_reading(initial_device, 7);
    print(updated_device.sensor.reading);
}
```
