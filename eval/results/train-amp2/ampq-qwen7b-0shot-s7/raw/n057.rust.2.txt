fn main() {
    let names = vec!["fig", "banana", "kiwi"];
    let mut max_name = "";
    let mut max_len = 0;

    for name in names {
        let len = name.len();
        if len > max_len {
            max_name = name;
            max_len = len;
        }
    }

    println!("{} {}", max_name, max_len);
}
