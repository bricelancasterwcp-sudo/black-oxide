fn main() {
    let word = "keyboard";
    let mut vowels = 0;
    let mut consonants = 0;

    for c in word.chars() {
        match c.to_ascii_lowercase() {
            'a' | 'e' | 'i' | 'o' | 'u' => vowels += 1,
            _ if c.is_alphabetic() => consonants += 1,
            _ => {}
        }
    }

    println!("Vowels: {}, Consonants: {}", vowels, consonants);
}
