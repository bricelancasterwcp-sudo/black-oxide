```rust
fn divide(a: i32, b: i32) -> Result<i32, &'static str> {
    if b == 0 {
        Err("Cannot divide by zero")
    } else {
        Ok(a / b)
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("Result is {}", result),
        Err(err) => println!("Error: {}", err),
    }
}
```
