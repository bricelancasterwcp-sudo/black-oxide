    let mut stdout = stdout_lock()
    unsafe_print_char(stdout, c)
}

fn stdout_lock() -> UnsafeMutablePointer<Stdout> {
    let mut stdout = UnsafeMutablePointer_NULL
    unsafe_stdout_init_lock(stdout)
    stdout
}

    let stderr_ptr = unsafe_stderr()
    unsafe_init_stdout_lock(mut_ptr, stderr_ptr)
    mut_ptr
}

fn chars(s: Str) -> Vec<Str> {
    let mut result = vec()
    for i in 0 < s.len() {
        result.push(get(s, i)?)
    }
    result
}

fn get(s: Str, i: Int) -> Option<Str> {
    if i >= s.len() {
        None
    } else {
        Some(s.chars().nth(i).unwrap())
    }
}

fn int_to_str(mut_ptr: Int) -> Str {
    let mut result = str()
    while mut_ptr > 0 {
        let rem = mut_ptr % 10
        result = format("%c%s", rem + 48, result)
        mut_ptr = mut_ptr / 10
    }
    result
}
