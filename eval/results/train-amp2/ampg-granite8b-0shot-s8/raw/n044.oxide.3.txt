struct Deposit { amount: Int }

fn print_balance(total: Int) {
    print(total)
}

fn main() {
    let mut total = 0
    for deposit in vec(Deposit { amount: 3 }, Deposit { amount: 1 }, Deposit { amount: 4 }, Deposit { amount: 1 }, Deposit { amount: 5 }) {
        total = push(total, deposit.amount)
    }
    print_balance(total)
}
