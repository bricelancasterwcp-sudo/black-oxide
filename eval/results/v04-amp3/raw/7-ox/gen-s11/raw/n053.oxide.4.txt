fn reverse(v: Vec<Str>) -> Str {
    let mut result = vec()
    for x in v {
        result = push(result, chars(x))
    }
    concat(result)
}

fn str_chars(s: Str) -> Vec<Str> {
    let mut result = vec()
    for c in chars(s) {
        push(result, concat(c, ""))
    }
    result
}

main() {
    print_str(concat("drum", " ", reverse(str_chars("drum"))))
}
