fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?
    Some(x + 1)
}

fn main() {
    match second(push(vec(), 7)) {
        Some(n) => print(n),
        None => print_str("None"),
    }
    print_str("\n")
    match second(push(push(vec(), 8), 9)) {
        Some(n) => print(n),
        None => print_str("None"),
    }
}
