fn main() {
    let list = vec![10, 20];

    if let Some(pos1) = get(list, 0) {
        println!("{}", pos1);
    } else {
        println!("Position 1 not found");
    }

    if let Some(pos4) = get(list, 3) {
        println!("{}", pos4);
    } else {
        println!("Position 4 not found");
    }
}

fn get<T>(v: Vec<T>, i: Int) -> Option<T> {
    if v.len() < i as usize {
        None
    } else {
        Some(v[i as usize])
    }
}
