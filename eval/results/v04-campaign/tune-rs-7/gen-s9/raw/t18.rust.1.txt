fn main() {
    let shapes = vec![
        Rectangle { width: 3, height: 4 },
        Triangle { sides: [3, 4, 5] },
        Rectangle { width: 2, height: 5 },
    ];
    for shape in shapes {
        match shape {
            Shape::Rectangle(r) => println!("{}", r.perimeter()),
            Shape::Triangle(t) => println!("{}", t.perimeter()),
        }
    }
}

struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    fn perimeter(&self) -> u32 {
        2 * (self.width + self.height)
    }
}

struct Triangle {
    sides: [u32; 3],
}

impl Triangle {
    fn perimeter(&self) -> u32 {
        self.sides.iter().sum()
    }
}

enum Shape {
    Rectangle(Rectangle),
    Triangle(Triangle),
}
