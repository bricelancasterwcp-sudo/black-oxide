fn divide(a: f64, b: f64) -> Result<f64, &'static str> {
    if b == 0.0 {
        Err("Error: Division by zero")
    } else {
        Ok(a / b)
    }
}

fn main() {
    let result = divide(20.0, 4.0);
    match result {
        Ok(divide) => println!("Result is {}", divide),
        Err(err) => println!("{}", err),
    }
}
