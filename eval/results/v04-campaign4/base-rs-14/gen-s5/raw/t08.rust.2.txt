fn main() {
    let numbers = vec![3, 8, -2, 12, 7];
    let mut positive_count = 0;
    let mut max_value = i32::MIN;

    for &number in &numbers {
        if number > 0 {
            positive_count += 1;
        }
        if number > max_value {
            max_value = number;
        }
    }

    println!("Number of positive values: {}", positive_count);
    println!("Largest value: {}", max_value);
}
