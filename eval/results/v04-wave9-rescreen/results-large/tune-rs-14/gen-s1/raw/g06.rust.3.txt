fn main() {
    let morning = [94, 71, 88, 65, 92, 79, 83];
    let evening = [55, 90, 77, 84, 68, 99, 0]; // Add a dummy value to make the array size 7
    for (name, marks) in [("Morning", morning), ("Evening", evening)] {
        let mut count_90_plus = 0;
        let mut count_80_to_89 = 0;
        let mut count_70_to_79 = 0;
        let mut count_below_70 = 0;
        let mut max_mark = 0;
        let mut sum_marks = 0;
        for &mark in marks.iter().take(6) { // Exclude the dummy value
            if mark >= 90 {
                count_90_plus += 1;
            } else if mark >= 80 {
                count_80_to_89 += 1;
            } else if mark >= 70 {
                count_70_to_79 += 1;
            } else {
                count_below_70 += 1;
            }
            max_mark = max_mark.max(mark);
            sum_marks += mark;
        }
        println!("{}", name);
        println!("{}", count_90_plus);
        println!("{}", count_80_to_89);
        println!("{}", count_70_to_79);
        println!("{}", count_below_70);
        println!("{}", max_mark);
        println!("{}", sum_marks / 6); // Divide by 6 instead of marks.len() as u32
    }
    if morning.iter().sum::<u32>() / 7 > evening.iter().take(6).sum::<u32>() / 6 { // Exclude the dummy value
        println!("Morning");
    } else {
        println!("Evening");
    }
}
