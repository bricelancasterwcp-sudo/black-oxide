fn main() {
    let s = "mississippi"
    print(count_chars(s, 's'))
    print(count_distinct_chars(s))
}
fn count_chars(s: Str, c: Char) -> Int {
    let mut count = 0
    for ch in chars(s) {
        if ch == c {
            count = count + 1
        }
    }
    count
}
fn count_distinct_chars(s: Str) -> Int {
    let mut distinct = vec()
    for ch in chars(s) {
        if !contains(distinct, ch) {
            push(distinct, ch)
        }
    }
    len(distinct)
}
