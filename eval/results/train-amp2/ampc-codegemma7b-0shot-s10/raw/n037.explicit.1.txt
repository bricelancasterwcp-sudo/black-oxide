fn chars(s: &Str) -> Vec<Str> {
    let mut v = Vec_Impl { len: 0, cap: 0, data: 0 }
    v.len = 0
    v.cap = 0
    v.data = 0
    for i in 0 < s.len {
        push(v, s.data + i)
    }
    v
}

fn parse_int(s: &Str) -> Option<Int> {
    let mut n = 0
    for c in chars(s) {
        n = n * 10 + int_to_digit(c)
    }
    Some(n)
}

    }

    new_v.data = new_v.data + v.len

}
