fn main() {
    let mut sum = 0;
    let mut total = 0;
    for num in [7, 4, 9, 10] {
        if num % 2 != 0 {
            println!("{}", num * 2);
        } else {
            sum += num;
            total += 1;
        }
    }
    println!("Total of even entries: {}", sum);
    println!("Count of even entries: {}", total - 1);
}
