fn sort(mut v: Vec<Int>) -> Vec<Int> {
    let n = len(v)
    for i in range(0, n) {
        for j in range(i + 1, n) {
            if get(v, i)? > get(v, j)? {
                let temp = get(v, i)?
                v = set(v, i, get(v, j)?)
                v = set(v, j, temp)
            }
        }
    }
    v
}

fn set(mut v: Vec<Int>, i: Int, x: Int) -> Vec<Int> {
    v = push(vec(), x)
    for j in range(0, i) {
        v = push(v, get(v, j)?)
    }
    for j in range(i + 1, len(v)) {
        v = push(v, get(v, j)?)
    }
    v
}
