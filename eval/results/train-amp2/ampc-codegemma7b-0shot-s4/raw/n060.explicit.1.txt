fn digit_count(s: &Str) -> Int {
    let mut total = 0
    let mut digits = 0
    for c in chars(s) {
        if c >= "0" && c <= "9" {
            total = total + int_to_str(parse_int(c).unwrap()).parse().unwrap()
            digits = digits + 1
        }
    }
    print(total)
    print(digits)
    total
}

fn main() {
    print(digit_count("a1b22"))
    print(digit_count("42"))
    print(digit_count("a"))
}
