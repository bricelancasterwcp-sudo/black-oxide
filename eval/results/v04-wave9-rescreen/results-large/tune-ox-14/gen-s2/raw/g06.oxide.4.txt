struct Sitting { name: Str, scores: Vec<Int> }
fn count_range(v: Vec<Int>, lo: Int, hi: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= lo && x <= hi {
            count += 1
        }
    }
    count
}
fn main() {
    let morning = Sitting { name: "Morning", scores: vec(94, 71, 88, 65, 92, 79, 83) }
    let evening = Sitting { name: "Evening", scores: vec(55, 90, 77, 84, 68, 99) }
    for s in vec(morning, evening) {
        print_str(s.name)
        print(count_range(clone(s.scores), 90, 100))
        print(count_range(clone(s.scores), 80, 89))
        print(count_range(clone(s.scores), 70, 79))
        print(count_range(clone(s.scores), 0, 69))
        let max_score = match max(clone(s.scores)) {
            Some(n) => n,
            None => -1,
        }
        print(max_score)
        let total = sum(clone(s.scores))
        let mean = total / len(clone(s.scores))
        print(mean)
    }
    let morning_mean = sum(clone(morning.scores)) / len(clone(morning.scores))
    let evening_mean = sum(clone(evening.scores)) / len(clone(evening.scores))
    if morning_mean > evening_mean {
        print_str("Morning")
    } else {
        print_str("Evening")
    }
}
