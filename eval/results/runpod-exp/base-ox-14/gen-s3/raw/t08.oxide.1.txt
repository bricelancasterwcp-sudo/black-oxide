struct Reading { label: Str, values: Vec<Int> }

fn count_positives(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count = count + 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max == None || x > max? {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let values = vec!([3, 8, -2, 12, 7])
    let positives_count = count_positives(values.clone())
    print(positives_count)
    let max_value = find_max(values)?
    print(max_value)
}
