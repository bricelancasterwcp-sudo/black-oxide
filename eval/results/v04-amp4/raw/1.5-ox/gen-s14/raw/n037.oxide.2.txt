struct Rectangle {
    width: Int,
    height: Int,
}

impl Rectangle {
    fn set_width(&mut self, w: Int) {
        if w > 0 {
            self.width = w;
        } else {
            panic!("Width must be positive");
        }
    }

    fn area(&self) -> Int {
        self.width * self.height
    }
}

fn main() {
    let mut r = Rectangle { width: 2, height: 3 };
    r.set_width(5);
    println(r.area()); // prints 15
}
