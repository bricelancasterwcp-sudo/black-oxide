struct Log { entries: Vec<Int> }

fn first_entry(log: Log) -> Option<Int> {
    get(log.entries, 0)?
}

fn last_entry(log: Log) -> Option<Int> {
    let len = len(log.entries)
    if len == 0 {
        None // or handle empty log appropriately
    } else {
        get(log.entries, len - 1)?
    }
}

fn main() {
    let log = Log { entries: vec(6, 7, 8, 9) }
    match first_entry(log.clone()) {
        Some(first) => print(first),
        None => print_str("empty log"),
    }
    match last_entry(log) {
        Some(last) => print(last),
        None => print_str("empty log"),
    }
}
