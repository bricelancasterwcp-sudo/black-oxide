struct Item { name: Str, unit_price: Int, quantity: Int }
fn calculate_cost(item: Item) -> Int {
    item.unit_price * item.quantity
}
fn find_highest_cost_item(items: Vec<Item>) -> Option<Item> {
    let mut max_cost = -1
    let mut highest_cost_item = None
    for item in items {
        let cost = calculate_cost(item)
        if cost > max_cost {
            max_cost = cost
            highest_cost_item = Some(item)
        }
    }
    highest_cost_item
}
fn main() {
    let pen = Item { name: "pen", unit_price: 3, quantity: 4 }
    let pad = Item { name: "pad", unit_price: 9, quantity: 2 }
    let ink = Item { name: "ink", unit_price: 5, quantity: 3 }
    let items = vec(pen, pad, ink)
    let total_cost = 0
    for item in clone(items) {
        let cost = calculate_cost(item)
        print(cost)
        total_cost += cost
    }
    print(total_cost)
    match find_highest_cost_item(clone(items)) {
        Some(item) => print_str(item.name),
        None => print_str("no items"),
    }
}
