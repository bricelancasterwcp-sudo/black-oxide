fn main() {
    let numbers = [7, 4, 9, 10];
    let mut even_total = 0;

    for &number in &numbers {
        if number % 2 == 1 {
            println!("{}", number * 2);
        } else {
            even_total += number;
        }
    }

    println!("Total of even entries: {}", even_total);
}
