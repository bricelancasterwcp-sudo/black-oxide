fn count_under_cutoff(values: Vec<Int>, cutoff: Int) -> Int {
    let mut under = 0
    for x in values {
        if x < cutoff {
            under += 1
        }
    }
    under
}

fn count_at_or_over_cutoff(values: Vec<Int>, cutoff: Int) -> Int {
    let mut at_or_over = 0
    for x in values {
        if x >= cutoff {
            at_or_over += 1
        }
    }
    at_or_over
}

fn main() {
    let values = vec()
    push(push(push(push(push(values, 5), 12), 3), 18), 9)
    let cutoff = 10
    let under = count_under_cutoff(values.clone(), cutoff)
    let at_or_over = count_at_or_over_cutoff(values, cutoff)
    print(under)
    print(at_or_over)
}
