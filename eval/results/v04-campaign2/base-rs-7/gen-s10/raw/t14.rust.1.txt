```rust
fn main() {
    let text = "mississippi";
    let count_s = text.matches('s').count();
    let distinct_chars: std::collections::HashSet<_> = text.chars().collect();
    let count_distinct_chars = distinct_chars.len();

    println!("Number of 's': {}", count_s);
    println!("Distinct characters: {}", count_distinct_chars);
}
```
