struct Item { name: Str, unit_price: Int, quantity: Int }
fn main() {
    let items = vec(Item { name: "pen", unit_price: 3, quantity: 4 },
                   Item { name: "pad", unit_price: 9, quantity: 2 },
                   Item { name: "ink", unit_price: 5, quantity: 3 })
    let mut total_cost = 0
    for item in items {
        total_cost = total_cost + (item.unit_price * item.quantity)
    }
    print(total_cost)
    let mut max_item = None
    for item in items {
        if max_item.is_none() || item.unit_price * item.quantity > max_item.unwrap().unit_price * max_item.unwrap().quantity {
            max_item = Some(item)
        }
    }
    match max_item {
        Some(item) => print(item.name),
        None => print_str("no items"),
    }
}
