```rust
fn divide(x: f64, y: f64) -> Result<f64, &'static str> {
    if y == 0.0 {
        Err("Cannot divide by zero")
    } else {
        Ok(x / y)
    }
}

fn main() {
    match divide(20.0, 4.0) {
        Ok(result) => println!("Result of division is: {}", result),
        Err(error) => println!("Error: {}", error),
    }
}
```
