struct Reading { label: Str, temps: Vec<Int> }
fn main() {
    let alpha = Reading { label: "alpha", temps: vec(-5, 12, 30, 22, 0, 35, 18) }
    let beta = Reading { label: "beta", temps: vec(8, -12, 5, 41, 27) }
    for r in vec(alpha, beta) {
        print_str(r.label)
        let below_zero = count_if(r.temps, |t| t <= 0)
        let above_thirty = count_if(r.temps, |t| t >= 30)
        let max_fahrenheit = match max(r.temps) { Some(t) => to_fahrenheit(t), None => -1 }
        let min_fahrenheit = match min(r.temps) { Some(t) => to_fahrenheit(t), None => -1 }
        let mean_celsius = if len(r.temps) > 0 {
            let sum = sum(r.temps)
            div(sum, len(r.temps))
        } else {
            -1
        }
        print(below_zero)
        print(above_thirty)
        print(max_fahrenheit)
        print(min_fahrenheit)
        print(mean_celsius)
    }
}
fn to_fahrenheit(c: Int) -> Int {
    let f = mul(c, 9)
    let f = div(f, 5)
    add(f, 32)
}
