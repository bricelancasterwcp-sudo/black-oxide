```rust
fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut max_name = "";
    let mut max_length = 0;

    for name in names.iter() {
        if name.len() > max_length {
            max_length = name.len();
            max_name = name;
        }
    }

    println!("The name with the most letters is: {}", max_name);
    println!("It has {} letters.", max_length);
}
```
