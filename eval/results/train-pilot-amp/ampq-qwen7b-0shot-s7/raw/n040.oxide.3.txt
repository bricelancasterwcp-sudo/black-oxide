struct Square { side: Int }

struct Circle { radius: Float }

fn area_square(s: Square) -> Float {
    let pi = 3.0
    pi * s.side.to_float() * s.side.to_float()
}

fn area_circle(c: Circle) -> Float {
    let pi = 3.0
    pi * c.radius * c.radius
}

fn main() {
    let s = Square { side: 4 }
    print(area_square(s))
}
