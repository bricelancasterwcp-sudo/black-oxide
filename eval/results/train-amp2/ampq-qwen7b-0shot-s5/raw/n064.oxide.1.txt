struct List { a: Int, b: Int }

fn get(list: List, i: Int) -> Option<Int> {
    match i {
        I0 => Some(list.a),
        I1 => Some(list.b),
        _ => None,
    }
}

fn main() {
    let list = List { a: 10, b: 20 }
    match get(list, I0) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match get(list, I1) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
