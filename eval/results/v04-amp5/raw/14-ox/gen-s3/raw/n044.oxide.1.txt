```rust
struct Account { balance: Int }

fn deposit(acc: Account, amount: Int) -> Account {
    let new_balance = acc.balance + amount
    Account { balance: new_balance }
}

fn print_balance(acc: Account) {
    print(acc.balance)
}

fn main() {
    let acc = Account { balance: 0 }
    let acc = deposit(acc, 3)
    print_balance(acc)
    let acc = deposit(acc, 1)
    print_balance(acc)
    let acc = deposit(acc, 4)
    print_balance(acc)
    let acc = deposit(acc, 1)
    print_balance(acc)
    let acc = deposit(acc, 5)
    print_balance(acc)
}
```
