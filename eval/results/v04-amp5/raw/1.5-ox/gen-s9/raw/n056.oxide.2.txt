```rust
fn main() {
    let texts = vec![
        String::from("7"), 
        String::from("x"), 
        String::from("9"), 
        " ?"
    ];
    
    let mut total = 0;
    let count_not_numbers = 0;

    for text in texts {
        match parse_int(&text) {
            Ok(number) => {
                total += number;
            },
            Err(_) => {
                count_not_numbers += 1;
            }
        }
    }

    println!("Total of numbers: {}", total);
    println!("Count of non-number strings: {}", count_not_numbers);
}
```
