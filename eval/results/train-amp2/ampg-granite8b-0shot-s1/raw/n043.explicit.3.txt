fn enqueue(mutQueue: &Vec<Int>, a: Int) -> Vec<Int> {
    push(clone(&mutQueue), a)
}

fn dequeue(mutQueue: &Vec<Int>) -> Vec<Int> {
    let first = get(mutQueue, 0)
    let last = get(mutQueue, len(mutQueue) - 1)
    let mutQueue = mutQueue // drop_mutQueue
    push(clone(&mutQueue), last)
}
