fn find_n_positions(s: Str) -> Vec<Int> {
    let positions = vec()
    let i = 0
    while i < str_len(s) {
        if get(s, i)? == 'n' {
            positions = push(positions, i)
        }
        i = i + 1
    }
    Some(positions) // Return an Option<Vec<Int>>
}

fn main() {
    let word = "banana"
    match find_n_positions(word.to_str()) { // Handle the Option<Vec<Int>> return
        Some(positions) => {
            for pos in positions {
                print(pos)
            }
        },
        None => {} // Handle None case if needed
    }
}
