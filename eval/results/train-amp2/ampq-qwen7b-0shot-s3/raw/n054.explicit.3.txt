fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnmQWERTYUIOPASDFGHJKLZXCVBNM"
    let mut vowels = 0
    let mut consonants = 0
    for c in keyboard.chars() {
        if "aeiouAEIOU".contains(c) {
            vowels = vowels + 1
        } else if c.is_alphabetic() {
            consonants = consonants + 1
        }
    }
    print(vowels)
    print_str(" ")
    print(consonants)
}
