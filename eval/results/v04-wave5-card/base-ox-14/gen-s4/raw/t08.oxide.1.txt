fn count_positives(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count += 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    if len(v) == 0 {
        None
    } else {
        let mut max = get(v, 0)?
        for x in v {
            if x > max {
                max = x
            }
        }
        Some(max)
    }
}

fn main() {
    let values = vec!([3, 8, -2, 12, 7])
    let positives = count_positives(values)
    print(positives)
    
    let values = vec!([3, 8, -2, 12, 7]) // Reassign to use again
    match find_max(values) {
        Some(max) => print(max),
        None => print_str("No values"),
    }
}
