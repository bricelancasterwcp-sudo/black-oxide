```rust
fn main() {
    match input() {
        "green" => println!("go"),
        "red" => println!("stop"),
        "yellow" => println!("wait"),
        _ => println!("unknown signal"),
    }
}

fn input() -> String {
    let mut input = String::new();
    std::io::stdin().read_line(&mut input).expect("failed to read line");
    input.trim().to_string()
}
```
