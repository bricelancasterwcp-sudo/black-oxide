fn split(s: Str, c: Char) -> Vec<Str> {
    let mut acc = vec()
    let mut current = ""
    for ch in chars(s) {
        if ch == c {
            acc = push(acc, current)
            current = ""
        } else {
            current = concat(current, ch)
        }
    }
    push(acc, current)
}
