fn main() {
    let names = vec("fig", "banana", "kiwi")
    let mut max_len = 0
    let mut longest_name = ""
    for name in names {
        let len = str_len(name)
        if len > max_len {
            max_len = len
            longest_name = name
        }
    }
    print_str(longest_name)
    print(max_len)
}
