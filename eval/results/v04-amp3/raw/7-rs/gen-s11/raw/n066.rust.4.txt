fn main() {
    let batch = vec![9, 2, 12, 5, 16];
    let mut kept: Vec<i32> = Vec::new();

    for &number in &batch {
        if number < 10 {
            kept.push(number);
        }
    }

    for &number in &kept {
        println!("{}", number);
    }

    println!("Kept {} entries", kept.len());
}
