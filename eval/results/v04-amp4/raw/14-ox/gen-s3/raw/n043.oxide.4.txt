struct Queue { values: Vec<Int> }

fn swap_first_last(q: Queue) -> Option<Queue> {
    let len = len(q.values)
    if len < 2 {
        return Some(q)
    }
    let first = get(q.values, 0)?
    let last = get(q.values, len - 1)?
    let middle = range(1, len - 1)
    let new_values = push(push(push(vec(), first), middle), last)
    Some(Queue { values: new_values })
}

fn main() {
    let q = Queue { values: vec(1, 2, 3, 4, 5) }
    match swap_first_last(q) {
        Some(swapped_q) => {
            for x in swapped_q.values {
                print(x)
                print_str("\n")
            }
        },
        None => {}
    }
}
