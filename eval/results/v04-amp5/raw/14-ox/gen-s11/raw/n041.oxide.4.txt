struct Reading { label: Str, values: Vec<Int> }

fn print_sorted(v: Vec<Int>) {
    let sorted = sort(v);
    for x in sorted {
        print(x)
    }
}

fn main() {
    let readings = vec();
    readings = push(readings, Reading { label: "lab1", values: vec() });
    readings = push(readings, Reading { label: "lab2", values: vec() });
    readings = push(get(readings, 0).unwrap().values, 9);
    readings = push(get(readings, 0).unwrap().values, 2);
    readings = push(get(readings, 1).unwrap().values, 7);
    readings = push(get(readings, 1).unwrap().values, 4);

    for reading in readings {
        print_str(reading.label);
        print_sorted(reading.values)
    }
}
