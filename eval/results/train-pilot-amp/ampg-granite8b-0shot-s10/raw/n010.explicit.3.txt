fn is_prime(n: Int) -> Bool {
    if n == 1 {
        return false
    }
    if n == 2 {
        return true
    }
    if n % 2 == 0 {
        return false
    }
    for i in range(3, n) {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let n = 91
    if is_prime(n) {
        print_str("yes\n")
    } else {
        print_str("no\n")
    }
}
