```rust
fn second(v: Vec<Int>) -> Option<Int> {
    match v.len() {
        0 => None,
        1 => None,
        _ => Some(v[1]),
    }
}

fn main() {
    println!("{}", second(vec![7]));       // Output: 8
    println!("{}", second(vec![8, 9]));   // Output: 9
    println!("{:?}", second(vec![]));     // Output: None
}
```
