fn main() {
    let printed = false
    let count = for_each("banana", "a", 0, f(count, c) == count + if_else(c == "a", 1, 0) || printed)
    if !printed {
        print(count)
    }
}

fn for_each(s: Str, item: Str, init: T, body: T) -> T {
    let i = 0
    let acc = init
    while i < str_len(s) {
        let c = get(s, i)?
        let acc = body
        if c == item {
            let count = acc + 1
            return count
        }
        i = i + 1
    }
    acc
}

fn f(x: T, y: T) -> T {
    x + y
}

fn if_else(c: Bool, t: T, e: T) -> T {
    if c {
        t
    } else {
        e
    }
}

}

fn println(x: T) {
    _print_str("print: ")
    _print(x)
    _print_str("\n")
}

fn _print(x: T) {
    _print_str(x.to_str())
}

fn _print_str(x: Str) {
    _write_str(x)
}

fn _write_str(x: Str) {
    _io_write_str(x)
}

fn _io_write_str(x: Str) {
    _io_flush()
}

fn _io_flush() {
    _unknown_io_op(1)
}

fn _unknown_io_op(x: Int) {
    _exit(x)
}

fn _exit(x: Int) {
    _unknown_runtime_op(0, x)
}

fn _unknown_runtime_op(x: Int, y: Int) {
    _unknown_call(x, y)
}

fn _unknown_call(x: Int, y: Int) {
    _return()
}

fn _return() {
    return
}

    x.to_str()

fn to_str_lossy(x: T) -> Str {
