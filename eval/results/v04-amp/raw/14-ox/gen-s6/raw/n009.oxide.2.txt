fn reverse_digits(n: Int) -> Vec<Int> {
    let mut result = vec()
    let mut num = n
    while num > 0 {
        let digit = num % 10
        result = push(result, digit)
        num = trunc(num / 10)
    }
    result
}

fn main() {
    let number = 1234
    let reversed_digits = reverse_digits(number)
    for digit in reversed_digits {
        print(digit)
    }
}
