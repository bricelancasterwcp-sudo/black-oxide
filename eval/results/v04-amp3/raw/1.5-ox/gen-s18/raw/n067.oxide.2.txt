struct Reading { label: Str, values: Vec<Int> }

fn first_big(v: Vec<Int>) -> Int {
    let mut found = -1;
    for x in v {
        if x > 10 {
            found = x;
            break;
        }
    }
    found
}

fn second(v: Vec<Int>) -> Option<Int> {
    let x = get(v, 1)?;
    Some(x + 1)
}

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) };
    print(first_big(r.values));            // field access copies: r stays usable
    let mut total_even_sum = 0;
    for i in 0..r.values.len() {
        if i % 2 == 1 {
            let even_value = get(r.values, i)?;
            total_even_sum += even_value * 2;
        }
    }
    println_str(total_even_sum.to_string());
}
