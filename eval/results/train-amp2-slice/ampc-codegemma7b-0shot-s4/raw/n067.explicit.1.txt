fn walk(nums: &Vec<Int>) -> Int {
    let mut total = 0
    let mut doubled = 0
    for x in nums {
        if x % 2 == 1 {
            doubled = doubled * 10 + x
        } else {
            total = total + x
        }
    }
    print(doubled)
    print(total)
    total
}

fn main() {
    let nums = vec(7, 4, 9, 10)
    print(walk(&nums))
    drop nums
}
