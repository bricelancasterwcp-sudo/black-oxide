fn main() {
    let inputs = vec("7", "x", "9", "?");
    let mut sum = 0;
    let mut count_non_numbers = 0;
    for input in inputs {
        match parse_int(input) {
            Some(n) => sum += n,
            None => count_non_numbers += 1,
        }
    }
    print(sum);
    print(count_non_numbers);
}
