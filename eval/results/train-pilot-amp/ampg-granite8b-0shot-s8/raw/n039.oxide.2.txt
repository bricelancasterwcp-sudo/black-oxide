    cons: Option<Int>,
    cars: Option<List>,
}

fn cons(head: Int, tail: List) -> List {
    List  // return type can be omitted because it's
    (Some(head) == Some(tail.cons) && tail.cars == None)
}

fn nil() -> List {
    List(None, None)
}
