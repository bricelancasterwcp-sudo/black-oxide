fn second(v: Vec<Int>) -> Option<Int> {
    if len(v) > 1 {
        let x = get(v, 1)?
        Some(x + 1)
    } else {
        None
    }
}

fn main() {
    print(second(vec(7)))
    print(second(vec(7, 8)))
    print(second(vec(7, 8, 9)))
}
