struct Reading { label: Str, values: Vec<Int> }
fn analyze(reading: Reading) -> Option<Str> {
    let mut increases = 0
    let mut decreases = 0
    let mut max_increase = 0
    let mut longest_streak = 0
    let mut current_streak = 0
    for i in range(1, len(reading.values)) {
        let prev = reading.values[i - 1]
        let curr = reading.values[i]
        if curr > prev {
            increases += 1
            max_increase = max(max_increase, curr - prev)
            current_streak += 1
        } else if curr < prev {
            decreases += 1
            current_streak = 0
        }
        longest_streak = max(longest_streak, current_streak)
    }
    print(reading.label)
    print(increases)
    print(decreases)
    print(max_increase)
    print(longest_streak)
    None
}
fn main() {
    let north = Reading { label: "north", values: vec(12, 15, 19, 14, 14, 20, 31, 30) }
    let south = Reading { label: "south", values: vec(40, 38, 41, 45, 50, 44) }
    analyze(north)
    analyze(south)
    let north_streak = longest_streak(north.values)
    let south_streak = longest_streak(south.values)
    if north_streak > south_streak {
        print_str("north")
    } else {
        print_str("south")
    }
}
fn longest_streak(values: Vec<Int>) -> Int {
    let mut current_streak = 0
    let mut longest_streak = 0
    for i in range(1, len(values)) {
        if values[i] > values[i - 1] {
            current_streak += 1
        } else {
            current_streak = 0
        }
        longest_streak = max(longest_streak, current_streak)
    }
    longest_streak + 1
}
