struct Reading { label: Str, values: Vec<Int> }

fn count_exceed_five(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 5 {
            count += 1
        }
    }
    count
}

fn find_highest(v: Vec<Int>) -> Option<Int> {
    max(v)
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14)
    let readings = push(readings, 9)
    let r = Reading { label: "lab", values: readings }
    print(count_exceed_five(r.values))            // field access copies: r stays usable
    let highest = find_highest(r.values)
    match highest {
        Some(n) => print(n),
        None => print_str("no readings"),
    }
}
