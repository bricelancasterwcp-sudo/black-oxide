fn sum_of_numbers(values: Vec<Str>) -> Int {
    let mut sum = 0
    for v in values {
        match parse_int(v) {
            Some(n) => sum += n,
            None => (),
        }
    }
    sum
}

fn count_non_numbers(values: Vec<Str>) -> Int {
    let mut count = 0
    for v in values {
        if parse_int(v).is_none() {
            count += 1
        }
    }
    count
}

fn main() {
    let inputs = vec("7", "x", "9", "?")
    print(sum_of_numbers(inputs))
    print(count_non_numbers(inputs))
}
