```rust
struct Item {
    label: Str,
    price: Float,
    quantity: Int,
}

fn main() {
    let pen = Item { label: "pen", price: 3.0, quantity: 4 };
    let pad = Item { label: "pad", price: 9.0, quantity: 2 };
    let ink = Item { label: "ink", price: 5.0, quantity: 3 };

    let total_cost = pen.price * pen.quantity + pad.price * pad.quantity + ink.price * ink.quantity;
    print(total_cost);

    let mut highest_cost_item = &pen;
    for item in [pad, ink] {
        if item.price * item.quantity > highest_cost_item.price * highest_cost_item.quantity {
            highest_cost_item = &item;
        }
    }
    print_str(highest_cost_item.label);
}
```
