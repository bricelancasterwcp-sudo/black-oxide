fn main() {
    let samples = vec![5, 12, 3, 18, 9];
    let cutoff = 10;
    let under_count = samples.iter().filter(|&&x| x < cutoff).count();
    let over_or_equal_count = samples.iter().filter(|&&x| x >= cutoff).count();
    println!("Under {}: {}", cutoff, under_count);
    println!("At or over {}: {}", cutoff, over_or_equal_count);
}
