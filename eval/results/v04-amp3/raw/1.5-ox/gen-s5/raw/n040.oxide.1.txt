```rust
enum Shape {
    Square { side: Int },
    Circle { radius: Float },
}

fn main() {
    let shape = Shape::Square { side: 4 };
    match shape {
        Shape::Square { side } => print(int_to_str(side * side)),
        Shape::Circle { radius } => print(int_to_str((radius * radius) * pi)),
    }
}
```
