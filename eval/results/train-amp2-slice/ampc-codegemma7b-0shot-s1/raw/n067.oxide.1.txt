fn walk(nums: Vec<Int>) -> Int {
    let mut seen = 0
    let mut sum = 0
    for num in nums {
        if num % 2 == 1 {
            seen = seen + 1
            print(num * 2)
            println()
        } else {
            sum = sum + num
        }
    }
    sum
}

fn main() {
    let numbers = vec(7, 4, 9, 10)
    print(walk(numbers))
}
