fn main() {
    let values = vec![Some(7), None, Some(9), Some(42)];
    let mut sum = 0;
    let mut count_non_numbers = 0;

    for value in values {
        match value {
            Some(n) => sum += n,
            None => count_non_numbers += 1,
        }
    }

    print(sum);
    print(count_non_numbers);
}
