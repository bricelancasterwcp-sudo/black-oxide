struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding_five(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 5 {
            count += 1
        }
    }
    count
}

fn find_highest(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if match max {
            Some(current_max) => x > current_max,
            None => true,
        } {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = vec!(6, 11, 3, 14, 9)
    print(count_exceeding_five(clone(readings)))
    print("\n")
    match find_highest(readings) {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
