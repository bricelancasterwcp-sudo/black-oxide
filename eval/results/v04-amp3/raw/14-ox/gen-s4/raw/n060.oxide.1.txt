fn count_digits(s: Str) -> Int {
    let mut count = 0
    let mut sum = 0
    for ch in chars(s) {
        if parse_int(ch).is_some() {
            count += 1
            sum += parse_int(ch).unwrap()
        }
    }
    print(sum)
    print(count)
}

fn main() {
    count_digits("a1b22")
}
