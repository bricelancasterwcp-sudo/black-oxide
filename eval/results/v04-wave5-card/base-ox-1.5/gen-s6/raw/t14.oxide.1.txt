```rust
fn main() {
    let s = "mississippi";
    let mut count_s = 0;
    let mut seen_chars: Vec<char> = vec![];

    for c in s.chars() {
        if c == 's' {
            count_s += 1;
        }
        if !seen_chars.contains(&c) {
            seen_chars.push(c);
        }
    }

    println!("The character 's' occurs {} times.", count_s);
    println!("There are {} distinct characters in the string.", seen_chars.len());
}
```
