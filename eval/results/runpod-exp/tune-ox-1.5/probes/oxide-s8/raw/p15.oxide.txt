fn f(o: Option<Vec<Int>>) -> Option<Int> {
    let v = match o {
        Some(v) => v,
        None => return None,
    }
    let w = match o {
        Some(w) => w,
        None => return None,
    }
    Some(len(v) * 10 + len(w))
}
fn main() {
    match f(Some(push(push(vec(), 1), 2))) {
        Some(n) => print(n),
        None => print(0),
    }
    match f(None) {
        Some(n) => print(n),
        None => print(99),
    }
}
