fn count_s(s: Str) -> Int {
    let mut count = 0;
    for c in chars(s) {
        if c == "s" {
            count += 1;
        }
    }
    count
}

fn distinct_chars(s: Str) -> Int {
    let mut set = Vec<Str>();
    for c in chars(s) {
        if !contains(set, c) {
            push(set, c);
        }
    }
    len(set)
}

    let s = "mississippi";
