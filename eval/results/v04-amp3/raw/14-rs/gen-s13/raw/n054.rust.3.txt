fn main() {
    let word = "keyboard";
    let mut vowels_count = 0;
    let mut consonants_count = 0;

    for c in word.chars() {
        if is_vowel(c) {
            vowels_count += 1;
        } else if c.is_alphabetic() {
            consonants_count += 1;
        }
    }

    println!("Vowels: {}", vowels_count);
    println!("Consonants: {}", consonants_count);
}

fn is_vowel(c: char) -> bool {
    matches!(c.to_ascii_lowercase(), 'a' | 'e' | 'i' | 'o' | 'u')
}
