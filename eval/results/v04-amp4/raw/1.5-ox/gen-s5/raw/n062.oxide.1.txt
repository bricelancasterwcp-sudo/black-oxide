```rust
struct Signal { color: Str }

fn main() {
    let mut signals = vec![
        Signal { color: "green" },
        Signal { color: "red" },
        Signal { color: "yellow" }
    ];

    for s in signals.iter_mut() {
        match s.color.as_str() {
            "green" => print("go for green\n"),
            "red" => print("stop for red\n"),
            "yellow" => print("wait for yellow\n"),
            _ => println!("unknown color: {}", s.color)
        }
    }
}
```
