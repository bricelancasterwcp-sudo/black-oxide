```rust
fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut sum = 0;
    let mut non_numeric_count = 0;

    for text in texts {
        if let Ok(number) = text.parse::<i32>() {
            sum += number;
        } else {
            non_numeric_count += 1;
        }
    }

    println!("Total of numbers: {}", sum);
    println!("Number of non-numeric entries: {}", non_numeric_count);
}
```
