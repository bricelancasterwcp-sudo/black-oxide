fn main() {
    let s = "a1b22"
    let mut sum = 0
    let mut count = 0
    for c in chars(s) {
        if parse_int(c).is_some() {
            let num = unwrap_or(parse_int(c), 0)
            sum += to_float(num).trunc()
            count += 1
        }
    }
    print(sum)
    print(count)
}
