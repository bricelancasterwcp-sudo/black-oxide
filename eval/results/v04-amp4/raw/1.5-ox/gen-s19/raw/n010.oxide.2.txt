```rust
fn main() {
    let n = 91;
    if n == 2 || n == 3 {
        println!("yes");
    } else if n % 2 == 0 || n % 3 == 0 {
        println!("no");
    } else {
        for i in (5..=n as f64).step_by(6.0) {
            if n % i == 0 || n % (i + 2) == 0 {
                println!("no");
                return;
            }
        }
    }
}
```
