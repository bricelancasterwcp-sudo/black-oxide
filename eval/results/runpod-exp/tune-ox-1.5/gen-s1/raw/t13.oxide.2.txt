fn split_str(s: Str) -> Vec<Str> {
    let mut result = vec()
    let chars = chars(s)
    for c in chars {
        result = push(result, str(c))
    }
    result
}
fn get_char(chars: Vec<Str>, index: Int) -> Str {
    let s = get_str(chars, index)
    str(s)
}
fn is_palindrome(s: Str) -> Bool {
    let chars = split_str(s)
    let len = len(chars)
    for i in range(0, len / 2) {
        if get_char(chars, i) != get_char(chars, len - i - 1) {
            return false
        }
    }
    true
}
fn main() {
    print(is_palindrome("level"))
    print(is_palindrome("stereo"))
    print(is_palindrome("rotor"))
}
