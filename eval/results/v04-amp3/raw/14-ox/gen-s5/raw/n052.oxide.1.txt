struct Reading { label: Str, values: Vec<Int> }

fn find_n_positions(s: Str) -> Vec<Int> {
    let positions = vec()
    for (i, c) in chars(s).enumerate() {
        if c == "n" {
            positions = push(positions, i)
        }
    }
    positions
}

fn main() {
    let r = Reading { label: "banana", values: find_n_positions("banana") }
    for pos in r.values {
        print(pos)
    }
}
