```rust
fn main() {
    let input = "keyboard";
    let mut vowels_count = 0;
    let mut consonants_count = 0;

    for c in input.chars() {
        if matches!(c.to_ascii_lowercase(), 'a' | 'e' | 'i' | 'o' | 'u') {
            vowels_count += 1;
        } else if c.is_alphabetic() && !matches!(c.to_ascii_lowercase(), 'a' | 'e' | 'i' | 'o' | 'u') {
            consonants_count += 1;
        }
    }

    println!("Vowels: {}", vowels_count);
    println!("Consonants: {}", consonants_count);
}
```
