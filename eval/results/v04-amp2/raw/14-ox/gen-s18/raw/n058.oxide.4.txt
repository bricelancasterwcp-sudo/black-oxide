struct LetterCount { letter: Str, count: Int }

fn main() {
    let word = "coffee"
    let mut counts = vec()
    for c in chars(word) {
        let found = false
        for i in range(0, len(counts)) {
            let count = get(counts, i)
            if count.is_some() && count.unwrap().letter == c {
                let new_count = LetterCount { letter: c, count: count.unwrap().count + 1 }
                counts = push(counts, new_count)
                found = true
                break
            }
        }
        if !found {
            let new_count = LetterCount { letter: c, count: 1 }
            counts = push(counts, new_count)
        }
    }
    for count in counts {
        if count.count > 1 {
            print(count.letter)
        }
    }
}
