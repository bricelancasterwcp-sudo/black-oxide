```rust
fn main() {
    let figures = vec![8, 3, 11, 6];
    let min_figure = min(figures.clone())?;
    let max_figure = max(figures)?;
    print(min_figure);
    print(max_figure - min_figure);
}

fn min(v: Vec<Int>) -> Option<Int> {
    v.into_iter().min()
}

fn max(v: Vec<Int>) -> Option<Int> {
    v.into_iter().max()
}
```
