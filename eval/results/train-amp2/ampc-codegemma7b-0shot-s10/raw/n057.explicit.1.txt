fn find_longest(fruits: Vec<String>) -> String {
    let mut longest = String { len: 0, fruit: "".to_string() }
    for fruit in fruits {
        if fruit.len() > longest.len {
            longest.len = fruit.len()
            longest.fruit = fruit
        }
    }
    longest.fruit
}

fn main() {
    let fruits = vec("fig".to_string(), "banana".to_string(), "kiwi".to_string())
    print(find_longest(fruits))
}
