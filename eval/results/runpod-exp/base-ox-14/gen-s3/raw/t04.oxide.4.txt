fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        false
    } else {
        for i in range(2, n) {
            if n % i == 0 {
                return false
            }
        }
        true
    }
}

fn count_primes_below(limit: Int) -> Int {
    let mut count = 0
    for i in range(2, limit) {
        if is_prime(i) {
            count = count + 1
        }
    }
    count
}

fn main() {
    let primes_count = count_primes_below(100)
    print(primes_count)
}
