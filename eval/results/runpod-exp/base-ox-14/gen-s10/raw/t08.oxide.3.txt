struct Reading { label: Str, values: Vec<Int> }

fn count_positives(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count = count + 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max.is_none() || x > max.unwrap() {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let values = push(push(push(push(push(vec(), 3), 8), -2), 12), 7)
    let r = Reading { label: "numbers", values: clone(values) }
    print(count_positives(clone(r.values)))
    match find_max(r.values) {
        Some(n) => print(n),
        None => print_str("no values"),
    }
}
