```rust
fn main() {
    let a1b22 = "a1b22";
    let mut total = 0;
    let mut count = 0;

    for c in a1b22.chars() {
        if c.is_ascii_digit() {
            total += c.to_digit(10).unwrap() as i32;
            count += 1;
        }
    }

    println!("total: {}", total);
    println!("count: {}", count);
}
```
