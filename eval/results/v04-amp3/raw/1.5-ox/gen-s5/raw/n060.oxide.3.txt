```rust
use std::io;

fn main() {
    let mut sum = 0;
    let mut count = 0;
    for c in io::stdin().read_line()?.chars() {
        if c.is_digit(10) {
            sum += c.to_digit(10).unwrap();
            count += 1;
        }
    }
    println!("Sum: {}", sum);
    println!("Count: {}", count);
}
```
