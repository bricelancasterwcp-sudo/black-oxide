fn is_prime(n: Int) -> Bool {
    if n <= 1 {
        return false
    }
    for i in range(2, n) {
        if n % i == 0 {
            return false
        }
    }
    true
}

fn main() {
    let x = parse_int(input())
    if is_prime(x) {
        print("yes")
    } else {
        print("no")
    }
}
