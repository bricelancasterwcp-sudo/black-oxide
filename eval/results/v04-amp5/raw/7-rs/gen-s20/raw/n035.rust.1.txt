```rust
fn divide(x: i32, y: i32) -> Result<i32, &'static str> {
    if y == 0 {
        Err("bad")
    } else {
        Ok(x / y)
    }
}

fn main() {
    match divide(20, 4) {
        Ok(result) => println!("{}", result),
        Err(e) => println!("{}", e),
    }

    match divide(20, 0) {
        Ok(result) => println!("{}", result),
        Err(e) => println!("{}", e),
    }
}
```
