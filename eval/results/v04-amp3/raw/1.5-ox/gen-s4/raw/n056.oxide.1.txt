fn main() {
    let texts = vec!["7", "x", "9", "?"];
    let mut count = 0;
    let mut sum = 0;

    for text in texts {
        if let Some(number) = parse_int(text) {
            sum += number;
            count += 1;
        }
    }

    print!("Total of the ones that do: {}", sum);
    print!("\nNumber of ones that do not: {}", texts.len() - count);
}
