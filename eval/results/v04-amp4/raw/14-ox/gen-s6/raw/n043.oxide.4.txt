struct Queue { values: Vec<Int> }

fn swap_first_last(q: Queue) -> Queue {
    let len = len(q.values)
    if len < 2 {
        return q
    }
    let first = unwrap_or(get(q.values, 0), -1)
    let last = unwrap_or(get(q.values, len - 1), -1)
    let middle = vec()
    for i in range(1, len - 1) {
        middle = push(middle, unwrap_or(get(q.values, i), -1))
    }
    Queue { values: push(push(push(middle, last), first), unwrap_or(get(q.values, len - 2), -1)) }
}

fn main() {
    let q = Queue { values: vec(1, 2, 3, 4, 5) }
    let q = swap_first_last(q)
    for x in q.values {
        print(x)
        print_str("\n")
    }
}
