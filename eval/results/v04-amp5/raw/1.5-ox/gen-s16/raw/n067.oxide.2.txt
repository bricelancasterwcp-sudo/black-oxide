```rust
fn main() {
    let mut list = vec![7, 4, 9, 10];
    let mut sum_of_evens = 0;

    for &x in list.iter() { // Changed `&list` to `list.iter()` for borrow checker compatibility
        if x % 2 != 0 {
            println!("{}", 2 * x);
        } else {
            sum_of_evens += x;
        }
    }

    println!("Sum of even entries: {}", sum_of_evens);
}
```
