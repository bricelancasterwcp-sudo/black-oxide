fn divide(a: Int, b: Int) -> Option<Int> {
    if b == 0 {
        None
    } else {
        Some(a / b)
    }
}

fn main() {
    let answer = divide(20, 4);
    match answer {
        Some(result) => print_str(&format!("The result is {}", result)),
        None => print_str("Division by zero"),
    }
}
