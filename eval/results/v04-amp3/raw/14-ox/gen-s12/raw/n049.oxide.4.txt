fn main() {
    let figures = push(push(push(vec(), 8), 3), push(push(vec(), 11), 6))
    let min_figure = unwrap_or(min(figures.clone()), -1)
    let max_figure = unwrap_or(max(figures), -1)
    let gap = to_float(max_figure) - to_float(min_figure)
    print(min_figure)
    print(gap)
}
