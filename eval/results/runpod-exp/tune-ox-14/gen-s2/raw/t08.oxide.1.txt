fn main() {
    let nums = vec(3, 8, -2, 12, 7)
    let mut pos_count = 0
    let mut max_val = -1000000
    for num in nums {
        if num > 0 {
            pos_count = pos_count + 1
        }
        if num > max_val {
            max_val = num
        }
    }
    print(pos_count)
    print(max_val)
}
