struct Reading { label: Str, values: Vec<Int> }

fn max_readings(v: Vec<Int>) -> Int {
    let mut max = -1
    for x in v {
        if x > max {
            max = x
        }
    }
    max
}

fn main() {
    let r = Reading { label: "lab", values: push(push(push(push(vec(), 5), 3), 9), 1), ..Reading(label == "lab", values == push(vec(), 7)) } // Corrected to push multiple values
    print(max_readings(r.values))
}
