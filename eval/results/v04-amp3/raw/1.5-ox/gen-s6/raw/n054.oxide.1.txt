```rust
fn is_vowel(c: Str) -> Bool {
    let vowels = vec!['a', 'e', 'i', 'o', 'u'];
    vowels.contains(&c)
}

fn main() {
    let mut vowel_count = 0;
    let mut consonant_count = 0;

    for ch in "keyboard".chars() {
        if is_vowel(ch) {
            vowel_count += 1;
        } else {
            consonant_count += 1;
        }
    }

    println!("Vowels: {}", vowel_count);
    println!("Consonants: {}", consonant_count);
}
```
