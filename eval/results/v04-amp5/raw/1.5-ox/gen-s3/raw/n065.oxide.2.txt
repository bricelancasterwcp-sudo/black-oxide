struct Reading { label: Str, values: Vec<Int> }

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    println(first_big(r.values));            // field access copies: r stays usable
    match second(r.values) {
        Some(n) => print(n),
        None => print_str("too short"),
    }

    let v = vec![6, 11, 3, 14, 9];
    println(count_if(&v, |&x| x > 5));    // count how many values exceed five
}
