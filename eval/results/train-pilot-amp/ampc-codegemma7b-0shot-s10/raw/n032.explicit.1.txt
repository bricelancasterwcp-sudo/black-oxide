    print(start.y + end.x)

struct Point {
    x: Int,
    y: Int,
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    let mut result = v
    result.push(x)
    result
}

fn print(x: Int) {
    print_str(int_to_str(x))
}

fn int_to_str(n: Int) -> Str {
    if n < 0 {
        return concat(Str("-"), int_to_str(-n))
    }
    if n >= 10 {
        return concat(int_to_str(n / 10), Str(char_from_int((n % 10) + 48)))
    }
    Str(char_from_int(n + 48))
}

fn char_from_int(n: Int) -> Str {
    let mut result = Str("")
    for i in 0.to_int() {
        if i == n {
            result = concat(result, Str(char(i)))
        }
    }
    result
}

fn concat(a: Str, b: Str) -> Str {
    let mut result = a
    for c in b.chars() {
        result = concat(result, Str(c))
    }
    result
}

fn print_str(s: Str) {
    for c in s.chars() {
        print_char(c)
    }
}

fn print_char(c: Str) {
    if c == Str("\n") {
        print_str("\\n")
    } else {
        print_str(c)
    }
}

fn clone(x: &Point) -> Point {
    Point(x.x, x.y)
}

fn len(v: &Vec<Int>) -> Int {
    let mut result = 0
    for _ in v {
        result = result + 1
    }
    result
}

fn get(v: &Vec<Int>, i: Int) -> Option<Int> {
    if i < len(v) {
        return Some(v.get(i).unwrap())
    }
    None
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec()
    for i in a.to_int() {
        if i < b {
            result.push(i)
        }
    }
    result
}

fn bool(b: Int) -> Bool {
    if b != 0 {
        return True
    }
    False
}
