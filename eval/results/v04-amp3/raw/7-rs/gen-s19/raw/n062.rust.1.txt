```rust
fn main() {
    match std::env::args().nth(1) {
        Some(signal) => {
            match signal.as_str() {
                "green" => println!("go"),
                "red" => println!("stop"),
                "yellow" => println!("wait"),
                _ => println!("unknown signal"),
            }
        },
        None => println!("no signal provided"),
    }
}
```
