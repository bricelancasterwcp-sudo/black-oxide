```rust
enum Shape {
    Square(Int),
    Circle(Float)
}

fn area(s: Shape) -> Float {
    match s {
        Shape::Square(side) => to_float(side * side),
        Shape::Circle(radius) => 3.14159 * radius * radius
    }
}

fn main() {
    let square = Shape::Square(4);
    print(area(square));
}
```
