```rust
fn main() {
    let s = "keyboard";
    let mut count_vowels = 0;
    let mut count_consonants = 0;

    for c in s.chars() {
        match c.to_lowercase().next() {
            Some('a') | Some('e') | Some('i') | Some('o') | Some('u') => count_vowels += 1,
            _ => count_consonants += 1,
        }
    }

    print(count_vowels);
    print_str(" ");
    print(count_consonants);
}
```
