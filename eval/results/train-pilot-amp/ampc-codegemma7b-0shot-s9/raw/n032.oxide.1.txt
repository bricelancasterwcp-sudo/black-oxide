fn print(n: Int) {
    print_str(to_str(n))
}

fn print_str(s: Str) {
    print(s)
}

fn to_str(n: Int) -> Str {
    int_to_str(n)
}

fn int_to_str(n: Int) -> Str {
    "0123456789abcdefghijklmnopqrstuvwxyz"
    let mut res = ""
    let mut n = n
    while n > 0 {
        let c = char(n % 10 + ord("0"))
        res = c + res
        n = n / 10
    }
    res
}

fn char(n: Int) -> Str {
    let mut buf = vec(0)
    buf.push(n)
    Str(buf)
}

fn ord(c: Str) -> Int {
    c.chars().next().unwrap() - ord("0")
}

fn ord(c: Str) -> Int {
    let mut chars = c.chars()
    let first = chars.next().unwrap()
    let mut sum = 0
    for c in chars {
        sum = sum * 26 + int(c) - int("a")
    }
    sum + int(first) - int("a")
}

fn int(c: Str) -> Int {
    ord(c) - ord("0")
}

fn push(v: Vec<T>, x: T) -> Vec<T> {
    let mut v = v
    v.push(x)
    v
}

fn len(v: Vec<T>) -> Int {
    let mut n = 0
    for _ in v {
        n = n + 1
    }
    n
}

fn get(v: Vec<T>, i: Int) -> Option<T> {
    if i >= len(v) {
        None
    } else {
        Some(v.get(i).unwrap())
    }
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut res = vec()
    for i in a + 1 < b {
        res.push(i)
    }
    res
}

    x
