struct Queue { entries: Vec<Int> }

fn main() {
    let mut q = Queue { entries: push(push(push(push(vec(), 1), 2), 3), 4) }
    let first = get(q.entries, 0)?.clone()
    let last = get(q.entries, len(q.entries) - 1)?.clone()
    let mut entries = q.entries
    entries = push(push(pop(entries), last), first)
    for e in entries {
        print(e)
    }
}
