fn main() {
    let log = push(push(vec(), 6), 7)
    let first = pop(&log)
    drop log
    let last = pop(&log)
    drop log
    print(first)
    print(last)
}

fn pop(log: &Vec<Int>) -> Int {
    let mut last = 0
    for i in 0.upto(len(&log) - 1) {
        last = get(&log, i).unwrap()
    }
    last
}
