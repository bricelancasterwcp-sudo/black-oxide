fn main() {
    let s = "keyboard";
    let (mut vowel_count, mut non_vowel_count) = (0, 0);
    for c in chars(s) {
        if c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' ||
           c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U' {
            vowel_count += 1;
        } else {
            non_vowel_count += 1;
        }
    }
    print(vowel_count);
    print_str(" ");
    print(non_vowel_count);
}
