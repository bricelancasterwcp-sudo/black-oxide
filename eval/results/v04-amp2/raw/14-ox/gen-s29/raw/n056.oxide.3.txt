struct NumberHolder { value: Option<Int> }

fn sum_and_count_unknowns(numbers: Vec<NumberHolder>) -> (Int, Int) {
    let mut sum = 0
    let mut unknown_count = 0
    for holder in numbers {
        match holder.value {
            Some(n) => sum += n,
            None => unknown_count += 1,
        }
    }
    (sum, unknown_count)
}

fn main() {
    let numbers = vec![
        NumberHolder { value: Some(7) },
        NumberHolder { value: None },
        NumberHolder { value: Some(9) },
        NumberHolder { value: None },
    ]
    let (total, unknowns) = sum_and_count_unknowns(numbers)
    print(total)
    print(unknowns)
}
