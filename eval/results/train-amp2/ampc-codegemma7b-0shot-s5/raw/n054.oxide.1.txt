fn main() {
    let word = "keyboard".to_lowercase().chars().collect_vec()
    let mut vowels = 0
    let mut not_vowels = 0
    for letter in word {
        if letter == "a" || letter == "e" || letter == "i" || letter == "o" || letter == "u" {
            vowels = vowels + 1
        } else {
            not_vowels = not_vowels + 1
        }
    }
    print(vowels)
    print(not_vowels)
}
