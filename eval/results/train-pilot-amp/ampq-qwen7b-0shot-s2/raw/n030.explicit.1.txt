fn count_vowels(words: Vec<Str>) -> Int {
    let mut count = 0
    for word in &words {
        for c in chars(word) {
            if c == "a" || c == "e" || c == "i" || c == "o" || c == "u" {
                count = count + 1
            }
        }
    }
    count
}

fn main() {
    let words = vec("hello", "world")
    let vowels = count_vowels(words)
    print(vowels)
}
