```rust
fn main() {
    let unit_price = 4;
    let price_per_five_units = 5;

    let four_units_value = (unit_price * 3) * 4;
    println!("Value of the first line: {}", four_units_value);

    let five_units_value = (price_per_five_units * 2) * 5;
    println!("Value of the second line: {}", five_units_value);

    let total_value = four_units_value + five_units_value;
    println!("Total value of both lines: {}", total_value);
}
```
