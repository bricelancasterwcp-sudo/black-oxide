struct Phrase { label: Str, words: Vec<Str> }
fn main() {
    let first = "the harbour lights flickered"
    let second = "a small boat drifted past silently"
    let p1 = parse_phrase(first)
    let p2 = parse_phrase(second)
    print(p1.label)
    print(len(p1.words))
    print(max_len(p1.words))
    print(min_len(p1.words))
    print(count_long_words(p1.words, 4))
    print(p2.label)
    print(len(p2.words))
    print(max_len(p2.words))
    print(min_len(p2.words))
    print(count_long_words(p2.words, 4))
    if len(p1.words) > len(p2.words) {
        print_str(p1.label)
    } else {
        print_str(p2.label)
    }
}
fn parse_phrase(s: Str) -> Phrase {
    let words = split(s, " ")
    Phrase { label: s, words: words }
}
fn max_len(words: Vec<Str>) -> Int {
    let mut max_len = 0
    for word in words {
        let len = str_len(word)
        if len > max_len {
            max_len = len
        }
    }
    max_len
}
fn min_len(words: Vec<Str>) -> Int {
    let mut min_len = 1000000 // Arbitrary large number
    for word in words {
        let len = str_len(word)
        if len < min_len {
            min_len = len
        }
    }
    min_len
}
fn count_long_words(words: Vec<Str>, threshold: Int) -> Int {
    let mut count = 0
    for word in words {
        if str_len(word) > threshold {
            count += 1
        }
    }
    count
}
