fn count_exceed_five(v: Vec<Int>) -> Int {
    count_if(v, x -> x > 5)
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    max(v)
}

fn main() {
    let readings = push(push(push(push(push(vec(), 6), 11), 3), 14), 9)
    print(count_exceed_five(readings))
    let max_value = max_reading(readings)?
    print(max_value)
}
