struct Log { entries: Vec<Int> }

fn first(log: Log) -> Int {
    let entry = get(log.entries, 0)?
    print(entry)
    entry
}

fn last(log: Log) -> Int {
    let entry = get(log.entries, len(log.entries) - 1)?
    print(entry)
    entry
}

fn main() {
    let log = Log { entries: push(push(push(vec(), 6), 7), 8) }
    print(last(log))
}
