fn main() {
    let values = vec![5, 12, 3, 18, 9];
    let count_under_cutoff = values.iter().filter(|&x| x < 10).count();
    let count_at_or_over_cutoff = values.iter().filter(|&x| x >= 10).count();

    println!("Number of values under the cutoff: {}", count_under_cutoff);
    println!("Number of values at or over the cutoff: {}", count_at_or_over_cutoff);
}
