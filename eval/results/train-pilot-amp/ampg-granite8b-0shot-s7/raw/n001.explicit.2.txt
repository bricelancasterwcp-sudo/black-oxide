fn sum_cubes(v: &Vec<Int>, limit: Int) -> Int {
    let total = 0
    for x in &v {
        if x > limit {
            total = total + x * x * x
        }
    }
    total
}

fn main() {
    let nums = push(push(vec(), 1), 2)
    let bigger = extend(clone(&nums), 3)
    print(sum_cubes(&nums, 1))
    drop nums
    print(sum_cubes(&bigger, 1))
    drop bigger
}
