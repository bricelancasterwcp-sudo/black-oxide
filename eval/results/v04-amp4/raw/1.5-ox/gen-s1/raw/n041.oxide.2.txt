struct Reading { label: Str, values: Vec<Int> }

fn main() {
    let r = Reading { label: "lab", values: push(push(vec(), 3), 42) }
    println!("{}", *r.values.first().unwrap()); // print the smallest
    println!("{}", *r.values.get(1).unwrap());   // print the second smallest
    println!("{}", *r.values.get(2).unwrap());   // print the third smallest
    println!("{}", *r.values.last().unwrap());  // print the largest
}
