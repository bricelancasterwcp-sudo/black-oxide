struct Reading { label: Str, values: Vec<Int> }

fn count_exceeding_five(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 5 {
            count += 1
        }
    }
    count
}

fn find_highest(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        if max.is_none() || x > unwrap_or(max, -1) {
            max = Some(x)
        }
    }
    max
}

fn main() {
    let readings = push(push(push(push(push(vec(), 6), 11), 3), 14), 9)
    print(count_exceeding_five(clone(readings)))
    let highest = find_highest(readings)
    match highest {
        Some(n) => print(n),
        None => print_str("No readings"),
    }
}
