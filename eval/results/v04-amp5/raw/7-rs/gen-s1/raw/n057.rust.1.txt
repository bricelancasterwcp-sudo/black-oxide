fn main() {
    let names = vec!["fig", "banana", "kiwi"];
    let mut max_length = 0;
    let mut longest_name = "";

    for name in names.iter() {
        if name.len() > max_length {
            max_length = name.len();
            longest_name = name;
        }
    }

    println!("{} {}", longest_name, max_length);
}
