struct Queue { values: Vec<Int> }

fn main() {
    let q = Queue { values: vec(1, 2, 3, 4, 5) }
    let first = get(q.values, 0)?
    let last = get(q.values, len(q.values) - 1)?
    let q2 = Queue { values: push(push(push(push(push(vec(), last), first), 4), 3), 2) }
    for x in q2.values {
        print(x)
    }
}
