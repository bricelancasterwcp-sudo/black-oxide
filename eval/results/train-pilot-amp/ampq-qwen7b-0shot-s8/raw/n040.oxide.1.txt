struct Square { side: Int }

struct Circle { radius: Int }

fn area_of_square(s: Square, pi: Float) -> Float {
    pi * (s.side * s.side).to_float()
}

fn area_of_circle(c: Circle, pi: Float) -> Float {
    pi * (c.radius * c.radius).to_float()
}

fn main() {
    let square = Square { side: 4 }
    let circle = Circle { radius: 3 }
    let pi = 3.0
    print(area_of_square(square, pi))
    print(area_of_circle(circle, pi))
}
