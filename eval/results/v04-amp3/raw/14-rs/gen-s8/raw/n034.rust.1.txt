fn main() {
    let list1 = vec![7, 8, 9];
    let list2: Vec<i32> = vec![];
    let list3 = vec![42];

    println!("{:?}", second_entry(&list1)); // Some(8)
    println!("{:?}", second_entry(&list2)); // None
    println!("{:?}", second_entry(&list3)); // None
}

fn second_entry<T>(list: &[T]) -> Option<&T> {
    if list.len() > 1 {
        Some(&list[1])
    } else {
        None
    }
}
