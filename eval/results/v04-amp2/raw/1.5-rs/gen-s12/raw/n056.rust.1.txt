```rust
fn main() {
    let numbers = vec![7, 10, 9, '?'];
    let mut sum = 0;
    let mut count = 0;

    for &num in &numbers {
        if num != '?' {
            sum += num;
            count += 1;
        }
    }

    println!("Sum of numbers: {}", sum);
    println!("Count of non-? numbers: {}", count);
}
```
