```rust
fn main() {
    let readings = [6, 11, 3, 14, 9];
    let mut count = 0;
    let mut highest = readings[0];

    for &reading in readings.iter() {
        if reading > 5 {
            count += 1;
        }
        if reading > highest {
            highest = reading;
        }
    }

    println!("{}", count);
    println!("{}", highest);
}
```
