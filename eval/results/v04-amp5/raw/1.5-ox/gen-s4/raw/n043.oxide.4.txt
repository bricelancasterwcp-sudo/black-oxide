```rust
struct Queue {
    values: Vec<Int>,
}

fn main() {
    let mut q = Queue { values: vec![1, 2, 3, 4, 5] };
    
    // Exchange the first and last entries
    let temp = q.values.remove(0);
    q.values.insert(0, temp);
    
    // Print the queue one entry per line
    for val in &q.values {
        println!("{}", val);
    }
}
```
