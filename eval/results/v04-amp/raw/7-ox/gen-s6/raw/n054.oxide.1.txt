fn main() {
    let vowels = "aeiouAEIOU";
    let mut count_vowels = 0;
    let mut count_non_vowels = 0;

    for c in "keyboard" {
        if vowels.contains(c) {
            count_vowels += 1;
        } else {
            count_non_vowels += 1;
        }
    }

    print(count_vowels);
    print_str(" ");
    print(count_non_vowels);
}
