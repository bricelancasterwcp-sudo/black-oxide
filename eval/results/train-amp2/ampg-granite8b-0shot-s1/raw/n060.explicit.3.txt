fn count_and_sum_digits(s: Str) -> Str {
    let mut total = 0
    let mut count = 0
    for c in chars(&s) {
        if c >= "0" && c <= "9" {
            total = total + (c - "0")
            count = count + 1
        }
    }
    print_str("Total: ")
    print(total)
    print_str(" Count: ")
    print(count)
    print_str("\n")
    (total + " " + count).to_string()
}

fn main() {
    let nums = "12345"
    let total = count_and_sum_digits(&nums)
    drop nums
    drop total
}
