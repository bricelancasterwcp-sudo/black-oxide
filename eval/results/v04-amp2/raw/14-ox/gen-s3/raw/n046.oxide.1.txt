fn count_below_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut below = 0
    for x in v {
        if x < cutoff {
            below += 1
        }
    }
    below
}

fn count_at_or_above_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut at_or_above = 0
    for x in v {
        if x >= cutoff {
            at_or_above += 1
        }
    }
    at_or_above
}

fn main() {
    let values = vec![5, 12, 3, 18, 9]
    let cutoff = 10
    let below = count_below_cutoff(values.clone(), cutoff)
    let at_or_above = count_at_or_above_cutoff(values, cutoff)
    print(below)
    print(at_or_above)
}
