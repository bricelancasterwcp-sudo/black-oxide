fn main() {
    let names = ["fig", "banana", "kiwi"];
    let mut longest_name = "";
    let mut max_length = 0;

    for &name in names.iter() {
        let length = name.len();
        if length > max_length {
            longest_name = name;
            max_length = length;
        }
    }

    println!("{} {}", longest_name, max_length);
}
