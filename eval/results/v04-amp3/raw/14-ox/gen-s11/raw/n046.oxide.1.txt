struct Reading { label: Str, values: Vec<Int> }

fn below_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x < cutoff {
            count += 1
        }
    }
    count
}

fn above_or_equal_cutoff(v: Vec<Int>, cutoff: Int) -> Int {
    let mut count = 0
    for x in v {
        if x >= cutoff {
            count += 1
        }
    }
    count
}

fn main() {
    let readings = vec![5, 12, 3, 18, 9]
    let cutoff = 10
    print(below_cutoff(readings, cutoff))
    print(above_or_equal_cutoff(readings, cutoff))
}
