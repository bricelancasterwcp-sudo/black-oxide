enum Signal {
    Green,
    Red,
    Yellow,
}

fn interpret(signal: Signal) -> Unit {
    match signal {
        Green => print_str("go"),
        Red => print_str("stop"),
        Yellow => print_str("wait"),
    }
}

fn main() -> Unit {
    let signals = vec![Green, Red, Yellow];
    for signal in signals {
        interpret(signal);
    }
}
