fn count_and_max(v: Vec<Int>) -> (Int, Int) {
    let mut count = 0;
    let mut max_val = -1000;
    for x in v {
        if x > 0 {
            count += 1;
        }
        if x > max_val {
            max_val = x;
        }
    }
    (count, max_val)
}

fn main() {
    let values = push(push(push(push(vec(), 3), 8), -2), 12);
    values.push(7);
    let (positive_count, max_value) = count_and_max(values);
    print(positive_count);
    print_str(" ");
    print(max_value);
}
