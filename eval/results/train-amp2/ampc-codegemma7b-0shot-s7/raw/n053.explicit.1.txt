fn drum() -> Str {
    let mut r = Str { chars: vec() }
    for c in "drum".chars() {
        r = push(r, c)
    }
    r
}

fn main() {
    let drum = drum()
    print(drum)
    print(reverse(&drum))
    drop drum
}

fn reverse(s: &Str) -> Str {
    let mut r = Str { chars: vec() }
    for c in &s {
        r = push(r, c)
    }
    r
}

fn push(mutv: Str, c: Char) -> Str {
    mutv.chars.push(c)
    mutv
}

struct Str {
    chars: Vec<Char>,
}

fn print(s: Str) {
    for c in &s {
        print(c)
    }
    print("\n")
}

fn drop_str(s: &Str) {
    for c in &s {
        drop c
    }
}

fn len(s: &Str) -> Int {
    s.chars.len()
}

fn get(s: &Str, i: Int) -> Option<Char> {
    if i >= len(s) {
        None
    } else {
        Some(s.chars.get(i).unwrap())
    }
}

fn chars(s: &Str) -> Vec<Char> {
    let mut v = vec()
    for c in &s {
        v.push(c)
    }
    v
}

fn str_len(s: &Str) -> Int {
    s.chars.len()
}

fn concat(a: Str, b: Str) -> Str {
    let mut v = a.chars
    for c in b.chars {
        v.push(c)
    }
    Str(v)
}

fn clone(s: &Str) -> Str {
    Str(chars(&s))
}

fn to_float(n: Int) -> Float {
    n
}

fn trunc(x: Float) -> Int {
    x
}

fn int_to_str(n: Int) -> Str {
    let mut v = vec()
    for c in n.to_string().chars() {
        v.push(c)
    }
    Str(v)
}
