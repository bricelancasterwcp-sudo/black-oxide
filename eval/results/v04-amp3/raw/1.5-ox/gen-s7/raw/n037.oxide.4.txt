```rust
struct Rectangle { width: Int, height: Int }

fn main() {
    let mut rect = Rectangle { width: 2, height: 3 };
    println("Initial area: {}", rect.area());

    // To simulate a long running loop
    for _ in 0..1000 {} 

    let new_width = 5;
    // Move the assignment outside the loop to avoid ownership issues
    rect.width = new_width;

    println("Updated area: {}", rect.area());
}
```
