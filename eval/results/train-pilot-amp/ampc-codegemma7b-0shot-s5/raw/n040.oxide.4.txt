fn square_area(side: Float) -> Float {
    side * side
}

fn area(shape: Shape) -> Float {
    match shape {
        Shape(variant) => match variant {
            Variant(square) => square_area(square),
            Variant(circle) => circle * circle * 3.14159,
        },
    }
}

fn main() {
    print(area(Shape(Variant(Square(4.0)))))
}
