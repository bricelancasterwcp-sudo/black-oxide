use std::io;

fn is_prime(num: u32) -> bool {
    if num <= 1 {
        return false;
    }
    for i in 2..=(num / 2) {
        if num % i == 0 {
            return false;
        }
    }
    true
}

fn main() {
    let num = 91;
    if is_prime(num) {
        println!("yes");
    } else {
        println!("no");
    }
}
