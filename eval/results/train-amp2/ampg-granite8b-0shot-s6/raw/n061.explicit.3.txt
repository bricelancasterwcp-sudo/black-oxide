fn main() {
    let rect = Rect { across: 9, down: 3 }
    print(area(&rect))
    print(perimeter(&rect))
    drop rect
}

struct Rect {
    across: Int,
    down: Int,
}

fn area(r: &Rect) -> Int {
    r.across * r.down
}

fn perimeter(r: &Rect) -> Int {
    2 * (r.across + r.down)
}

}
