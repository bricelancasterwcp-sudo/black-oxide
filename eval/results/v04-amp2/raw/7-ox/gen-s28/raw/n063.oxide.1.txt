struct Line { items: Vec<Int>, price_per_item: Int }

fn total_value(line: Line) -> Int {
    let mut total = 0;
    for _ in range(0, line.items.len()) {
        total += line.price_per_item;
    }
    total
}

fn main() {
    let line1 = Line { items: push(push(vec(), 4), 4), price_per_item: 3 };
    let line2 = Line { items: push(push(vec(), 5), 5), price_per_item: 2 };
    print(total_value(line1))
    print_str("\n")
    print(total_value(line2))
    print_str("\n")
    print(total_value(Line { items: push(line1.items, line2.items), price_per_item: 0 }))
}
