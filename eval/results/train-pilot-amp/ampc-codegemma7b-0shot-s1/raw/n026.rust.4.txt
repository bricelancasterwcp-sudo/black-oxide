use std::fmt::Write;

fn main() {
    let mut x = 2026;
    let mut s = String::new();

    loop {
        let r = x % 10;
        x /= 10;

        match r {
            0 => s.push('0'),
            1 => s.push('1'),
            2 => s.push('2'),
            3 => s.push('3'),
            4 => s.push('4'),
            5 => s.push('5'),
            6 => s.push('6'),
            7 => s.push('7'),
            8 => s.push('8'),
            9 => s.push('9'),
            _ => unreachable!(),
        }

        if x == 0 {
            break;
        }
    }

    // Reverse the string
    let mut temp = String::new();
    for c in s.chars().rev() {
        temp.push(c);
    }
    s = temp;

    let y = s.len();
    println!("The number of characters is {}", y);
}
