fn len(s: &Str) -> Int {
    let mut total = 0
    for c in chars(s) {
        total = total + 1
    }
    total
}

struct Str {
    ptr: RawPtr,
    len: Int,
}

fn str_lit(s: &Str) -> Str {
    let s = str_len(s)
    let mut chars = vec()
    for i in 0 < s {
        chars.push(get(s, i))
    }
    Str(chars.ptr, s)
}

fn chars(s: &Str) -> Vec<Char> {
    let mut chars = vec()
    for i in 0 < s.len {
        chars.push(get(s.ptr, i))
    }
    chars
}

fn get(ptr: RawPtr, i: Int) -> Char {
    let raw = get_raw(ptr, i)
    Char(raw)
}

fn push(vec: Vec<T>, x: T) -> Vec<T> {
    let mut vec = vec
    vec.ptr = realloc(vec.ptr, vec.len + 1)
    vec.len = vec.len + 1
    write(vec.ptr + ((vec.len - 1) * SIZE_T), x)
    vec
}

fn realloc(ptr: RawPtr, n: Int) -> RawPtr {
    let new_ptr = malloc(n)
    copy(ptr, new_ptr, n)
    free(ptr)
    new_ptr
}

fn copy(from: RawPtr, to: RawPtr, n: Int) {
    for i in 0 < n {
        write(to + (i * SIZE_T), read(from + (i * SIZE_T)))
    }
}

    unsafe
    ptr.write(x)
}

fn malloc(n: Int) -> RawPtr {
    unsafe
    RawPtr(alloc(n * SIZE_T))
}

fn alloc(n: Int) -> Raw {
    let raw = malloc(n)
    raw.set_zero()
    raw
}

fn free(ptr: RawPtr) {
    unsafe
    dealloc(ptr)
}

enum Raw {
    Ptr(RawPtr),
    Val(T),
}

fn size(raw: &Raw) -> Int {
    match raw {
        RawPtr(ptr) => SIZE_PTR,
        RawVal(x) => SIZE_T,
    }
}

fn free(raw: &Raw) {
    match raw {
        RawPtr(ptr) => drop_ptr(ptr),
        RawVal(x) => drop_t(x),
    }
}

fn copy(src: &Raw, dst: &Raw) {
    for i in 0 < size(src) {
        write(dst + i, read(src + i))
    }
}

fn drop_t(t: &T) {
    unsafe
    drop_raw(t)
}

fn drop_raw(raw: &Raw) {
    match raw {
        RawPtr(ptr) => drop_ptr(ptr),
        RawVal(x) => drop_ref(x),
    }
}
