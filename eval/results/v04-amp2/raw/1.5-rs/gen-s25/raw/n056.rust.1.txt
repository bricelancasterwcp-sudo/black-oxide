```rust
fn main() {
    let mut numbers = [7, 0, 9, ?];
    let mut count = 0;

    for &num in &numbers {
        if num != 0 {
            println!("{}", num);
            count += 1;
        }
    }

    println!("Total of the ones: {}", count);
    println!("Number of zeros: {}", numbers.len() - count);
}
```
