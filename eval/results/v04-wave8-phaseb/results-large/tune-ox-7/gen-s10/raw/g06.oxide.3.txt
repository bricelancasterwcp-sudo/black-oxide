struct Sitting { name: Str, scores: Vec<Int> }
fn count_in_range(scores: Vec<Int>, min: Int, max: Int) -> Int {
    let mut count = 0
    for score in scores {
        if score >= min && score <= max {
            count += 1
        }
    }
    count
}
fn main() {
    let morning_sitting = Sitting { name: "Morning", scores: vec(94, 71, 88, 65, 92, 79, 83) }
    let evening_sitting = Sitting { name: "Evening", scores: vec(55, 90, 77, 84, 68, 99) }
    for sitting in vec(clone(morning_sitting), clone(evening_sitting)) {
        print_str(sitting.name)
        let count_90_or_above = count_in_range(sitting.scores, 90, 100)
        let count_80_to_89 = count_in_range(sitting.scores, 80, 89)
        let count_70_to_79 = count_in_range(sitting.scores, 70, 79)
        let count_below_70 = count_in_range(sitting.scores, 0, 69)
        print(count_90_or_above)
        print(count_80_to_89)
        print(count_70_to_79)
        print(count_below_70)
        let max_score = match max(sitting.scores) {
            Some(n) => n,
            None => -1,
        }
        print(max_score)
        let mean_score = sum(sitting.scores) / len(sitting.scores)
        print(mean_score)
    }
    let morning_mean = sum(morning_sitting.scores) / len(morning_sitting.scores)
    let evening_mean = sum(evening_sitting.scores) / len(evening_sitting.scores)
    if morning_mean > evening_mean {
        print_str("Morning")
    } else {
        print_str("Evening")
    }
}
