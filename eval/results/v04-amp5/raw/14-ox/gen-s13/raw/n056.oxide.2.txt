fn sum_numbers_and_count_non_numbers(numbers: Vec<Str>) -> (Int, Int) {
    let mut sum = 0;
    let mut non_number_count = 0;
    for n in numbers {
        match parse_int(n) {
            Some(num) => sum += num,
            None => non_number_count += 1,
        }
    }
    (sum, non_number_count)
}

fn main() {
    let inputs = vec("7", "x", "9", "?");
    let (total, count) = sum_numbers_and_count_non_numbers(inputs);
    print(total);
    print(count);
}
