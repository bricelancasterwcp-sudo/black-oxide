```rust
fn reverse_number(n: Int) -> Str {
    let mut num = n;
    let mut reversed = int_to_str(0);
    
    while num > 0 {
        let digit = num % 10;
        reversed = concat(int_to_str(digit), reversed);
        num = num / 10;
    }
    
    reversed
}

fn main() {
    let number = 1234;
    let reversed_str = reverse_number(number);
    print_str(reversed_str);
}
```
