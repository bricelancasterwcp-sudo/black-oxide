```rust
fn main() {
    let numbers: Vec<&str> = vec!["7", "x", "9", "?"];
    let mut sum = 0;
    let mut count = 0;

    for &num in numbers.iter() {
        match num.parse::<i32>() {
            Ok(n) => {
                sum += n;
                count += 1;
            }
            Err(_) => {}
        }
    }

    println!("The total of the ones that do: {}", sum);
    println!("How many do not: {}", numbers.len() - count);
}
```
