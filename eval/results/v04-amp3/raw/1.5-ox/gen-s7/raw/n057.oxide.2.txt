```rust
fn main() {
    let names = vec!["fig", "banana", "kiwi"];
    let mut max_name = "";
    let mut max_length = 0;

    for name in &names {  // Use & to borrow the vector elements
        let length = name.len();
        if length > max_length {
            max_name = *name;  // Assign to a mutable reference
            max_length = length;
        }
    }

    println!("The longest name is '{}'", max_name);
    println!("It has {} letters", max_length);
}
```
