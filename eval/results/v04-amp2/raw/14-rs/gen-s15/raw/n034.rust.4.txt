```rust
fn main() {
    let numbers = vec![7, 8, 9];
    match get_second_entry(&numbers) {
        Some(second) => println!("The second entry is: {}", second),
        None => println!("There is no second entry."),
    }
}

fn get_second_entry<T>(list: &[T]) -> Option<&T> {
    if list.len() > 1 {
        Some(&list[1])
    } else {
        None
    }
}
```
