fn get_second_entry<T>(list: &[T]) -> Option<&T> {
    list.get(1)
}

fn main() {
    let list7 = vec![1, 2, 3];
    let list8 = vec![4, 5];
    let list9 = vec![6];

    println!("{:?}", get_second_entry(&list7)[..]); // Some([2])
    println!("{:?}", get_second_entry(&list8)[..]); // Some([5])
    println!("{:?}", get_second_entry(&list9)); // None
}
