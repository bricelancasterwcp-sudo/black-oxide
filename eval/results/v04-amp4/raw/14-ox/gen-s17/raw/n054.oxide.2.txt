struct LetterCount { vowels: Int, consonants: Int }

fn count_letters(s: Str) -> LetterCount {
    let vowels = "aeiou"
    let mut count = LetterCount { vowels: 0, consonants: 0 }
    for ch in chars(s) {
        let lower_ch = to_lower(ch)
        if contains(vowels, lower_ch) {
            count.vowels += 1
        } else if is_alpha(lower_ch) {
            count.consonants += 1
        }
    }
    count
}

fn to_lower(c: Str) -> Str {
    if 'A' <= c && c <= 'Z' {
        concat(chr(sub(c, chr('A'))), "a")
    } else {
        c
    }
}

fn is_alpha(c: Str) -> Bool {
    ('a' <= c && c <= 'z') || ('A' <= c && c <= 'Z')
}

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    let count = count_letters(keyboard)
    print(count.vowels)
    print(count.consonants)
}
