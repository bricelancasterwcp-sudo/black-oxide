fn count_and_max(v: Vec<Int>) -> (Int, Option<Int>) {
    let mut pos_count = 0;
    let mut max_val = None;
    for x in v {
        if x > 0 {
            pos_count += 1;
        }
        max_val = match max_val {
            Some(m) => Some(if m < x { x } else { m }),
            None => Some(x),
        };
    }
    (pos_count, max_val)
}

fn main() {
    let v = push(push(push(push(vec(), 3), 8), -2), 12);
    let v = push(v, 7);
    let (count, max) = count_and_max(v);
    print(count);
    match max {
        Some(m) => print(m),
        None => print_str("no values"),
    }
}
