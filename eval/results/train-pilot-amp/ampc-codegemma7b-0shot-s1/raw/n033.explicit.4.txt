fn is_green() -> Bool {
    true
}

fn wait_for_yellow() -> Unit {
    print("waiting for yellow\n")
}

fn stop_for_red() -> Unit {
    print("stopping\n")
}

fn main() {
    if is_green() {
        print("go for green\n")
        if wait_for_yellow() {
            drop wait_for_yellow
            stop_for_red()
        }
    }
}
