```rust
use std::cmp;

struct Point { x: Int, y: Int }

fn min_point(a: Point, b: Point) -> Point {
    if a.x < b.x || (a.x == b.x && a.y < b.y) {
        a
    } else {
        b
    }
}

fn max_point(a: Point, b: Point) -> Point {
    if a.x > b.x || (a.x == b.x && a.y > b.y) {
        a
    } else {
        b
    }
}

fn main() {
    let fig1 = Point { x: 5, y: 3 };
    let fig2 = Point { x: -4, y: -6 };
    let fig3 = Point { x: 0, y: 0 };
    let fig4 = Point { x: 10, y: 2 };

    let smallest = min_point(fig1, min_point(fig2, min_point(fig3, fig4)));
    let largest = max_point(fig1, max_point(fig2, max_point(fig3, fig4)));

    print_str(format!("Smallest figure: ({}, {})", smallest.x, smallest.y));
    print_str(format!("Gap between largest and smallest: {}", largest.y - smallest.y));
}
```
