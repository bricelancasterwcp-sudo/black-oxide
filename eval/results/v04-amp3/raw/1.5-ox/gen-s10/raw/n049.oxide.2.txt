```rust
struct Figure {
    name: Str,
    value: Int,
}

fn main() {
    let figures = vec![
        Figure { name: "fig8".to_string(), value: 12 },
        Figure { name: "fig3".to_string(), value: 7 },
        Figure { name: "fig11".to_string(), value: 9 },
        Figure { name: "fig6".to_string(), value: 5 },
    ];

    let smallest = figures.iter().map(|f| f.value).min().unwrap();
    let largest = figures.iter().map(|f| f.value).max().unwrap();
    println!("Smallest figure: {}", smallest);
    println!("Gap between largest and smallest: {}", largest - smallest);
}
```
