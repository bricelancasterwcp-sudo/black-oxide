fn struct_field_copy(seven: Int) {
    struct_field_copy_inner(seven)
}

fn struct_field_copy_inner(mut_seven: Int) {
    let reading = Reading { label: "sensor".to_string(), values: vec_push(vec_push(vec(), mut_seven), 7) }
    print(get(reading.values, 0).unwrap())
}

struct Reading { label: Str, values: Vec<Int> }

fn vec_push(v: Vec<Int>, x: Int) -> Vec<Int> {
    return push(v, x)
}

fn get(v: Vec<Int>, i: Int) -> Option<Int> {
    return match v.get(i) {
        Some(x) => Some(x.clone()),
        None => None,
    }
}

fn push(v: Vec<Int>, x: Int) -> Vec<Int> {
    return vec_push(v, x)
}
