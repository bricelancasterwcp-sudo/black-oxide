struct Reading { label: Str, values: Vec<Int> }
fn analyze(reading: Reading) -> Option<Str> {
    let mut increases = 0
    let mut decreases = 0
    let mut max_increase = 0
    let mut longest_streak = 0
    let mut current_streak = 0
    for i in range(1, len(reading.values)) {
        let prev = get(reading.values, i - 1)?
        let curr = get(reading.values, i)?
        if curr > prev {
            increases += 1
            max_increase = max(max_increase, curr - prev)
            current_streak += 1
        } else if curr < prev {
            decreases += 1
            current_streak = 0
        }
        longest_streak = max(longest_streak, current_streak)
    }
    print_str(reading.label)
    print(increases)
    print(decreases)
    print(max_increase)
    print(longest_streak)
    Some(reading.label)
}
fn main() {
    let north = Reading { label: "north", values: vec(12, 15, 19, 14, 14, 20, 31, 30) }
    let south = Reading { label: "south", values: vec(40, 38, 41, 45, 50, 44) }
    let north_result = analyze(north)?
    let south_result = analyze(south)?
    if longest_streak(north_result) > longest_streak(south_result) {
        print_str(north_result)
    } else {
        print_str(south_result)
    }
}
