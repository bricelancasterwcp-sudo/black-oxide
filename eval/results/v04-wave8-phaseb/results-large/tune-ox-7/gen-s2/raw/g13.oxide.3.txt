fn main() {
    let mut counter = 9
    let mut halved = 0
    let mut added_five = 0
    let mut tripled = 0
    let mut largest = 9
    let mut first_largest = 1
    let mut smallest = 9
    let mut running_total = 0
    let mut below_ten = 0
    for i in range(1, 31) {
        if counter % 2 == 0 {
            counter /= 2
            halved += 1
        } else if counter % 3 == 0 {
            counter += 5
            added_five += 1
        } else {
            counter = (counter * 3 + 1) % 1000
            tripled += 1
        }
        running_total += counter
        if counter > largest {
            largest = counter
            first_largest = i
        }
        if counter < smallest {
            smallest = counter
        }
        if counter < 10 {
            below_ten += 1
        }
    }
    print(largest)
    print(first_largest)
    print(smallest)
    print(running_total)
    print(below_ten)
    if largest > 100 {
        print_str("high")
    } else {
        print_str("low")
    }
}
