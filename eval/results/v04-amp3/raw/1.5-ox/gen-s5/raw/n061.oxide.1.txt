fn main() {
    let width = 9;
    let height = 3;

    // Calculate area (int multiplication)
    let area = width * height;
    println!("Area: {}", area);

    // Calculate perimeter (2 * length + 2 * width)
    let perimeter = 2 * (width + height);
    println!("Perimeter: {}", perimeter);
}
