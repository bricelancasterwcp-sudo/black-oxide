fn is_prime(n: Int) -> Bool {
    if n < 2 { return false }
    for i in range(2, trunc(to_float(n).sqrt()) + 1) {
        if to_int(trunc(to_float(n) / to_float(i))) * i == n {
            return false
        }
    }
    true
}

fn main() {
    print(if is_prime(91) { "yes" } else { "no" })
}
