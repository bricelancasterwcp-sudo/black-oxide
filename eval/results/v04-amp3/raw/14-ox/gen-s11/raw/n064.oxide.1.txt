struct Reading { label: Str, values: Vec<Int> }

fn look_up(v: Vec<Int>, pos: Int) -> Option<Int> {
    get(v, pos)
}

fn main() {
    let r = Reading { label: "numbers", values: push(push(vec(), 10), 20) }
    match look_up(r.values, 1) {
        Some(n) => print(n),
        None => print_str("none"),
    }
    match look_up(r.values, 4) {
        Some(n) => print(n),
        None => print_str("none"),
    }
}
