fn str_len(s: &Str) -> Int {
    let mut count = 0
    for _ in chars(s) {
        count = count + 1
    }
    count
}

fn str_lit(s: Str) -> Str {
    s
}

fn main() {
    let str_lit_str_hello_world = str_lit("hello world")
    print(str_len(&str_lit_str_hello_world))
    drop str_lit_str_hello_world
}
