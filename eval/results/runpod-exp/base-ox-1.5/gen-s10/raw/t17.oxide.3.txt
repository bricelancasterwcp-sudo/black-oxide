fn main() {
    let list = vec![10, 20, 30];

    if list.len() > 1 {
        print(list[1]); // element at position 1
    } else {
        print(-1); // position 1 is outside the list
    }

    if list.len() > 4 {
        print(list[4]); // element at position 5
    } else {
        print(-1); // position 5 is outside the list
    }
}
