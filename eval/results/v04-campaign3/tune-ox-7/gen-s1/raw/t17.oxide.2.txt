struct Vec<Int> { items: Vec<Int> }
fn get(v: Vec<Int>, pos: Int) -> Option<Int> {
    if pos < 0 || pos >= len(v.items) {
        None
    } else {
        Some(get(v.items, pos).unwrap_or(-1))
    }
}
