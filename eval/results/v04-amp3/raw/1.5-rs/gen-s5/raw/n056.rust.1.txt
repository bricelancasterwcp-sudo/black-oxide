```rust
fn main() {
    let texts = ["7", "x", "9", "?"];
    let mut total_numbers = 0;
    let mut count_not_numbers = 0;

    for &text in texts.iter() {
        if let Ok(number) = text.parse::<i32>() {
            total_numbers += number;
        } else {
            count_not_numbers += 1;
        }
    }

    println!("Total of numbers: {}", total_numbers);
    println!("Number of not numbers: {}", count_not_numbers);
}
```
