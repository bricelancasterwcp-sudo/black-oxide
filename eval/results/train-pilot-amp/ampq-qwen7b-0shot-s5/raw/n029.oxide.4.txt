struct TextBuilder { content: Str }

fn to_string(s: Str) -> Str {
    s
}

fn build_text() -> TextBuilder {
    let mut builder = TextBuilder { content: "one" }
    builder.content = builder.content + "two"
    builder.content = builder.content + "three"
    builder
}

fn main() {
    let text = build_text()
    print_str(text.content)
}
