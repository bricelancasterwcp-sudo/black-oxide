```rust
use std::io;

fn main() {
    let mut sum = 0;
    let mut count = 0;
    io::stdin().read_line(&mut sum).expect("Failed to read line");
    io::stdin().read_line(&mut count).expect("Failed to read line");

    let numbers: Vec<&str> = sum.split_whitespace().collect();
    for &s in numbers.iter() {
        if s.parse::<i32>().is_ok() {
            sum += s.parse::<i32>().unwrap();
            count += 1;
        }
    }

    println!("Total: {}", sum);
    println!("Count: {}", count);
}
```
