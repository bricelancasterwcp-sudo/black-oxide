fn double_if_odd(x: Int) -> Option<Int> {
    if x % 2 == 1 { Some(x * 2) } else { None }
}

fn sum_evens(v: Vec<Int>) -> Int {
    let mut sum = 0;
    for x in v {
        if x % 2 == 0 {
            sum += x;
        }
    }
    sum
}

fn main() {
    let numbers = vec![7, 4, 9, 10];
    for num in numbers.clone() {
        match double_if_odd(num) {
            Some(doubled) => print_str(&doubled.to_string()),
            None => (),
        }
    }
    let evens_sum = sum_evens(numbers);
    print(evens_sum)
}
