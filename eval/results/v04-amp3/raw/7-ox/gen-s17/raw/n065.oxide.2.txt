fn count_above_five(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 5 {
            count += 1
        }
    }
    count
}

fn max_reading(v: Vec<Int>) -> Option<Int> {
    let sorted_v = sort(v)?;
    Some(sorted_v[len(sorted_v) - 1])
}

fn main() {
    let readings = push(push(push(push(vec(), 6), 11), 3), 14)
    let readings = push(readings, 9);
    print(count_above_five(readings));
    let max_val = max_reading(readings)?;
    print(max_val)
}
