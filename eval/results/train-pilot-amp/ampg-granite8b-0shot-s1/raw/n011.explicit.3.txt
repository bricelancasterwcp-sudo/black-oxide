fn main() {
    let mut multiples = vec()
    for i in range(0, 6) {
        push(multiples, 4 * i)
    }
    let total = add(&mutmultiples, &mutmultiples)
    print(total)
    drop multiples
}

fn add(a: &Vec<Int>, b: &Vec<Int>) -> Int {
    let mut result = 0
    for i in range(0, a.len()) {
        result = result + a.get(i).unwrap()
    }
    for i in range(0, b.len()) {
        result = result + b.get(i).unwrap()
    }
    result
}

fn range(a: Int, b: Int) -> Vec<Int> {
    let mut result = vec()
    for i in a.to(b) {
        push(result, i)
    }
    result
}

fn push(mutvec: &Vec<Int>, new_item: Int) -> Vec<Int> {
    let mut v = clone(&mutvec)
    for i in range(0, v.len()) {
        v = functional_update(v, i, v.get(i).unwrap())
    }
    functional_update(v, v.len(), new_item)
}

fn functional_update(vec: Vec<Int>, i: Int, new_item: Int) -> Vec<Int> {
    let mut result = vec
    result.set(i, new_item)
    result
}

fn clone(v: &Vec<Int>) -> Vec<Int> {
    push(vec(), v.get(0).unwrap())
}

fn print(x: Int) {
    print_str(format_int(x))
}

fn format_int(x: Int) -> Str {
    let mut s = str()
    if x == 0 {
        push(s, "0")
    } else {
        let mut n = x
        let mut digits = vec()
        while n != 0 {
            push(digits, (n % 10).to_string())
            n = n / 10
        }
        for i in range(0, digits.len()) {
            push(s, digits.get(digits.len() - i - 1).unwrap())
        }
    }
    s
}
