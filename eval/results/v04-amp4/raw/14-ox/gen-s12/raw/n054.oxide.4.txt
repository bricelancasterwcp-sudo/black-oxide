struct LetterCount { vowels: Int, consonants: Int }

fn is_vowel(c: Str) -> Bool {
    match c {
        "a" => true,
        "e" => true,
        "i" => true,
        "o" => true,
        "u" => true,
        "A" => true,
        "E" => true,
        "I" => true,
        "O" => true,
        "U" => true,
        _ => false
    }
}

fn count_letters(s: Str) -> LetterCount {
    let mut count = LetterCount { vowels: 0, consonants: 0 }
    for c in chars(s) {
        if is_vowel(c) {
            count.vowels += 1
        } else if str_len(c) > 0 { // ensure it's a letter
            count.consonants += 1
        }
    }
    count
}

fn main() {
    let keyboard = "qwertyuiopasdfghjklzxcvbnm"
    let result = count_letters(keyboard)
    print(result.vowels)
    print(result.consonants)
}
