fn count_positives(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count = count + 1
        }
    }
    count
}

fn find_max(v: Vec<Int>) -> Option<Int> {
    let mut max = None
    for x in v {
        max = match max {
            Some(current_max) => {
                if x > current_max {
                    Some(x)
                } else {
                    Some(current_max)
                }
            },
            None => {
                Some(x)
            },
        }
    }
    max
}

fn main() {
    let values = vec()
    values.push(3)
    values.push(8)
    values.push(-2)
    values.push(12)
    values.push(7)

    let positives_count = count_positives(values.clone())
    print(positives_count)

    match find_max(clone(values)) {
        Some(max_value) => print(max_value),
        None => print_str("No values"),
    }
}
