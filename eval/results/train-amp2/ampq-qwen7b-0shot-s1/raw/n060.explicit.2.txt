fn sum_and_count(digits: &Vec<Str>) -> Int {
    let total = 0
    let count = 0
    for d in digits {
        let num = parse_int(d).unwrap()
        total = total + num
        count = count + 1
    }
    total + count
}

fn main() {
    let nums = vec("1", "2", "3", "4", "5")
    let result = sum_and_count(&nums)
    print(result)
    drop nums
}
