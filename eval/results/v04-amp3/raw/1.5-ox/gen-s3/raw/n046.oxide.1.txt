```rust
fn main() {
    let values = vec![5, 12, 3, 18, 9];
    let mut count_below_cutoff = 0;
    let mut count_at_or_over_cutoff = 0;

    for &value in &values {
        if value < 10 {
            count_below_cutoff += 1;
        } else if value >= 10 {
            count_at_or_over_cutoff += 1;
        }
    }

    println!("Number of values below cutoff: {}", count_below_cutoff);
    println!("Number of values at or over cutoff: {}", count_at_or_over_cutoff);
}
```
