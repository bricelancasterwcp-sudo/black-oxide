fn main() {
    let n = 91
    if n == 1 {
        print_str("no")
    } else {
        let mut i = 2
        while i * i <= n {
            if n % i == 0 {
                print_str("no")
                break
            }
            i = i + 1
        }
        if i * i > n {
            print_str("yes")
        }
    }
}
