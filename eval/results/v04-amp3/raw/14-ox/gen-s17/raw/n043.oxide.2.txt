struct Queue { items: Vec<Int> }

fn exchange_first_last(q: Queue) -> Option<Queue> {
    let len = len(q.items)
    if len < 2 {
        return Some(q)
    }
    let first = get(q.items, 0)?
    let last = get(q.items, len - 1)?
    let middle = vec()
    for i in range(1, len - 1) {
        push(middle, get(q.items, i)?)
    }
    Some(Queue { items: push(push(middle, last), first) })
}

fn main() {
    let q = Queue { items: vec(1, 2, 3, 4, 5) }
    match exchange_first_last(q) {
        Some(exchanged) => {
            for x in exchanged.items {
                print(x)
            }
        },
        None => print_str("Queue is too short"),
    }
}
