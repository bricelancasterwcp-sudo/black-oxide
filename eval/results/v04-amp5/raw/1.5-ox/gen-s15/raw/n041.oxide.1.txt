```rust
struct Measurement { value: Int }

fn main() {
    let measurements = vec![Measurement { value: 9 }, Measurement { value: 2 },
                            Measurement { value: 7 }, Measurement { value: 4 }];
    measurements.sort_by(|a, b| a.value.cmp(&b.value));
    for m in measurements {
        print!("{}", m.value);
    }
}
```
