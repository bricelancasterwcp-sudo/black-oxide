```rust
fn main() {
    let input = "a1b22";

    let mut total = 0;
    let mut count = 0;

    for c in input.chars() {
        if c.is_digit(10) {
            total += c.to_digit(10).unwrap();
            count += 1;
        }
    }

    print!("Total digits: {}\n", total);
    print!("Count of digits: {}\n", count);
}
```
