fn count(s: &Str) -> Result<Int, String> {
    let mut digits = 0
    let mut total = 0
    for c in s.chars() {
        if c.is_digit(10) {
            digits = digits + 1
            total = total + c.to_digit(10).unwrap()
        }
    }
    Ok((total))
}

fn main() {
    let text = "a1b22"
    let result = count(text)
    match result {
        Ok(total) => println("Total: {}", total),
        Err(e) => println("Error: {}", e),
    }
}
