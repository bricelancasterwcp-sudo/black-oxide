```rust
struct Queue { items: Vec<Int> }

fn push(q: Queue, item: Int) -> Queue {
    Queue { items: push(q.items, item) }
}

fn pop(q: Queue) -> (Int, Queue) {
    let (item, rest) = pop(q.items);
    (item, Queue { items: rest })
}

fn first(q: &Queue) -> Int {
    q.items[0]
}

fn last(q: &Queue) -> Int {
    q.items[q.items.len() - 1]
}

fn swap_first_last(mut q: Queue) -> Queue {
    let n = q.items.len();
    if n > 0 {
        let (first, mut rest) = pop(q);
        let last = last(&q);
        for i in 0..n-2 {
            rest = push(rest, first);
            let (item, new_rest) = pop(rest);
            rest = new_rest;
            first = item;
        }
        q = push(Queue { items: vec![last] }, first);
    }
    q
}

fn main() {
    let mut q = Queue { items: vec![1, 2, 3, 4, 5] };
    q = swap_first_last(q);
    while let Some(item) = pop(q).0 {
        print(item);
    }
}
```
