fn main() {
    let mut n = 44
    let mut seen = vec()
    let mut max_val = 0
    let mut min_val = 999999999
    let mut sum_vals = 0
    let mut even_count = 0
    let mut digit_sum = 0
    let mut rounds = 0
    while n != 1 && rounds < 50 {
        seen = push(seen, n)
        max_val = max(max_val, n)
        min_val = min(min_val, n)
        sum_vals += n
        if n % 2 == 0 {
            even_count += 1
        }
        let mut next_n = 0
        for c in int_to_str(n).chars() {
            let d = parse_int(c).unwrap()
            digit_sum += d
            next_n += d * d
        }
        n = next_n
        rounds += 1
    }
    print(rounds)
    print(n)
    if rounds > 0 { print(seen[rounds - 1]) } else { print_str("none") }
    print(max_val)
    print(min_val)
    print(sum_vals)
    print(even_count)
    print(digit_sum)
    print(n == 1)
}
