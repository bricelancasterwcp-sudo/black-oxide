```rust
fn main() {
    let mut count_vowels = 0;
    let mut count_not_vowels = 0;

    for c in "keyboard".chars() {
        if c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' || c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U' {
            count_vowels += 1;
        } else {
            count_not_vowels += 1;
        }
    }

    println!("Vowels: {}", count_vowels);
    println!("Consonants: {}", count_not_vowels);
}
```
