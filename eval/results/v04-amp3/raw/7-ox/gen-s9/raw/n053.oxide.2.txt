fn reverse(s: Str) -> Str {
    let mut chars = chars(s)
    let mut reversed = vec()
    while let Some(c) = get(chars, 0)? {
        reversed = push(reversed, concat("", c))
        chars = drop(get(chars, 1)?)
    }
    concat(reversed, "")
}
