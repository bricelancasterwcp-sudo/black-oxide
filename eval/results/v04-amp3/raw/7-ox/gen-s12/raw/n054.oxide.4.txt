fn main() {
    let vowels = ['a', 'e', 'i', 'o', 'u'];
    let mut count_vowels = 0;
    let mut count_not_vowels = 0;

    for c in "keyboard".chars() {
        if vowels.contains(&c) {
            count_vowels += 1;
        } else {
            count_not_vowels += 1;
        }
    }

    print(count_vowels);
    print_str(" ");
    print(count_not_vowels);
}
