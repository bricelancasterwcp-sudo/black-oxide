```rust
fn main() {
    let mut count_vowels = 0;
    let mut count_consonants = 0;

    for letter in "keyboard".chars() {
        match letter.to_ascii_lowercase() {
            'a' | 'e' | 'i' | 'o' | 'u' => count_vowels += 1,
            _ => count_consonants += 1,
        }
    }

    println!("Vowel count: {}", count_vowels);
    println!("Consonant count: {}", count_consonants);
}
```
