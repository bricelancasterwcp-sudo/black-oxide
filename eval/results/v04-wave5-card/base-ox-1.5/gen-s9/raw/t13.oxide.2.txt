```rust
fn main() {
    println!("{}", is_palindrome("level"));  // true
    println!("{}", is_palindrome("stereo")); // true
    println!("{}", is_palindrome("rotor"));  // false
}

fn is_palindrome(s: &str) -> bool {
    let mut i = 0;
    let mut j = s.len() - 1;
    while i < j {
        if s.chars().nth(i).unwrap() != s.chars().nth(j).unwrap() {
            return false;
        }
        i += 1;
        j -= 1;
    }
    true
}
```
