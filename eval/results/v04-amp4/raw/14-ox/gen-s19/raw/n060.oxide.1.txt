fn count_digits(s: Str) -> Int {
    let mut sum = 0
    let mut count = 0
    for ch in chars(s) {
        if parse_int(ch).is_some() {
            sum += to_float(parse_int(ch).unwrap())
            count += 1
        }
    }
    print(sum)
    print(count)
}

fn main() {
    let input = "a1b22"
    count_digits(input)
}
