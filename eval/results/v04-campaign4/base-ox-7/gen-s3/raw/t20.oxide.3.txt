struct Item { name: Str, unit_price: Int, quantity: Int }

fn main() {
    let pen = Item { name: "pen", unit_price: 3, quantity: 4 };
    let pad = Item { name: "pad", unit_price: 9, quantity: 2 };
    let ink = Item { name: "ink", unit_price: 5, quantity: 3 };

    let total_cost = pen.unit_price * pen.quantity + pad.unit_price * pad.quantity + ink.unit_price * ink.quantity;
    print(total_cost);

    let mut max_cost_item = pen.clone();
    if pad.unit_price * pad.quantity > max_cost_item.unit_price * max_cost_item.quantity {
        max_cost_item = pad.clone();
    }
    if ink.unit_price * ink.quantity > max_cost_item.unit_price * max_cost_item.quantity {
        max_cost_item = ink.clone();
    }

    print_str(max_cost_item.name);
}
