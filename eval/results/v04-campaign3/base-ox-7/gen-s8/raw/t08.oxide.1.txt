fn count_positive(v: Vec<Int>) -> Int {
    let mut count = 0
    for x in v {
        if x > 0 {
            count += 1
        }
    }
    count
}

fn max_value(v: Vec<Int>) -> Option<Int> {
    let mut max_val = None
    for x in v {
        if let Some(m) = max_val {
            if x > m {
                max_val = Some(x)
            }
        } else {
            max_val = Some(x)
        }
    }
    max_val
}

fn main() {
    let values = push(push(push(push(vec(), 3), 8), -2), 12)
    let values = push(values, 7)
    print(count_positive(values))
    match max_value(values) {
        Some(n) => print(n),
        None => print_str("empty list"),
    }
}
