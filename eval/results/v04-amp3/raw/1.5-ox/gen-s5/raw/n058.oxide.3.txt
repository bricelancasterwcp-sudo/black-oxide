struct Coffee {
    c: Int,
    o: Int,
    f: Int,
    e: Int,
    r: Int,
}

fn main() {
    let mut coffee = Coffee { c: 0, o: 0, f: 0, e: 0, r: 0 };
    for letter in "coffee".chars().map(|l| l.to_ascii_lowercase()) {
        match letter {
            'c' => coffee.c += 1,
            'o' => coffee.o += 1,
            'f' => coffee.f += 1,
            'e' => coffee.e += 1,
            'r' => coffee.r += 1,
            _ => (),
        }
    }

    for letter in "coffee".chars().map(|l| l.to_ascii_lowercase()) {
        if coffee.letter == 0 {
            print!("{}", letter);
            coffee.letter = 1;
        }
    }
}
